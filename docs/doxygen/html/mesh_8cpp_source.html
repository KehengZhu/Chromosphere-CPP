28RefineProfile env_profile(const char* key, RefineProfile fallback) {
29 const char* e = std::getenv(key);
30 if (!e) return fallback;
31 std::string v(e);
32 for (char& c : v) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
33 if (v == "outer") return RefineProfile::Outer;
34 if (v == "lower") return RefineProfile::Lower;
35 return fallback;

38// Append a UNIFORM segment (seg_start, seg_end] whose cells are AT MOST
39// `target_ds` wide (ceil, so a segment is never coarser than its target — that is
40// what keeps the ratio at a junction with a graded segment bounded). Both
41// endpoints land on faces; the starting face is assumed already present as
42// faces.back() and the last appended face is snapped to seg_end exactly.
43void append_uniform(std::vector<double>& faces, double seg_start, double seg_end,
44 double target_ds) {
45 const double span = seg_end - seg_start;
46 const arma::uword n = static_cast<arma::uword>(
47 std::max(1.0, std::ceil(span / std::max(target_ds, 1.0e-30) - 1.0e-9)));
48 const double w = span / static_cast<double>(n);
49 for (arma::uword j = 1; j <= n; ++j)
50 faces.push_back(seg_start + w * static_cast<double>(j));
51 faces.back() = seg_end;
52}

54// An exponential (geometric) grade bridging cell width `fine_ds` to `coarse_ds`.
55//
56// Cell widths are fine_ds·r^m, m = 0…n-1, with r = (coarse_ds/fine_ds)^{1/n}, so
57// the narrow end matches `fine_ds` EXACTLY and the wide end is coarse_ds/r. Every
58// adjacent ratio — inside the grade and at BOTH junctions — is then r, which is
59// ≤ MESH_MAX_RATIO by construction.
60//
61// That fixes both the width and the cell count: `width` is an OUTPUT, and the
62// requested transition width acts as a MINIMUM. `n` is the smallest count whose
63// grade is at least `tau_min_m` wide (the width fine_ds·(f−1)/(r−1) grows
64// monotonically with n), so a requested width the ratio cap cannot support is
65// widened to the nearest one it can, instead of silently breaking the cap.
66struct Grade {
67 arma::uword n = 0;
68 double ratio = 1.0;
69 double width = 0.0;
70};
71
72Grade make_grade(double fine_ds, double coarse_ds, double tau_min_m) {
73 Grade g;
74 const double f = coarse_ds / fine_ds;
75 if (!(f > 1.0 + 1.0e-9) || !(fine_ds > 0.0)) return g; // no grade needed
76 auto width_of = [&](arma::uword n) {
77 const double r = std::pow(f, 1.0 / static_cast<double>(n));
78 return fine_ds * (f - 1.0) / (r - 1.0); // Σ fine·r^m, m<n
79 };
80 // Ratio cap: r = f^{1/n} ≤ MESH_MAX_RATIO ⇒ n ≥ ln f / ln cap.
81 arma::uword n = static_cast<arma::uword>(
82 std::max(1.0, std::ceil(std::log(f) / std::log(MESH_MAX_RATIO))));
83 while (width_of(n) < tau_min_m && n < 1000000u) ++n;
84 g.n = n;
85 g.ratio = std::pow(f, 1.0 / static_cast<double>(n));
86 g.width = width_of(n);
87 return g;
88}

90// Append the graded transition of width `g.width` starting at `seg_start`, with
91// the cells running fine→coarse when `outward_coarsening` and coarse→fine
92// otherwise. Node offsets use the closed-form partial sums (no accumulation
93// drift); the final face is snapped to seg_start + g.width exactly.
94void append_grade(std::vector<double>& faces, double seg_start, double fine_ds,
95 const Grade& g, bool outward_coarsening) {
96 const double c = fine_ds / (g.ratio - 1.0); // Σ_{m<j} fine·r^m = c·(r^j − 1)
97 for (arma::uword j = 1; j <= g.n; ++j) {
98 const double from_fine_end =
99 c * (std::pow(g.ratio, static_cast<double>(j)) - 1.0);
100 faces.push_back(seg_start + (outward_coarsening ? from_fine_end
101 : g.width - c *
102 (std::pow(g.ratio, static_cast<double>(g.n - j)) - 1.0)));
103 }
104 faces.back() = seg_start + g.width;
105}

111 rp.factor = env_double("GRID_REFINE_FACTOR", rp.factor);
112 rp.s_lo_km = env_double("GRID_REFINE_S_LO_KM", rp.s_lo_km);

117 rp.factor = env_double("ISO_REFINE_FACTOR", rp.factor);
118 rp.s_lo_km = env_double("ISO_REFINE_S_LO_KM", rp.s_lo_km);
119 rp.s_hi_km = env_double("ISO_REFINE_S_HI_KM", rp.s_hi_km);
120 rp.transition_km = env_double("ISO_REFINE_TRANSITION_KM", rp.transition_km);
121 rp.profile = env_profile("ISO_REFINE_PROFILE", rp.profile);

126std::vector<double> build_static_mesh_faces(double length_m, arma::uword ns_coarse,
127 const RefineParams& rp) {
128 const double coarse_ds = length_m / static_cast<double>(ns_coarse);
129
130 const double s_lo = rp.s_lo_km * 1000.0;
131 const double s_hi = rp.s_hi_km * 1000.0;
132 const double fine_ds = coarse_ds / rp.factor;
133 const bool outer = (rp.profile == RefineProfile::Outer);
134
135 // The graded transition's width is set by the ratio cap; the requested
136 // ISO_REFINE_TRANSITION_KM is its MINIMUM (see make_grade).
137 const Grade grade = (rp.enabled() && fine_ds > 0.0)
138 ? make_grade(fine_ds, coarse_ds, rp.transition_km * 1000.0)
139 : Grade{};
140 const double tau = grade.width;
141 const double s_end = s_hi + tau; // top of transition (Lower)
142 const double s_mid = s_lo - tau; // foot of transition (Outer)
143
144 // Uniform fallback: refinement off, or the refined+transition window does not
145 // fit strictly inside the domain, or a degenerate window. Byte-for-byte the
146 // k·coarse_ds grid a scenario builds when refinement is disabled.
147 const bool fits = rp.enabled() && fine_ds > 0.0 && grade.n > 0 &&
148 (outer ? (s_mid > 0.0 && s_lo < length_m)
149 : (s_lo >= 0.0 && s_hi > s_lo && s_end < length_m));
150 if (!fits) {
151 std::vector<double> faces(ns_coarse + 1);
152 for (arma::uword k = 0; k <= ns_coarse; ++k)
153 faces[k] = coarse_ds * static_cast<double>(k);
154 faces.back() = length_m;
155 return faces;
156 }
157
158 std::vector<double> faces;
159 faces.push_back(0.0);
160
161 if (outer) {
162 // OUTER profile: coarse → grade coarse→fine → fine up to the domain top.
163 append_uniform(faces, 0.0, s_mid, coarse_ds); // [0, s_lo − τ]
164 append_grade(faces, s_mid, fine_ds, grade, /*outward_coarsening=*/false);
165 append_uniform(faces, s_lo, length_m, fine_ds); // [s_lo, length]
166 return faces;
167 }
168
169 // [0, s_lo] — coarse (only present when the refined region does not start at
170 // the inner face). Retains the outer coarse spacing below s_lo.
171 if (s_lo > 0.0) append_uniform(faces, 0.0, s_lo, coarse_ds);
172
173 // [s_lo, s_hi] — fine, ≈ coarse_ds/factor, exact endpoints.
174 append_uniform(faces, s_lo, s_hi, fine_ds);
175
176 // [s_hi, s_hi + τ] — exponential transition fine→coarse.
177 append_grade(faces, s_hi, fine_ds, grade, /*outward_coarsening=*/true);
178
179 // [s_hi + τ, length] — coarse, retains the original outer spacing.
180 append_uniform(faces, s_end, length_m, coarse_ds);
181
182 return faces;
183}
