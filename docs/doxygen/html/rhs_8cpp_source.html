18// Well-balanced gravity correction (chromosphere.hpp::well_balanced).
19// cons2prim of a SPATIALLY-SHIFTED conserved state recovers the wrong thermal
20// pressure: it subtracts the LOCAL-index φ_g, but the shifted cell's total energy
21// carries the SHIFTED cell's ρφ_g, so each neighbour pressure is biased by
22// ⅔ρ·Δφ_g = ⅔ρg·Δs. That flattens the reconstructed hydrostatic slope dp/ds = −ρg
23// to −⅓ρg, giving a hydrostatic atmosphere a spurious resolution-independent
24// (γ−1)g downforce. Undo it: subtract ⅔ρ(φ_src − φ_local) from p_i and p_n, where
25// φ_src is the cell-centred potential actually baked into the shifted data's
26// energy. p_e carries no φ_g (shift-invariant) and is left untouched. Stage-D
27// conduction already does the analogous φ_g shift (rhs_implicit_state).
28static void wb_correct_shifted(const Grid& grid, Vec& prim,
29 const Vec& phi_src, const Vec& phi_cell) {
30 const Vec d = grid.gm1() * (phi_src - phi_cell);
31 const Vec rho_i = get_scalar(grid, prim, prim::RHO_I);
32 const Vec rho_n = get_scalar(grid, prim, prim::RHO_N);
33 prim -= scalar_to(grid, rho_i % d, prim::P_I);
34 prim -= scalar_to(grid, rho_n % d, prim::P_N);
35}

37// Log-space reconstruction (chromosphere.hpp::log_reconstruct). The exponentially-
38// stratified, strictly-positive primitive slots — the densities and pressures — are
39// transformed to log before the limiter/extrapolation arithmetic and back (exp) on
40// every reconstructed face before prim2cons, so an isothermal-hydrostatic column is
41// piecewise-linear and minmod no longer clips its smooth gradient to first order
42// (the lower-boundary diffusion source). The SIGNED velocities {V, U} are left linear
43// (log undefined for them, and they carry no large dynamic range). A scalar block for
44// primitive variable v occupies the contiguous subvec [v·ns, v·ns+ns-1] (cons2prim /
45// scalar_to packing). The pre-log clamp guards log(0) at a vacuum cell; exp is safe.
46static const arma::uword LOG_PRIM_VARS[5] =
47 {prim::RHO_I, prim::RHO_N, prim::P_I, prim::P_N, prim::P_E};
48
49static void prim_to_log(const Grid& grid, Vec& prim) {
50 const arma::uword ns = grid.ns;
51 for (arma::uword v : LOG_PRIM_VARS) {
52 const arma::uword lo = v * ns, hi = lo + ns - 1;
53 const Vec blk = prim.subvec(lo, hi);
54 prim.subvec(lo, hi) = arma::log(arma::clamp(blk, 1.0e-30f, arma::datum::inf));
55 }
56}
57
58static void prim_from_log(const Grid& grid, Vec& prim) {
59 const arma::uword ns = grid.ns;
60 for (arma::uword v : LOG_PRIM_VARS) {
61 const arma::uword lo = v * ns, hi = lo + ns - 1;
62 const Vec blk = prim.subvec(lo, hi);
63 prim.subvec(lo, hi) = arma::exp(blk);
64 }
65}

67// Broadcast a length-ns cell field across all num_of_eq equation slots — the
68// same packing scalar_to/get_scalar use — so a per-cell metric vector can
69// multiply a packed primitive state elementwise. Used by the non-uniform MUSCL
70// reconstruction below.
71static Vec bcast_state(const Grid& grid, const Vec& v_i) {
72 Vec s(grid.n_state, arma::fill::zeros);
73 for (arma::uword k = 0; k < num_of_eq; ++k) s += scalar_to(grid, v_i, k);
74 return s;
75}

78// Explicit RHS: TVD-MUSCL reconstruction + predictor-corrector +
79// Rusanov flux differencing, minus the explicit source.
80// (1.5D field-aligned; z-direction not considered. CoMFi Re_MUSCL analogue.)

82
83Vec rhs_explicit_state(const Grid& grid, const Vec& xn_state) {
84 if (!grid.eos_gamma_table.empty())
85 throw std::logic_error(
86 "the legacy two-fluid RHS cannot advance a Gamma1-table Grid; "
87 "use the release solver in mixture.hpp");
88 // ---- shifted neighbours (conserved) ---------------------------------
89 Vec cons_xn_state_ip1 = ip1(grid, xn_state);
90 Vec cons_xn_state_im1 = im1(grid, xn_state);
91 Vec cons_xn_state_ip2 = ip2(grid, xn_state);
92 Vec cons_xn_state_im2 = im2(grid, xn_state);
93
94 // ---- limit on primitive variables -----------------------------------
95 Vec prim_xn_state = cons2prim(grid, xn_state);
96 Vec prim_xn_state_ip1 = cons2prim(grid, cons_xn_state_ip1);
97 Vec prim_xn_state_im1 = cons2prim(grid, cons_xn_state_im1);
98 Vec prim_xn_state_ip2 = cons2prim(grid, cons_xn_state_ip2);
99 Vec prim_xn_state_im2 = cons2prim(grid, cons_xn_state_im2);
100
101 // Well-balanced gravity: correct the φ_g inconsistency in every shifted-state
102 // pressure (wb_correct_shifted). φ_src is the cell-centred potential baked into
103 // the shifted data's energy, matching what the ip*/im* shifts pull in: interior
104 // → neighbour φ_cell; the outer/inner ghosts → the face potentials the BCs used
105 // when packing the ghost energies (as rhs_implicit_state does for conduction).
106 // The same correction is reused for the predictor-shifted states below.
107 const Vec phi_cell = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
108 Vec phi_ip1, phi_im1;
109 if (grid.well_balanced) {
110 const arma::uword ns = grid.ns;
111 phi_ip1 = ip1(grid, phi_cell, SLICE); phi_ip1[ns - 1] = grid.phi_g_iph[ns - 1];
112 phi_im1 = im1(grid, phi_cell, SLICE); phi_im1[0] = grid.phi_g_imh[0];
113 Vec phi_ip2 = ip1(grid, phi_ip1, SLICE); phi_ip2[ns - 1] = grid.phi_g_iph[ns - 1];
114 Vec phi_im2 = im1(grid, phi_im1, SLICE); phi_im2[0] = grid.phi_g_imh[0];
115 wb_correct_shifted(grid, prim_xn_state_ip1, phi_ip1, phi_cell);
116 wb_correct_shifted(grid, prim_xn_state_im1, phi_im1, phi_cell);
117 wb_correct_shifted(grid, prim_xn_state_ip2, phi_ip2, phi_cell);
118 wb_correct_shifted(grid, prim_xn_state_im2, phi_im2, phi_cell);
119 }
120
121 // Log-space reconstruction: transform the stratified ρ/p slots to log (AFTER the
122 // well-balanced pressure correction above, which needs linear p). All the slope
123 // differences, limiter ratios and edge extrapolations below then run in log space
124 // for those slots; the faces are exp'd back before prim2cons. V, U stay linear.
125 if (grid.log_reconstruct) {
126 prim_to_log(grid, prim_xn_state);
127 prim_to_log(grid, prim_xn_state_ip1);
128 prim_to_log(grid, prim_xn_state_im1);
129 prim_to_log(grid, prim_xn_state_ip2);
130 prim_to_log(grid, prim_xn_state_im2);
131 }
132
133 const Vec dxn_state_iph = prim_xn_state_ip1 - prim_xn_state;
134 const Vec dxn_state_imh = prim_xn_state - prim_xn_state_im1;
135
136 if ((dxn_state_iph.has_nan() + dxn_state_imh.has_nan()) > 0) {
137 std::cout << "XN" << std::endl;
138 print_xn(grid, xn_state);
139 std::cout << "XNPRIM" << std::endl;
140 print_xn(grid, prim_xn_state);
141 std::cout << "dxn_iph has nan: " << dxn_state_iph.has_nan() << std::endl
142 << "dxn_imh has nan: " << dxn_state_imh.has_nan() << std::endl;
143 std::exit(1);
144 }
145
146 // limiter inputs
147 Vec r_state = dxn_state_imh / dxn_state_iph;
148 Vec r_state_ip1 = dxn_state_iph / (prim_xn_state_ip2 - prim_xn_state_ip1);
149 Vec r_state_im1 = (prim_xn_state_im1 - prim_xn_state_im2) / dxn_state_imh;
150
151 // ---- non-uniform-mesh metric corrections ----------------------------
152 // The MUSCL slope ratios above compare raw primitive DIFFERENCES (correct
153 // only when the cells are equally spaced). On a refined mesh the limiter must
154 // instead compare CENTER-TO-CENTER GRADIENTS, and each face is reconstructed
155 // with the owning cell's HALF-WIDTH — so the difference terms carry the
156 // factor 0.5·ds_i/ds_{i±½}. Both reduce to the legacy form on a uniform mesh
157 // (ratio metric = 1, weight = 0.5), so the reconstruction stays byte-for-byte
158 // identical there (the uniform branch below runs the original literals).
159 // r_i gradient ratio ×= ds_{i+½}/ds_{i-½}
160 // r_{i+1} ×= ds_{i+3/2}/ds_{i+½} (mirrored-ghost width shift)
161 // r_{i-1} ×= ds_{i-½}/ds_{i-3/2}
162 // Reconstruction weights (state-packed):
163 // W1 = 0.5·ds_i/ds_{i+½} (owner cell i: Lxn_iph, Rxn_imh)
164 // W3 = ip1(W1) = 0.5·ds_{i+1}/ds_{i+3/2} (owner i+1: Rxn_iph)
165 // W4 = im1(W1) = 0.5·ds_{i-1}/ds_{i-½} (owner i-1: Lxn_imh)
166 Vec W1, W3, W4;
167 if (!grid.uniform_mesh) {
168 const Vec w1_i = 0.5f * grid.ds_i / grid.ds_iph_i;
169 W1 = bcast_state(grid, w1_i);
170 W3 = bcast_state(grid, ip1(grid, w1_i, SLICE));
171 W4 = bcast_state(grid, im1(grid, w1_i, SLICE));
172 r_state %= bcast_state(grid, grid.ds_iph_i / grid.ds_imh_i);
173 r_state_ip1 %= bcast_state(grid, ip1(grid, grid.ds_iph_i, SLICE) / grid.ds_iph_i);
174 r_state_im1 %= bcast_state(grid, grid.ds_imh_i / im1(grid, grid.ds_imh_i, SLICE));
175 }
176
177 // Flare scenario: the positivity floors can pin several adjacent cells to the
178 // SAME density/pressure value (flat plateaus in the evaporated column), making
179 // the slope ratio r = 0/0 = NaN, which propagates through flux_lim into the
180 // reconstruction and the Rusanov spectral radius. Replace any non-finite ratio
181 // with 0 → flux_lim(0)=0 → first-order (no reconstructed slope) at those flat
182 // cells, which is the correct TVD behavior at extrema. Gated on the floors
183 // (beam heating or the C7 vacuum floor) so untouched scenarios / the test
184 // suite are unchanged.
185 if (grid.floors_active()) {
186 r_state.elem(arma::find_nonfinite(r_state)).zeros();
187 r_state_ip1.elem(arma::find_nonfinite(r_state_ip1)).zeros();
188 r_state_im1.elem(arma::find_nonfinite(r_state_im1)).zeros();
189 }
190
191 // Slope limiter. Default: symmetric minmod (slimL=slimR ⇒ one φ per cell).
192 // grid.mc3_limiter ⇒ the MC3 / Koren limiter, which is ASYMMETRIC: the '+'
193 // (right-face, Lxn = u+½φΔ₊) and '−' (left-face, Rxn = u−½φΔ₊) reconstruction
194 // terms take different limited slopes, so they get different limiter functions.
195 // With mc3 off both lambdas are flux_lim ⇒ byte-identical to the minmod baseline.
196 const float beta = grid.limiter_beta;
197 auto lim_plus = [&](const Vec& r) { return grid.mc3_limiter ? flux_lim_mc3_plus (r, beta) : flux_lim(r); };
198 auto lim_minus = [&](const Vec& r) { return grid.mc3_limiter ? flux_lim_mc3_minus(r, beta) : flux_lim(r); };
199
200 // extrapolated cell-edge variables (MUSCL paper eq 4.5). Uniform mesh keeps
201 // the exact legacy 0.5 half-cell weight; the refined mesh uses the owning
202 // cell's half-width factor (W1/W3/W4, = 0.5 when uniform).
203 Vec Rxn_state_iph, Lxn_state_iph, Rxn_state_imh, Lxn_state_imh;
204 if (grid.uniform_mesh) {
205 Rxn_state_iph = prim_xn_state_ip1 - 0.5 * lim_minus(r_state_ip1) % (prim_xn_state_ip2 - prim_xn_state_ip1);
206 Lxn_state_iph = prim_xn_state + 0.5 * lim_plus (r_state) % (prim_xn_state_ip1 - prim_xn_state);
207 Rxn_state_imh = prim_xn_state - 0.5 * lim_minus(r_state) % (prim_xn_state_ip1 - prim_xn_state);
208 Lxn_state_imh = prim_xn_state_im1 + 0.5 * lim_plus (r_state_im1) % (prim_xn_state - prim_xn_state_im1);
209 } else {
210 Rxn_state_iph = prim_xn_state_ip1 - W3 % lim_minus(r_state_ip1) % (prim_xn_state_ip2 - prim_xn_state_ip1);
211 Lxn_state_iph = prim_xn_state + W1 % lim_plus (r_state) % (prim_xn_state_ip1 - prim_xn_state);
212 Rxn_state_imh = prim_xn_state - W1 % lim_minus(r_state) % (prim_xn_state_ip1 - prim_xn_state);
213 Lxn_state_imh = prim_xn_state_im1 + W4 % lim_plus (r_state_im1) % (prim_xn_state - prim_xn_state_im1);
214 }
215
216 if ((Lxn_state_iph.has_nan() + Rxn_state_iph.has_nan() +
217 Lxn_state_imh.has_nan() + Rxn_state_imh.has_nan()) > 0) {
218 std::cout << "Lxn_iph has nan: " << Lxn_state_iph.has_nan() << std::endl
219 << "Rxn_iph has nan: " << Rxn_state_iph.has_nan() << std::endl
220 << "Lxn_imh has nan: " << Lxn_state_imh.has_nan() << std::endl
221 << "Rxn_imh has nan: " << Rxn_state_imh.has_nan() << std::endl;
222 std::exit(1);
223 }
224
225 // Undo the log transform on the reconstructed faces before mapping to conserved.
226 if (grid.log_reconstruct) {
227 prim_from_log(grid, Rxn_state_iph);
228 prim_from_log(grid, Lxn_state_iph);
229 prim_from_log(grid, Rxn_state_imh);
230 prim_from_log(grid, Lxn_state_imh);
231 }
232
233 Rxn_state_iph = prim2cons(grid, Rxn_state_iph);
234 Lxn_state_iph = prim2cons(grid, Lxn_state_iph);
235 Rxn_state_imh = prim2cons(grid, Rxn_state_imh);
236 Lxn_state_imh = prim2cons(grid, Lxn_state_imh);
237
238 // ---- prediction step (half-step) ------------------------------------
239 Vec prim_xt_state = cons2prim(grid,
240 xn_state - grid.dt_state / grid.ds_state %
241 (cal_flux_state(grid, Lxn_state_iph) - cal_flux_state(grid, Rxn_state_imh)));
242 // Positivity safeguard (flare scenario only): the half-step predictor can
243 // undershoot density/pressure to ≤0 at the steep evaporation front on the
244 // coarse grid, which then poisons the 2nd-order reconstruction and the
245 // Rusanov spectral radius (c_s = √(γp/ρ) → NaN). Floor the predicted density
246 // and pressure slots to small positive values. Gated on floors_active() so
247 // untouched scenarios and the existing test suite are byte-for-byte unchanged
248 // (a floor that only clips negatives never fires on those runs anyway).
249 if (grid.floors_active()) {
250 const auto sz = arma::size(grid.ns, num_of_eq);
251 const float RHO_FLOOR = grid.m_i * 1.0e10f;
252 const float P_FLOOR = 1.0e-8f;
253 for (arma::uword i = 0; i < grid.ns; ++i) {
254 float& ri = prim_xt_state(arma::sub2ind(sz, i, prim::RHO_I));
255 float& rn = prim_xt_state(arma::sub2ind(sz, i, prim::RHO_N));
256 float& pi = prim_xt_state(arma::sub2ind(sz, i, prim::P_I));
257 float& pn = prim_xt_state(arma::sub2ind(sz, i, prim::P_N));
258 if (ri < RHO_FLOOR) ri = RHO_FLOOR;
259 if (rn < RHO_FLOOR) rn = RHO_FLOOR;
260 if (pi < P_FLOOR) pi = P_FLOOR;
261 if (pn < P_FLOOR) pn = P_FLOOR;
262 }
263 }
264
265 Vec prim_xt_state_ip1 = cons2prim(grid, ip1(grid, prim2cons(grid, prim_xt_state)));
266 Vec prim_xt_state_im1 = cons2prim(grid, im1(grid, prim2cons(grid, prim_xt_state)));
267 // Same well-balanced φ_g correction for the predictor-shifted neighbours, which
268 // feed the 2nd-order corrector below (otherwise they re-introduce the ⅓-of-truth
269 // hydrostatic slope and a residual (γ−1)/2 g downforce).
270 if (grid.well_balanced) {
271 wb_correct_shifted(grid, prim_xt_state_ip1, phi_ip1, phi_cell);
272 wb_correct_shifted(grid, prim_xt_state_im1, phi_im1, phi_cell);
273 }
274
275 // Log the predictor states too, so the 2nd-order blend below (which mixes them
276 // with the still-logged prim_xn_state_* and reuses the log-space limiter ratios)
277 // is consistent. After the positivity floor / wb correction, which need linear p.
278 if (grid.log_reconstruct) {
279 prim_to_log(grid, prim_xt_state);
280 prim_to_log(grid, prim_xt_state_ip1);
281 prim_to_log(grid, prim_xt_state_im1);
282 }
283
284 // ---- 2nd-order-in-time reconstruction -------------------------------
285 // The 0.5·(xn+xt) prefactor is the predictor-corrector TIME average (mesh
286 // independent); only the SPATIAL slope term carries the half-cell weight,
287 // so on a refined mesh it becomes W1/W3/W4 (= 0.5 when uniform).
288 if (grid.uniform_mesh) {
289 Rxn_state_iph = 0.5 * (prim_xn_state_ip1 + prim_xt_state_ip1) - 0.5 * lim_minus(r_state_ip1) % (prim_xn_state_ip2 - prim_xn_state_ip1);
290 Lxn_state_iph = 0.5 * (prim_xn_state + prim_xt_state) + 0.5 * lim_plus (r_state) % (prim_xn_state_ip1 - prim_xn_state);
291 Rxn_state_imh = 0.5 * (prim_xn_state + prim_xt_state) - 0.5 * lim_minus(r_state) % (prim_xn_state_ip1 - prim_xn_state);
292 Lxn_state_imh = 0.5 * (prim_xn_state_im1 + prim_xt_state_im1) + 0.5 * lim_plus (r_state_im1) % (prim_xn_state - prim_xn_state_im1);
293 } else {
294 Rxn_state_iph = 0.5 * (prim_xn_state_ip1 + prim_xt_state_ip1) - W3 % lim_minus(r_state_ip1) % (prim_xn_state_ip2 - prim_xn_state_ip1);
295 Lxn_state_iph = 0.5 * (prim_xn_state + prim_xt_state) + W1 % lim_plus (r_state) % (prim_xn_state_ip1 - prim_xn_state);
296 Rxn_state_imh = 0.5 * (prim_xn_state + prim_xt_state) - W1 % lim_minus(r_state) % (prim_xn_state_ip1 - prim_xn_state);
297 Lxn_state_imh = 0.5 * (prim_xn_state_im1 + prim_xt_state_im1) + W4 % lim_plus (r_state_im1) % (prim_xn_state - prim_xn_state_im1);
298 }
299
300 // Undo the log transform before the positivity floor (linear thresholds) and
301 // prim2cons. With log reconstruction the ρ/p faces are exp(·) > 0 by construction,
302 // so the floor only ever fires on the linear V/U slots' companions — harmless.
303 if (grid.log_reconstruct) {
304 prim_from_log(grid, Rxn_state_iph);
305 prim_from_log(grid, Lxn_state_iph);
306 prim_from_log(grid, Rxn_state_imh);
307 prim_from_log(grid, Lxn_state_imh);
308 }
309
310 // Positivity safeguard (flare scenario only): floor the reconstructed face
311 // primitives' density/pressure before prim2cons, so the Rusanov spectral
312 // radius c_s = √(γp/ρ) at the steep evaporation front stays real. Gated on
313 // floors_active() → untouched scenarios / test suite unchanged.
314 if (grid.floors_active()) {
315 const auto sz = arma::size(grid.ns, num_of_eq);
316 const float RHO_FLOOR = grid.m_i * 1.0e10f;
317 const float P_FLOOR = 1.0e-8f;
318 Vec* faces[4] = {&Rxn_state_iph, &Lxn_state_iph, &Rxn_state_imh, &Lxn_state_imh};
319 for (Vec* fp : faces) {
320 Vec& F = *fp;
321 for (arma::uword i = 0; i < grid.ns; ++i) {
322 float& ri = F(arma::sub2ind(sz, i, prim::RHO_I));
323 float& rn = F(arma::sub2ind(sz, i, prim::RHO_N));
324 float& pi = F(arma::sub2ind(sz, i, prim::P_I));
325 float& pn = F(arma::sub2ind(sz, i, prim::P_N));
326 if (ri < RHO_FLOOR) ri = RHO_FLOOR;
327 if (rn < RHO_FLOOR) rn = RHO_FLOOR;
328 if (pi < P_FLOOR) pi = P_FLOOR;
329 if (pn < P_FLOOR) pn = P_FLOOR;
330 }
331 }
332 }
333
334 Rxn_state_iph = prim2cons(grid, Rxn_state_iph);
335 Lxn_state_iph = prim2cons(grid, Lxn_state_iph);
336 Rxn_state_imh = prim2cons(grid, Rxn_state_imh);
337 Lxn_state_imh = prim2cons(grid, Lxn_state_imh);
338
339 // ---- Rusanov / local Lax–Friedrichs flux ----------------------------
340 const Vec a_state_imh = arma::max(cal_spectral_radius_state(grid, Lxn_state_imh),
341 cal_spectral_radius_state(grid, Rxn_state_imh));
342 const Vec a_state_iph = arma::max(cal_spectral_radius_state(grid, Lxn_state_iph),
343 cal_spectral_radius_state(grid, Rxn_state_iph));
344
345 const Vec F_state_imh = 0.5 * (cal_flux_state(grid, Lxn_state_imh) + cal_flux_state(grid, Rxn_state_imh)
346 - a_state_imh % (Rxn_state_imh - Lxn_state_imh));
347 const Vec F_state_iph = 0.5 * (cal_flux_state(grid, Lxn_state_iph) + cal_flux_state(grid, Rxn_state_iph)
348 - a_state_iph % (Rxn_state_iph - Lxn_state_iph));
349
350 if ((a_state_imh.has_nan() + a_state_iph.has_nan() +
351 F_state_imh.has_nan() + F_state_iph.has_nan()) > 0) {
352 std::cout << "a_imh has nan: " << a_state_imh.has_nan() << std::endl
353 << "a_iph has nan: " << a_state_iph.has_nan() << std::endl
354 << "Fximh has nan: " << F_state_imh.has_nan() << std::endl
355 << "Fxiph has nan: " << F_state_iph.has_nan() << std::endl;
356 std::exit(1);
357 }
358
359 Vec rhs = -1.0 * grid.B_state % (F_state_iph / grid.B_state_iph - F_state_imh / grid.B_state_imh) / grid.ds_state
360 + cal_source_state(grid, xn_state);
361
362 // Equilibrium-reference well-balancing (chromosphere.hpp::eq_wb): subtract the
363 // frozen scheme residual at the reference equilibrium so eq_state is an exact
364 // discrete steady state. The empty() guard means the one-time computation of
365 // eq_residual itself (integrators::ensure_eq_residual, which calls this on
366 // eq_state while eq_residual is still empty) is NOT self-corrected, and the
367 // default-off path (eq_residual never populated) is byte-identical.
368 if (grid.eq_wb && !grid.eq_residual.is_empty()) {
369 rhs -= grid.eq_residual;
370 }
371 return rhs;
372}

374// ============================================================================
375// Implicit RHS (writeup §4.3, R_I): ion-neutral drag, collisional + frictional
376// heating, conservative field-aligned heat conduction. Pressure-area and
377// gravity live in cal_source_state (R_E).
378// ============================================================================
379
380Vec rhs_implicit_state(const Grid& grid, const Vec& xn_state) {
381 Vec RI_state(arma::size(xn_state), arma::fill::zeros);
382
383 // ---- cell-centered primitives ---------------------------------------
384 const Vec rho_i = get_scalar(grid, xn_state, cons::RHO_I);
385 const Vec rho_n = get_scalar(grid, xn_state, cons::RHO_N);
386 const Vec rhoV_i = get_scalar(grid, xn_state, cons::MOM_I);
387 const Vec rhoU_n = get_scalar(grid, xn_state, cons::MOM_N);
388 const Vec e_i = get_scalar(grid, xn_state, cons::E_I);
389 const Vec e_n = get_scalar(grid, xn_state, cons::E_N);
390
391 const Vec n_i = rho_i / grid.m_i;
392 const Vec n_n = rho_n / grid.m_n;
393 const Vec V = rhoV_i / rho_i;
394 const Vec U = rhoU_n / rho_n;
395 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
396 const Vec p_i = grid.gm1() * e_i - grid.half_gm1() * rho_i % V % V - grid.gm1() * rho_i % phi_g;
397 const Vec p_n = grid.gm1() * e_n - grid.half_gm1() * rho_n % U % U - grid.gm1() * rho_n % phi_g;
398 const Vec T_i = p_i / (2.0 * n_i * grid.k_b);
399 const Vec T_n = p_n / (n_n * grid.k_b);
400
401 const Vec Ke = kappa_e(n_i, n_n, T_i);
402 const Vec Ki(arma::size(T_i), arma::fill::zeros);
403 const Vec Kn = kappa_n(n_i, n_n, T_i, T_n);
404
405 // ---- i+1 neighbour (uses outer ghost at i = ns-1) -------------------
406 const Vec xn_state_ip1 = ip1(grid, xn_state);
407 const Vec rho_i_ip1 = get_scalar(grid, xn_state_ip1, cons::RHO_I);
408 const Vec rho_n_ip1 = get_scalar(grid, xn_state_ip1, cons::RHO_N);
409 const Vec rhoV_i_ip1 = get_scalar(grid, xn_state_ip1, cons::MOM_I);
410 const Vec rhoU_n_ip1 = get_scalar(grid, xn_state_ip1, cons::MOM_N);
411 const Vec e_i_ip1 = get_scalar(grid, xn_state_ip1, cons::E_I);
412 const Vec e_n_ip1 = get_scalar(grid, xn_state_ip1, cons::E_N);
413
414 const Vec n_i_ip1 = rho_i_ip1 / grid.m_i;
415 const Vec n_n_ip1 = rho_n_ip1 / grid.m_n;
416 const Vec V_ip1 = rhoV_i_ip1 / rho_i_ip1;
417 const Vec U_ip1 = rhoU_n_ip1 / rho_n_ip1;
418 Vec phi_g_ip1 = ip1(grid, 0.5 * (grid.phi_g_iph + grid.phi_g_imh), SLICE);
419 phi_g_ip1[grid.ns - 1] = grid.phi_g_iph[grid.ns - 1];
420 const Vec p_i_ip1 = grid.gm1() * e_i_ip1 - grid.half_gm1() * rho_i_ip1 % V_ip1 % V_ip1 - grid.gm1() * rho_i_ip1 % phi_g_ip1;
421 const Vec p_n_ip1 = grid.gm1() * e_n_ip1 - grid.half_gm1() * rho_n_ip1 % U_ip1 % U_ip1 - grid.gm1() * rho_n_ip1 % phi_g_ip1;
422 const Vec T_i_ip1 = p_i_ip1 / (2.0 * n_i_ip1 * grid.k_b);
423 const Vec T_n_ip1 = p_n_ip1 / (n_n_ip1 * grid.k_b);
424
425 const Vec Ke_ip1 = kappa_e(n_i_ip1, n_n_ip1, T_i_ip1);
426 const Vec Ki_ip1(arma::size(T_i), arma::fill::zeros);
427 const Vec Kn_ip1 = kappa_n(n_i_ip1, n_n_ip1, T_i_ip1, T_n_ip1);
428
429 // ---- i-1 neighbour (uses inner ghost at i = 0) ----------------------
430 const Vec xn_state_im1 = im1(grid, xn_state);
431 const Vec rho_i_im1 = get_scalar(grid, xn_state_im1, cons::RHO_I);
432 const Vec rho_n_im1 = get_scalar(grid, xn_state_im1, cons::RHO_N);
433 const Vec rhoV_i_im1 = get_scalar(grid, xn_state_im1, cons::MOM_I);
434 const Vec rhoU_n_im1 = get_scalar(grid, xn_state_im1, cons::MOM_N);
435 const Vec e_i_im1 = get_scalar(grid, xn_state_im1, cons::E_I);
436 const Vec e_n_im1 = get_scalar(grid, xn_state_im1, cons::E_N);
437
438 const Vec n_i_im1 = rho_i_im1 / grid.m_i;
439 const Vec n_n_im1 = rho_n_im1 / grid.m_n;
440 const Vec V_im1 = rhoV_i_im1 / rho_i_im1;
441 const Vec U_im1 = rhoU_n_im1 / rho_n_im1;
442 Vec phi_g_im1 = im1(grid, 0.5 * (grid.phi_g_iph + grid.phi_g_imh), SLICE);
443 phi_g_im1[0] = grid.phi_g_imh[0];
444 const Vec p_i_im1 = grid.gm1() * e_i_im1 - grid.half_gm1() * rho_i_im1 % V_im1 % V_im1 - grid.gm1() * rho_i_im1 % phi_g_im1;
445 const Vec p_n_im1 = grid.gm1() * e_n_im1 - grid.half_gm1() * rho_n_im1 % U_im1 % U_im1 - grid.gm1() * rho_n_im1 % phi_g_im1;
446 const Vec T_i_im1 = p_i_im1 / (2.0 * n_i_im1 * grid.k_b);
447 const Vec T_n_im1 = p_n_im1 / (n_n_im1 * grid.k_b);
448
449 const Vec Ke_im1 = kappa_e(n_i_im1, n_n_im1, T_i_im1);
450 const Vec Ki_im1(arma::size(T_i), arma::fill::zeros);
451 const Vec Kn_im1 = kappa_n(n_i_im1, n_n_im1, T_i_im1, T_n_im1);
452
453 // ---- ion-neutral drag (α = ρ_i ν_in) --------------------------------
454 const Vec nu_in_v = nu_in(grid, n_n, T_i, T_n);
455 const Vec alpha = rho_i % nu_in_v;
456 const Vec w = V - U;
457 RI_state += scalar_to(grid, -alpha % w, cons::MOM_I);
458 RI_state += scalar_to(grid, alpha % w, cons::MOM_N);
459
460 // ---- collisional + frictional heating (R_I energy rows) -------------
461 RI_state += scalar_to(grid,
462 alpha / (grid.m_i + grid.m_n) % (3.0 * grid.k_b * (T_n - T_i) + grid.m_n * w % w) - alpha % V % w,
463 cons::E_I);
464 RI_state += scalar_to(grid,
465 alpha / (grid.m_i + grid.m_n) % (3.0 * grid.k_b * (T_i - T_n) + grid.m_i * w % w) + alpha % U % w,
466 cons::E_N);
467
468 // ---- conservative field-aligned heat conduction (writeup §4.4) ----
469 // C = (B_i/Δs_i) [ K_{i+1/2}/B_{i+1/2} · (T_{i+1}-T_i)/Δs_{i+1/2}
470 // - K_{i-1/2}/B_{i-1/2} · (T_i-T_{i-1})/Δs_{i-1/2} ]
471 // Face spacings use Neumann BC at both ends (mirror Δs_i across the
472 // outermost/innermost interior cell).
473 const Vec ds_i_ip1 = ip1(grid, grid.ds_i, SLICE);
474 const Vec ds_i_im1 = im1(grid, grid.ds_i, SLICE);
475 const Vec ds_iph = 0.5 * (grid.ds_i + ds_i_ip1);
476 const Vec ds_imh = 0.5 * (ds_i_im1 + grid.ds_i);
477
478 // Face conductivities. Uniform mesh keeps the legacy arithmetic average;
479 // a refined mesh uses the width-weighted series-resistance combination
480 // (physics.hpp::face_conductivity_series) so unequal half-cells conduct
481 // correctly. Ki ≡ 0, so (Ke+Ki) is the charged-row conductivity.
482 Vec Kei_iph, Kei_imh, Kn_iph, Kn_imh;
483 if (grid.uniform_mesh) {
484 Kei_iph = 0.5 * ((Ke + Ki) + (Ke_ip1 + Ki_ip1));
485 Kei_imh = 0.5 * ((Ke + Ki) + (Ke_im1 + Ki_im1));
486 Kn_iph = 0.5 * (Kn + Kn_ip1);
487 Kn_imh = 0.5 * (Kn + Kn_im1);
488 } else {
489 Kei_iph = face_conductivity_series(Ke + Ki, Ke_ip1 + Ki_ip1, grid.ds_i, ds_i_ip1);
490 Kei_imh = face_conductivity_series(Ke + Ki, Ke_im1 + Ki_im1, grid.ds_i, ds_i_im1);
491 Kn_iph = face_conductivity_series(Kn, Kn_ip1, grid.ds_i, ds_i_ip1);
492 Kn_imh = face_conductivity_series(Kn, Kn_im1, grid.ds_i, ds_i_im1);
493 }
494
495 // Same face-local artificial conduction as the production implicit solver:
496 // chi_num,f = C_num*ds_face and K_num,f = chi_num,f*C_V,f. The charged
497 // single-temperature capacity is 3 n_i k_B; the neutral capacity is
498 // 1.5 n_n k_B, both face-averaged with the existing ghost geometry.
499 const Vec chi_iph = numerical_diffusivity_at_face(grid, ds_iph);
500 const Vec chi_imh = numerical_diffusivity_at_face(grid, ds_imh);
501 Kei_iph += (1.5f*grid.k_b)*(n_i+n_i_ip1)%chi_iph;
502 Kei_imh += (1.5f*grid.k_b)*(n_i+n_i_im1)%chi_imh;
503 Kn_iph += (0.75f*grid.k_b)*(n_n+n_n_ip1)%chi_iph;
504 Kn_imh += (0.75f*grid.k_b)*(n_n+n_n_im1)%chi_imh;
505
506 const Vec q_i_iph = Kei_iph / grid.B_iph % (T_i_ip1 - T_i) / ds_iph;
507 const Vec q_i_imh = Kei_imh / grid.B_imh % (T_i - T_i_im1) / ds_imh;
508 const Vec q_n_iph = Kn_iph / grid.B_iph % (T_n_ip1 - T_n) / ds_iph;
509 const Vec q_n_imh = Kn_imh / grid.B_imh % (T_n - T_n_im1) / ds_imh;
510
511 const Vec C_i = grid.B_i % (q_i_iph - q_i_imh) / grid.ds_i;
512 const Vec C_n = grid.B_i % (q_n_iph - q_n_imh) / grid.ds_i;
513 RI_state += scalar_to(grid, C_i, cons::E_I);
514 RI_state += scalar_to(grid, C_n, cons::E_N);
515
516 return RI_state;
517}
