2/*!
3 * @file run_control.hpp
4 * @brief Driver run-control helpers: step-cap selection and snapshot scheduling.
5 * @ingroup driver
6 *
7 * Both are pure, header-only, and free of OpenMP/Armadillo so they can be unit
8 * tested directly without running a simulation.
9 *
10 * ### Step cap (`CHROMO_STEP_CAP` / `CHROMO_T_END`)
11 *
12 * Historically the cap was derived from the positional `time_mult` argument
13 * (`max(10000, 10000*time_mult)`). That is correct for legacy `time_mult`-driven
14 * runs but silently truncates a run whose end time comes from `CHROMO_T_END`:
15 * `CHROMO_T_END=1000` with `time_mult=20` caps at 200000 steps, roughly 140 of
16 * the 1000 requested physical seconds. select_step_cap() keeps the legacy
17 * behavior only when `CHROMO_T_END` is absent.
18 *
19 * ### Snapshot cadence (`CHROMO_FRAME_DT`)
20 *
21 * The legacy cadence is a step stride, also derived from `time_mult`, so a long
22 * `CHROMO_T_END` run keeps a short-run stride and can emit tens of thousands of
23 * snapshots. OutputSchedule adds a physical-time cadence; with `CHROMO_FRAME_DT`
24 * unset it reproduces the stride behavior exactly.
25 */

39/// Effectively unbounded, but far below LLONG_MAX so neither the cap nor the
40/// step counter can overflow. At 1 ms/step this is ~31 years of wall time.
41constexpr long long kUnboundedStepCap = 1000000000000LL; // 1e12
42
43/// Which rule produced the step cap in force. select_step_cap() sets exactly one
44/// of these, and the run banner prints its step_cap_source_name() so a finished run
45/// records why it was allowed to stop.
46enum class StepCapSource {
47 Explicit, ///< CHROMO_STEP_CAP
48 LegacyTimeMult, ///< max(10000, 10000*time_mult), no CHROMO_T_END
49 EndTime, ///< CHROMO_T_END set, no explicit cap -> unbounded
50 Invalid ///< CHROMO_STEP_CAP present but unparseable or non-positive; see StepCapSelection::error
51};
52
53/// Result of select_step_cap(): the cap to enforce plus the rule that produced it.
54struct StepCapSelection {
55 /// Maximum number of steps the run may take, kUnboundedStepCap when no rule
56 /// bounds it. Meaningless unless valid().
57 long long cap = kUnboundedStepCap;
58 StepCapSource source = StepCapSource::EndTime; ///< rule that produced #cap
59 std::string error; ///< non-empty iff source == Invalid
60 /// False only for a rejected CHROMO_STEP_CAP; the driver then reports #error
61 /// and aborts instead of substituting a cap of its own.
62 bool valid() const { return source != StepCapSource::Invalid; }
63};
64
65/// step_cap_env / end_time_env are the raw getenv() results (nullptr when unset).
66/// An explicit CHROMO_STEP_CAP always wins; an unparseable or non-positive value
67/// is rejected rather than silently replaced by a dangerous cap.
68inline StepCapSelection select_step_cap(const char* step_cap_env,
69 const char* end_time_env,
70 float time_mult) {
71 StepCapSelection out;
72 if (step_cap_env && *step_cap_env) {
73 errno = 0;
74 char* tail = nullptr;
75 const long long requested = std::strtoll(step_cap_env, &tail, 10);
76 while (tail && (*tail == ' ' || *tail == '\t' || *tail == '\n')) ++tail;
77 if (errno == ERANGE || !tail || *tail != '\0') {
78 out.source = StepCapSource::Invalid;
79 out.error = "CHROMO_STEP_CAP is not a valid integer: '"
80 + std::string(step_cap_env) + "'";
81 return out;
82 }
83 if (requested <= 0) {
84 out.source = StepCapSource::Invalid;
85 out.error = "CHROMO_STEP_CAP must be a positive integer, got '"
86 + std::string(step_cap_env) + "'";
87 return out;
88 }
89 out.cap = requested < kUnboundedStepCap ? requested : kUnboundedStepCap;
90 out.source = StepCapSource::Explicit;
91 return out;
92 }
93 if (end_time_env && *end_time_env) {
94 // Explicit end time, no explicit cap: the end time governs.
95 out.cap = kUnboundedStepCap;
96 out.source = StepCapSource::EndTime;
97 return out;
98 }
99 // Legacy path, bit-for-bit the historical expression, guarded against a
100 // float overflow / out-of-range cast for absurd time_mult values.
101 const float legacy = std::max(10000.0f, 10000.0f * time_mult);
102 out.cap = (!std::isfinite(legacy) || legacy >= 1.0e12f)
103 ? kUnboundedStepCap : static_cast<long long>(legacy);
104 out.source = StepCapSource::LegacyTimeMult;
105 return out;
106}

108/// Stable lowercase token for a StepCapSource, as printed in the run banner:
109/// "explicit_step_cap", "legacy_time_mult", "end_time", or "invalid".
110inline const char* step_cap_source_name(StepCapSource s) {
111 switch (s) {
112 case StepCapSource::Explicit: return "explicit_step_cap";
113 case StepCapSource::LegacyTimeMult: return "legacy_time_mult";
114 case StepCapSource::EndTime: return "end_time";
115 default: return "invalid";
116 }
117}

119/// Why the time loop stopped. Every run prints termination_name() of this, and a
120/// long production run is only trusted when it reports `end_time`.
121enum class Termination {
122 EndTime, ///< the requested physical end time was reached
123 StepCap, ///< the step cap was hit first, so the run is truncated
124 Other ///< neither condition was met (e.g. the loop exited some other way)
125};
126
127/// Stable lowercase token for a Termination: "end_time", "step_cap", or "other".
128inline const char* termination_name(Termination t) {
129 switch (t) {
130 case Termination::EndTime: return "end_time";
131 case Termination::StepCap: return "step_cap";
132 default: return "other";
133 }
134}

136/// Reached end time wins over the cap when both fire on the same step.
137inline Termination classify_termination(double final_time, double requested_end_time,
138 long long final_step, long long cap) {
139 if (final_time >= requested_end_time) return Termination::EndTime;
140 if (final_step >= cap) return Termination::StepCap;
141 return Termination::Other;
142}

144/// Snapshot scheduler. Two modes:
145/// stride mode (legacy): write when step % stride == 0
146/// time mode: write when the simulation time reaches
147/// frame_index * frame_dt
148/// The scheduled time is always recomputed from an integer frame index, so the
149/// cadence cannot drift no matter how many intervals elapse. A step that crosses
150/// several nominal output times still writes the state only once.
151struct OutputSchedule {
152 /// Selects the mode: false = legacy step stride, true = physical-time cadence.
153 bool time_based = false;
154 /// Physical seconds between snapshots in time mode (`CHROMO_FRAME_DT`); unused
155 /// in stride mode.
156 double frame_dt = 0.0;
157 long long frame_index = 0; ///< index of the NEXT scheduled output time
158 /// Steps between snapshots in stride mode; always >= 1.
159 long long stride = 1;
160 /// Step number of the most recent snapshot, -1 before anything is written.
161 /// Guards against writing the same step twice.
162 long long last_written_step = -1;
163
164 /// Legacy stride mode: write every @p s steps. A non-positive stride is
165 /// clamped to 1 (write every step) rather than dividing by zero.
166 static OutputSchedule from_stride(long long s) {
167 OutputSchedule o;
168 o.stride = s > 0 ? s : 1;
169 return o;
170 }
171 /// Time mode: write when the simulation time reaches each multiple of @p dt
172 /// physical seconds.
173 static OutputSchedule from_frame_dt(double dt) {
174 OutputSchedule o;
175 o.time_based = true;
176 o.frame_dt = dt;
177 return o;
178 }
179
180 /// Physical time (seconds) of the next scheduled snapshot in time mode.
181 /// Recomputed as frame_index * frame_dt from the integer frame index rather
182 /// than accumulated, so the cadence cannot drift over a long run.
183 double next_output_time() const {
184 return static_cast<double>(frame_index) * frame_dt;
185 }
186 /// Slack, in seconds, allowed when testing whether the scheduled time has been
187 /// reached: 1e-9 of one frame interval, so a step landing a rounding error
188 /// short of a frame time still writes that frame.
189 double tolerance() const { return 1.0e-9 * frame_dt; }
190
191 /// Should a snapshot be written for the state at (step, t)? Never true twice
192 /// for the same step.
193 bool due(long long step, double t) const {
194 if (step == last_written_step) return false;
195 if (!time_based) return step % stride == 0;
196 return t + tolerance() >= next_output_time();
197 }
198
199 /// Record that (step, t) was written and advance past every scheduled time
200 /// this state already covers.
201 void note_written(long long step, double t) {
202 last_written_step = step;
203 if (!time_based) return;
204 const long long crossed =
205 static_cast<long long>(std::floor((t + tolerance()) / frame_dt));
206 if (crossed + 1 > frame_index) frame_index = crossed + 1;
207 }
208
209 /// Whether a snapshot for @p step has already been written, so the driver can
210 /// skip a redundant final-state write.
211 bool already_written(long long step) const { return step == last_written_step; }
212};
