28/// TRAC broadening factor ε(T) (Johnston et al. 2020 §2.2): ε = (T_c/T)^{5/2}
29/// inside the TRAC region T_b ≤ T < T_c, and ε = 1 elsewhere (chromosphere
30/// below T_b and corona at/above T_c are untouched). The conduction stage
31/// multiplies κ_e by ε (κ' = κ(T_c), broadening the TR) and the cooling stage
32/// divides the optically-thin loss by ε (Λ' = Λ(T/T_c)^{5/2}), so κ'Λ' = κΛ and
33/// the TR-integrated radiation is preserved. Returns all-ones when TRAC is off.
34inline Vec trac_broadening_factor(const Grid& grid, const Vec& T) {
35 Vec eps(T.n_elem, arma::fill::ones);
36 if (!grid.enable_trac) return eps;
37 const float Tc = grid.trac_cutoff_T;
38 const float Tb = grid.trac_T_chrom;
39 if (!(Tc > Tb)) return eps; // degenerate cutoff → no broadening
40 for (arma::uword i = 0; i < T.n_elem; ++i) {
41 const float t = T(i);
42 if (t >= Tb && t < Tc) eps(i) = std::pow(Tc / t, 2.5f);
43 }
44 return eps;
45}

47/// Electron heat conductivity in SI units (writeup eq 53). The Spitzer e–i
48/// numerator (9.2048e-12) is unchanged; the n_n T_e² term in the denominator
49/// carries the electron–neutral collision frequency ν_en = σ_en n_n √(8k_BT_e/πm_e),
50/// and its coefficient scales with σ_en. We use the accurate quantum-mechanical
51/// (Krstic & Schultz) e–H cross section σ_en ≈ 2.5e-19 m² at chromospheric
52/// energies, tabulated by Vranjes & Krstic (2013, A&A 554 A22, Fig. 4 / Eq. 3),
53/// ≈8× the earlier ad-hoc value (3.5609e-12 → 2.836e-11). e–n matters only
54/// where n_n ≳ 10²–10³ n_e (dense lower chromosphere); κ_n dominates there.
55inline Vec kappa_e(const Vec& n_e, const Vec& n_n, const Vec& T_e) {
56 return (9.2048e-12f * n_e % arma::pow(T_e, 2.5)
57 / (n_e + 2.836e-11f * n_n % T_e % T_e));
58}

60/// Width-weighted series-resistance face conductivity for a NON-UNIFORM mesh.
61/// The conductive flux from cell center i to center j crosses two half-cells in
62/// series, so the effective face conductivity is the resistance-weighted harmonic
63/// mean over the center-to-center distance ds_i+ds_j:
64/// K_face = (ds_i + ds_j) / (ds_i/K_i + ds_j/K_j).
65/// On a uniform mesh this is the harmonic mean 2 K_i K_j/(K_i+K_j) — NOT the
66/// arithmetic average — so callers keep the legacy arithmetic average
67/// 0.5(K_i+K_j) on uniform meshes and only switch to this on refined ones.
68inline Vec face_conductivity_series(const Vec& K_here, const Vec& K_there,
69 const Vec& ds_here, const Vec& ds_there) {
70 const Vec kh = arma::clamp(K_here, 1.0e-30f, arma::datum::inf);
71 const Vec kt = arma::clamp(K_there, 1.0e-30f, arma::datum::inf);
72 return (ds_here + ds_there) / (ds_here / kh + ds_there / kt);
73}

75/// Face-local artificial thermal diffusivity [m^2/s]. C_num is stored on Grid in
76/// [m/s]; ds_face is the actual centre-to-centre distance, including the mirrored
77/// ghost geometry at domain boundaries.
78template <typename FaceSpacing>
79inline auto numerical_diffusivity_at_face(const Grid& grid,
80 const FaceSpacing& ds_face) {
81 return grid.numerical_diffusivity_per_length * ds_face;

84/// Neutral heat conductivity in SI units (writeup eq 59).
85inline Vec kappa_n(const Vec& n_i, const Vec& n_n, const Vec& T_i, const Vec& T_n) {
86 return (0.0342006f * n_n % T_n)
87 / (1.20613f * n_i % arma::sqrt(T_n + T_i)
88 + 1.70573f * n_n % arma::sqrt(T_n));
89}

91/// Scalar common-temperature conductivity components used by EOS diagnostics and
92/// by the single-fluid release conduction operator, which is PHYSICAL only:
93/// these deliberately contain no TRAC and no numerical-diffusion contribution.
94inline double physical_kappa_e(double n_e, double n_n, double temperature) {
95 return 9.2048e-12*n_e*std::pow(temperature, 2.5)
96 / (n_e + 2.836e-11*n_n*temperature*temperature);
97}

99/// Neutral-hydrogen heat conductivity [W m^-1 K^-1] at a single common
100/// temperature: the scalar counterpart of kappa_n() above with T_i = T_n = T,
101/// which is why the ion-neutral term carries sqrt(2 T). `n_i` is the ion
102/// (= electron) density [m^-3] and `n_n` the neutral density [m^-3]. This
103/// channel dominates the total conductivity in the dense, weakly-ionized lower
104/// chromosphere where n_n greatly exceeds n_e.
105inline double physical_kappa_n(double n_i, double n_n, double temperature) {
106 return 0.0342006*n_n*temperature
107 / (1.20613*n_i*std::sqrt(2.0*temperature)
108 + 1.70573*n_n*std::sqrt(temperature));
109}

111/// Total physical field-aligned heat conductivity [W m^-1 K^-1] at one common
112/// temperature: kappa = kappa_e + kappa_n, the Spitzer electron channel plus the
113/// neutral-hydrogen channel. `n_e` is the electron density [m^-3] and `n_n` the
114/// neutral density [m^-3]; both conduction channels are evaluated at the same T.
115/// This is the closure the single-fluid release conduction operator uses, so it
116/// deliberately contains no TRAC broadening and no artificial diffusivity.
117inline double physical_conductivity(double n_e, double n_n, double temperature) {
118 return physical_kappa_e(n_e, n_n, temperature)
119 + physical_kappa_n(n_e, n_n, temperature);
120}

122/// Ion-neutral collision frequency, target-density form (writeup eq 57):
123/// ν_in = (2 a₀)² n_n √(8 π k_b (T_i + T_n) / m_i)
124inline Vec nu_in(const Grid& grid, const Vec& n_n, const Vec& T_i, const Vec& T_n) {
125 const float bohr_r = 53e-12f;
126 return (2.0f * bohr_r) * (2.0f * bohr_r) * n_n
127 % arma::sqrt(arma::abs(8.0f * static_cast<float>(arma::datum::pi)
128 * grid.k_b * (T_i + T_n) / grid.m_i));
129}

131/// Coulomb logarithm lnΛ_ei for electron–ion collisions (NRL Plasma Formulary).
132/// Two regimes about T_e = 10 Z² eV (≈ 1.16×10⁵ K for hydrogen), in cgs/eV:
133/// T_e < 10 eV: lnΛ = 23 − ln(n_e^{1/2} T_e^{-3/2})
134/// T_e > 10 eV: lnΛ = 24 − ln(n_e^{1/2} / T_e)
135/// (n_e in cm⁻³, T_e in eV, Z = 1). Clamped to [5,30] so the rate below stays
136/// well-behaved over the full chromosphere→flare range.
137inline Vec coulomb_log_ei(const Vec& n_e, const Vec& T_e) {
138 constexpr float k_b_eV = 8.617333262e-5f; // eV/K
139 const Vec ne_cm3 = arma::clamp(n_e, 1.0f, arma::datum::inf) * 1.0e-6f;
140 const Vec T_eV = k_b_eV * arma::clamp(T_e, 1.0f, arma::datum::inf);
141 const Vec sqrt_ne = arma::sqrt(ne_cm3);
142 Vec lnL(n_e.n_elem);
143 for (arma::uword i = 0; i < n_e.n_elem; ++i) {
144 const float t = T_eV(i);
145 lnL(i) = (t < 10.0f)
146 ? 23.0f - std::log(sqrt_ne(i) * std::pow(t, -1.5f))
147 : 24.0f - std::log(sqrt_ne(i) / t);
148 }
149 return arma::clamp(lnL, 5.0f, 30.0f);
150}

152/// Electron–ion temperature equilibration frequency ν_ei [s⁻¹] such that
153/// dT_e/dt = ν_ei (T_i − T_e). Spitzer (1962) / NRL Plasma Formulary energy
154/// relaxation in the m_e ≪ m_i, T_i m_e/m_i ≪ T_e limit:
155/// ν_ei = (8√(2π)/3) · m_e^{1/2} e⁴ /(m_p (4πε₀)²) · n_e lnΛ /(k_B T_e)^{3/2}
156/// ≈ 2.030×10⁻⁴³ · n_e lnΛ /(k_B T_e)^{3/2} (SI; the bracket evaluated
157/// for hydrogen Z=1). This is the load-bearing decoupling parameter
158/// (docs/electron_temperature_plan.md): ν_ei ∝ n_e T_e^{-3/2} is fast (T_e≈T_i)
159/// in the dense cool chromosphere (~4×10³ s⁻¹) and slow (decoupling) at the hot
160/// tenuous loop-top / flare onset (~4×10⁻² s⁻¹ at n_e=10¹⁵, T_e=10⁶ K), matching
161/// Bradshaw (2006) and Manchester (2012).
162inline Vec nu_ei(const Grid& grid, const Vec& n_e, const Vec& T_e) {
163 constexpr float C_ei = 2.030e-43f;
164 const Vec kT = grid.k_b * arma::clamp(T_e, 1.0f, arma::datum::inf);
165 return C_ei * n_e % coulomb_log_ei(n_e, T_e) / arma::pow(kT, 1.5f);
166}

168/// Electron–neutral temperature equilibration frequency ν_en [s⁻¹] such that
169/// dT_e/dt = ν_en (T_n − T_e). Elastic light-on-heavy energy transfer carries a
170/// fraction 2 m_e/m_n of the energy per collision, so ν_en = (2 m_e/m_n) ν_en^coll
171/// where ν_en^coll is the electron–neutral COLLISION frequency. We use the same
172/// ν_en^coll = 1.55×10⁻¹⁵ n_n √T_e [SI] adopted for the electron heat conductivity
173/// κ_e (kappa_e above), built from the accurate quantum-mechanical (Krstic &
174/// Schultz) e–H cross section σ_en ≈ 2.5×10⁻¹⁹ m² at chromospheric energies,
175/// tabulated by Vranjes & Krstic (2013, A&A 554 A22, Fig. 4 / Eq. 3). So the
176/// three-temperature equilibration and the conduction closure use one consistent,
177/// physically-accurate e–n cross-section. Sub-dominant to ν_ei wherever the gas is
178/// appreciably ionized, but it equilibrates T_e to T_n deep in the weakly-ionized
179/// chromosphere where n_n ≫ n_e.
180inline Vec nu_en(const Grid& grid, const Vec& n_n, const Vec& T_e) {
181 const Vec nu_en_coll = 1.55e-15f * n_n
182 % arma::sqrt(arma::clamp(T_e, 1.0f, arma::datum::inf));
183 return (2.0f * grid.m_e / grid.m_n) * nu_en_coll;
184}

186/// Hydrogen electron-impact ionization rate coefficient S_i(T_e) [m^3/s].
187/// Voronov (1997) fit in SI; cgs original is
188/// S_i = A (1+P√U) U^K e^{-U} / (X+U),
189/// with A = 0.291×10^{-7} cm^3/s, P = 0, X = 0.232, K = 0.39, and
190/// U = χ_H / (k_b T_e). The cm^3 → m^3 conversion gives 2.91×10^{-14}.
191inline Vec ionization_rate_S(const Grid& grid, const Vec& T_e) {
192 const Vec U = grid.chi_H_J / (grid.k_b * arma::clamp(T_e, 1.0f, arma::datum::inf));
193 return 2.91e-14f * arma::pow(U, 0.39f) % arma::exp(-U) / (0.232f + U);
194}

196/// Hydrogen case-B radiative recombination rate coefficient α_r(T_e) [m^3/s].
197/// Low-T power-law fit to Hummer (1994); accurate to ~10% for 10^3–10^5 K.
198/// Case B is the consistent choice with the optically-thin χ_H bookkeeping
199/// used by Stage E (writeup §6.5).
200inline Vec recombination_rate_alpha(const Grid& /*grid*/, const Vec& T_e) {
201 return 2.7e-19f * arma::pow(arma::clamp(T_e, 1.0f, arma::datum::inf) / 1.0e4f, -0.75f);
202}

204/// Saha ionization function Φ(T_e) [m^-3] for hydrogen (writeup eq:Seff):
205/// Φ = (2π m_e k_B T_e / h^2)^{3/2} e^{-χ_H / k_B T_e}.
206/// At LTE the ground-state Saha relation is n_e n_i / n_n = Φ, so Φ ties the
207/// multilevel collisional ionization S_CR to its three-body inverse α_c.
208/// NOTE: the prefactor coefficient 2π m_e k_B / h² ≈ 1.79985×10^{14} m^-2 K^-1
209/// is lumped into a single constant (computed in double precision) because the
210/// intermediates m_e k_B (~8×10^-53) and h² (~4.4×10^-67) both underflow to
211/// zero in float32 — forming them separately would divide by zero.
212inline Vec saha_phi(const Grid& grid, const Vec& T_e) {
213 constexpr float SAHA_PREF_C = 1.79985e14f; // 2π m_e k_B / h² [m^-2 K^-1]
214 const Vec Tc = arma::clamp(T_e, 1.0f, arma::datum::inf);
215 const Vec pref = arma::pow(SAHA_PREF_C * Tc, 1.5f);
216 return pref % arma::exp(-grid.chi_H_J / (grid.k_b * Tc));
217}

219/// Three-body (collisional–radiative) recombination, density coefficient
220/// κ_c(T_e) [m^6/s] such that α_c = κ_c · n_e. Hinnov & Hirschberg (1962) /
221/// Stevefelt (1975) low-T form; cgs original
222/// α_c = 5.6×10^{-27} (k_B[eV] T_e)^{-4.5} n_e[cm^-3] [cm^3/s].
223/// The cm^3→m^3 (×1e-6) and cm^-3→m^-3 (×1e-6 on n_e) conversions give the
224/// 5.6×10^{-39} SI prefactor below. Keeping κ_c (the n_e-independent part)
225/// separate lets Stage E carry the three-body channel as the cubic f^3 term.
226inline Vec recombination_rate_kappa_c(const Grid& /*grid*/, const Vec& T_e) {
227 constexpr float k_b_eV = 8.617333262e-5f; // Boltzmann constant [eV/K]
228 const Vec kT_eV = k_b_eV * arma::clamp(T_e, 1.0f, arma::datum::inf);
229 // κ_c = 5.6e-27 (k_B[eV] T)^{-4.5} cm^3/s · 1e-12 (cm^3·cm^-3 → m^6) [m^6/s].
230 // Apply the 1e-12 conversion last so the 5.6e-39 product never forms as a
231 // (low-precision) float32 denormal.
232 return (5.6e-27f * arma::pow(kT_eV, -4.5f)) * 1.0e-12f; // [m^6/s]
233}
234
235/// Three-body recombination coefficient α_c(T_e, n_e) = κ_c · n_e [m^3/s].
236inline Vec recombination_rate_alpha_c(const Grid& grid, const Vec& T_e, const Vec& n_e) {
237 return recombination_rate_kappa_c(grid, T_e) % n_e;
238}

240/// Multilevel (Rydberg-ladder) collisional ionization coefficient
241/// S_CR(T_e) [m^3/s], the detailed-balance inverse of three-body
242/// recombination: S_CR = α_c Φ / n_e = κ_c Φ. Because α_c ∝ n_e, S_CR is
243/// independent of n_e (a function of T_e alone). This is the dominant
244/// collisional ionization channel in the chromosphere — orders of magnitude
245/// larger than the direct ground-state Voronov S_i, since it proceeds through
246/// high-n levels (writeup §3.1 eq:Seff).
247inline Vec ionization_rate_S_CR(const Grid& grid, const Vec& T_e) {
248 return recombination_rate_kappa_c(grid, T_e) % saha_phi(grid, T_e);
249}

251/// Effective hydrogen photoionization rate P_phot [s^-1] (per neutral atom).
252/// Single-rate closure for the Balmer-continuum (n=2) channel that sets
253/// chromospheric H ionization (Carlsson & Stein 2002 ApJ 572, 626; the Lyman
254/// continuum is negligible). For the 2-state reduction adopted here this is
255/// the radiative complement to the collisional Voronov rate:
256/// dn_i/dt |_phot = +P_phot · n_n (does not deplete electron thermal pool
257/// — energy comes from the radiation field,
258/// not from local kinetic temperature).
259/// Returns `grid.photoionization_rate_i` if the scenario IC populated a
260/// per-cell profile. model_c7_ic populates it with the height-structured
261/// frozen-radiation-field rate R_ik(h) from Chae (2021) Table 1 (FAL-C),
262/// via photoionization_rate_chae() below; otherwise this falls back to the
263/// uniform scalar `grid.photoionization_rate`.
264inline Vec photoionization_rate_P(const Grid& grid) {
265 if (grid.photoionization_rate_i.n_elem == grid.ns) {
266 return grid.photoionization_rate_i;
267 }
268 return Vec(grid.ns, arma::fill::value(grid.photoionization_rate));

326/// 1-D linear interpolation: piecewise linear, constant extrapolation.
327inline float lin_interp(float x, const float* xs, const float* ys, int n) {
328 if (x <= xs[0]) return ys[0];
329 if (x >= xs[n-1]) return ys[n-1];
330 int k = 0;
331 while (k < n-1 && xs[k+1] < x) ++k;
332 const float t = (x - xs[k]) / (xs[k+1] - xs[k]);
333 return ys[k] + t * (ys[k+1] - ys[k]);
334}

336/// Vectorized look-up wrapper. Returns one Vec same size as q.
337inline Vec lookup_vec(const Vec& q, const float* xs, const float* ys, int n) {
338 Vec out(q.n_elem);
339 for (arma::uword i = 0; i < q.n_elem; ++i) out[i] = lin_interp(q[i], xs, ys, n);
340 return out;
341}

345/*!
346 * Frozen-radiation-field hydrogen photoionization rate R_ik(z)
347 * Chae (2021) J. Astron. Space Sci. 38, 83, Table 1 — the FAL-C
348 * (Fontenla, Avrett & Loeser 1993) "frozen radiation field" photoionization
349 * rate tabulated vs height z [km] on the tau_500=1 scale, as log10 R_ik

388/// Optically-thick chromospheric radiative cooling Q_rad [W/m^3].
389/// Positive = net cooling (caller subtracts from energy row).
390/// CL2012 Eq. 1; tables in namespace cl2012 above.
391/// `s_up_is_outer` matches the grid convention: index 0 = inner (photosphere),
392/// index ns-1 = outer (TR). Column integrals are accumulated downward from
393/// the TR (top of column = zero column mass).
394inline Vec radiative_loss_thick(const Grid& grid,
395 const Vec& n_i, const Vec& n_n,
396 const Vec& T_e) {
397 using namespace cl2012;
398 const arma::uword ns = grid.ns;
399
400 // ---- column integrals (CGS) ----
401 // m_c [g/cm^2] = 0.1 * integral_{s}^{s_top} rho_SI ds_SI (SI -> cgs surface)
402 // N_HI [/cm^2] = 1e-4 * integral n_n[/m^3] ds[m]
403 const Vec rho = (n_i * grid.m_i + n_n * grid.m_n); // SI kg/m^3
404 Vec m_c(ns, arma::fill::zeros); // g/cm^2
405 Vec N_HI(ns, arma::fill::zeros); // /cm^2
406 // Top cell: half-cell column above the midpoint.
407 m_c[ns-1] = 0.5f * rho[ns-1] * grid.ds_i[ns-1] * 0.1f;
408 N_HI[ns-1] = 0.5f * n_n[ns-1] * grid.ds_i[ns-1] * 1.0e-4f;
409 for (int i = static_cast<int>(ns) - 2; i >= 0; --i) {
410 // Trapezoid: mean density times cell-center separation.
411 const float ds_face = 0.5f * (grid.ds_i[i] + grid.ds_i[i+1]);
412 const float rho_dl = 0.5f * (rho[i] + rho[i+1]) * ds_face;
413 const float nn_dl = 0.5f * (n_n[i] + n_n[i+1]) * ds_face;
414 m_c[i] = m_c[i+1] + rho_dl * 0.1f;
415 N_HI[i] = N_HI[i+1] + nn_dl * 1.0e-4f;
416 }
417
418 // ---- log coordinates for table lookup ----
419 const Vec T_kK = T_e / 1000.0f;
420 const Vec log_mc = arma::log10(arma::clamp(m_c, 1.0e-12f, 1.0e6f));
421 const Vec tau_Lya = 4.0e-14f * N_HI;
422 const Vec log_tau = arma::log10(arma::clamp(tau_Lya, 1.0e-6f, 1.0e10f));
423
424 // ---- per-species factors ----
425 constexpr float LN10 = 2.302585092994046f;
426 const Vec L_H = arma::exp(LN10 * lookup_vec(T_kK, L_T_kK, L_logL_H, N_L));
427 const Vec L_Ca = arma::exp(LN10 * lookup_vec(T_kK, L_T_kK, L_logL_Ca, N_L));
428 const Vec L_Mg = arma::exp(LN10 * lookup_vec(T_kK, L_T_kK, L_logL_Mg, N_L));
429
430 const Vec E_H_ = lookup_vec(log_tau, EH_logTau, EH_E, N_EH);
431 const Vec E_Ca_ = lookup_vec(log_mc, ECa_logMc, ECa_E, N_ECa);
432 const Vec E_Mg_ = lookup_vec(log_mc, EMg_logMc, EMg_E, N_EMg);
433
434 const Vec f_HI = lookup_vec(T_kK, F_T_kK, F_HI, N_F);
435 const Vec f_Ca = lookup_vec(T_kK, F_T_kK, F_Ca, N_F);
436 const Vec f_Mg = lookup_vec(T_kK, F_T_kK, F_Mg, N_F);
437
438 // ---- assemble Q_X [erg/cm^3/s] ----
439 // n_e_cgs = n_i [SI /m^3] * 1e-6 ; rho_cgs = rho [kg/m^3] * 1e-3
440 const Vec n_e_cgs = 1.0e-6f * n_i;
441 const Vec rho_cgs = 1.0e-3f * rho;
442 const Vec NH_n_e_rho_cgs = NH_over_rho_cgs * n_e_cgs % rho_cgs; // = N_H * n_e [/cm^6]
443
444 const Vec Q_H = L_H % E_H_ % f_HI % NH_n_e_rho_cgs * A_H;
445 const Vec Q_Ca = L_Ca % E_Ca_ % f_Ca % NH_n_e_rho_cgs * A_Ca;
446 const Vec Q_Mg = L_Mg % E_Mg_ % f_Mg % NH_n_e_rho_cgs * A_Mg;
447
448 // cgs (erg/cm^3/s) -> SI (W/m^3): factor 0.1
449 return 0.1f * (Q_H + Q_Ca + Q_Mg);
450}

452/*!
453 * Optically-thin transition-region / coronal radiative losses
454 * Q_thin = n_e n_H Λ(T) [W/m^3], positive = net cooling.
455 * Λ(T) is the standard coronal-abundance optically-thin radiative loss
456 * function (CHIANTI-class; peak ≈ 4×10^-35 W m^3 ≈ 10^-21.35 erg cm^3 s^-1
457 * near log T ≈ 5.1). It covers the 10^4–10^7 K regime that the optically-THICK
458 * chromospheric CL2012 recipe (H I/Ca II/Mg II, effective only for T ≲ 3×10^4 K)
459 * does not. This is the radiative sink the lower TR needs to re-radiate the
460 * imposed downward coronal conductive flux q(T) (writeup §Boundary conditions,
461 * RTV 1978); without it the imposed flux had no TR-temperature sink and ran
462 * the top up to coronal temperatures. The enthalpy/advective flux — the other
463 * dominant TR energy term (Bradshaw & Cargill 2010; Klimchuk et al. 2008) —
464 * is already carried by the hydrodynamic advection, so adding this radiative
465 * term completes the leading-order TR energy balance.
466 *
467 * A smooth low-T switch-on (~2×10^4 K) stitches the two recipes: CL2012
468 * governs the chromosphere, the thin loss governs the TR/corona, with a tanh
469 * blend across the overlap so neither double-counts the other. Λ(T) tabulated
470 * from standard sources (Cook et al. 1989; Klimchuk et al. 2008; CHIANTI,
471 * Dere et al. 1997+); refine against CHIANTI v11 (Dufresne 2024) if needed.
472 * NOTE: equilibrium Λ(T) underestimates lower-TR losses by up to ~×3 under
473 * non-equilibrium ionization (Landi et al. 2012) — a known conservative bias.
474 */
475namespace optthin {
476constexpr int N_L = 17; ///< Temperature-sample count of the optically-thin Lambda(T) table.
477// log10 T [K]
478static constexpr float logT_K[N_L] = {
479 4.0f, 4.1f, 4.2f, 4.3f, 4.5f, 4.7f, 4.9f, 5.1f, 5.3f,
480 5.5f, 5.7f, 5.9f, 6.1f, 6.3f, 6.5f, 6.7f, 7.0f};
481// log10 Λ [erg cm^3 s^-1]
482static constexpr float logLambda[N_L] = {
483 -25.0f, -23.8f, -22.9f, -22.35f, -21.85f, -21.55f, -21.4f, -21.35f, -21.45f,
484 -21.6f, -21.75f, -21.9f, -22.05f, -22.2f, -22.35f, -22.45f, -22.5f};
485} // namespace optthin

487/// Optically-thin radiative loss Q_thin [W/m^3] (positive = cooling). `n_i` is
488/// the electron/proton density (n_e = n_i), `n_n` the neutral density so that
489/// n_H = n_i + n_n is total hydrogen; Q = n_e n_H Λ(T). The 1e-13 factor is the
490/// cgs→SI conversion for Λ (erg cm^3 s^-1 → W m^3).
491inline Vec radiative_loss_thin(const Grid& /*grid*/, const Vec& n_i,
492 const Vec& n_n, const Vec& T_e) {
493 constexpr float LN10 = 2.302585092994046f;
494 const Vec Tc = arma::clamp(T_e, 1.0f, arma::datum::inf);
495 const Vec logTc = arma::log10(Tc);
496 const Vec logLam = cl2012::lookup_vec(logTc, optthin::logT_K,
497 optthin::logLambda, optthin::N_L);
498 const Vec Lambda_SI = arma::exp(LN10 * logLam) * 1.0e-13f; // W m^3
499 const Vec n_H = n_i + n_n; // total hydrogen
500 // Smooth switch-on across ~2e4 K (stitch to optically-thick CL2012 below).
501 const Vec w = 0.5f * (1.0f + arma::tanh((Tc - 2.0e4f) / 5.0e3f));
502 // Multiply by the tiny loss coefficient before n_H. At the high-density,
503 // low-ionization edge of the EOS domain, n_e*n_H can overflow float even
504 // though n_e*Lambda(T)*n_H is finite and physically small.
505 return w % ((n_i % Lambda_SI) % n_H); // W m^-3
506}

508/*!
509 * Flare nonthermal-electron-beam heating (chromospheric evaporation driver)
510 * Q_beam(s,t) = beam_flux · g(t) · φ(s) [W/m³], positive = heating.
511 * The beam is injected from the corona at the outer boundary and stops in the
512 * upper chromosphere. Its true thick-target stopping column (~10²⁰ cm⁻² for
513 * tens of keV) is unresolved on the coarse C7 grid — a single top cell already
514 * holds ~10²⁶ m⁻² — so we deposit the flux over a finite layer of thickness
515 * beam_deposition_height [km] below the top, weighted by the local total
516 * density n_tot (the densest reachable layer absorbs the most, as a stopping
517 * beam does). φ(s) is normalized so ∫φ ds = 1, hence the column-integrated
518 * heating equals beam_flux [W/m²]. g(t) is a flat-top window over
519 * [t_on, t_on+τ] with cosine ramps of half-width beam_ramp.
520 *
521 * This is the standard Fisher/RADYN explosive-evaporation driver expressed as
522 * an interior source term (see docs/explosive_evaporation_plan.md). Explosive
523 * evaporation requires beam_flux above the Fisher (1985) threshold
524 * F_crit ≈ 7×10⁶ W m⁻² together with an active radiative sink.
525 */
526inline Vec beam_heating_rate(const Grid& grid, const Vec& n_i, const Vec& n_n) {
527 Vec Q = arma::zeros<Vec>(grid.ns);
528 if (!grid.enable_beam_heating || grid.beam_flux <= 0.0f) return Q;
529
530 // Temporal window g(t): cosine ramp up over [t_on, t_on+ramp], flat top,
531 // cosine ramp down over [t_off-ramp, t_off]. Zero outside.
532 const float t = grid.sim_time;
533 const float t_on = grid.beam_t_on;
534 const float toff = grid.beam_t_on + grid.beam_duration;
535 const float r = (grid.beam_ramp > 1.0e-6f) ? grid.beam_ramp : 1.0e-6f;
536 float g;
537 if (t <= t_on || t >= toff) {
538 return Q; // beam off
539 } else if (t < t_on + r) {
540 const float x = (t - t_on) / r; // 0→1
541 g = 0.5f * (1.0f - std::cos(static_cast<float>(arma::datum::pi) * x));
542 } else if (t > toff - r) {
543 const float x = (toff - t) / r; // 1→0
544 g = 0.5f * (1.0f - std::cos(static_cast<float>(arma::datum::pi) * x));
545 } else {
546 g = 1.0f; // flat top
547 }
548 if (g <= 0.0f) return Q;
549
550 const Vec ds = grid.ds_i; // [m]
551 const Vec ntot = n_i + n_n; // [m^-3], collisional target
552
553 // --- Self-consistent thick-target deposition (Emslie 1978) ----------------
554 // Inject at the loop apex (max φ_g) and attenuate down each leg by collisional
555 // stopping. An electron of energy E stops at column N_stop ∝ E²; for an
556 // injected power-law (index δ, cutoff E_c) the heating per unit volume at
557 // column depth N (from the apex) is Q ∝ n_tot · N^{-δ/2} for N>N_c, flattening
558 // to Q ∝ n_tot for N≪N_c (the full beam still in flight). We use the smooth
559 // form (1+N/N_c)^{-δ/2} and normalize to ∫Q ds = beam_flux (thick target).
560 // The column N(s) = ∫n_tot ds is recomputed from the CURRENT state every call,
561 // so as evaporation fills the loop the stopping depth migrates UP the loop —
562 // the feedback the fixed window misses.
563 if (grid.beam_thick_target) {
564 const Vec phi_g = 0.5f * (grid.phi_g_imh + grid.phi_g_iph);
565 const arma::uword a = phi_g.index_max(); // apex = potential maximum
566 Vec N = arma::zeros<Vec>(grid.ns); // column depth from apex [m^-2]
567 for (arma::uword i = a; i-- > 0; ) // apex → inner footpoint
568 N(i) = N(i + 1) + 0.5f * (ntot(i + 1) * ds(i + 1) + ntot(i) * ds(i));
569 for (arma::uword i = a + 1; i < grid.ns; ++i) // apex → outer footpoint
570 N(i) = N(i - 1) + 0.5f * (ntot(i) * ds(i) + ntot(i - 1) * ds(i - 1));
571 // N_c[m^-2] ≈ 2.0e21 · (E_c/keV)² (stopping column of cutoff electrons,
572 // ≈ 2e17 cm^-2 keV^-2 for Coulomb log ~20).
573 const float N_c = 2.0e21f * grid.beam_E_cut_keV * grid.beam_E_cut_keV;
574 const float p = 0.5f * grid.beam_delta; // δ/2
575 Vec w = ntot % arma::pow(1.0f + N / N_c, -p);
576 const float norm = arma::dot(w, ds);
577 if (norm <= 0.0f) return Q;
578 Q = (grid.beam_flux * g / norm) * w; // W m^-3, ∫Q ds = beam_flux·g
579 return Q;
580 }
581
582 // Deposition window: cells whose center height (absolute km, C7 base =
583 // 1003 km) lies in [beam_h_lo_km, beam_h_hi_km] — the upper chromosphere
584 // below the TR. Weight ∝ n_tot (thick-target: densest reachable layer
585 // absorbs the most).
586 const float H_BASE_KM = 1.003e3f; // C7 base offset (chromo_main)
587 Vec w = arma::zeros<Vec>(grid.ns);
588 float h_lo = 0.0f;
589 for (arma::uword i = 0; i < grid.ns; ++i) {
590 const float h_ctr_km = H_BASE_KM + (h_lo + 0.5f * ds(i)) * 1.0e-3f;
591 if (h_ctr_km >= grid.beam_h_lo_km && h_ctr_km <= grid.beam_h_hi_km)
592 w(i) = ntot(i);
593 h_lo += ds(i);
594 }
595 // Normalize so ∫ φ ds = 1 ⇒ Σ w_i ds_i = 1 (φ = w / Σ w ds).
596 const float norm = arma::dot(w, ds);
597 if (norm <= 0.0f) return Q; // layer empty (shouldn't happen)
598
599 Q = (grid.beam_flux * g / norm) * w; // W m^-3
600 return Q;
601}

603/*!
604 * Ambient coronal (footpoint) heating — gentle conduction-driven evaporation
605 * H(s) = E_H0 · exp(−d(s)/s_H) · enhance(t) [W/m³], positive = heating.
606 * The steady, footpoint-anchored exponential heating of Aschwanden & Schrijver
607 * (2002, ApJS 142, 269, §3): E_H0 is the volumetric heating rate at the loop
608 * base and s_H the field-aligned scale length over which it decays into the
609 * corona. This is the H(s) term in the RTV (1978) static loop balance
610 * d/ds(κ_e T^{5/2} dT/ds) + H − n²Λ = 0 — the ingredient the code previously
611 * lacked, without which a resolved corona simply drains. d(s) is the distance
612 * to the nearest chromospheric footpoint (two-sided for a full loop so the
613 * heating is weakest at the apex; one-sided from the inner footpoint for an
614 * open line / half loop). The s_H → ∞ limit gives uniform heating (Martens
615 * 2010: T(s) is only weakly sensitive to the shape).
616 *
617 * enhance(t) is the Phase-3 driver: 1 during relaxation, ramped up to
618 * coronal_heat_enhance to drive the gentle upflow. A modest, slow increment
619 * keeps it in the Antiochos & Sturrock (1978) gentle regime (v ≪ c_s); a
620 * large/fast jump would tip into the Fisher (1985) explosive regime.
621 */
622inline Vec coronal_heating_rate(const Grid& grid) {
623 Vec H = arma::zeros<Vec>(grid.ns);
624 if (!grid.enable_coronal_heating || grid.coronal_heat_E0 <= 0.0f) return H;
625
626 // Time ramp enhance(t): hold at 1 until t_on, cosine-rise to enhance over
627 // [t_on, t_on+ramp], hold. enhance ≡ 1 (relaxation default) ⇒ purely steady.
628 float amp = 1.0f;
629 if (grid.coronal_heat_enhance != 1.0f) {
630 const float t = grid.sim_time;
631 const float ton = grid.coronal_heat_t_on;
632 const float r = (grid.coronal_heat_ramp > 1.0e-6f) ? grid.coronal_heat_ramp : 1.0e-6f;
633 float w;
634 if (t <= ton) w = 0.0f;
635 else if (t >= ton + r) w = 1.0f;
636 else w = 0.5f * (1.0f - std::cos(static_cast<float>(arma::datum::pi) * (t - ton) / r));
637 amp = 1.0f + (grid.coronal_heat_enhance - 1.0f) * w;
638 }
639
640 // Field-aligned distance to the nearest footpoint. Arc length is accumulated
641 // from grid.ds_i with the cell-0 inner face at s = 0.
642 const Vec& ds = grid.ds_i;
643 const float s_end = arma::sum(ds);
644 const float sH = (grid.coronal_heat_sH > 1.0e-3f) ? grid.coronal_heat_sH : 1.0e-3f;
645 const float E0amp = grid.coronal_heat_E0 * amp;
646 float s_lo = 0.0f;
647 for (arma::uword i = 0; i < grid.ns; ++i) {
648 const float s_c = s_lo + 0.5f * ds(i);
649 float d = grid.coronal_heat_two_sided
650 ? std::min(s_c, s_end - s_c) - grid.coronal_heat_s0
651 : s_c - grid.coronal_heat_s0;
652 if (d < 0.0f) d = 0.0f;
653 H(i) = E0amp * std::exp(-d / sH);
654 s_lo += ds(i);
655 }
656 return H;
657}
