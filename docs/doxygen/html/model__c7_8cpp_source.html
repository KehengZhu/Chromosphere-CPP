68// Natural cubic spline interpolation, port of interp1 in ModChromosphereTest.f90
69static Vec interp1_spline(const Vec& xData, const Vec& yData, const Vec& xVal) {
70 const arma::uword n = xData.n_elem;
71 Vec h(n - 1);
72 Vec alpha(n, arma::fill::zeros);
73 Vec a(n), b(n - 1), c(n, arma::fill::zeros), d(n - 1);
74 Vec l(n, arma::fill::zeros), mu(n, arma::fill::zeros), z(n, arma::fill::zeros);
75
76 for (arma::uword i = 0; i < n - 1; ++i) h(i) = xData(i + 1) - xData(i);
77 for (arma::uword i = 1; i < n - 1; ++i)
78 alpha(i) = 3.0f / h(i) * (yData(i + 1) - yData(i))
79 - 3.0f / h(i - 1) * (yData(i) - yData(i - 1));
80
81 l(0) = 1.0f;
82 for (arma::uword i = 1; i < n - 1; ++i) {
83 l(i) = 2.0f * (xData(i + 1) - xData(i - 1)) - h(i - 1) * mu(i - 1);
84 mu(i) = h(i) / l(i);
85 z(i) = (alpha(i) - h(i - 1) * z(i - 1)) / l(i);
86 }
87 l(n - 1) = 1.0f;
88
89 for (long j = (long)n - 2; j >= 0; --j) {
90 c(j) = z(j) - mu(j) * c(j + 1);
91 b(j) = (yData(j + 1) - yData(j)) / h(j) - h(j) * (c(j + 1) + 2.0f * c(j)) / 3.0f;
92 d(j) = (c(j + 1) - c(j)) / (3.0f * h(j));
93 a(j) = yData(j);
94 }
95
96 Vec yVal(xVal.n_elem);
97 for (arma::uword i = 0; i < xVal.n_elem; ++i) {
98 arma::uword j = 0;
99 while (j < n - 1 && xVal(i) > xData(j + 1)) ++j;
100 if (j >= n - 1) j = n - 2;
101 const float dx = xVal(i) - xData(j);
102 yVal(i) = a(j) + b(j) * dx + c(j) * dx * dx + d(j) * dx * dx * dx;
103 }
104 return yVal;
105}

250// Sample the COMPLETE Model C7 atmosphere (photosphere → corona, h ≈ −10 km … 68 Mm)
251// at height h_km: T linearly, n_e and n_HI log-linearly (they span many decades).
252// Concatenates MODEL_C7_PHOTO (h < 1003 km) with MODEL_C7 (h ≥ 1003 km); clamps to
253// the table ends. (model_c7 itself only ever uses MODEL_C7 from h = 1003 km up, so
254// this helper does not change any existing scenario.)
255void c7_full_profile(float h_km, float& T, float& n_e, float& n_HI) {
256 auto row = [](int k) -> const float* {
257 return (k < N_C7_PHOTO) ? MODEL_C7_PHOTO[k] : MODEL_C7[k - N_C7_PHOTO];
258 };
259 const int N = N_C7_PHOTO + N_C7;
260 auto unpack = [&](int k) { const float* r = row(k); T = r[3]; n_e = r[1]; n_HI = r[2]; };
261 if (h_km <= row(0)[0]) { unpack(0); return; }
262 if (h_km >= row(N - 1)[0]) { unpack(N - 1); return; }
263 int i = 0;
264 while (i < N - 1 && row(i + 1)[0] < h_km) ++i;
265 const float* a = row(i);
266 const float* b = row(i + 1);
267 const float t = (h_km - a[0]) / (b[0] - a[0]);
268 T = a[3] + t * (b[3] - a[3]);
269 n_e = std::exp(std::log(a[1]) + t * (std::log(b[1]) - std::log(a[1])));
270 n_HI = std::exp(std::log(a[2]) + t * (std::log(b[2]) - std::log(a[2])));
271}

273double c7_full_temperature_pchip(double h_km) {
274 // Fritsch-Carlson/Fritsch-Butland monotone cubic Hermite slopes, including
275 // the standard one-sided endpoint limiter. The complete C7 temperature is
276 // not globally monotone (it has a photospheric minimum), so slopes are
277 // limited interval-by-interval and vanish at sign-changing extrema.
278 struct TemperaturePchip {
279 std::vector<double> h;
280 std::vector<double> temperature;
281 std::vector<double> slope;
282
283 TemperaturePchip() {
284 auto row = [](int k) -> const float* {
285 return (k < N_C7_PHOTO) ? MODEL_C7_PHOTO[k]
286 : MODEL_C7[k-N_C7_PHOTO];
287 };
288 const int n = N_C7_PHOTO+N_C7;
289 h.resize(n); temperature.resize(n); slope.assign(n, 0.0);
290 for (int i = 0; i < n; ++i) {
291 h[i] = row(i)[0];
292 temperature[i] = row(i)[3];
293 }
294 std::vector<double> width(n-1), secant(n-1);
295 for (int i = 0; i+1 < n; ++i) {
296 width[i] = h[i+1]-h[i];
297 secant[i] = (temperature[i+1]-temperature[i])/width[i];
298 }
299 for (int i = 1; i+1 < n; ++i) {
300 if (secant[i-1]*secant[i] <= 0.0) continue;
301 const double w1 = 2.0*width[i]+width[i-1];
302 const double w2 = width[i]+2.0*width[i-1];
303 slope[i] = (w1+w2)/(w1/secant[i-1]+w2/secant[i]);
304 }
305 auto endpoint = [](double h0, double h1, double d0, double d1) {
306 double value = ((2.0*h0+h1)*d0-h0*d1)/(h0+h1);
307 if (value*d0 <= 0.0) return 0.0;
308 if (d0*d1 < 0.0 && std::abs(value) > 3.0*std::abs(d0))
309 return 3.0*d0;
310 return value;
311 };
312 slope.front() = endpoint(width[0], width[1], secant[0], secant[1]);
313 slope.back() = endpoint(width[n-2], width[n-3],
314 secant[n-2], secant[n-3]);
315 }
316 };
317 static const TemperaturePchip data;
318 if (h_km <= data.h.front()) return data.temperature.front();
319 if (h_km >= data.h.back()) return data.temperature.back();
320 const std::size_t i = static_cast<std::size_t>(
321 std::upper_bound(data.h.begin(), data.h.end(), h_km)-data.h.begin()-1);
322 const double width = data.h[i+1]-data.h[i];
323 const double s = (h_km-data.h[i])/width;
324 const double s2 = s*s, s3 = s2*s;
325 return (2.0*s3-3.0*s2+1.0)*data.temperature[i]
326 + (s3-2.0*s2+s)*width*data.slope[i]
327 + (-2.0*s3+3.0*s2)*data.temperature[i+1]
328 + (s3-s2)*width*data.slope[i+1];
329}

331// Route-B photoionization closure (docs/photoionization_c7_inversion_plan.md;
332// writeup §3.1), shared by model_c7 and model_column. Given cell-centered
333// (T, n_i, n_n) and heights, returns the per-cell photoionization rate P_phot:
334// invert the LOCAL ionization-equilibrium balance of EXACTLY the active Stage-E
335// network so the C7 profile is a fixed point,
336// P_phot = (n_i/n_n)(α_r + α_c) n_i − (S_CR + S_i) n_i , clipped ≥ 0,
337// with the α_c / S_i terms present only when their flags are set. Above the
338// equilibrium↔NEQ crossover (h ≈ 1500 km) blend to the frozen FAL-C rate (Chae
339// 2021) — forcing equilibrium there is what blew up the earlier calibration.
340// n_n is floored at 1 m^-3 so the (n_i/n_n) ratio stays finite where the corona
341// ionizes n_n → 0 (the w_eq → 0 weight then discards the irrelevant value); on
342// the C7 chromosphere n_n ≥ 1e8, so the floor is a no-op there.
343Vec c7_route_b_photoionization(const Grid& grid, const Vec& T_c,
344 const Vec& ni_c, const Vec& nn_c,
345 const Vec& h_cell) {
346 const Vec nn_safe = arma::clamp(nn_c, 1.0f, arma::datum::inf);
347 Vec a_tot = recombination_rate_alpha(grid, T_c); // α_r, always on
348 if (grid.enable_threebody_recombination)
349 a_tot += recombination_rate_alpha_c(grid, T_c, ni_c); // α_c = κ_c n_e
350 Vec S_tot = ionization_rate_S_CR(grid, T_c); // S_CR, always on
351 if (grid.enable_direct_collisional_ionization)
352 S_tot += ionization_rate_S(grid, T_c); // S_i, Voronov
353 Vec P_inv = (ni_c / nn_safe) % (a_tot % ni_c) - S_tot % ni_c;
354 P_inv = arma::clamp(P_inv, 0.0f, arma::datum::inf);
355 const Vec P_falc = photoionization_rate_chae(h_cell);
356 const float H_EQ_NEQ = 1500.0f; // km, equilibrium ↔ NEQ crossover (Chae marker)
357 const float BLEND_W = 120.0f; // km, blend width
358 const Vec w_eq = 0.5f * (1.0f - arma::tanh((h_cell - H_EQ_NEQ) / BLEND_W));
359 return w_eq % P_inv + (1.0f - w_eq) % P_falc;
360}

362Vec model_c7_ic(Grid& grid, bool extended, bool tr_jump_bc) {
363 // Domain top: standard column (rows 0..N_C7_BASE-1, h ≈ 2.153 Mm) or the real
364 // Table-26 corona extension (rows 0..N_C7-1, up to h ≈ 68 Mm) when the
365 // corona-as-boundary gentle-evaporation config asks for it. When extended and
366 // grid.corona_top_km > 0, truncate the table at the first row reaching that
367 // height — so the upper boundary sits high in a resolved coronal volume at a
368 // chosen altitude (default set by model_gentle ≈ 10 Mm) rather than at the very
369 // top of the table.
370 int n_rows = extended ? N_C7 : N_C7_BASE;
371 if (extended && grid.corona_top_km > 0.0f) {
372 int nr = N_C7_BASE; // never go below the standard column
373 while (nr < N_C7 && MODEL_C7[nr - 1][0] < grid.corona_top_km) ++nr;
374 n_rows = nr;
375 }
376 // Phase-1 geometry choices (quiet-sun relaxation):
377 // * Uniform B = 1, ∂(1/B)/∂s = 0 — single straight field line; the
378 // flux-tube expansion term in the source/flux drops out trivially.
379 // A non-uniform B(s) (e.g. Spruit 1981 thin-flux-tube) is the
380 // follow-on; the integrator already supports it.
381 // * Gravity ON: φ_g(s) = g · (h(s) - h_base) populated below.
382 //
383 // HSE strategy (Option A from the Phase-2 plan): we leave C7's tabulated
384 // (n, T) as the IC even though C7 was built with NLTE radiative balance,
385 // not ideal-gas HSE. Under the code's EOS the C7 profile is off HSE by a
386 // nearly uniform 20–26%: |dp/dh + ρg| / |ρg| ≈ 0.25 across the chromosphere
387 // (dp/dh is more negative than -ρg, i.e. C7 is "lighter at the top" than
388 // a pure-HSE atmosphere with the same T(h) would be). The run absorbs
389 // this imbalance during the first few acoustic crossings; the residual
390 // drift away from C7 is dominated by missing radiative cooling, not by
391 // the IC choice.
392 grid.B_imh.ones();
393 grid.B_iph.ones();
394 grid.B_i.ones();
395 grid.dinvB_ds_i.zeros();
396
397 // Tabulated profile (n_rows: chromosphere + TR, Table 26).
398 Vec h_tab(n_rows), ne_tab(n_rows), nn_tab(n_rows), T_tab(n_rows);
399 for (arma::uword i = 0; i < (arma::uword)n_rows; ++i) {
400 h_tab(i) = MODEL_C7[i][0];
401 ne_tab(i) = MODEL_C7[i][1];
402 nn_tab(i) = MODEL_C7[i][2];
403 T_tab(i) = MODEL_C7[i][3];
404 }
405
406 // Faces: indices -1..ns+3 (size ns+5), uniformly over [h_tab(0), h_tab(top)].
407 const arma::uword nF = grid.ns + 5;
408 Vec h_F(nF);
409 const float h0 = h_tab(0), h1 = h_tab(n_rows - 1);
410 for (arma::uword k = 0; k < nF; ++k) {
411 const float frac = (float)k / (float)(nF - 1);
412 h_F(k) = h0 + (h1 - h0) * frac;
413 }
414
415 // φ_g at each face, measured relative to the C7 base height (h0 = 1003 km)
416 // so the absolute potential stays comparable to thermal energy in float32.
417 // Only ∂φ_g/∂s matters physically; the constant offset is gauge.
418 Vec phi_g_F(nF);
419 for (arma::uword k = 0; k < nF; ++k) {
420 phi_g_F(k) = grid.g * (h_F(k) - h0) * 1000.0f; // km -> m
421 }
422 for (arma::uword i = 0; i < grid.ns; ++i) {
423 grid.phi_g_imh(i) = phi_g_F(i);
424 grid.phi_g_iph(i) = phi_g_F(i + 1);
425 }
426
427 Vec ne_F = interp1_spline(h_tab, ne_tab, h_F);
428 Vec nn_F = interp1_spline(h_tab, nn_tab, h_F);
429 Vec T_F = interp1_spline(h_tab, T_tab, h_F);
430 // The natural cubic spline can overshoot to negative values across the steep
431 // TR where n_HI drops ~2 orders of magnitude over a few km; clamp to small
432 // positive floors so densities/temperatures stay physical on every face.
433 ne_F = arma::clamp(ne_F, 1.0e10f, arma::datum::inf);
434 nn_F = arma::clamp(nn_F, 1.0e8f, arma::datum::inf);
435 T_F = arma::clamp(T_F, 3.0e3f, arma::datum::inf);
436
437 // Extended-cell quantities: averages of adjacent faces, indices 0..ns+2.
438 const arma::uword nE = grid.ns + 3;
439 Vec len_E(nE), ne_E(nE), nn_E(nE), T_E(nE);
440 for (arma::uword k = 0; k < nE; ++k) {
441 len_E(k) = (h_F(k + 1) - h_F(k)) * 1000.0f; // km -> m
442 ne_E(k) = 0.5f * (ne_F(k + 1) + ne_F(k));
443 nn_E(k) = 0.5f * (nn_F(k + 1) + nn_F(k));
444 T_E(k) = 0.5f * (T_F(k + 1) + T_F(k));
445 }
446
447 // Cell-centered interior: first ns of the extended array.
448 for (arma::uword i = 0; i < grid.ns; ++i) grid.ds_i(i) = len_E(i);
449
450 // Build the conserved-variable state vector. Total energy includes the
451 // ρ·φ_g term so that cons2prim (state.cpp:108) recovers the right pressure.
452 Vec xn = arma::zeros<Vec>(grid.n_state);
453 for (arma::uword i = 0; i < grid.ns; ++i) {
454 const float n_e = ne_E(i);
455 const float n_n = nn_E(i);
456 const float T = T_E(i);
457 const float phi_g = 0.5f * (grid.phi_g_imh(i) + grid.phi_g_iph(i));
458 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::RHO_I)) = n_e * grid.m_i;
459 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::RHO_N)) = n_n * grid.m_n;
460 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::MOM_I)) = 0.0f;
461 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::MOM_N)) = 0.0f;
462 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::E_I)) = grid.inv_gm1() * grid.k_b * n_e * 2.0f * T + n_e * grid.m_i * phi_g;
463 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::E_N)) = grid.inv_gm1() * grid.k_b * n_n * T + n_n * grid.m_n * phi_g;
464 // Electron internal energy ε_e = p_e/(γ−1) with T_e = T at IC (no KE/φ).
465 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::E_E)) = grid.inv_gm1() * grid.k_b * n_e * T;
466 }

468 // Outer ghost cells: the fixed lower-TR reservoir. Both ghosts hold the
469 // same TR-base state taken from the top of the extended C7 table (h ≈ 2153
470 // km, T ≈ 2.3×10⁴ K) — RTV (1978) constant-pressure TR base. This is the
471 // coronal heat reservoir; model_c7_update_bc keeps it FIXED each step (it
472 // does NOT track the cooling interior), which is what supplies the net
473 // downward conductive flux q(T) that holds the chromosphere top against
474 // ionization cooling. φ_g is the upper face of the last interior cell.
475 {
476 const float phi_g_outer = grid.phi_g_iph(grid.ns - 1);
477 kOuterTtr = T_tab(n_rows - 1); // standard 2.310×10⁴ K; extended 2.925×10⁵ K
478 kOuterNeTr = ne_tab(n_rows - 1); // electron/proton density at the top
479 kOuterNnTr = nn_tab(n_rows - 1); // neutral density at the top (slaved, ~fully ionized)
480 for (arma::uword k = 0; k < 2; ++k) {
481 Vec& ob = (k == 0 ? grid.outer_boundary0_i : grid.outer_boundary1_i);
482 ob(cons::RHO_I) = kOuterNeTr * grid.m_i;
483 ob(cons::RHO_N) = kOuterNnTr * grid.m_n;
484 ob(cons::MOM_I) = 0.0f;
485 ob(cons::MOM_N) = 0.0f;
486 ob(cons::E_I) = grid.inv_gm1() * grid.k_b * kOuterNeTr * 2.0f * kOuterTtr + kOuterNeTr * grid.m_i * phi_g_outer;
487 ob(cons::E_N) = grid.inv_gm1() * grid.k_b * kOuterNnTr * kOuterTtr + kOuterNnTr * grid.m_n * phi_g_outer;
488 ob(cons::E_E) = grid.inv_gm1() * grid.k_b * kOuterNeTr * kOuterTtr;
489 }
490 }

492 // Inner ghost cells replicate the innermost extended cell. φ_g taken as
493 // the lower face of cell 0 (which is 0 by gauge choice), matching the
494 // im1 path in rhs.cpp:178 and integrators.cpp:189.
495 //
496 // The (n_n, T) snapshot here is what model_c7_update_bc later uses as
497 // the Dirichlet-pinned photospheric reservoir state for the inner ghost.
498 {
499 const float n_e = ne_E(0);
500 const float n_n = nn_E(0);
501 const float T = T_E(0);
502 const float ds0 = len_E(0);
503 const float phi_g_inner = grid.phi_g_imh(0);
504 grid.inner_boundary0_i(cons::RHO_I) = n_e * grid.m_i;
505 grid.inner_boundary0_i(cons::RHO_N) = n_n * grid.m_n;
506 grid.inner_boundary0_i(cons::MOM_I) = 0.0f;
507 grid.inner_boundary0_i(cons::MOM_N) = 0.0f;
508 grid.inner_boundary0_i(cons::E_I) = grid.inv_gm1() * grid.k_b * n_e * 2.0f * T + n_e * grid.m_i * phi_g_inner;
509 grid.inner_boundary0_i(cons::E_N) = grid.inv_gm1() * grid.k_b * n_n * T + n_n * grid.m_n * phi_g_inner;
510 grid.inner_boundary0_i(cons::E_E) = grid.inv_gm1() * grid.k_b * n_e * T;
511 grid.inner_boundary1_i = grid.inner_boundary0_i;
512
513 kInnerRhoIPinned = n_e * grid.m_i;
514 kInnerRhoNPinned = n_n * grid.m_n;
515 kInnerTnPinned = T;

517 // Discrete-HSE ghost pressures (Pandey 2024 §3.1): the inner ghost sits
518 // a cell-spacing ds0 below cell 0, so a hydrostatic reservoir carries the
519 // extra weight ρg·ds0 → p_ghost = p_0 + ρ_0 g ds0. Imposing these (rather
520 // than a zero-gradient copy of p_0) makes the pressure gradient across
521 // the inner face balance gravity discretely, so V=U=0 is an exact fixed
522 // point of the momentum update and the base does not launch a wave.
523 const float p_i_0 = 2.0f * n_e * grid.k_b * T;
524 const float p_n_0 = n_n * grid.k_b * T;
525 kInnerPiGhost = p_i_0 + (n_e * grid.m_i) * grid.g * ds0;
526 kInnerPnGhost = p_n_0 + (n_n * grid.m_n) * grid.g * ds0;
527 }

529 // Height-dependent photoionization rate P_phot(h): Route B closure
530 // (docs/photoionization_c7_inversion_plan.md; writeup §3.1).
531 //
532 // Equilibrium region (dense lower/mid chromosphere): set P_phot by
533 // inverting the *local ionization-equilibrium balance of exactly the Stage E
534 // network* on C7's tabulated (n_e, n_n, T), so C7 is a self-consistent
535 // fixed point of the rate network Stage E integrates under whatever flags
536 // are active:
537 // P_phot(z) = (n_e/n_n)(α_r + α_c) n_e − (S_i + S_CR) n_e , clipped ≥ 0,
538 // where the α_c (three-body) and S_i (Voronov) terms are present only when
539 // their flags are set. The default network is P_phot + S_CR (ionization) and
540 // α_r (recombination); S_i is 2–3 orders below S_CR everywhere and α_c is
541 // within 1–2 orders of α_r only below ~700 km (outside C7's 1003 km floor),
542 // so dropping them barely perturbs the inverted P_phot — but the inversion
543 // must still match the active network exactly, or C7 drifts off its own IC.
544 // This closes the over-ionization gap where it physically should (the
545 // ionization timescale is short and equilibrium holds). The inverted rate
546 // runs a few × above the ground-state FAL-C rate of Chae (2021) — the
547 // expected case-B-vs-ground-state recombination difference, not a defect.
548 //
549 // NEQ region (rarefied upper chromosphere, z ≳ 1500 km): the recombination
550 // timescale exceeds the dynamical time, so the gas is genuinely out of
551 // ionization equilibrium (real over-ionization). There we keep the frozen
552 // FAL-C radiation-field rate (photoionization_rate_chae) and let the gas
553 // depart from C7 — forcing C7 as a fixed point here is what blew up the
554 // old calibration. A tanh window in height blends the two.
555 {
556 Vec h_cell(grid.ns), Tc(grid.ns), nec(grid.ns), nnc(grid.ns);
557 for (arma::uword i = 0; i < grid.ns; ++i) {
558 h_cell(i) = 0.5f * (h_F(i) + h_F(i + 1)); // km, tau_500=1 scale
559 Tc(i) = T_E(i);
560 nec(i) = ne_E(i);
561 nnc(i) = nn_E(i);
562 }
563 grid.photoionization_rate_i = c7_route_b_photoionization(grid, Tc, nec, nnc, h_cell);
564 }

566 // Upper-boundary coronal conductive flux q(T) (RTV 1978). The static
567 // conduction-dominated corona delivers F_c = (2/7) κ₀ T_cor^{7/2} / L to the
568 // TR base, where it is re-radiated; we impose that flux as the Stage-D
569 // Neumann outer BC. κ₀ = 10⁻¹¹ W m⁻¹ K⁻⁷ᐟ² (Johnston 2020 SI), and a quiet-Sun
570 // corona T_cor ≈ 8×10⁵ K over a half-length L ≈ 15 Mm gives F_c ≈ 90 W m⁻².
571 //
572 // q(T) heating and radiative cooling are a physical pair: the coronal flux is
573 // re-radiated at/below the TR base. Impose q(T) only when Stage R cooling is
574 // active — otherwise the injected heat has no sink and the top runs away. So
575 // the complete physical model_c7 is run WITH cooling; the no-cooling mode is
576 // a diagnostic baseline (no coronal heating imposed).
577 {
578 const float kappa0 = 1.0e-11f; // W m^-1 K^-7/2
579 if (extended) {
580 // Corona-as-boundary (top at h ≈ 2.628 Mm, T ≈ 0.29 MK): impose the
581 // conductive flux C7 ITSELF carries across the new top, computed from
582 // its own gradient q = κ₀ T^{5/2} dT/ds using the top two Table-26 rows
583 // (≈130 W m⁻² downward) — self-consistent with the model, no assumed
584 // T_cor / L. This is the steady quiet value; the gentle-evaporation
585 // driver (model_gentle) ramps it up via outer_heat_flux_enhance.
586 const float T_top = T_tab(n_rows - 1);
587 const float dTds = (T_tab(n_rows - 1) - T_tab(n_rows - 2))
588 / ((h_tab(n_rows - 1) - h_tab(n_rows - 2)) * 1.0e3f); // km→m
589 grid.outer_heat_flux = kappa0 * std::pow(T_top, 2.5f) * dTds;
590 } else {
591 const float T_cor = 8.0e5f; // K, quiet-Sun coronal apex
592 const float L_cor = 4.5e7f; // m, coronal loop half-length (quiet-Sun)
593 // ≈ 29 W m⁻². With the optically-thin TR radiative sink present, this
594 // sits in the stable window q ∈ [~30, 90] W m⁻²: the TR/corona radiates
595 // the imposed flux and the chromosphere proper stays cool. Below ~10 W m⁻²
596 // the weak heating drops the upper chromosphere into the radiative-loss-peak
597 // thermal-instability (condensation) regime and the run fails.
598 grid.outer_heat_flux = (2.0f / 7.0f) * kappa0
599 * std::pow(T_cor, 3.5f) / L_cor;
600 }
601 // Steady base for the q(T) ramp (model_gentle); enhance = 1 ⇒ q ≡ base.
602 grid.outer_heat_flux_base = grid.outer_heat_flux;
603 grid.impose_outer_heat_flux = grid.enable_radiative_cooling;
604 }
605
606 // Isotropic numerical diffusivity (Pandey 2024 §3.3), grid-scaled. Pandey's
607 // benchmark is χ ≳ 30×10⁸ m² s⁻¹ at Δz = 40 km; here it only needs to damp the
608 // base wave and any TR-gradient ringing (the coronal heating is carried by
609 // q(T), not by χ), so a modest value suffices. Scale ∝ Δs so coarser grids
610 // diffuse over a comparable number of cells.
611 {
612 grid.numerical_diffusivity_per_length = 2.0e3f; // C_num [m/s]
613 }
614
615 // TRAC (Johnston et al. 2020): broaden the under-resolved TR so the coarse
616 // 100-cell grid recovers the grid-converged coronal density response while
617 // conserving the TR-integrated radiation/heating (the Eq.-27 jump balance).
618 // Paired with the radiative sink (q(T)/Λ(T) need a resolved TR to balance).
619 grid.enable_trac = grid.enable_radiative_cooling;
620 grid.trac_T_chrom = 2.0e4f;
621
622 // Vacuum / positivity floor (the same machinery the flare scenario uses).
623 // As the C7 column relaxes, the upper chromosphere/TR fully ionizes and the
624 // neutral density n_n collapses toward zero there; the 0/0 neutral-velocity
625 // decode then makes the spectral radius blow up → NaN (the run otherwise dies
626 // around step ~2000). The floor clips ρ,p positive and, where f = n_i/(n_i+n_n)
627 // ≥ 0.99, slaves the trace neutral onto the charged fluid (U=V, T_n=T_e) — the
628 // single-fluid limit the outer BC already imposes. It is a no-op below the TR
629 // where the gas is partially neutral and well-conditioned.
630 grid.enable_vacuum_floor = true;
631
632 // "New explanation" upper BC (docs/gentle_evaporation_downflow.md), HYBRID
633 // form, gated to the bare model_c7 scenario (model_flare / analytic_canopy /
634 // model_gentle keep the default RTV reservoir):
635 // * well-balanced explicit reconstruction (removes the spurious (γ−1)g
636 // downforce that the old run damped into the persistent ~1 km/s TR
637 // downflow — see the doc's "load-bearing discovery"),
638 // * the top face imposes a HYDROSTATIC ghost pressure + EOS ghost density
639 // (model_c7_update_bc, c7_tr_jump_bc branch) so V=0 is a boundary fixed
640 // point and there is no boundary downflow.
641 // The coronal heat keeps entering via the imposed Neumann flux q(T)
642 // (impose_outer_heat_flux left ON with cooling) — NOT a Dirichlet temperature
643 // jump, which over-conducts on the coarse C7 grid. So we do NOT touch
644 // impose_outer_heat_flux here.
645 if (tr_jump_bc) {
646 auto env_f = [](const char* key, float fallback) -> float {
647 if (const char* e = std::getenv(key)) {
648 try { return std::stof(std::string(e)); } catch (...) {}
649 }
650 return fallback;
651 };
652 kC7Ajump = env_f("C7_TJUMP_A", 1.0f); // ghost-T multiplier on the live top T
653 kC7Bjump = env_f("C7_TJUMP_B", 1.0f); // (default 1 ⇒ continuous extrapolation)
654 grid.c7_tr_jump_bc = true;
655 grid.well_balanced = true;
656 }
657
658 grid.broadcast();
659 return xn;
660}

662void model_c7_update_bc(Grid& grid, const Vec& xn) {
663 // Gentle-evaporation driver (docs/gentle_evaporation_plan.md v2): ramp the
664 // imposed coronal conductive flux q(T) = q_base · enhance(t). enhance(t) holds
665 // at 1 until t_on, then cosine-rises to outer_heat_flux_enhance over
666 // [t_on, t_on+ramp]. enhance = 1 (default) ⇒ q stays at the steady base, so
667 // model_c7 / model_flare are byte-for-byte unchanged. The stronger downward
668 // flux heats the upper TR → conduction-driven upflow (Antiochos & Sturrock 1978).
669 if (grid.outer_heat_flux_base > 0.0f) {
670 float amp = 1.0f;
671 if (grid.outer_heat_flux_enhance != 1.0f) {
672 const float t = grid.sim_time;
673 const float ton = grid.outer_heat_flux_t_on;
674 const float r = (grid.outer_heat_flux_ramp > 1.0e-6f) ? grid.outer_heat_flux_ramp : 1.0e-6f;
675 float w;
676 if (t <= ton) w = 0.0f;
677 else if (t >= ton + r) w = 1.0f;
678 else w = 0.5f * (1.0f - std::cos(static_cast<float>(arma::datum::pi) * (t - ton) / r));
679 amp = 1.0f + (grid.outer_heat_flux_enhance - 1.0f) * w;
680 }
681 grid.outer_heat_flux = grid.outer_heat_flux_base * amp;
682 }

684 // Physically-grounded boundary closures (docs/boundary_conditions_plan.md):
685 // * Outer face — fixed lower-TR reservoir (RTV 1978 + Bradshaw & Emslie
686 // 2020 + the single-fluid slaving of Gómez Míguez et al. 2024).
687 // * Inner face — discrete-HSE photospheric reservoir, V=U=0 (Pandey 2024).
688 // apply_open_bcs lays down a default ghost layer; we overwrite both faces.
689 apply_open_bcs(grid, xn);
690
691 const auto sz = arma::size(grid.ns, num_of_eq);
692 const float m_i = grid.m_i;
693 const float m_n = grid.m_n;
694 const float k_b = grid.k_b;
695
696 // ====================================================================
697 // Outer face — lower transition region (h ≈ 2153 km, T ≈ 2.3×10⁴ K).
698 // ====================================================================
699 if (grid.c7_tr_jump_bc) {
700 // "New explanation" upper BC (docs/gentle_evaporation_downflow.md), HYBRID
701 // form. The boundary downflow is removed by the well-balanced
702 // reconstruction + a hydrostatic/EOS ghost; the coronal HEAT keeps coming
703 // from the imposed Neumann flux q(T) (not a Dirichlet T-jump, which over-
704 // conducts on the coarse C7 grid). Three steps:
705 // (1) PRESSURE — hydrostatic ghost so the centered gradient at the
706 // boundary cell balances gravity, (p_{nl-1} − p_ghost)/(2Δs) = ρ_nl g
707 // ⇒ V=0 is a fixed point (paired with well_balanced, no boundary
708 // downflow). Applied per fluid (ions carry the electron pressure).
709 // (2) TEMPERATURE — CONTINUOUS from the top cell, T_ghost0 = a·T_top,
710 // T_ghost1 = b·T_ghost0 with a=b=1 by default (no jump). The ghost-T
711 // sets only the EOS density and the hydro flux; it does NOT drive
712 // conduction (the imposed flux does), so there is no over-conduction.
713 // (3) DENSITY — from the EOS at (p_ghost, T_ghost): p_i = 2 n_i k T
714 // (protons+electrons), p_n = n_n k T.
715 // Velocity: Mach-capped OUTFLOW (lets a conduction-driven upflow leave; the
716 // cap |V| ≤ outer_mach_cap·c_s prevents the ill-posed-inflow runaway).
717 const float phi_g_out = grid.phi_g_iph(grid.ns - 1);
718 const float g = grid.g;
719
720 // Decode an interior cell's per-fluid pressures / densities (cons2prim).
721 auto decode = [&](arma::uword i, float& p_i, float& p_n,
722 float& rho_i, float& rho_n) {
723 rho_i = xn(arma::sub2ind(sz, i, cons::RHO_I));
724 rho_n = xn(arma::sub2ind(sz, i, cons::RHO_N));
725 const float Vi = xn(arma::sub2ind(sz, i, cons::MOM_I)) / rho_i;
726 const float Un = xn(arma::sub2ind(sz, i, cons::MOM_N)) / rho_n;
727 const float E_i = xn(arma::sub2ind(sz, i, cons::E_I));
728 const float E_n = xn(arma::sub2ind(sz, i, cons::E_N));
729 const float phi_g = 0.5f * (grid.phi_g_imh(i) + grid.phi_g_iph(i));
730 p_i = grid.gm1() * E_i - grid.half_gm1() * rho_i * Vi * Vi - grid.gm1() * rho_i * phi_g;
731 p_n = grid.gm1() * E_n - grid.half_gm1() * rho_n * Un * Un - grid.gm1() * rho_n * phi_g;
732 };
733
734 const arma::uword nl = grid.ns - 1;
735 const float ds = grid.ds_i(nl);
736 float p_i_top, p_n_top, rho_i_top, rho_n_top;
737 decode(nl, p_i_top, p_n_top, rho_i_top, rho_n_top);
738 float p_i_2, p_n_2, rho_i_2, rho_n_2;
739 decode(grid.ns >= 2 ? nl - 1 : nl, p_i_2, p_n_2, rho_i_2, rho_n_2);
740
741 // (2) ghost temperature, continuous from the top cell (a=b=1 default).
742 const float T_top = p_i_top / (2.0f * (rho_i_top / m_i) * k_b); // charged-fluid T
743 const float T_g0 = kC7Ajump * T_top;
744 const float T_g1 = kC7Bjump * T_g0;
745
746 // (1)+(3) per-fluid hydrostatic ghost pressure → EOS density at (p, T).
747 auto build = [&](float p_i_lo, float p_n_lo, float rho_i_w, float rho_n_w,
748 float T_g, float& rho_i_g, float& rho_n_g,
749 float& p_i_g, float& p_n_g) {
750 p_i_g = p_i_lo - 2.0f * ds * rho_i_w * g;
751 p_n_g = p_n_lo - 2.0f * ds * rho_n_w * g;
752 if (p_i_g < 1.0e-6f) p_i_g = 1.0e-6f; // keep positive at the tenuous top
753 if (p_n_g < 1.0e-12f) p_n_g = 1.0e-12f;
754 rho_i_g = (p_i_g / (2.0f * k_b * T_g)) * m_i; // p_i = 2 n_i k T
755 rho_n_g = (p_n_g / (k_b * T_g)) * m_n; // p_n = n_n k T
756 };
757 float rho_i_g0, rho_n_g0, p_i_g0, p_n_g0;
758 build(p_i_2, p_n_2, rho_i_top, rho_n_top, T_g0, rho_i_g0, rho_n_g0, p_i_g0, p_n_g0);
759 float rho_i_g1, rho_n_g1, p_i_g1, p_n_g1;
760 build(p_i_top, p_n_top, rho_i_g0, rho_n_g0, T_g1, rho_i_g1, rho_n_g1, p_i_g1, p_n_g1);
761
762 // velocity: Mach-capped outflow (lifted for flare's free outflow).
763 const float c_s = std::sqrt(2.0f * grid.gamma_mono * k_b * kOuterTtr / m_i);
764 const float V_cap = grid.outer_mach_cap * c_s;
765 float V_g = xn(arma::sub2ind(sz, nl, cons::MOM_I)) / rho_i_top;
766 if (!grid.outer_free_outflow) {
767 if (V_g > V_cap) V_g = V_cap;
768 if (V_g < -V_cap) V_g = -V_cap;
769 }
770
771 auto pack = [&](Vec& ob, float rho_i, float rho_n, float p_i, float p_n) {
772 ob(cons::RHO_I) = rho_i;
773 ob(cons::RHO_N) = rho_n;
774 ob(cons::MOM_I) = rho_i * V_g;
775 ob(cons::MOM_N) = rho_n * V_g; // neutral locked to charge fluid
776 ob(cons::E_I) = grid.inv_gm1() * p_i + 0.5f * rho_i * V_g * V_g + rho_i * phi_g_out;
777 ob(cons::E_N) = grid.inv_gm1() * p_n + 0.5f * rho_n * V_g * V_g + rho_n * phi_g_out;
778 ob(cons::E_E) = 0.5f * grid.inv_gm1() * p_i; // ε_e = p_e/(γ−1), p_e = ½ p_i ⇒ T_e = T_i
779 };
780 pack(grid.outer_boundary0_i, rho_i_g0, rho_n_g0, p_i_g0, p_n_g0);
781 pack(grid.outer_boundary1_i, rho_i_g1, rho_n_g1, p_i_g1, p_n_g1);
782 } else {
783 // Default: fixed lower-TR reservoir (RTV 1978). The TR base is a constant-
784 // pressure layer: pressure is continuous across it while temperature jumps
785 // to the coronal value over the (here unresolved) TR. We do NOT impose a
786 // hot dense ghost wall — on the coarse grid a fixed coronal T would over-
787 // conduct (κ_e∝T^{5/2}) and over-pressurize the top cell, driving a
788 // spurious supersonic downflow. Instead:
789 // * ρ, T — Neumann (continuous pressure across the constant-pressure TR;
790 // apply_open_bcs already laid these down).
791 // * V, U — mass-flux outflow capped at a slow Mach number M ≤ 0.05 of the
792 // TR sound speed c_s = √(2γ k_B T/m_p) (Bradshaw & Emslie 2020;
793 // |V| ≲ 1 km/s, sign/magnitude as in Hansteen 1993). U = V locks
794 // the trace neutral to the charged fluid (Gómez Míguez 2024,
795 // single-fluid TR) — no ill-conditioned U_n = ρU_n/ρ_n.
796 // The TR thermal structure / coronal support enters entirely through the
797 // imposed downward conductive flux q(T) (grid.outer_heat_flux, set in
798 // model_c7_ic and applied as the Stage-D Neumann BC) — the load-bearing
799 // closure that balances Route-B ionization cooling.
800 const float phi_g_out = grid.phi_g_iph(grid.ns - 1);
801 const float c_s = std::sqrt(2.0f * grid.gamma_mono * k_b * kOuterTtr / m_i);
802 const float V_cap = grid.outer_mach_cap * c_s;
803
804 auto cap_ghost = [&](Vec& ob) {
805 const float rho_i = ob(cons::RHO_I);
806 const float rho_n = ob(cons::RHO_N);
807 const float n_i = rho_i / m_i;
808 const float n_n = rho_n / m_n;
809 const float V_old = ob(cons::MOM_I) / rho_i;
810 const float U_old = ob(cons::MOM_N) / rho_n;
811 // Decode Neumann T_i, T_n from the energy apply_open_bcs packed.
812 const float T_i_in = (ob(cons::E_I) - 0.5f * rho_i * V_old * V_old
813 - rho_i * phi_g_out) / (2.0f * grid.inv_gm1() * k_b * n_i);
814 const float T_n_in = (ob(cons::E_N) - 0.5f * rho_n * U_old * U_old
815 - rho_n * phi_g_out) / (grid.inv_gm1() * k_b * n_n);
816 // Mach-0.05 cap for the quiet-Sun TR (Bradshaw & Emslie 2020). For
817 // flare scenarios (grid.outer_free_outflow) the cap is lifted so a
818 // supersonic evaporation upflow can leave the domain through a
819 // transparent (Neumann) outer face — clamping it would choke the
820 // ~100s km/s evaporated outflow.
821 float V_g = V_old;
822 if (!grid.outer_free_outflow) {
823 if (V_g > V_cap) V_g = V_cap;
824 if (V_g < -V_cap) V_g = -V_cap;
825 }
826 const float U_g = V_g; // neutral locked to charge fluid
827 ob(cons::MOM_I) = rho_i * V_g;
828 ob(cons::MOM_N) = rho_n * U_g;
829 ob(cons::E_I) = grid.inv_gm1() * k_b * n_i * 2.0f * T_i_in
830 + 0.5f * rho_i * V_g * V_g + rho_i * phi_g_out;
831 ob(cons::E_N) = grid.inv_gm1() * k_b * n_n * T_n_in
832 + 0.5f * rho_n * U_g * U_g + rho_n * phi_g_out;
833 ob(cons::E_E) = grid.inv_gm1() * k_b * n_i * T_i_in; // T_e = T_charged ghost
834 };
835 cap_ghost(grid.outer_boundary0_i);
836 cap_ghost(grid.outer_boundary1_i);
837 } // end outer-face mode branch
838
839 // ====================================================================
840 // Inner face — photosphere, discrete-HSE V=0 reservoir (Pandey 2024).
841 // ====================================================================
842 // Dirichlet ρ_i, ρ_n at the C7 base; V=U=0; and hydrostatic ghost pressures
843 // p_ghost = p_0 + ρ_0 g Δs (captured in model_c7_ic) so the pressure
844 // gradient across the inner face balances gravity discretely and V=U=0 is an
845 // exact fixed point — removing the spurious base wave the old "pin-to-IC"
846 // ghost launched. Numerical diffusion (Stage D isotropic floor) damps any
847 // residual ringing rather than reflecting it.
848 {
849 const float phi_g_inner = grid.phi_g_imh(0);
850 const float rho_i_g = kInnerRhoIPinned;
851 const float rho_n_g = kInnerRhoNPinned;
852
853 Vec& ob0 = grid.inner_boundary0_i;
854 ob0(cons::RHO_I) = rho_i_g;
855 ob0(cons::RHO_N) = rho_n_g;
856 ob0(cons::MOM_I) = 0.0f; // V = 0 reservoir
857 ob0(cons::MOM_N) = 0.0f; // U = 0 reservoir
858 ob0(cons::E_I) = grid.inv_gm1() * kInnerPiGhost + rho_i_g * phi_g_inner;
859 ob0(cons::E_N) = grid.inv_gm1() * kInnerPnGhost + rho_n_g * phi_g_inner;
860 ob0(cons::E_E) = 0.5f * grid.inv_gm1() * kInnerPiGhost; // p_e = ½ p_total ⇒ T_e = T_i
861 grid.inner_boundary1_i = ob0;
862 }
863
864 grid.broadcast();
865}
