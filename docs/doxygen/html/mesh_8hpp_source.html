6 * Shared static local-refinement mesh construction.
7 *
8 * Builds a metric-aware, STATIC non-uniform finite-volume face grid used by the
9 * built-in static scenarios. This is static local refinement, NOT AMR: the mesh
10 * is generated once at IC time and never regridded, remapped, or adapted at
11 * runtime. PFSS field lines carry their own externally-specified mesh and never
12 * go through this builder.
13 *
14 * The refinement is opt-in and reduces to the exact uniform grid when disabled

30enum class RefineProfile {
31 /// coarse → fine [s_lo,s_hi] → graded fine→coarse transition → coarse.

36 /// A compact refined band that reaches the OUTER boundary (the upper TR).
37 /// `s_hi_km` is unused: the fine region always ends at the domain top, and

39 Outer,

42/// Static local-refinement controls. Read from the environment; the documented
43/// lower-domain diagnostic profile refines 0–700 km by 4× and transitions back to
44/// the outer coarse spacing smoothly over 700–800 km, while the documented
45/// outer/TR profile refines from s_lo up to the domain top with the transition in
46/// [s_lo−τ, s_lo]. Coordinates are field-line distance [km] from the domain INNER
47/// face (s = 0 at the inner boundary).
48struct RefineParams {
49 double factor = 1.0; ///< fine/coarse resolution ratio; 1 ⇒ disabled
50 double s_lo_km = 0.0; ///< inner edge of the refined region [km]
51 double s_hi_km = 700.0; ///< outer edge of the refined region [km] (Lower only)
52 double transition_km = 100.0; ///< width of the graded coarse↔fine transition [km]
53 /// Which end of the domain the refined band is anchored to, i.e. how
54 /// s_lo_km / s_hi_km / transition_km are laid out (see RefineProfile).
55 RefineProfile profile = RefineProfile::Lower;
56
57 /// True when refinement is actually requested. `factor` is compared against
58 /// 1 with a small tolerance, so a nominally-1 factor read from the
59 /// environment still yields the exact uniform mesh.
60 bool enabled() const { return factor > 1.0 + 1.0e-9; }
61};

63/// Read `GRID_REFINE_FACTOR` / `_S_LO_KM` / `_S_HI_KM` / `_TRANSITION_KM` /
64/// `_PROFILE` (`lower` | `outer`, case-insensitive). When `iso_alias` is true the
65/// ISO_REFINE_* variables OVERRIDE their GRID_REFINE_* counterparts
66/// (model_column). Missing variables keep the struct defaults.
67RefineParams refine_params_from_env(bool iso_alias);

69/// Build canonical cell FACES [m], measured from the inner face: the returned
70/// vector has (ncells+1) entries, faces.front() == 0 and faces.back() == length_m
71/// exactly. `ns_coarse` is the coarse-grid-equivalent cell count; the coarse
72/// spacing is length_m/ns_coarse and refinement only ADDS cells (it never coarsens
73/// the rest of the column). When rp is disabled — or the refined window does not
74/// fit inside the domain — a uniform ns_coarse grid is returned.
75///
76/// RefineProfile::Lower: [0,s_lo] coarse, [s_lo,s_hi] fine (≈coarse/factor),
77/// [s_hi,s_hi+transition] an exponential (geometric) grade fine→coarse, then
78/// [s_hi+transition,length] coarse. Exact faces at 0, s_lo, s_hi, s_hi+transition.
79///
80/// RefineProfile::Outer: [0,s_lo−transition] coarse, [s_lo−transition,s_lo] an
81/// exponential grade coarse→fine, then [s_lo,length] fine (≈coarse/factor) all the
82/// way to the outer boundary. Exact faces at 0, s_lo−transition, s_lo, length.
83///
84/// Both grades keep the adjacent cell-width ratio ≤ MESH_MAX_RATIO.
85std::vector<double> build_static_mesh_faces(double length_m,
86 arma::uword ns_coarse,
87 const RefineParams& rp);
88
89/// Adjacent cell-width ratio cap enforced across the transition segment.
90constexpr double MESH_MAX_RATIO = 1.1;
