106double gamma_density_from_pressure(const Grid& grid, double pressure, double temperature) {
107 if (!(pressure > 0.0) || !(temperature > 0.0))
108 throw std::domain_error("gamma ghost/HSE pressure and temperature must be positive");
109 const double lo = grid.eos_gamma_table.min_n_h();
110 const double hi = grid.eos_gamma_table.max_n_h();
111 auto pressure_at = [&](double n_h) {
112 const double x = saha_ionization_fraction_n_h(n_h, temperature);
113 return (1.0+x)*n_h*eos_constants::k_b*temperature;
114 };
115 // EOS density-domain guard, unchanged: p(n_H) is monotone in n_H at fixed T,
116 // so bracketing the request between the two axis endpoints is exactly the
117 // in-table test the former bisection performed before iterating.
118 if (pressure < pressure_at(lo) || pressure > pressure_at(hi))
119 throw std::out_of_range("gamma ghost/HSE pressure implies density outside EOS table");
120 // The 100 geometric bisections this replaced (200 Saha evaluations per call)
121 // solved the same closure the algebraic inverse below solves in closed form.
122 return equilibrium_density_from_pressure(pressure, temperature);
123}

125// One rung of an EOS-consistent hydrostatic ladder taken DOWNWARD by `ds`
126// metres from an anchor (p_anchor, rho_anchor) at fixed temperature T_gh:
127// p = p_anchor + Δs·½(ρ_anchor + ρ)·g, ρ = ρ_EOS(p, T_gh),
128// iterated to a fixed point. Trapezoidal in ρ, so the ladder reproduces the
129// exponential stratification to O(Δs²) instead of the O(Δs) of p + ρ_anchor gΔs.
130void gamma_hse_rung_down(const Grid& grid, double p_anchor, double rho_anchor,
131 double T_gh, double ds, double& p_out, double& rho_out) {
132 double rho_gh = rho_anchor;
133 for (int it = 0; it < 8; ++it) {
134 const double p = p_anchor + ds*0.5*(rho_anchor + rho_gh)*grid.g;
135 const double rho_next = gamma_density_from_pressure(grid, p, T_gh);
136 const bool done = std::abs(rho_next - rho_gh) <= 1.0e-14*rho_next;
137 p_out = p;
138 rho_gh = rho_next;
139 if (done) break;
140 }
141 rho_out = rho_gh;
142}

144// Pack a LEGACY two-fluid ghost cell from explicit ion/neutral densities,
145// temperature, velocity and the gravitational potential (p_i = 2 n_i k T incl.
146// electrons, p_n = n_n k T, ε_e = 3/2 p_e ⇒ T_e = T_i). Taking (rho_i, rho_n)
147// explicitly lets each ghost carry the LOCAL ionization fraction (ionized corona
148// ↔ neutral chromosphere). The release solver packs its ghosts through
149// mixture_pack_ghost instead, which needs only the total density.
150void pack_ghost(Vec& ob, const Grid& grid, float rho_i, float rho_n, float T, float V, float phi_g) {
151 const float n_i = rho_i / grid.m_i;
152 const float n_n = rho_n / grid.m_n;
153 const float p_i = 2.0f * n_i * grid.k_b * T; // protons + electrons
154 const float p_n = n_n * grid.k_b * T;
155 const float p_e = n_i * grid.k_b * T; // ε_e = 3/2 p_e ⇒ T_e = T
156 ob(cons::RHO_I) = rho_i;
157 ob(cons::RHO_N) = rho_n;
158 ob(cons::MOM_I) = rho_i * V;
159 ob(cons::MOM_N) = rho_n * V;
160 ob(cons::E_I) = grid.inv_gm1() * p_i + 0.5f * rho_i * V * V + rho_i * phi_g;
161 ob(cons::E_N) = grid.inv_gm1() * p_n + 0.5f * rho_n * V * V + rho_n * phi_g;
162 ob(cons::E_E) = grid.inv_gm1() * p_e;
163}

174// Cell-average of the prescribed C7/PCHIP temperature for the finite-volume
175// Gamma/Saha IC. Two-point Gauss-Legendre quadrature is enough for the smooth
176// PCHIP segments and removes the O(ds) top-cell centre-sampling shift under
177// refinement without introducing a general quadrature framework.
178double c7_temperature_cell_average(double h_lo_km, double h_hi_km) {
179 constexpr double inv_sqrt3 = 0.57735026918962576451;
180 const double mid = 0.5*(h_lo_km+h_hi_km);
181 const double half = 0.5*(h_hi_km-h_lo_km);
182 const double h0 = mid-half*inv_sqrt3;
183 const double h1 = mid+half*inv_sqrt3;
184 return 0.5*(c7_full_temperature_pchip(h0)+c7_full_temperature_pchip(h1));
185}

319 // --- finite-volume profile (T, n_i, n_n) -------------------------------
320 // Real C7 T(h) and ionization FRACTION with the total density re-integrated
321 // hydrostatically from the base (p(h)=p_base·exp(−∫ g/(R_s T) dh), R_s local),
322 // so V≈0 is a clean discrete fixed point and any flow is heat-flux driven.
323 // The production Gamma/Saha path stores a cell-average C7/PCHIP temperature
324 // rather than a centre point sample, so refinement represents the same
325 // continuum upper thermal profile instead of shifting the top-cell datum.
326 Vec T_c(grid.ns), ni_c(grid.ns), nn_c(grid.ns);
327 double ln_p = 0.0, prev_h = h_base, prev_invH = 0.0;
328 {
329 const double nbase = static_cast<double>(ni_b)+nn_b;
330 const double xb = gamma_mode ? saha_ionization_fraction_n_h(nbase, T_base)
331 : ni_b/nbase;
332 ln_p = std::log((1.0+xb)*nbase*grid.k_b*T_base);
333 prev_invH = grid.g*grid.m_i/((1.0+xb)*grid.k_b*T_base);
334 }
335 for (arma::uword i = 0; i < grid.ns; ++i) {
336 const double h_c = 0.5 * (h_F(i) + h_F(i + 1)); // km
337 float T, n_i, n_n;
338 c7_cell(static_cast<float>(h_c), T, n_i, n_n);
339 if (gamma_mode) {
340 T = static_cast<float>(c7_temperature_cell_average(h_F(i), h_F(i+1)));
341 }
342 const double dh_m = (h_c-prev_h)*1000.0;
343 double x = n_i/(n_i+n_n);
344 if (gamma_mode) {
345 const double previous_ln_p = ln_p;
346 double trial = ln_p-prev_invH*dh_m;
347 bool converged = false;
348 for (int it = 0; it < 60; ++it) {
349 const double rho = gamma_density_from_pressure(grid, std::exp(trial), T);
350 x = saha_ionization_fraction_n_h(rho/eos_constants::m_h, T);
351 const double invH = grid.g*grid.m_i/((1.0+x)*grid.k_b*T);
352 const double next = previous_ln_p-0.5*(prev_invH+invH)*dh_m;
353 if (std::abs(next-trial) < 2.0e-13) {
354 trial = next;
355 converged = true;
356 break;
357 }
358 trial = next;
359 }
360 if (!converged) {
361 throw std::runtime_error(
362 "gamma Saha-HSE iteration did not converge at cell "
363 + std::to_string(i)+", h_km="+std::to_string(h_c)
364 +", T="+std::to_string(T)+", p="+std::to_string(std::exp(trial)));
365 }
366 ln_p = trial;
367 } else {
368 const double invH = grid.g*grid.m_i/((1.0+x)*grid.k_b*T);
369 ln_p -= 0.5*(prev_invH+invH)*dh_m;
370 }
371 const double n_tot = std::exp(ln_p)/((1.0+x)*grid.k_b*T);
372 if (gamma_mode) x = saha_ionization_fraction_n_h(n_tot, T);
373 n_i = x * n_tot;
374 n_n = (1.0-x)*n_tot;
375 prev_h = h_c;
376 prev_invH = grid.g*grid.m_i/((1.0+x)*grid.k_b*T);
377 T_c(i) = T;
378 ni_c(i) = n_i;
379 nn_c(i) = n_n;
380 }

410 // IC top value: the unresolved TR/corona above sits at ~constant T, so
411 // T_ghost = a·T_ref is a fixed hot reservoir and the downward conductive flux
412 // q ∝ (T_ghost − T_top) SELF-LIMITS as the chromosphere heats.

420 // --- one-time IC coronal superheat (ISO_TBOOST > 1) -------------------
421 // HISTORICAL two-fluid option (rejected in release mode).
422 // Heat-flux-driven evaporation with NO heating ramp: raise the IC coronal
423 // temperature above its own conduction+radiation balance so its own Spitzer
424 // flux conducts down the TR and drives evaporation from t=0 (Antiochos &
425 // Sturrock 1978). Switched on smoothly ABOVE the TR (tanh in T) so the
426 // chromosphere/TR base is untouched; density held fixed (heat raises p).
427 const float T_boost = env_f("ISO_TBOOST", 1.0f);
428 if (T_boost > 1.0f) {
429 const float Tc = env_f("ISO_TBOOST_TC", 3.0e5f); // selector center [K]
430 const float Tw = env_f("ISO_TBOOST_W", 1.0e5f); // selector width [K]
431 for (arma::uword i = 0; i < grid.ns; ++i) {
432 const float rho_i = ni_c(i) * grid.m_i;
433 const float rho_n = nn_c(i) * grid.m_n;
434 const float phi_g = 0.5f * (grid.phi_g_imh(i) + grid.phi_g_iph(i));
435 const float n_i = rho_i / grid.m_i;
436 const float n_n = rho_n / grid.m_n;
437 const float T = T_c(i);
438 const float w = 0.5f * (1.0f + std::tanh((T - Tc) / Tw)); // ~0 chromo → ~1 corona
439 const float Tb = (1.0f + (T_boost - 1.0f) * w) * T;
440 if (gamma_mode) {
441 mixture_pack_cell(grid, xn, i, static_cast<double>(rho_i)+rho_n,
442 0.0, Tb, phi_g);
443 continue;
444 }
445 const float p_i_b = 2.0f * n_i * grid.k_b * Tb; // protons + electrons
446 const float p_n_b = n_n * grid.k_b * Tb;
447 xn(arma::sub2ind(sz, i, cons::E_I)) = grid.inv_gm1() * p_i_b + rho_i * phi_g;
448 xn(arma::sub2ind(sz, i, cons::E_N)) = grid.inv_gm1() * p_n_b + rho_n * phi_g;
449 xn(arma::sub2ind(sz, i, cons::E_E)) = grid.inv_gm1() * n_i * grid.k_b * Tb; // T_e = Tb
450 }

453 // --- lower truncation reservoir (h_base = 1600 km, NOT the photosphere) ---
454 // The ghosts sit one and two Δs below cell 0 and stand in for the unmodeled
455 // lower chromosphere, which on these timescales is a quasi-static stratified
456 // reservoir. The RELEASE ladder (gamma_mode) is trapezoidal and EOS-closed at
457 // the truncation temperature T₀; the legacy two-fluid ladder below it keeps
458 // the historical first-order form p_0 + kρ_0gΔs with ρ ∝ p.

460 const float T0 = T_c(0);
461 kInnerTRef = T0;
462 const float n_i0 = ni_c(0);
463 const float n_n0 = nn_c(0);
464 kInnerRhoI = n_i0 * grid.m_i;
465 kInnerRhoN = n_n0 * grid.m_n;
466 const float p_i0 = 2.0f * n_i0 * grid.k_b * T0;
467 const float p_n0 = n_n0 * grid.k_b * T0;
468 // Center-to-ghost distance = the BASE cell's own width (mirrored ghost). On a
469 // refined mesh cell 0 is a fine cell, so this is the fine Δs, NOT the coarse ds_m.
470 const float ds_base = grid.ds_i(0);
471 if (gamma_mode) {
472 const double rho0 = kInnerRhoI+kInnerRhoN;
473 const double p0 = (1.0+saha_ionization_fraction_n_h(
474 rho0/eos_constants::m_h, T0))*rho0/eos_constants::m_h*eos_constants::k_b*T0;
475 double pg0, rg0, pg1, rg1;
476 gamma_hse_rung_down(grid, p0, rho0, T0, ds_base, pg0, rg0);
477 gamma_hse_rung_down(grid, pg0, rg0, T0, ds_base, pg1, rg1);
478 kInnerPRes0 = pg0; kInnerRhoRes0 = rg0;
479 kInnerPRes1 = pg1; kInnerRhoRes1 = rg1;
480 kInnerPiGh = pg0; kInnerPnGh = 0.0f;
481 kInnerPiGh2 = pg1; kInnerPnGh2 = 0.0f;
482 kInnerRhoIGh = rg0;
483 kInnerRhoNGh = 0.0f;
484 kInnerRhoIGh2 = rg1;
485 kInnerRhoNGh2 = 0.0f;
486 } else {
487 kInnerPiGh = p_i0 + kInnerRhoI * grid.g * ds_base;
488 kInnerPnGh = p_n0 + kInnerRhoN * grid.g * ds_base;
489 kInnerPiGh2 = p_i0 + 2.0f * kInnerRhoI * grid.g * ds_base;
490 kInnerPnGh2 = p_n0 + 2.0f * kInnerRhoN * grid.g * ds_base;
491 // ρ ∝ p at fixed T₀ (isothermal-hydrostatic ghost densities).
492 kInnerRhoIGh = kInnerRhoI * (kInnerPiGh / p_i0);
493 kInnerRhoIGh2 = kInnerRhoI * (kInnerPiGh2 / p_i0);
494 kInnerRhoNGh = kInnerRhoN * (kInnerPnGh / p_n0);
495 kInnerRhoNGh2 = kInnerRhoN * (kInnerPnGh2 / p_n0);
496 }

624 // HISTORICAL two-fluid option (rejected in release mode).
625 // Alternative to the ghost-T jump: impose the downward coronal conductive flux
626 // as a Stage-D Neumann outer BC (RTV 1978 / model_c7 closure), optionally ramped

635 / (0.5f * (grid.ds_i(nl) + grid.ds_i(nl - 1))); // K/m

646 // --- ambient volumetric coronal heating H(s) (ISO_CHEAT=1) ------------
647 // HISTORICAL two-fluid option (rejected in release mode).
648 // The RTV (1978) static-balance term the resolved corona otherwise lacks:
649 // H(s) = E_H0 · exp(−(s−s0)/s_H) [W/m³] (Aschwanden & Schrijver 2002),
650 // deposited through the corona so it balances the radiative loss locally and
651 // pins V≈0; ramping it up (ISO_CHEAT_ENHANCE>1, slow) drives gentle evaporation.
652 // Calibration inverts the RTV/AS2002 static scaling for the E_H0 that relaxes a
653 // column of length L to apex temperature T_max. When on, the boundary q(T) is
654 // turned OFF (the corona is heated inside the domain; no double-heating).
655 if (env_f("ISO_CHEAT", 0.0f) != 0.0f) {
656 const float L = arma::sum(grid.ds_i); // column length [m]
657 const float Tmax = env_f("ISO_CHEAT_TMAX_MK", 1.5f) * 1.0e6f;
658 const float sHf = env_f("ISO_CHEAT_SH_FRAC", 0.4f);
659 const float s_H = sHf * L;
660 const float L_cm = L * 100.0f;
661 const float p0 = std::pow(Tmax / 1400.0f, 3.0f) / L_cm;
662 const float Eun = 9.8e4f * std::pow(p0, 7.0f / 6.0f) * std::pow(L_cm, -5.0f / 6.0f);
663 const float E0 = 0.1f * Eun * std::exp(0.5f * L / s_H); // erg/cm³/s → W/m³
664
665 grid.enable_coronal_heating = true;
666 grid.coronal_heat_E0 = env_f("ISO_CHEAT_E0", E0);
667 grid.coronal_heat_sH = s_H;
668 grid.coronal_heat_s0 = env_f("ISO_CHEAT_S0_MM", 0.0f) * 1.0e3f;
669 grid.coronal_heat_two_sided = false; // open column, base-anchored
670 grid.coronal_heat_enhance = env_f("ISO_CHEAT_ENHANCE", 1.0f);
671 grid.coronal_heat_t_on = env_f("ISO_CHEAT_T_ON", 0.0f);
672 grid.coronal_heat_ramp = env_f("ISO_CHEAT_RAMP", 30.0f);
673 // Corona heated internally ⇒ drop the boundary q(T) crutch.
674 grid.impose_outer_heat_flux = false;
675 grid.outer_heat_flux_base = 0.0f;
676 grid.outer_heat_flux = 0.0f;
677 kQflux = false;
678 }

680 // Optional relaxation phase: run Stage-1 adiabatic relaxation (conduction off,
681 // no jump, V=0 reservoir) for sim_time < kRelaxTime so the C7 IC settles to a
682 // near-hydrostatic V≈0 equilibrium, THEN start Stage 2 (switched per step in
683 // model_column_update_bc off grid.sim_time).
684 kRelaxTime = env_f("ISO_RELAX_TIME", 0.0f);
685 kHeatFluxTarget = heat_flux_on;
686 const bool relaxing0 = (kRelaxTime > 0.0f);
687 kStage2Started = !relaxing0; // recapture kTtopRef at the relax→Stage-2 edge
688 kOuterPRefSet = false; // fresh IC ⇒ recapture the outer back-pressure
689 kHeatFluxOn = relaxing0 ? false : heat_flux_on;

691 // Stage 1: conduction OFF (adiabatic Euler steady state). Stage 2: ON, driven
692 // by the top T jump (or the imposed q(T) flux / coronal heating).
693 grid.enable_conduction = relaxing0 ? false : heat_flux_on;
694 // The release conduction operator is structurally physical-only and never
695 // reads this field. The historical two-fluid Stage D still folds a
696 // mesh-scaled artificial diffusivity into its conduction operator; keep it
697 // there behind an explicit diagnostic multiplier for controlled legacy
698 // comparisons, but never enable it implicitly.
699 // ISO_NUMERICAL_DIFFUSIVITY_MULT=1 restores the historical C_num=2000 m/s
700 // (4x in the resolved-corona testbed); unset/0 is the two-fluid baseline.
701 if (gamma_mode) {
702 grid.numerical_diffusivity_per_length = 0.0f;
703 } else {
704 const float diff_mult = (kCorona ? 4.0f : 1.0f)
705 * env_f("ISO_NUMERICAL_DIFFUSIVITY_MULT", 0.0f);
706 grid.numerical_diffusivity_per_length = heat_flux_on
707 ? (diff_mult * 2.0e3f) : 0.0f;
708 }

710 model_column_update_bc(grid, xn);
711 grid.broadcast();

713 // The release is REFERENCE-FREE: it does not know a hydrostatic equilibrium
714 // in advance and does not subtract a frozen residual. The release formerly
715 // froze this (V=0, HSE) IC as grid.eq_state and subtracted R(U_eq) every
716 // step, which made that one state an exact discrete fixed point but bound
717 // the solver to a pre-known global equilibrium and kept applying a residual
718 // evaluated at t = 0 (and at the first step's Δt) long after the transition
719 // region had restructured. Completing the MUSCL-Hancock source treatment and
720 // making the lower ghost ladder second order dropped the raw discrete
721 // hydrostatic face mass-flux defect of the release mesh by ~560x, to within
722 // ~1.3x of the float32 round-off floor of the stored state, so the frozen
723 // reference no longer has anything material to remove.
724 //
725 // The non-gamma (two-fluid) leg of this scenario is a different solver that
726 // never received the predictor source term, so it keeps the frozen
727 // reference exactly as before. eq_wb is now a two-fluid-only mechanism.
728 grid.eq_wb = !gamma_mode;
729 if (grid.eq_wb) grid.eq_state = xn;
730 return xn;

733void model_column_update_bc(Grid& grid, const Vec& xn) {
734 const auto sz = arma::size(grid.ns, num_of_eq);
735 const float m_i = grid.m_i;
736 const float m_n = grid.m_n;
737 const float k_b = grid.k_b;
738 const float g = grid.g;
739
740 // Decode total density / pressure / temperature / velocity / ionization fraction
741 // x = ρ_i/ρ_tot of an interior cell (cons2prim convention). Single-fluid T uses
742 // the LOCAL ionization (p = (2 n_i + n_n) k T).
743 // Release state: decode the mixture cell straight out of (rho, rho u, E).
744 auto mixture_cell_state = [&](arma::uword i, float& rho_tot, float& p_tot,
745 float& T, float& V, float& x) {
746 const auto msz = arma::size(grid.ns, num_of_mixture_eq);
747 const MixtureThermo th = decode_equilibrium_mixture(
748 grid.eos_gamma_table,
749 xn(arma::sub2ind(msz, i, mix::RHO)),
750 xn(arma::sub2ind(msz, i, mix::MOM)),
751 xn(arma::sub2ind(msz, i, mix::ENERGY)),
752 mixture_cell_phi(grid, i),
753 std::numeric_limits<double>::quiet_NaN(), grid.eos_gamma_debug_clamp);
754 rho_tot = th.rho; p_tot = th.p; T = th.T;
755 V = static_cast<double>(xn(arma::sub2ind(msz, i, mix::MOM)))/th.rho;
756 x = th.x;
757 };
758
759 // Legacy two-fluid state (single-fluid T from the local ionization).
760 auto two_fluid_cell_state = [&](arma::uword i, float& rho_tot, float& p_tot,
761 float& T, float& V, float& x) {
762 const float rho_i = xn(arma::sub2ind(sz, i, cons::RHO_I));
763 const float rho_n = xn(arma::sub2ind(sz, i, cons::RHO_N));
764 const float momI = xn(arma::sub2ind(sz, i, cons::MOM_I));
765 const float momN = xn(arma::sub2ind(sz, i, cons::MOM_N));
766 const float E_i = xn(arma::sub2ind(sz, i, cons::E_I));
767 const float E_n = xn(arma::sub2ind(sz, i, cons::E_N));
768 const float phi_g = 0.5f * (grid.phi_g_imh(i) + grid.phi_g_iph(i));
769 const float Vi = momI / rho_i;
770 const float Un = momN / rho_n;
771 const float p_i = grid.gm1() * E_i - grid.half_gm1() * rho_i * Vi * Vi - grid.gm1() * rho_i * phi_g;
772 const float p_n = grid.gm1() * E_n - grid.half_gm1() * rho_n * Un * Un - grid.gm1() * rho_n * phi_g;
773 const float n_i = rho_i / m_i;
774 const float n_n = rho_n / m_n;
775 rho_tot = rho_i + rho_n;
776 p_tot = p_i + p_n;
777 T = p_tot / ((2.0f * n_i + n_n) * k_b); // single-fluid T, local ionization
778 V = Vi;
779 x = rho_i / rho_tot; // ionization fraction
780 };
781
782 const bool release = !grid.eos_gamma_table.empty();
783 auto cell_state = [&](arma::uword i, float& rho_tot, float& p_tot,
784 float& T, float& V, float& x) {
785 if (release) mixture_cell_state(i, rho_tot, p_tot, T, V, x);
786 else two_fluid_cell_state(i, rho_tot, p_tot, T, V, x);
787 };

789 // --- imposed q(T) flux ramp (ISO_QFLUX) --------------------------------
790 // q(T) = q_base · enhance(t): holds at 1 until t_on, then cosine-rises to
791 // outer_heat_flux_enhance over [t_on, t_on+ramp]. Applied to the Stage-D
792 // Neumann outer flux (impose_outer_heat_flux). No-op when kQflux is off.
793 if (kQflux && grid.outer_heat_flux_base > 0.0f) {
794 float amp = 1.0f;
795 if (grid.outer_heat_flux_enhance != 1.0f) {
796 const float t = grid.sim_time;
797 const float ton = grid.outer_heat_flux_t_on;
798 const float r = (grid.outer_heat_flux_ramp > 1.0e-6f) ? grid.outer_heat_flux_ramp : 1.0e-6f;
799 float w;
800 if (t <= ton) w = 0.0f;
801 else if (t >= ton + r) w = 1.0f;
802 else w = 0.5f * (1.0f - std::cos(static_cast<float>(arma::datum::pi) * (t - ton) / r));
803 amp = 1.0f + (grid.outer_heat_flux_enhance - 1.0f) * w;
804 }
805 grid.outer_heat_flux = grid.outer_heat_flux_base * amp;
806 }

808 // --- relaxation → Stage-2 phase switch (off grid.sim_time) -------------
809 // While relaxing: conduction OFF, no jump, V=0 reservoir top. At the transition
810 // recapture kTtopRef from the RELAXED top cell (anchor the FIXED jump to the
811 // equilibrium), then run Stage 2 (conduction ON + jump + Mach-capped outflow).
812 const bool relaxing = (kRelaxTime > 0.0f) && (grid.sim_time < kRelaxTime);
813 kHeatFluxOn = relaxing ? false : kHeatFluxTarget;
814 grid.enable_conduction = relaxing ? false : kHeatFluxTarget;
815 const float a_live = relaxing ? 1.0f : kAjump;
816 const float b_live = relaxing ? 1.0f : kBjump;
817 if (!relaxing && !kStage2Started) {
818 if (!kTtopFixedAbs) { // an ABSOLUTE imposed top T (ISO_T_TOP) stays put
819 float rho_t, p_t, T_t, V_t, x_t;
820 cell_state(grid.ns - 1, rho_t, p_t, T_t, V_t, x_t);
821 kTtopRef = T_t; // else anchor the jump to the relaxed top T
822 }
823 kOuterPRefSet = false; // re-anchor the back-pressure on the relaxed top
824 kStage2Started = true;
825 }

827 // ====================================================================
828 // Outer face — bottom of the transition region (new upper BC):
829 // 1. pressure — hydrostatic, ONE-SIDED from the top cell (see below).
830 // 2. temperature — imposed jump T_ghost1 = a·T_ref, T_ghost2 = b·T_ghost1.
831 // 3. density — from the EOS at (p_ghost, T_ghost).
832 // ====================================================================
833 {
834 const arma::uword nl = grid.ns - 1;
835 const float ds = grid.ds_i(nl);
836 const float phi_g_out = grid.phi_g_iph(nl); // ghost decode uses this gauge
837
838 float rho_top, p_top, T_top, V_top, x_top;
839 cell_state(nl, rho_top, p_top, T_top, V_top, x_top);
840
841 // (2) temperature. TWO distinct roles, kept explicitly separate:
842 // * T_cond_wall — the fixed hot wall that DRIVES Stage-D conduction, always
843 // the imposed jump a·T_ref off the reference top T. With ISO_CORONA the
844 // corona is resolved and a=b=1 (the ghost just extends it).
845 // * T_hydro_g0/g1 — what the HYDRO ghost carries (its EOS density and
846 // internal energy, hence the outer-face reconstruction and Riemann state).
847 // Baseline (default): the hydro ghost IS the wall, T_hydro = a·T_ref, b·a·T_ref
848 // — one ghost state serving both roles, so the outer face fixes pressure AND
849 // temperature. ISO_HYDRO_T_DECOUPLE=1: the hydro ghost instead zero-gradient-
850 // extrapolates the live top cell (T_hydro_g0 = T_hydro_g1 = T_top) and only the
851 // conduction rows keep the wall, via grid.outer_conduction_temperature.
852 const float T_cond_wall = a_live * kTtopRef;
853 const float T_g0 = kHydroTDecouple ? T_top : T_cond_wall;
854 const float T_g1 = kHydroTDecouple ? T_g0 : b_live * T_g0;
855 grid.outer_conduction_temperature_override = kHydroTDecouple;
856 grid.outer_conduction_temperature = T_cond_wall;
857
858 // (1)+(3) Hydrostatic ghost ladder: each rung one-sided on the rung below,
859 // p_{k+1} = p_k − Δs·½(ρ_k + ρ_{k+1})·g,
860 // with ρ closed by the EOS at the imposed T. The ladder is ANCHORED on the
861 // fixed reservoir back-pressure kOuterPRef, NOT on a live interior cell.
862 //
863 // Both live-cell anchors are wrong, in opposite directions:
864 // * p[ns-2] (the previous CENTERED form, p_g0 = p[ns-2] − 2Δs·ρ_top·g)
865 // extrapolates ACROSS the very cell it bounds. That is well balanced only
866 // while the interior satisfies the same discrete HSE; inside the transition
867 // region it does not — the measured residual (dp/ds+ρg)/(ρg) runs +0.03 at
868 // 2140 km, +0.21 at 2145 km and −5.0 in the top cell — so the ghost landed
869 // ABOVE p_top and INVERTED the pressure gradient across the outer face
870 // (+4.1e-5 Pa where it should be −4.3e-6): a standing ~9 g downward push
871 // that reversed the top-cell velocity and seeded a ~25 km boundary layer.
872 // * p_top (a one-sided anchor) makes pressure zero-gradient. Combined with the
873 // already-extrapolated V that leaves the outer face transmissive in EVERY
874 // variable but T, so nothing sets the back-pressure and the column drains
875 // out the top (measured: V → 1.9 km/s, p falling monotonically, crash at
876 // t ≈ 12 s). Subsonic outflow has one incoming characteristic and therefore
877 // needs exactly one externally imposed condition.
878 // A fixed reference back-pressure supplies that one condition, cannot invert
879 // against the interior, and is exactly hydrostatic at the V=0 IC by
880 // construction. Physically: a corona of fixed pressure and T sits above 2153 km.
881 auto ghost_density = [&](float p_gh, float T_gh) {
882 return release ? gamma_density_from_pressure(grid, p_gh, T_gh)
883 : p_gh*m_i/((1.0f+x_top)*k_b*T_gh);
884 };
885 // ρ_ghost depends on p_ghost, so take one fixed-point pass seeded with the
886 // rung below (the correction is O(Δs·Δρ/ρ) and converges in a single sweep).
887 auto hse_rung = [&](float p_below, float rho_below, float T_gh,
888 float& p_out, float& rho_out) {
889 float rho_gh = rho_below;
890 for (int it = 0; it < 2; ++it) {
891 p_out = p_below - ds * 0.5f * (rho_below + rho_gh) * g;
892 if (p_out < 1.0e-12f) p_out = 1.0e-12f;
893 rho_gh = ghost_density(p_out, T_gh);
894 }
895 rho_out = rho_gh;
896 };
897 // Capture the reservoir back-pressure once, one-sided from the (hydrostatic,
898 // V=0) reference top cell. Recaptured at the relax→Stage-2 edge alongside
899 // kTtopRef so a Stage-1 relaxation re-anchors on the settled equilibrium.
900 // The capture uses the HYDRO ghost temperature T_g0 (not the conduction wall),
901 // so the ladder is exactly the hydrostatic one this BC will later reproduce:
902 // at the reference state T_g0 = T_top there, and BC(q_ref) = q_g,ref holds in
903 // decoupled mode exactly as it does in baseline mode.
904 if (!kOuterPRefSet) {
905 float p_ref, rho_ref;
906 hse_rung(p_top, rho_top, T_g0, p_ref, rho_ref);
907 kOuterPRef = p_ref;
908 kOuterPRefSet = true;
909 // The reservoir back-pressure is the one externally imposed condition at
910 // the outer face; log it (once per capture) so post-run diagnostics can
911 // form the top-cell pressure deficit p_top − p_g0 without re-deriving it.
912 std::cerr << "[model_column] outer BC anchor: kOuterPRef=" << kOuterPRef
913 << " Pa p_top=" << p_top << " Pa T_top=" << T_top
914 << " K T_hydro_g0=" << T_g0 << " K T_cond_wall="
915 << T_cond_wall << " K hydro_T_decouple="
916 << (kHydroTDecouple ? 1 : 0) << std::endl;
917 }
918 float p_g0 = kOuterPRef;
919 if (p_g0 < 1.0e-12f) p_g0 = 1.0e-12f;
920 const float rho_g0 = ghost_density(p_g0, T_g0);
921 float p_g1, rho_g1;
922 hse_rung(p_g0, rho_g0, T_g1, p_g1, rho_g1);
923
924 // Velocity. Stage 1 (relaxation): static reservoir V=0. Stage 2: Mach-capped
925 // OUTFLOW so the evaporation upflow can leave, while |V| ≤ kVcapMach·c_s
926 // prevents the ill-posed-inflow runaway.
927 float V_g = 0.0f;
928 if (kHeatFluxOn) {
929 const float c_s = release
930 ? std::sqrt(gamma_state(grid.eos_gamma_table, rho_top, T_top,
931 grid.eos_gamma_debug_clamp).gamma_sound*p_top/rho_top)
932 : std::sqrt(2.0f*grid.gamma_mono*k_b*T_top/m_i);
933 const float vcap = kVcapMach * c_s;
934 V_g = V_top;
935 if (V_g > vcap) V_g = vcap;
936 if (V_g < -vcap) V_g = -vcap;
937 }
938 if (release) {
939 mixture_pack_ghost(grid, grid.mix_outer_boundary0, rho_g0, V_g, T_g0,
940 phi_g_out);
941 mixture_pack_ghost(grid, grid.mix_outer_boundary1, rho_g1, V_g, T_g1,
942 phi_g_out);
943 } else {
944 pack_ghost(grid.outer_boundary0_i, grid, x_top * rho_g0,
945 (1.0f - x_top) * rho_g0, T_g0, V_g, phi_g_out);
946 pack_ghost(grid.outer_boundary1_i, grid, x_top * rho_g1,
947 (1.0f - x_top) * rho_g1, T_g1, V_g, phi_g_out);
948 }
949 }
950
951 // ====================================================================
952 // Inner face — the LOWER TRUNCATION of the modeled domain at h_base
953 // (1600 km). Not the photosphere: it is the height below which this
954 // solver's Saha/LTE equilibrium closure is not intended to be trusted, so
955 // the atmosphere underneath is deliberately outside the model and the two
956 // ghosts are a numerical closure for the MUSCL stencil.
957 //
958 // The physical content is one statement: the lower chromosphere below the
959 // cut is a quasi-static stratified reservoir at temperature T₀. That fixes
960 // the ghost thermal state, the hydrostatic ghost pressures kInnerPRes0/1
961 // (trapezoidal, EOS-closed — see gamma_hse_rung_down) and V = 0.
962 //
963 // A gravity-aware CHARACTERISTIC alternative was prototyped and measured:
964 // impose only the incoming (upward, u+c) acoustic invariant δu + δp/Z = 0
965 // about the base reference state with impedance Z = ρc, and take the
966 // outgoing (u−c) one from the live base cell, giving ghost deviations
967 // V_g = −(p_0 − p_0,ref)/Z and p_g = p_g,ref − Z·V_0. It degenerates
968 // exactly to this wall at equilibrium. Over 20 s conduction-off, 100 s and
969 // 1000 s conduction-driven runs it changed the hydrostatic residual not at
970 // all, the evaporation window velocity by ≤0.4 %, and the detrended
971 // base-region oscillation content not at all: the disturbance that reaches
972 // 1600 km is a slow quasi-static pressure adjustment, not acoustic ringing,
973 // so this face is not an important reflector for this model. It was
974 // therefore NOT promoted — see main.tex for the measurement.
975 //
976 // The lower-BC TEMPERATURE choice (Dirichlet vs Neumann/zero-flux) is a
977 // separate CONDUCTION boundary condition imposed inside the Stage-D solver
978 // (grid.inner_conduction_neumann); the hydro closure is independent of it.
979 // ====================================================================
980 {
981 const float phi_g_in = grid.phi_g_imh(0);
982 if (release) {
983 mixture_pack_ghost(grid, grid.mix_inner_boundary0, kInnerRhoRes0, 0.0,
984 kInnerTRef, phi_g_in);
985 mixture_pack_ghost(grid, grid.mix_inner_boundary1, kInnerRhoRes1, 0.0,
986 kInnerTRef, phi_g_in);
987 grid.broadcast();
988 return;
989 }
990 Vec& ob0 = grid.inner_boundary0_i;
991 ob0(cons::RHO_I) = kInnerRhoIGh;
992 ob0(cons::RHO_N) = kInnerRhoNGh;
993 ob0(cons::MOM_I) = 0.0f;
994 ob0(cons::MOM_N) = 0.0f;
995 ob0(cons::E_I) = grid.inv_gm1() * kInnerPiGh + kInnerRhoIGh * phi_g_in;
996 ob0(cons::E_N) = grid.inv_gm1() * kInnerPnGh + kInnerRhoNGh * phi_g_in;
997 ob0(cons::E_E) = 0.5f * grid.inv_gm1() * kInnerPiGh; // p_e = ½ p_i ⇒ T_e = T_i
998
999 Vec& ob1 = grid.inner_boundary1_i;
1000 ob1(cons::RHO_I) = kInnerRhoIGh2;
1001 ob1(cons::RHO_N) = kInnerRhoNGh2;
1002 ob1(cons::MOM_I) = 0.0f;
1003 ob1(cons::MOM_N) = 0.0f;
1004 ob1(cons::E_I) = grid.inv_gm1() * kInnerPiGh2 + kInnerRhoIGh2 * phi_g_in;
1005 ob1(cons::E_N) = grid.inv_gm1() * kInnerPnGh2 + kInnerRhoNGh2 * phi_g_in;
1006 ob1(cons::E_E) = 0.5f * grid.inv_gm1() * kInnerPiGh2;
1007 }
1008
1009 grid.broadcast();
1010}
