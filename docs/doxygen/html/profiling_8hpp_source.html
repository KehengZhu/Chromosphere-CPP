23/// Wall-clock timer buckets. One ProfileScope in each stage attributes the
24/// seconds it was alive to exactly one of these; the report prints them as
25/// `time.<name>_s`. The buckets are stage labels, not a partition of the step:
26/// they neither overlap nor necessarily cover the whole timestep.
27enum class ProfileRegion : unsigned {
28 Cfl, ///< Timestep sizing (mixture_timestep / the legacy cal_dt_i).
29 Decode, ///< Conserved-to-thermodynamic decode (mixture_decode_into).
30 Rhs, ///< Explicit MUSCL-Hancock/Roe hydro RHS (mixture_rhs_explicit).
31 Conduction, ///< Implicit backward-Euler conduction (mixture_apply_conduction).
32 Boundary, ///< Scenario boundary-condition update at the top of each step.
33 Output, ///< Formatting and writing one ASCII snapshot frame.
34 Count ///< Number of regions; array bound, never a timed region.
35};
36
37/// Which physical constraint set the timestep actually taken. Recorded once per
38/// timestep computation via profile_note_timestep_limiter() and reported as
39/// `limiter.<name>` counts, so a run shows what was limiting it and for how many
40/// steps. The release solver always reports Acoustic — with no volumetric source
41/// and an unconditionally stable implicit conduction stage, the acoustic CFL
42/// condition is its only constraint. The remaining values come from the legacy
43/// two-fluid timestep, which lowers dt when a source term's heating time is
44/// shorter than the acoustic step.
45enum class TimestepLimiter : unsigned {
46 Acoustic, ///< Acoustic CFL, CFL*ds/(|u|+c_s). The only release limiter.
47 BeamHeating, ///< Flare beam energy-gain cap (two-fluid, enable_beam_heating).
48 CoronalHeating, ///< Coronal heating energy-gain cap (two-fluid, enable_coronal_heating).
49 Cooling, ///< Radiative-cooling category; no current rule selects it.
50 ElectronEnergy, ///< Electron-energy category; no current rule selects it.
51 Other, ///< Catch-all category; no current rule selects it.
52 Count ///< Number of limiters; array bound, never a recorded value.
53};

55/// Counters for the (rho,p) -> T face inversion of the pressure-based MUSCL
56/// reconstruction ONLY. Deliberately separate from EosInversionProfile, which
57/// counts the (rho,E) -> T energy inversion of the decode/projection stages.
58struct PressureInversionProfile {
59 /// Safeguarded Newton/bisection solves that returned a temperature. One is
60 /// recorded per successful inversion; a solve that exhausts its 128-iteration
61 /// budget throws instead and is never counted.
62 std::uint64_t calls = 0;
63 std::uint64_t hinted_calls = 0; ///< a caller-supplied guess was usable
64 std::uint64_t evaluations = 0; ///< Saha/caloric evaluations consumed
65 /// Largest number of Saha/caloric evaluations any single solve needed — a
66 /// maximum over calls (and over threads), not a sum, so it exposes the worst
67 /// face rather than the typical one.
68 std::uint64_t maximum_evaluations = 0;
69 /// Newton steps that were discarded because the candidate temperature fell
70 /// outside the current bracket and had to be replaced by the bracket
71 /// midpoint. Summed over steps, so it can exceed #calls.
72 std::uint64_t bisection_fallbacks = 0;
73};
74
75/// Counters for the (rho,E) -> T ENERGY inversion, i.e. the safeguarded
76/// Newton/bisection solve of e_int(rho,T) = e_int that the decode and projection
77/// stages run per cell. Deliberately separate from PressureInversionProfile,
78/// which counts the (rho,p) -> T face inversion of the MUSCL reconstruction.
79struct EosInversionProfile {
80 /// Energy inversions attempted: one per call that passes argument validation,
81 /// counted before any fast path, so it is the denominator for the averages in
82 /// the report.
83 std::uint64_t calls = 0;
84 /// Calls whose supplied temperature guess already satisfied the energy
85 /// tolerance, so the root was returned with zero Newton updates.
86 std::uint64_t initial_guess_accepts = 0;
87 /// Calls that converged after exactly one Newton update — the measured common
88 /// case on the production column, and what the fused fast path targets.
89 std::uint64_t one_update_convergences = 0;
90 /// Sum over calls of the Newton updates each one reported; the report divides
91 /// it by #calls to print `inversion.average_iterations`.
92 std::uint64_t total_iterations = 0;
93 /// Largest per-call Newton-update count — a maximum over calls (and over
94 /// threads at merge time), not a sum.
95 std::uint64_t maximum_iterations = 0;
96 /// Newton steps rejected by a safeguard (candidate outside the bracket, or
97 /// progress slower than halving the previous bracket) and replaced by the
98 /// bracket midpoint. Summed over steps, so it can exceed #calls.
99 std::uint64_t bisection_fallbacks = 0;
100 /// Caloric evaluations spent establishing the table-wide bracket: two per
101 /// call (the minimum and maximum table temperatures), added only when both
102 /// fast paths miss. An accepted fast path deliberately contributes zero, so
103 /// this counter measures how often the full solver was actually entered.
104 std::uint64_t bracket_evaluations = 0;
105};

156/// RAII wall-clock timer for one ProfileRegion: it samples a steady_clock stamp
157/// at construction and, at destruction, adds the elapsed seconds to that region's
158/// bucket for the calling thread. Whether profiling is on is latched at
159/// construction, so toggling it mid-scope cannot produce a one-sided measurement.
160/// Non-copyable, and a no-op beyond one branch when profiling is off.
161class ProfileScope {
162public:
163 /// Start timing @p region (no clock read at all when profiling is off).
164 explicit ProfileScope(ProfileRegion region) noexcept;
165 /// Stop timing and charge the elapsed seconds to the region.
166 ~ProfileScope();
167 ProfileScope(const ProfileScope&) = delete;
168 ProfileScope& operator=(const ProfileScope&) = delete;
169private:
170 ProfileRegion region_; ///< bucket the elapsed time is charged to
171 bool enabled_; ///< profiling state latched at construction
172 std::chrono::steady_clock::time_point start_; ///< set only when enabled_
173};
