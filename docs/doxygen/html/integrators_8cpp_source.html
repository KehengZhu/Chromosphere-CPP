3 * @brief Two-fluid time integrators and the operator-split source stages.
4 * @ingroup two_fluid_solver

45Vec cal_dt_i(const Grid& grid, const Vec& xn_state) {
46 if (!grid.eos_gamma_table.empty())
47 throw std::logic_error(
48 "the legacy two-fluid timestep cannot size a Gamma1-table Grid; "
49 "use mixture_timestep");
50 ProfileScope timer(ProfileRegion::Cfl);
51 Vec dt_i = grid.CFL * grid.ds_i / cal_max_v_i(grid, xn_state);
52 float dt_min_i = arma::min(dt_i);
53 TimestepLimiter active_limiter = TimestepLimiter::Acoustic;
54
55 // Beam-heating timescale limit. In the low-density chromosphere (n ~ 10^16
56 // m^-3) the heat capacity is tiny, so the flare beam's heating time
57 // τ_heat = ε / Q can be SHORTER than the hydro CFL step. Operator splitting
58 // then over-heats a cell in one step (ΔT ~ 10^5 K) and breaks. Cap dt so the
59 // beam raises the total thermal energy by at most BEAM_HEAT_CFL per step.
60 if (grid.enable_beam_heating) {
61 const Vec prim = cons2prim(grid, xn_state);
62 const Vec rho_i = get_scalar(grid, prim, prim::RHO_I);
63 const Vec rho_n = get_scalar(grid, prim, prim::RHO_N);
64 const Vec p_i = get_scalar(grid, prim, prim::P_I);
65 const Vec p_n = get_scalar(grid, prim, prim::P_N);
66 const Vec p_e = get_scalar(grid, prim, prim::P_E);
67 const Vec n_i = rho_i / grid.m_i;
68 const Vec n_n = rho_n / grid.m_n;
69 const Vec Q = beam_heating_rate(grid, n_i, n_n); // W/m^3
70 const float Qmax = Q.max();
71 if (Qmax > 0.0f) {
72 const float BEAM_HEAT_CFL = 0.1f;
73 // Three-T: the beam deposits into electrons alone, so the heating
74 // timescale that matters is ε_e/Q (a much smaller pool). Single-T:
75 // the deposited heat is shared, so use the total thermal energy.
76 const Vec eps = grid.enable_Te ? Vec(grid.inv_gm1() * p_e)
77 : Vec(grid.inv_gm1() * (p_i + p_n));
78 // dt ≤ BEAM_HEAT_CFL · min(ε/Q) over cells receiving the beam.
79 Vec tau = eps / arma::clamp(Q, 1.0e-30f, arma::datum::inf);
80 // Only constrain where the beam actually deposits (Q>0).
81 for (arma::uword i = 0; i < grid.ns; ++i)
82 if (Q(i) <= 0.0f) tau(i) = arma::datum::inf;
83 const float dt_beam = BEAM_HEAT_CFL * arma::min(tau);
84 if (dt_beam < dt_min_i) {
85 dt_min_i = dt_beam;
86 active_limiter = TimestepLimiter::BeamHeating;
87 }
88 }
89 }
90
91 // Coronal-heating timescale limit (same rationale as the beam guard). The
92 // steady footpoint heating is gentle (τ_heat = ε/H ~ tens of s in the tenuous
93 // corona, far longer in the dense chromosphere), so this rarely binds — but a
94 // ramped/localized enhancement (Phase 3) can shorten it, so cap dt to a small
95 // fractional energy gain per step where H > 0, exactly as for the beam.
96 if (grid.enable_coronal_heating) {
97 const Vec prim = cons2prim(grid, xn_state);
98 const Vec p_i = get_scalar(grid, prim, prim::P_I);
99 const Vec p_n = get_scalar(grid, prim, prim::P_N);
100 const Vec p_e = get_scalar(grid, prim, prim::P_E);
101 const Vec H = coronal_heating_rate(grid); // W/m^3
102 const float Hmax = H.max();
103 if (Hmax > 0.0f) {
104 const float HEAT_CFL = 0.1f;
105 const Vec eps = grid.enable_Te ? Vec(grid.inv_gm1() * p_e)
106 : Vec(grid.inv_gm1() * (p_i + p_n));
107 Vec tau = eps / arma::clamp(H, 1.0e-30f, arma::datum::inf);
108 for (arma::uword i = 0; i < grid.ns; ++i)
109 if (H(i) <= 0.0f) tau(i) = arma::datum::inf;
110 const float dt_heat = HEAT_CFL * arma::min(tau);
111 if (dt_heat < dt_min_i) {
112 dt_min_i = dt_heat;
113 active_limiter = TimestepLimiter::CoronalHeating;
114 }
115 }
116 }
117 profile_note_timestep_limiter(active_limiter);
118 return dt_min_i * arma::ones<Vec>(grid.ns);
119}

137// ----------------------------------------------------------------------------
138// TRAC adaptive cutoff temperature (Johnston et al. 2020 Eq. 8). T_c is the
139// highest charged-fluid temperature among cells where the TR is under-resolved
140// (L_R/L_T > δ = 1/2, with L_T = T/|dT/ds| the temperature length scale and
141// L_R = Δs the grid spacing), clamped to [trac_T_chrom, 0.2 T_peak]. When no
142// cell is under-resolved it defaults to the floor trac_T_chrom.
143// ----------------------------------------------------------------------------
144float compute_trac_cutoff_T(const Grid& grid, const Vec& prim_state) {
145 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
146 const Vec p_i = get_scalar(grid, prim_state, prim::P_I);
147 const Vec p_e = get_scalar(grid, prim_state, prim::P_E);
148 const Vec n_i = rho_i / grid.m_i;
149 // TRAC broadens the (electron) heat-conduction operator, so it keys off the
150 // temperature that drives it: T_e in the three-T model, T_charged otherwise.
151 const Vec T = grid.enable_Te ? Vec(p_e / (n_i * grid.k_b))
152 : Vec(p_i / (2.0f * n_i * grid.k_b));
153
154 const float Tpeak = arma::max(T);
155 const float Tc_upper = std::max(grid.trac_Tc_max_frac * Tpeak, grid.trac_T_chrom);
156 float Tc = grid.trac_T_chrom;
157 const arma::uword ns = grid.ns;
158 for (arma::uword i = 0; i < ns; ++i) {
159 float dTds;
160 // Center-to-center separations (grid.ds_iph_i/ds_imh_i mirror the boundary
161 // widths, so the one-sided ends match the legacy Δs on a uniform mesh).
162 if (i == 0) dTds = (T(1) - T(0)) / grid.ds_iph_i(0);
163 else if (i == ns - 1) dTds = (T(i) - T(i - 1)) / grid.ds_imh_i(i);
164 else dTds = (T(i + 1) - T(i - 1)) / (grid.ds_iph_i(i) + grid.ds_imh_i(i));
165 // Under-resolved ⟺ L_R/L_T = Δs|dT/ds|/T > 1/2.
166 if (std::fabs(dTds) * grid.ds_i(i) > 0.5f * T(i)) {
167 if (T(i) > Tc) Tc = T(i);
168 }
169 }
170 Tc = std::min(Tc, Tc_upper);
171 Tc = std::max(Tc, grid.trac_T_chrom);
172
173 // Cutoff-temperature limiter (Johnston 2020 Appendix A.2): a sudden jump in
174 // T_c changes κ/Λ over many cells at once, producing a conduction/radiation
175 // shock. Limit the per-step change to a small ratio (both directions) so the
176 // broadening region grows/shrinks gradually; the integrated quantities are
177 // unaffected. grid.trac_cutoff_T holds the previous step's value.
178 const float prev = grid.trac_cutoff_T;
179 const float max_ratio = 1.03f;
180 if (prev >= grid.trac_T_chrom) {
181 Tc = std::min(Tc, prev * max_ratio);
182 Tc = std::max(Tc, prev / max_ratio);
183 Tc = std::min(Tc, Tc_upper);
184 Tc = std::max(Tc, grid.trac_T_chrom);
185 }
186 return Tc;
187}

189// ----------------------------------------------------------------------------
190// Stage B (writeup §3.7 Eqs. (drag-w)--(drag-partition)): point-implicit drag
191// + frictional heating. Exact for the linear drift ODE with α frozen at U*.
192// Mutates V, U, p_i, p_n in place; ρ_i, ρ_n unchanged.
193// ----------------------------------------------------------------------------
194static void apply_drag_stage(const Grid& grid, Vec& prim_state, float dt) {
195 Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
196 Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
197 Vec V = get_scalar(grid, prim_state, prim::V);
198 Vec U = get_scalar(grid, prim_state, prim::U);
199 Vec p_i = get_scalar(grid, prim_state, prim::P_I);
200 Vec p_n = get_scalar(grid, prim_state, prim::P_N);
201 const Vec p_e = get_scalar(grid, prim_state, prim::P_E); // carried untouched
202
203 const Vec n_i = rho_i / grid.m_i;
204 const Vec n_n = rho_n / grid.m_n;
205 const Vec T_i = p_i / (2.0 * n_i * grid.k_b);
206 const Vec T_n = p_n / (n_n * grid.k_b);
207
208 const Vec alpha = rho_i % nu_in(grid, n_n, T_i, T_n);
209 const Vec w_old = V - U;
210 const Vec rho_tot = rho_i + rho_n;
211 const Vec V_cm = (rho_i % V + rho_n % U) / rho_tot;
212 const Vec mu = rho_i % rho_n / rho_tot;
213 const Vec lambda_drag = alpha % rho_tot / (rho_i % rho_n);
214
215 // Single-fluid limit: rigidly lock the drift to zero (V→U→V_cm), the exact
216 // dt·λ_drag→∞ limit. The full relative kinetic energy 0.5·μ·w_old² is then
217 // thermalized below via dE_drag, conserving total momentum and energy.
218 const Vec w_new = grid.single_fluid
219 ? Vec(arma::zeros<Vec>(w_old.n_elem))
220 : Vec(w_old / (1.0 + dt * lambda_drag));
221 const Vec V_new = V_cm + (rho_n / rho_tot) % w_new;
222 const Vec U_new = V_cm - (rho_i / rho_tot) % w_new;
223
224 const Vec dE_drag = 0.5 * mu % (w_old % w_old - w_new % w_new);
225 const float f_i = grid.m_n / (grid.m_i + grid.m_n);
226 const float f_n = grid.m_i / (grid.m_i + grid.m_n);
227 const Vec p_i_new = p_i + grid.gm1() * f_i * dE_drag;
228 const Vec p_n_new = p_n + grid.gm1() * f_n * dE_drag;
229
230 // Frictional heating goes to the combined charged pool (P_I = total charged
231 // pressure); with the electron pressure P_E carried unchanged through this
232 // stage, the heat lands on the protons (p_proton = P_I − P_E) — ion–neutral
233 // friction does not heat electrons. P_E is preserved verbatim.
234 prim_state.zeros();
235 prim_state += scalar_to(grid, rho_i, prim::RHO_I);
236 prim_state += scalar_to(grid, rho_n, prim::RHO_N);
237 prim_state += scalar_to(grid, V_new, prim::V);
238 prim_state += scalar_to(grid, U_new, prim::U);
239 prim_state += scalar_to(grid, p_i_new, prim::P_I);
240 prim_state += scalar_to(grid, p_n_new, prim::P_N);
241 prim_state += scalar_to(grid, p_e, prim::P_E);
242}

244// ----------------------------------------------------------------------------
245// Stage C (writeup §3.7 Eqs. (temp-relax)--(temp-reconstruct)): point-implicit
246// temperature equilibration between ions/electrons and neutrals. Exact for
247// the linear drift ODE with α frozen at the post-drag state.
248// ----------------------------------------------------------------------------
249static void apply_temperature_stage(const Grid& grid, Vec& prim_state, float dt) {
250 Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
251 Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
252 Vec V = get_scalar(grid, prim_state, prim::V);
253 Vec U = get_scalar(grid, prim_state, prim::U);
254 Vec p_i = get_scalar(grid, prim_state, prim::P_I);
255 Vec p_n = get_scalar(grid, prim_state, prim::P_N);
256 Vec p_e = get_scalar(grid, prim_state, prim::P_E);
257
258 const Vec n_i = rho_i / grid.m_i;
259 const Vec n_n = rho_n / grid.m_n;
260
261 if (!grid.enable_Te) {
262 // ---- Single-temperature baseline (unchanged 2-way T_charged ↔ T_n) ----
263 const Vec T_i = p_i / (2.0 * n_i * grid.k_b); // T_charged
264 const Vec T_n = p_n / (n_n * grid.k_b);
265
266 const Vec alpha = rho_i % nu_in(grid, n_n, T_i, T_n);
267 // Heat capacities per volume C = nk/(γ−1); the charged row carries ions
268 // + electrons (n_e = n_i ⇒ 2 n_i), the neutral row n_n. (β below is a
269 // kinetic collisional coefficient — see Grid::gm1 note — left at 3 k_B.)
270 const Vec C_i = 2.0f * grid.inv_gm1() * n_i * grid.k_b;
271 const Vec C_n = grid.inv_gm1() * n_n * grid.k_b;
272 const Vec C_sum = C_i + C_n;
273
274 const Vec beta = 3.0 * grid.k_b * alpha / (grid.m_i + grid.m_n);
275 const Vec lambda_T = beta % C_sum / (C_i % C_n);
276
277 const Vec T_cm = (C_i % T_i + C_n % T_n) / C_sum;
278 // Single-fluid limit: lock both temperatures to the common T_cm (dt·λ_T→∞).
279 const Vec dT = grid.single_fluid
280 ? Vec(arma::zeros<Vec>(T_i.n_elem))
281 : Vec((T_i - T_n) / (1.0 + dt * lambda_T));
282 const Vec T_i_new = T_cm + (C_n / C_sum) % dT;
283 const Vec T_n_new = T_cm - (C_i / C_sum) % dT;
284
285 const Vec p_i_new = 2.0 * n_i * grid.k_b % T_i_new;
286 const Vec p_n_new = n_n * grid.k_b % T_n_new;
287
288 prim_state.zeros();
289 prim_state += scalar_to(grid, rho_i, prim::RHO_I);
290 prim_state += scalar_to(grid, rho_n, prim::RHO_N);
291 prim_state += scalar_to(grid, V, prim::V);
292 prim_state += scalar_to(grid, U, prim::U);
293 prim_state += scalar_to(grid, p_i_new, prim::P_I);
294 prim_state += scalar_to(grid, p_n_new, prim::P_N);
295 prim_state += scalar_to(grid, p_e, prim::P_E);
296 return;
297 }

299 // ---- Three-temperature point-implicit relaxation (T_e, T_i, T_n) --------
300 // Generalizes the 2-way solve to a symmetric 3×3 backward-Euler relaxation
301 // with three pairwise conductances g_ab [W m⁻³ K⁻¹] (g_ab = g_ba ⇒ energy
302 // conserved exactly): electron–ion g_ei = C_e ν_ei, electron–neutral
303 // g_en = C_e ν_en, ion–neutral g_in = 3 k_B α/(m_i+m_n) (the same β as the
304 // 2-way solve, now connecting protons↔neutrals only — electrons reach the
305 // neutrals through the separate ν_en channel). Heat capacities split the
306 // old combined charged C = 3 n_i k_B into equal electron and proton halves.
307 const Vec p_proton = p_i - p_e; // derived proton pressure
308 const Vec T_e = p_e / (n_i * grid.k_b);
309 const Vec T_i = p_proton / (n_i * grid.k_b);
310 const Vec T_n = p_n / (n_n * grid.k_b);
311
312 const Vec C_e = grid.inv_gm1() * n_i * grid.k_b;
313 const Vec C_i = grid.inv_gm1() * n_i * grid.k_b;
314 const Vec C_n = grid.inv_gm1() * n_n * grid.k_b;
315
316 const Vec alpha = rho_i % nu_in(grid, n_n, T_i, T_n);
317 const Vec g_in = 3.0 * grid.k_b * alpha / (grid.m_i + grid.m_n);
318 const Vec g_ei = C_e % nu_ei(grid, n_i, T_e);
319 const Vec g_en = C_e % nu_en(grid, n_n, T_e);
320
321 Vec T_e_new(grid.ns), T_i_new(grid.ns), T_n_new(grid.ns);
322 for (arma::uword k = 0; k < grid.ns; ++k) {
323 // Symmetric 3×3 system M·[T_e,T_i,T_n]ᵀ = [C_e T_e, C_i T_i, C_n T_n]ᵀ.
324 // M_ee = C_e + dt(g_ei+g_en), M_ei = -dt g_ei, M_en = -dt g_en
325 // M_ii = C_i + dt(g_ei+g_in), M_in = -dt g_in
326 // M_nn = C_n + dt(g_en+g_in)
327 const float ge = g_ei(k), gn = g_en(k), gi = g_in(k);
328 const float a11 = C_e(k) + dt*(ge+gn), a12 = -dt*ge, a13 = -dt*gn;
329 const float a22 = C_i(k) + dt*(ge+gi), a23 = -dt*gi;
330 const float a33 = C_n(k) + dt*(gn+gi);
331 const float b1 = C_e(k)*T_e(k), b2 = C_i(k)*T_i(k), b3 = C_n(k)*T_n(k);
332 // Solve via cofactor expansion (symmetric, diagonally dominant ⇒ regular).
333 const float c11 = a22*a33 - a23*a23;
334 const float c12 = -(a12*a33 - a23*a13);
335 const float c13 = a12*a23 - a22*a13;
336 const float c22 = a11*a33 - a13*a13;
337 const float c23 = -(a11*a23 - a12*a13);
338 const float c33 = a11*a22 - a12*a12;
339 const float det = a11*c11 + a12*c12 + a13*c13;
340 const float inv = (std::fabs(det) > 0.0f) ? 1.0f/det : 0.0f;
341 T_e_new(k) = inv*(c11*b1 + c12*b2 + c13*b3);
342 T_i_new(k) = inv*(c12*b1 + c22*b2 + c23*b3);
343 T_n_new(k) = inv*(c13*b1 + c23*b2 + c33*b3);
344 }
345
346 const Vec p_e_new = n_i * grid.k_b % T_e_new;
347 const Vec p_proton_new = n_i * grid.k_b % T_i_new;
348 const Vec p_n_new = n_n * grid.k_b % T_n_new;
349 const Vec p_i_new = p_proton_new + p_e_new; // total charged pressure
350
351 prim_state.zeros();
352 prim_state += scalar_to(grid, rho_i, prim::RHO_I);
353 prim_state += scalar_to(grid, rho_n, prim::RHO_N);
354 prim_state += scalar_to(grid, V, prim::V);
355 prim_state += scalar_to(grid, U, prim::U);
356 prim_state += scalar_to(grid, p_i_new, prim::P_I);
357 prim_state += scalar_to(grid, p_n_new, prim::P_N);
358 prim_state += scalar_to(grid, p_e_new, prim::P_E);
359}

361// ----------------------------------------------------------------------------
362// Thomas algorithm (in-place destruction of b, d): solves
363// a[i] x[i-1] + b[i] x[i] + c[i] x[i+1] = d[i], i = 0..N-1,
364// where a[0] and c[N-1] are unused. No pivoting; the matrix is strictly
365// diagonally dominant for the backward-Euler diffusion operator below.
366// ----------------------------------------------------------------------------
367static Vec thomas_solve(const Vec& a, Vec b, const Vec& c, Vec d) {
368 const arma::uword N = b.n_elem;
369 for (arma::uword i = 1; i < N; ++i) {
370 const float m = a[i] / b[i - 1];
371 b[i] -= m * c[i - 1];
372 d[i] -= m * d[i - 1];
373 }
374 Vec x(N);
375 x[N - 1] = d[N - 1] / b[N - 1];
376 for (arma::uword i = N - 1; i > 0; --i) {
377 x[i - 1] = (d[i - 1] - c[i - 1] * x[i]) / b[i - 1];
378 }
379 return x;
380}

382// ----------------------------------------------------------------------------
383// Stage D (writeup §3.7 Eqs. (cond-tridiag)--(cond-coeffs)): backward-Euler
384// heat conduction with conductivities lagged at the post-equilibration state.
385// One tridiagonal solve per species (charged fluid uses κ_e + κ_i, neutrals
386// use κ_n). Ghost temperatures are folded into the RHS as Dirichlet data,
387// mirroring the φ_g extrapolation used by rhs_implicit_state.
388// ----------------------------------------------------------------------------
389static Vec ghost_extended_T(const Grid& grid,
390 const Vec& cons_state,
391 arma::uword cons_RHO,
392 arma::uword cons_MOM,
393 arma::uword cons_E,
394 float k_per_n, // 2*k_B for ions, k_B for neutrals
395 float m_species,
396 Vec& T_ghost_inner_out,
397 Vec& T_ghost_outer_out)
398{
399 // T at all ns cells, computed cell-locally from cons + cell-centered phi_g.
400 const Vec rho = get_scalar(grid, cons_state, cons_RHO);
401 const Vec mom = get_scalar(grid, cons_state, cons_MOM);
402 const Vec etot = get_scalar(grid, cons_state, cons_E);
403 const Vec vel = mom / rho;
404 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
405 const Vec p = grid.gm1() * etot - grid.half_gm1() * rho % vel % vel - grid.gm1() * rho % phi_g;
406 const Vec n = rho / m_species;
407 const Vec T = p / (k_per_n * n);
408
409 // T at ghost positions (i = -1 and i = ns). Same construction as the
410 // ip1/im1 paths in rhs_implicit_state: cons from the ghost cube, phi_g
411 // taken at the face (Neumann mirror beyond the boundary face).
412 const Vec cons_ip1 = ip1(grid, cons_state);
413 const Vec cons_im1 = im1(grid, cons_state);
414
415 auto T_at_shifted = [&](const Vec& cons_shift, bool outer) {
416 const Vec rho_s = get_scalar(grid, cons_shift, cons_RHO);
417 const Vec mom_s = get_scalar(grid, cons_shift, cons_MOM);
418 const Vec e_s = get_scalar(grid, cons_shift, cons_E);
419 const Vec v_s = mom_s / rho_s;
420 Vec phi_g_s = outer
421 ? ip1(grid, 0.5 * (grid.phi_g_iph + grid.phi_g_imh), SLICE)
422 : im1(grid, 0.5 * (grid.phi_g_iph + grid.phi_g_imh), SLICE);
423 if (outer) phi_g_s[grid.ns - 1] = grid.phi_g_iph[grid.ns - 1];
424 else phi_g_s[0] = grid.phi_g_imh[0];
425 const Vec p_s = grid.gm1() * e_s - grid.half_gm1() * rho_s % v_s % v_s - grid.gm1() * rho_s % phi_g_s;
426 const Vec n_s = rho_s / m_species;
427 return Vec(p_s / (k_per_n * n_s));
428 };
429
430 Vec T_ip1 = T_at_shifted(cons_ip1, true);
431 Vec T_im1 = T_at_shifted(cons_im1, false);
432
433 T_ghost_outer_out = T_ip1.tail(1); // value at i = ns (only slot used)
434 T_ghost_inner_out = T_im1.head(1); // value at i = -1 (only slot used)
435 return T;
436}

438// Electron-temperature variant of ghost_extended_T for the three-temperature
439// conduction path. The electron internal energy E_E carries NO kinetic or
440// gravitational part, so T_e = (2/3 E_E)/(n_i k_B) = p_e/(n_i k_B) directly —
441// no KE/φ subtraction. Ghost values come from the shifted cubes' E_E and ρ_i.
442static Vec ghost_extended_Te(const Grid& grid, const Vec& cons_state,
443 Vec& T_ghost_inner_out, Vec& T_ghost_outer_out) {
444 auto Te_of = [&](const Vec& cs) {
445 const Vec rho_i = get_scalar(grid, cs, cons::RHO_I);
446 const Vec e_e = get_scalar(grid, cs, cons::E_E);
447 const Vec n_i = rho_i / grid.m_i;
448 return Vec(grid.gm1() * e_e / (n_i * grid.k_b));
449 };
450 const Vec T = Te_of(cons_state);
451 const Vec T_ip1 = Te_of(ip1(grid, cons_state));
452 const Vec T_im1 = Te_of(im1(grid, cons_state));
453 T_ghost_outer_out = T_ip1.tail(1);
454 T_ghost_inner_out = T_im1.head(1);
455 return T;
456}

458static void apply_conduction_stage(const Grid& grid, Vec& prim_state, float dt) {
459 // Conduction operates on temperatures; rebuild cons once to reuse the
460 // existing ip1/im1 ghost machinery, then convert back.
461 Vec cons_state = prim2cons(grid, prim_state);
462
463 // Cell-centered densities/temperatures.
464 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
465 const Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
466 const Vec n_i = rho_i / grid.m_i;
467 const Vec n_n = rho_n / grid.m_n;
468
469 // The "charged row" conducts either the combined charged temperature
470 // T_charged = p_total/(2 n_i k_B) with heat capacity 3 n_i k_B (single-T
471 // baseline) or the electron temperature T_e = p_e/(n_i k_B) with heat
472 // capacity 1.5 n_i k_B (three-temperature model). Spitzer κ_e ∝ T^{5/2} is
473 // evaluated at whichever temperature drives the row; the neutral row is
474 // identical in both. The proton conduction κ_i ≈ κ_e/43 is dropped (plan §3).
475 const bool Te_on = grid.enable_Te;
476 // C_charged = cfac · n_i k_B. Single-T conducts the combined charged pool
477 // (ions+electrons ⇒ 2 n_i): cfac = 2/(γ−1). Three-T conducts electrons only
478 // (n_i): cfac = 1/(γ−1). (γ=5/3 ⇒ 3.0 and 1.5 respectively.)
479 const float cfac = Te_on ? grid.inv_gm1() : 2.0f * grid.inv_gm1();
480 const Vec p_i_orig = get_scalar(grid, prim_state, prim::P_I);
481 const Vec p_e_orig = get_scalar(grid, prim_state, prim::P_E);
482
483 Vec T_ghost_in_i, T_ghost_out_i, T_ghost_in_n, T_ghost_out_n;
484 Vec T_i = Te_on
485 ? ghost_extended_Te(grid, cons_state, T_ghost_in_i, T_ghost_out_i)
486 : ghost_extended_T(grid, cons_state, cons::RHO_I, cons::MOM_I, cons::E_I,
487 2.0f * grid.k_b, grid.m_i, T_ghost_in_i, T_ghost_out_i);
488 Vec T_n = ghost_extended_T(grid, cons_state,
489 cons::RHO_N, cons::MOM_N, cons::E_N,
490 grid.k_b, grid.m_n,
491 T_ghost_in_n, T_ghost_out_n);
492
493 // Conduction-only outer thermal wall (chromosphere.hpp
494 // ::outer_conduction_temperature_override). Replace the hydro outer ghost's
495 // temperature on BOTH conduction rows so the Dirichlet datum folded into the
496 // RHS below — and the outer ghost conductivity assembled from the shifted
497 // cubes further down — stay pinned to the wall while the hydro ghost's own
498 // temperature is free to float. Ghost DENSITIES are untouched.
499 if (grid.outer_conduction_temperature_override) {
500 T_ghost_out_i[0] = grid.outer_conduction_temperature;
501 T_ghost_out_n[0] = grid.outer_conduction_temperature;
502 }
503
504 // Face conductivities (lagged at current T). Reuse the same averaging as
505 // rhs_implicit_state, with ghost values from the shifted cubes.
506 const Vec K_e = kappa_e(n_i, n_n, T_i);
507 const Vec K_n = kappa_n(n_i, n_n, T_i, T_n);
508
509 // Build per-cell K_iph, K_imh by averaging cell-centered K with shifted
510 // K (the shifts pull in the ghost K through the ip1/im1 path).
511 const Vec cons_ip1 = ip1(grid, cons_state);
512 const Vec cons_im1 = im1(grid, cons_state);
513 const Vec rho_i_ip1 = get_scalar(grid, cons_ip1, cons::RHO_I);
514 const Vec rho_n_ip1 = get_scalar(grid, cons_ip1, cons::RHO_N);
515 const Vec rho_i_im1 = get_scalar(grid, cons_im1, cons::RHO_I);
516 const Vec rho_n_im1 = get_scalar(grid, cons_im1, cons::RHO_N);
517 const Vec n_i_ip1 = rho_i_ip1 / grid.m_i;
518 const Vec n_n_ip1 = rho_n_ip1 / grid.m_n;
519 const Vec n_i_im1 = rho_i_im1 / grid.m_i;
520 const Vec n_n_im1 = rho_n_im1 / grid.m_n;
521
522 // Shifted T at neighbour cells. ghost_extended_T already gave T at ns
523 // cells from the unshifted cube; use the shifted cubes here for the K_e
524 // and K_n inputs.
525 Vec T_ghost_unused_a, T_ghost_unused_b;
526 Vec T_i_ip1_full = Te_on
527 ? ghost_extended_Te(grid, cons_ip1, T_ghost_unused_a, T_ghost_unused_b)
528 : ghost_extended_T(grid, cons_ip1, cons::RHO_I, cons::MOM_I, cons::E_I,
529 2.0f * grid.k_b, grid.m_i, T_ghost_unused_a, T_ghost_unused_b);
530 Vec T_n_ip1_full = ghost_extended_T(grid, cons_ip1,
531 cons::RHO_N, cons::MOM_N, cons::E_N,
532 grid.k_b, grid.m_n,
533 T_ghost_unused_a, T_ghost_unused_b);
534 Vec T_i_im1_full = Te_on
535 ? ghost_extended_Te(grid, cons_im1, T_ghost_unused_a, T_ghost_unused_b)
536 : ghost_extended_T(grid, cons_im1, cons::RHO_I, cons::MOM_I, cons::E_I,
537 2.0f * grid.k_b, grid.m_i, T_ghost_unused_a, T_ghost_unused_b);
538 Vec T_n_im1_full = ghost_extended_T(grid, cons_im1,
539 cons::RHO_N, cons::MOM_N, cons::E_N,
540 grid.k_b, grid.m_n,
541 T_ghost_unused_a, T_ghost_unused_b);
542
543 // Last entry of the ip1-shifted temperatures IS the outer ghost; pin it to the
544 // conduction wall too so κ at the outer face matches the Dirichlet datum.
545 if (grid.outer_conduction_temperature_override) {
546 T_i_ip1_full[grid.ns - 1] = grid.outer_conduction_temperature;
547 T_n_ip1_full[grid.ns - 1] = grid.outer_conduction_temperature;
548 }
549
550 const Vec K_e_ip1 = kappa_e(n_i_ip1, n_n_ip1, T_i_ip1_full);
551 const Vec K_e_im1 = kappa_e(n_i_im1, n_n_im1, T_i_im1_full);
552 const Vec K_n_ip1 = kappa_n(n_i_ip1, n_n_ip1, T_i_ip1_full, T_n_ip1_full);
553 const Vec K_n_im1 = kappa_n(n_i_im1, n_n_im1, T_i_im1_full, T_n_im1_full);
554
555 const Vec ds_i_ip1 = ip1(grid, grid.ds_i, SLICE);
556 const Vec ds_i_im1 = im1(grid, grid.ds_i, SLICE);
557 const Vec ds_iph = 0.5 * (grid.ds_i + ds_i_ip1);
558 const Vec ds_imh = 0.5 * (ds_i_im1 + grid.ds_i);
559
560 // Isotropic numerical diffusion (Pandey 2024 §3.3): add K_num = χ·C to each
561 // face conductivity, where C is the face-averaged heat capacity per volume
562 // (ion C = 3 n_i k_B, neutral C = 1.5 n_n k_B). Folding it into K turns the
563 // existing backward-Euler tridiagonal into an unconditionally-stable solve
564 // for the combined Spitzer + isotropic-diffusion operator at no extra cost.
565 Vec chi_iph(grid.ns), chi_imh(grid.ns);
566 for (arma::uword i = 0; i < grid.ns; ++i) {
567 chi_iph(i) = numerical_diffusivity_at_face(grid, grid.ds_iph_i(i));
568 chi_imh(i) = numerical_diffusivity_at_face(grid, grid.ds_imh_i(i));
569 }
570 // K_num = χ · C_face. Charged-row capacity is cfac·n_i k_B (cfac = 3 single-T,
571 // 1.5 three-T), so the face value is 0.5·cfac·k_B·(n_i + n_i_shift).
572 const Vec Knum_iph_i = (0.5f * cfac * grid.k_b) * (n_i + n_i_ip1) % chi_iph;
573 const Vec Knum_imh_i = (0.5f * cfac * grid.k_b) * (n_i + n_i_im1) % chi_imh;
574 const Vec Knum_iph_n = (0.5f * grid.inv_gm1() * grid.k_b) * (n_n + n_n_ip1) % chi_iph;
575 const Vec Knum_imh_n = (0.5f * grid.inv_gm1() * grid.k_b) * (n_n + n_n_im1) % chi_imh;
576
577 // TRAC broadening (Johnston 2020): below the adaptive cutoff T_c the charged-
578 // fluid conductivity is enhanced by ε=(T_c/T)^{5/2} (→ κ' = κ(T_c), constant),
579 // spreading the unresolved TR onto the coarse grid. Applied per cell at the
580 // cell temperature (and at the shifted-neighbour temperatures) before face
581 // averaging. Only κ_e is broadened; the neutral row is chromospheric.
582 Vec Kb = K_e, Kb_ip1 = K_e_ip1, Kb_im1 = K_e_im1;
583 if (grid.enable_trac) {
584 Kb %= trac_broadening_factor(grid, T_i);
585 Kb_ip1 %= trac_broadening_factor(grid, T_i_ip1_full);
586 Kb_im1 %= trac_broadening_factor(grid, T_i_im1_full);
587 }
588
589 // ----- ion (charged) row ----- K_ion_total = κ_e (κ_i is zero in this code)
590 // Spitzer κ_e is combined across a face by the legacy arithmetic average on a
591 // uniform mesh, or the width-weighted series-resistance combination on a
592 // refined mesh (physics.hpp::face_conductivity_series). The isotropic
593 // numerical diffusivity K_num keeps its face average either way.
594 const Vec K_face_iph_i = (grid.uniform_mesh ? Vec(0.5 * (Kb + Kb_ip1))
595 : face_conductivity_series(Kb, Kb_ip1, grid.ds_i, ds_i_ip1))
596 + Knum_iph_i;
597 const Vec K_face_imh_i = (grid.uniform_mesh ? Vec(0.5 * (Kb + Kb_im1))
598 : face_conductivity_series(Kb, Kb_im1, grid.ds_i, ds_i_im1))
599 + Knum_imh_i;
600 const Vec C_i = cfac * n_i * grid.k_b;
601 const Vec g_iph_i = grid.B_i / grid.ds_i % (K_face_iph_i / grid.B_iph) / ds_iph;
602 const Vec g_imh_i = grid.B_i / grid.ds_i % (K_face_imh_i / grid.B_imh) / ds_imh;
603 Vec R_coef_i = dt * g_iph_i / C_i;
604 Vec L_coef_i = dt * g_imh_i / C_i;
605
606 Vec a_i(grid.ns, arma::fill::zeros);
607 Vec b_i(grid.ns);
608 Vec c_i(grid.ns, arma::fill::zeros);
609 Vec d_i = T_i;
610 for (arma::uword i = 0; i < grid.ns; ++i) {
611 a_i[i] = -L_coef_i[i];
612 c_i[i] = -R_coef_i[i];
613 b_i[i] = 1.0f + L_coef_i[i] + R_coef_i[i];
614 }
615 // Inner BC. Default Dirichlet at i = -1: fold L_0 * T_ghost into RHS, zero
616 // a[0]. Neumann (grid.inner_conduction_neumann): zero conductive flux through
617 // the base — drop the inner-face coupling entirely (no L_0 ghost term, no L_0
618 // in the diagonal), so dT/ds = 0 at the base and the photosphere is insulating.
619 a_i[0] = 0.0f;
620 if (grid.inner_conduction_neumann) {
621 b_i[0] = 1.0f + R_coef_i[0];
622 } else {
623 d_i[0] += L_coef_i[0] * T_ghost_in_i[0];
624 }
625 // Outer BC. Either:
626 // * Neumann fixed-flux (impose_outer_heat_flux): the top face carries the
627 // imposed coronal conductive flux q = outer_heat_flux [W/m²] into the
628 // cell. Drop the R_coef coupling (no Dirichlet ghost) and add the flux
629 // as a source ΔT = dt·q/(C·ds). Robust on the coarse grid — does not
630 // over-conduct through a hot T-wall. (RTV q(T); plan upper-BC step 2.)
631 // * Dirichlet (default): fold R_{ns-1}·T_ghost into the RHS. T_ghost_out is
632 // the hydro outer ghost's temperature, or the pinned conduction wall when
633 // outer_conduction_temperature_override is set (substituted above).
634 const arma::uword nlast = grid.ns - 1;
635 if (grid.impose_outer_heat_flux) {
636 b_i[nlast] = 1.0f + L_coef_i[nlast];
637 d_i[nlast] = T_i[nlast]
638 + dt * grid.outer_heat_flux / (C_i[nlast] * grid.ds_i[nlast]);
639 } else {
640 d_i[nlast] += R_coef_i[nlast] * T_ghost_out_i[0];
641 }
642 c_i[nlast] = 0.0f;
643 const Vec T_i_new = thomas_solve(a_i, b_i, c_i, d_i);
644
645 // ----- neutral row ----- (same series-resistance treatment on a refined mesh)
646 const Vec K_face_iph_n = (grid.uniform_mesh ? Vec(0.5 * (K_n + K_n_ip1))
647 : face_conductivity_series(K_n, K_n_ip1, grid.ds_i, ds_i_ip1))
648 + Knum_iph_n;
649 const Vec K_face_imh_n = (grid.uniform_mesh ? Vec(0.5 * (K_n + K_n_im1))
650 : face_conductivity_series(K_n, K_n_im1, grid.ds_i, ds_i_im1))
651 + Knum_imh_n;
652 const Vec C_n_vec = grid.inv_gm1() * n_n * grid.k_b;
653 const Vec g_iph_n = grid.B_i / grid.ds_i % (K_face_iph_n / grid.B_iph) / ds_iph;
654 const Vec g_imh_n = grid.B_i / grid.ds_i % (K_face_imh_n / grid.B_imh) / ds_imh;
655 Vec R_coef_n = dt * g_iph_n / C_n_vec;
656 Vec L_coef_n = dt * g_imh_n / C_n_vec;
657
658 Vec a_n(grid.ns, arma::fill::zeros);
659 Vec b_n(grid.ns);
660 Vec c_n(grid.ns, arma::fill::zeros);
661 Vec d_n = T_n;
662 for (arma::uword i = 0; i < grid.ns; ++i) {
663 a_n[i] = -L_coef_n[i];
664 c_n[i] = -R_coef_n[i];
665 b_n[i] = 1.0f + L_coef_n[i] + R_coef_n[i];
666 }
667 a_n[0] = 0.0f;
668 if (grid.inner_conduction_neumann) {
669 b_n[0] = 1.0f + R_coef_n[0];
670 } else {
671 d_n[0] += L_coef_n[0] * T_ghost_in_n[0];
672 }
673 d_n[grid.ns - 1] += R_coef_n[grid.ns - 1] * T_ghost_out_n[0];
674 c_n[grid.ns - 1] = 0.0f;
675 Vec T_n_new = thomas_solve(a_n, b_n, c_n, d_n);
676 Vec T_i_new2 = T_i_new;
677
678 // Flare robustness: at the explosive evaporation front the conduction
679 // tridiagonal can occasionally yield a non-finite temperature (extreme
680 // κ_e∝T^{5/2} contrasts across a near-discontinuity on the coarse grid). Fall
681 // back to the pre-conduction temperature for any such cell rather than
682 // letting the NaN propagate. Gated on beam heating → steady runs unchanged.
683 if (grid.enable_beam_heating) {
684 for (arma::uword i = 0; i < grid.ns; ++i) {
685 if (!std::isfinite(T_i_new2(i))) T_i_new2(i) = T_i(i);
686 if (!std::isfinite(T_n_new(i))) T_n_new(i) = T_n(i);
687 }
688 }
689
690 // Rebuild pressures and store back into prim_state.
691 const Vec p_n_new = n_n * grid.k_b % T_n_new;
692 Vec p_i_new, p_e_new;
693 if (!Te_on) {
694 // Single-T baseline: T_i_new2 is the combined charged temperature.
695 p_i_new = 2.0 * n_i * grid.k_b % T_i_new2; // total charged pressure
696 p_e_new = p_e_orig; // carried (re-slaved later)
697 } else {
698 // Three-T: T_i_new2 is the electron temperature. Only electrons conducted,
699 // so the total charged pressure shifts by exactly the electron pressure
700 // change (protons unchanged); E_I stays the energy-conserving total.
701 p_e_new = n_i * grid.k_b % T_i_new2;
702 p_i_new = p_i_orig + (p_e_new - p_e_orig);
703 }
704 const Vec V = get_scalar(grid, prim_state, prim::V);
705 const Vec U = get_scalar(grid, prim_state, prim::U);
706
707 prim_state.zeros();
708 prim_state += scalar_to(grid, rho_i, prim::RHO_I);
709 prim_state += scalar_to(grid, rho_n, prim::RHO_N);
710 prim_state += scalar_to(grid, V, prim::V);
711 prim_state += scalar_to(grid, U, prim::U);
712 prim_state += scalar_to(grid, p_i_new, prim::P_I);
713 prim_state += scalar_to(grid, p_n_new, prim::P_N);
714 prim_state += scalar_to(grid, p_e_new, prim::P_E);
715}

717// ----------------------------------------------------------------------------
718// Stage R (writeup §2.4, §5.6): optically-thick chromospheric radiative
719// cooling. Carlsson & Leenaarts 2012 recipe summing H I + Ca II + Mg II
720// losses; tables in physics.hpp::cl2012. Operates as a backward-Euler sink
721// on the folded ion+electron thermal pressure: p_i ← p_i / (1 + Δt Q/ε_i)
722// at lagged Q evaluated from the post-conduction T_e. This preserves
723// positivity unconditionally — when Δt Q ≫ ε_i, the cell relaxes asymptot-
724// ically to a small positive pressure rather than overshooting to a floor.
725// Mass / velocities are unaffected; only p_i is modified.
726// ----------------------------------------------------------------------------
727void apply_radiative_cooling_stage(const Grid& grid, Vec& prim_state, float dt) {
728 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
729 const Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
730 const Vec p_i = get_scalar(grid, prim_state, prim::P_I);
731 const Vec p_e = get_scalar(grid, prim_state, prim::P_E);
732
733 const Vec n_i = rho_i / grid.m_i;
734 const Vec n_n = rho_n / grid.m_n;
735 // Radiation is an electron-pool loss (Λ ∝ n_e and the bound-level excitation
736 // is electron-collisional). Single-T baseline: cool the combined charged pool
737 // at T_charged = p_total/(2 n_i k_B). Three-T: cool the electron pool at the
738 // true T_e = p_e/(n_i k_B), and mirror the loss into the total energy P_I.
739 const Vec T_e = grid.enable_Te ? Vec(p_e / (n_i * grid.k_b))
740 : Vec(p_i / (2.0f * n_i * grid.k_b));
741
742 // Total radiative sink: optically-thick chromospheric (CL2012, H/Ca/Mg) plus
743 // optically-thin TR/coronal losses n_e n_H Λ(T). The two are stitched by a
744 // ~2e4 K switch inside radiative_loss_thin so neither double-counts; together
745 // they span the chromosphere through the lower TR / corona, giving the TR the
746 // radiative sink to re-radiate the imposed coronal conductive flux q(T).
747 // Optically-thin loss is TRAC-broadened (divided by ε): Λ' = Λ(T/T_c)^{5/2}
748 // below T_c, so κ'Λ' = κΛ and the TR-integrated radiation is conserved on the
749 // coarse grid (Johnston 2020). The optically-thick CL2012 chromospheric term
750 // is left unbroadened (it lives below the TRAC region).
751 Vec Q_thin = radiative_loss_thin(grid, n_i, n_n, T_e);
752 if (grid.enable_trac) Q_thin /= trac_broadening_factor(grid, T_e);
753 const Vec Q = radiative_loss_thick(grid, n_i, n_n, T_e) // W/m^3, +ve = cooling
754 + Q_thin;
755 // Pool that radiates: combined charged (single-T) or electron-only (three-T).
756 const Vec p_pool = grid.enable_Te ? p_e : p_i;
757 const Vec eps = grid.inv_gm1() * p_pool; // thermal energy density
758 // Backward-Euler relaxation: ε^{n+1} = ε^n / (1 + Δt Q/ε^n) with Q frozen.
759 const Vec denom = 1.0f + dt * Q / arma::clamp(eps, 1.0e-30f, arma::datum::inf);
760 const Vec p_pool_new = p_pool / arma::clamp(denom, 1.0e-6f, arma::datum::inf);
761 const Vec dp = p_pool_new - p_pool;
762 if (grid.enable_Te) {
763 prim_state += scalar_to(grid, dp, prim::P_E); // electrons radiate
764 prim_state += scalar_to(grid, dp, prim::P_I); // mirror into total energy
765 } else {
766 prim_state += scalar_to(grid, dp, prim::P_I);
767 }
768}

770/// Vacuum-floor / single-fluid-collapse stage for the flare scenario. Explosive
771/// evaporation fully ionizes the heated gas, so the neutral density n_n collapses
772/// toward zero in the evaporated column. There the neutral velocity decode
773/// U = ρ_n U / ρ_n becomes 0/0 noise and the spectral radius |U| + c_s,n blows up
774/// → NaN (the ρ_n→0 ill-conditioning of two-fluid codes). We therefore floor both
775/// densities, floor the pressures positive, cap the per-fluid speeds at a generous
776/// physical ceiling, and — where the gas is ionized — COLLAPSE the neutral onto
777/// the charged fluid (U = V, T_n = T_e), the exact single-fluid limit (Gómez
778/// Míguez et al. 2024; the same slaving the C7 outer BC uses).
779///
780/// The collapse triggers on the ionization fraction f = n_i/(n_i+n_n) exceeding
781/// F_SLAVE, NOT on an absolute n_n floor. This is the key to a smooth neutral
782/// profile: f varies smoothly through the hot column, so the whole ionized region
783/// is slaved consistently. An absolute-floor trigger instead let neighbouring
784/// cells flip in and out of the slaved state every step — the explicit hydro
785/// re-noises the near-zero-mass neutral momentum each step, pushing ρ_n a hair
786/// above/below the floor — producing the cell-to-cell checkerboard (sawtooth) seen
787/// in u and p_n. Gated on enable_beam_heating so steady scenarios and the existing
788/// test suite are untouched. Mutates prim in place.
789///
790/// Internal to the historical two-fluid integrator: it is called from the
791/// operator-split step (guarded by Grid::floors_active()) and has no declaration
792/// in two_fluid.hpp.
793void apply_flare_floor_stage(const Grid& grid, Vec& prim_state) {
794 const float m_i = grid.m_i, m_n = grid.m_n, k_b = grid.k_b;
795 const float RHO_I_FLOOR = m_i * 1.0e10f; // n_i ≥ 10^10 m^-3
796 const float RHO_N_FLOOR = m_n * 1.0e10f; // n_n ≥ 10^10 m^-3
797 const float P_FLOOR = 1.0e-8f; // Pa (chromospheric p ~ 10^-2 Pa)
798 const float V_MAX = 1.5e6f; // m/s, above the 2.35 c_s evaporation ceiling
799 const float T_MAX = 5.0e7f; // K, ceiling — beyond coronal validity; bounds
800 // κ_e∝T^{5/2} & TRAC ε so conduction stays finite
801 const float F_SLAVE = 0.99f; // ionization fraction above which the neutral is a
802 // collisionally-locked trace species → single fluid
803
804 Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
805 Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
806 Vec V = get_scalar(grid, prim_state, prim::V);
807 Vec U = get_scalar(grid, prim_state, prim::U);
808 Vec p_i = get_scalar(grid, prim_state, prim::P_I);
809 Vec p_n = get_scalar(grid, prim_state, prim::P_N);
810
811 rho_i = arma::clamp(rho_i, RHO_I_FLOOR, arma::datum::inf);
812 p_i = arma::clamp(p_i, P_FLOOR, arma::datum::inf);
813 V = arma::clamp(V, -V_MAX, V_MAX);
814
815 const Vec n_i = rho_i / m_i;
816 // Temperature ceiling on the charged fluid: p_i ≤ 2 n_i k_B T_MAX.
817 p_i = arma::min(p_i, 2.0f * n_i * (k_b * T_MAX));
818 const Vec T_e = p_i / (2.0f * n_i * k_b); // electron/ion temperature (≤ T_MAX)
819 for (arma::uword i = 0; i < grid.ns; ++i) {
820 // Floor both densities positive so the V = ρV_i/ρ_i and U = ρU_n/ρ_n decodes
821 // and the spectral radii stay well-conditioned, then pick the regime.
822 if (rho_n(i) < RHO_N_FLOOR) rho_n(i) = RHO_N_FLOOR;
823 const float n_n_i = rho_n(i) / m_n;
824 const float f_ion = rho_i(i) / (rho_i(i) + rho_n(i)); // ionization fraction
825 if (f_ion >= F_SLAVE) { // ion-dominated → neutral is trace: slave neutral
826 U(i) = V(i); // U = V
827 p_n(i) = n_n_i * k_b * T_e(i); // T_n = T_e
828 } else { // partially/fully neutral → two-fluid, bound both
829 if (U(i) > V_MAX) U(i) = V_MAX;
830 if (U(i) < -V_MAX) U(i) = -V_MAX;
831 if (p_n(i) < P_FLOOR) p_n(i) = P_FLOOR;
832 float T_n_i = p_n(i) / (n_n_i * k_b);
833 if (T_n_i > T_MAX) { T_n_i = T_MAX; p_n(i) = n_n_i * k_b * T_MAX; } // T_n ≤ T_MAX
834 // Trace-ion overheating cap. When the ion is a minority species (f small)
835 // its T_i = p_i/(2 n_i k_B) blows up if n_i craters faster than p_i can
836 // relax — recombination at the condensation front. Cap the trace ion's
837 // temperature at the dominant neutral's T_n (the collisional single-fluid
838 // limit). DOWNWARD-only, with a 2× deadband, so it fires ONLY on genuine
839 // overheating (T_i > 2 T_n): the cool quiescent lower chromosphere
840 // (T_i ≈ T_n, also low f) is left untouched and V is never overwritten —
841 // unlike a full slave, which would rewrite the whole cool column and
842 // violate energy conservation (it crashed the run at beam onset).
843 if (f_ion <= 1.0f - F_SLAVE) {
844 const float p_i_cap = 2.0f * (rho_i(i) / m_i) * k_b * T_n_i;
845 if (p_i(i) > 2.0f * p_i_cap) p_i(i) = p_i_cap;
846 }
847 }
848 }
849
850 // Electron pressure (three-T only): keep T_e in (0, T_MAX] and p_e < p_i so the
851 // derived proton pressure p_proton = p_i − p_e stays positive. Single-T leaves
852 // P_E untouched (re-slaved at end of step).
853 Vec p_e = get_scalar(grid, prim_state, prim::P_E);
854 if (grid.enable_Te) {
855 for (arma::uword i = 0; i < grid.ns; ++i) {
856 const float n_i_i = rho_i(i) / m_i;
857 const float p_e_cap = n_i_i * (k_b * T_MAX); // T_e ≤ T_MAX
858 if (p_e(i) < P_FLOOR) p_e(i) = P_FLOOR;
859 if (p_e(i) > p_e_cap) p_e(i) = p_e_cap;
860 if (p_e(i) > p_i(i) - P_FLOOR) p_e(i) = p_i(i) - P_FLOOR;
861 if (p_e(i) < P_FLOOR) p_e(i) = P_FLOOR;
862 }
863 }
864
865 prim_state += scalar_to(grid, rho_i - get_scalar(grid, prim_state, prim::RHO_I), prim::RHO_I);
866 prim_state += scalar_to(grid, rho_n - get_scalar(grid, prim_state, prim::RHO_N), prim::RHO_N);
867 prim_state += scalar_to(grid, V - get_scalar(grid, prim_state, prim::V), prim::V);
868 prim_state += scalar_to(grid, U - get_scalar(grid, prim_state, prim::U), prim::U);
869 prim_state += scalar_to(grid, p_i - get_scalar(grid, prim_state, prim::P_I), prim::P_I);
870 prim_state += scalar_to(grid, p_n - get_scalar(grid, prim_state, prim::P_N), prim::P_N);
871 prim_state += scalar_to(grid, p_e - get_scalar(grid, prim_state, prim::P_E), prim::P_E);
872}

874// ----------------------------------------------------------------------------
875// Flare beam-heating stage. Adds the nonthermal-electron-beam volumetric
876// heating Q_beam [W/m³] (physics.hpp::beam_heating_rate) to the gas over one
877// step. The deposited energy is shared between the charged and neutral fluids in
878// proportion to their heat capacities per volume, C_i = 3 n_i k_b (ions +
879// electrons, since ε_i = 3/2 p_i = 3 n_i k_b T) and C_n = 3/2 n_n k_b, so both
880// fluids gain the SAME ΔT = Δt Q / (C_i + C_n). This is essential in the weakly
881// ionized chromosphere: there n_e ≪ n_n, so dumping the whole flux into the tiny
882// electron pool alone would spike T_e by many× in one step (numerically fatal,
883// and unphysical — a precipitating beam heats the bulk gas, which the fast
884// electron–neutral collisions then share; the temperature-equilibration stage
885// enforces the same on longer scales). Resulting pressure increments are
886// Δp_i = 2 n_i k_b ΔT and Δp_n = n_n k_b ΔT. Q_beam is T-independent so this
887// explicit additive update is exact and positivity-preserving. When TRAC is
888// active the heating is broadened by the same factor ε = (T_c/T)^{5/2} as the
889// optically-thin loss, conserving κQ across the unresolved TR (Johnston 2020).
890// Mutates p_i and p_n in place. (Nonthermal collisional ionization by the beam
891// is omitted; the thermal heating drives Stage E ionization as T rises.)
892// ----------------------------------------------------------------------------
893void apply_beam_heating_stage(const Grid& grid, Vec& prim_state, float dt) {
894 if (!grid.enable_beam_heating) return;
895 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
896 const Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
897 const Vec n_i = rho_i / grid.m_i;
898 const Vec n_n = rho_n / grid.m_n;
899 const float k_b = grid.k_b;
900
901 Vec Q_beam = beam_heating_rate(grid, n_i, n_n); // W/m^3, +ve = heating
902 if (grid.enable_trac) {
903 const Vec p_i = get_scalar(grid, prim_state, prim::P_I);
904 const Vec p_e = get_scalar(grid, prim_state, prim::P_E);
905 const Vec T_e = grid.enable_Te ? Vec(p_e / (n_i * k_b))
906 : Vec(p_i / (2.0f * n_i * k_b));
907 Q_beam /= trac_broadening_factor(grid, T_e);
908 }
909 if (grid.enable_Te) {
910 // Three-temperature: the thick-target beam deposits its energy into the
911 // ambient ELECTRONS (Q_beam → ε_e). The subsequent ν_ei equilibration
912 // (Stage C, next step) shares it with protons/neutrals; until then T_e can
913 // run well above T_i at the impulsive onset — the decoupling the plan
914 // targets. The total energy P_I gains the same Δε so E_I stays the
915 // conserved total. (cal_dt_i caps Δt on the electron pool so the spike is
916 // bounded.) ΔT_e = Δt Q /(1.5 n_i k_B); Δp_e = (2/3)·1.5 n_i k_B ΔT_e = Δt Q.
917 const Vec dp_e = grid.gm1() * dt * Q_beam; // = (γ−1) Δε_e
918 prim_state += scalar_to(grid, dp_e, prim::P_E);
919 prim_state += scalar_to(grid, dp_e, prim::P_I);
920 return;
921 }
922 // Single-temperature baseline: share Q_beam between the charged and neutral
923 // pools by heat capacity so both gain the SAME ΔT (essential in the weakly
924 // ionized chromosphere; see header note). ΔT = Δt Q / (C_i + C_n).
925 const Vec C_i = 2.0f * grid.inv_gm1() * n_i * k_b;
926 const Vec C_n = grid.inv_gm1() * n_n * k_b;
927 const Vec dT = (dt * Q_beam) / (C_i + C_n);
928 const Vec dp_i = 2.0f * n_i % (k_b * dT); // = (2/3) C_i ΔT
929 const Vec dp_n = n_n % (k_b * dT); // = (2/3) C_n ΔT
930 prim_state += scalar_to(grid, dp_i, prim::P_I);
931 prim_state += scalar_to(grid, dp_n, prim::P_N);
932}

934// ----------------------------------------------------------------------------
935// Ambient coronal heating stage. Adds the steady footpoint-anchored volumetric
936// heating H(s) [W/m³] (physics.hpp::coronal_heating_rate) over one step, with
937// the SAME pool partitioning and TRAC broadening as the flare beam: in the
938// single-temperature baseline the heat is shared between the charged and neutral
939// fluids by heat capacity so both gain the same ΔT = Δt H / (C_i + C_n); with
940// enable_Te it is deposited into the electron pool (electrons carry the Spitzer
941// conduction that delivers the flux into the TR), mirrored into the total energy
942// P_I so E_I stays the conserved total. H is T-independent, so this explicit
943// additive update is exact and positivity-preserving. No-op unless
944// grid.enable_coronal_heating. Mutates the thermal pressures in place.
945// ----------------------------------------------------------------------------
946void apply_coronal_heating_stage(const Grid& grid, Vec& prim_state, float dt) {
947 if (!grid.enable_coronal_heating) return;
948 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
949 const Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
950 const Vec n_i = rho_i / grid.m_i;
951 const Vec n_n = rho_n / grid.m_n;
952 const float k_b = grid.k_b;
953
954 Vec Q = coronal_heating_rate(grid); // W/m³, +ve = heating
955 if (arma::all(Q <= 0.0f)) return;
956 if (grid.enable_trac) {
957 const Vec p_i = get_scalar(grid, prim_state, prim::P_I);
958 const Vec p_e = get_scalar(grid, prim_state, prim::P_E);
959 const Vec T_e = grid.enable_Te ? Vec(p_e / (n_i * k_b))
960 : Vec(p_i / (2.0f * n_i * k_b));
961 Q /= trac_broadening_factor(grid, T_e);
962 }
963 if (grid.enable_Te) {
964 // Deposit into the electrons (Δp_e = Δt Q); mirror into the total P_I.
965 const Vec dp_e = grid.gm1() * dt * Q; // = (γ−1) Δε_e
966 prim_state += scalar_to(grid, dp_e, prim::P_E);
967 prim_state += scalar_to(grid, dp_e, prim::P_I);
968 return;
969 }
970 // Single-temperature baseline: share by heat capacity → equal ΔT.
971 const Vec C_i = 2.0f * grid.inv_gm1() * n_i * k_b;
972 const Vec C_n = grid.inv_gm1() * n_n * k_b;
973 const Vec dT = (dt * Q) / (C_i + C_n);
974 const Vec dp_i = 2.0f * n_i % (k_b * dT); // = (2/3) C_i ΔT
975 const Vec dp_n = n_n % (k_b * dT); // = (2/3) C_n ΔT
976 prim_state += scalar_to(grid, dp_i, prim::P_I);
977 prim_state += scalar_to(grid, dp_n, prim::P_N);
978}

980// ----------------------------------------------------------------------------
981// Stage E (writeup §5.3): point-implicit hydrogen ionization / recombination,
982// Route B network (docs/photoionization_c7_inversion_plan.md). The local ODE
983// for f ≡ ρ_i / (ρ_i + ρ_n) under quasi-neutrality (n_e = n_i = f n_tot) is
984// df/dt = (1-f) P_phot [photoionization, always on]
985// + f(1-f) n_tot S_CR [multilevel collisional, always on]
986// + f(1-f) n_tot S_i [Voronov direct, flag-gated, off by default]
987// - f² n_tot α_r [radiative recomb, always on]
988// - f³ n_tot² κ_c [three-body recomb, flag-gated, off by default]
989// Default network (grid.enable_direct_collisional_ionization = false,
990// grid.enable_threebody_recombination = false): P_phot + S_CR + α_r only.
991// S_i is subdominant by 2–3 orders vs. S_CR everywhere in the chromosphere.
992// α_c is within 1–2 orders of α_r below ~700 km but negligible above.
993// Backward Euler at lagged T_e reduces to the cell-local polynomial
994// a3 f³ + a2 f² + a1 f + a0 = 0,
995// a3 = Δt n_tot² κ_c, (= 0 when three-body is off → quadratic)
996// a2 = Δt n_tot (S_i + S_CR + α_r),
997// a1 = 1 + Δt P_phot - Δt n_tot (S_i + S_CR),
998// a0 = -(f^n + Δt P_phot).
999// g(0)=a0≤0 and g(1)≥0 always, so the physical root lies in [0,1]; it is
1000// found by a safeguarded Newton iteration seeded from the κ_c→0 quadratic
1001// root (which recovers the pre-Route-B solve exactly when κ_c=0).
1002// χ_H bookkeeping: collisional ionization (S_i and S_CR) drains χ_H from the
1003// electron thermal pool; three-body recombination is super-elastic and returns
1004// χ_H to it (the S_CR drain and α_c return cancel at LTE by detailed balance);
1005// photoionization adds ions but drains no χ_H (energy comes from the absorbed
1006// UV photon), and radiative recombination radiates χ_H away (case B, optically
1007// thin). Mutates V, U, p_i, p_n, ρ_i, ρ_n in place.
1008// ----------------------------------------------------------------------------
1009
1010// Solve a3 f³ + a2 f² + a1 f + a0 = 0 for the physical root in [0,1].
1011// The Stage-E coefficients guarantee g(0)=a0≤0 and g(1)=a3+a2+a1+a0≥0, so a
1012// root is bracketed by [0,1]. Safeguarded Newton (Newton step when it stays in
1013// the bracket, bisection otherwise) — robust through the degenerate quadratic
1014// and linear limits (a3,a2 → 0). `f_seed` is the κ_c→0 quadratic root.
1015static float solve_ion_cubic(float a3, float a2, float a1, float a0, float f_seed) {
1016 auto g = [&](float f) { return ((a3 * f + a2) * f + a1) * f + a0; };
1017 auto dg = [&](float f) { return (3.0f * a3 * f + 2.0f * a2) * f + a1; };
1018 if (g(0.0f) >= 0.0f) return 0.0f; // root at/below 0 (f^n = 0, no photoion)
1019 if (g(1.0f) <= 0.0f) return 1.0f; // root at/above 1
1020 float lo = 0.0f, hi = 1.0f; // g(lo) < 0 < g(hi)
1021 float f = (f_seed > 0.0f && f_seed < 1.0f) ? f_seed : 0.5f;
1022 for (int it = 0; it < 60; ++it) {
1023 const float gf = g(f);
1024 if (gf > 0.0f) hi = f; else lo = f;
1025 const float d = dg(f);
1026 float f_next = (d != 0.0f) ? f - gf / d : 0.5f * (lo + hi);
1027 if (!(f_next > lo && f_next < hi)) f_next = 0.5f * (lo + hi);
1028 if (std::fabs(f_next - f) < 1.0e-7f) return f_next;
1029 f = f_next;
1030 }
1031 return f;
1032}

1034void apply_ionization_stage(const Grid& grid, Vec& prim_state, float dt) {
1035 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
1036 const Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
1037 const Vec V = get_scalar(grid, prim_state, prim::V);
1038 const Vec U = get_scalar(grid, prim_state, prim::U);
1039 const Vec p_i = get_scalar(grid, prim_state, prim::P_I);
1040 const Vec p_n = get_scalar(grid, prim_state, prim::P_N);
1041 const Vec p_e_old = get_scalar(grid, prim_state, prim::P_E);
1042
1043 const float m = grid.m_i; // m_i = m_n for hydrogen
1044 const Vec n_i = rho_i / m;
1045 const Vec n_n = rho_n / m;
1046 const Vec T_i = p_i / (2.0 * n_i * grid.k_b); // rate temperature (T_charged proxy)
1047 const Vec T_n = p_n / (n_n * grid.k_b);
1048
1049 // Rate coefficients at lagged T_e (Route B network).
1050 // Default network: P_phot + S_CR (ionization), α_r (recombination).
1051 // S_i and κ_c are zeroed unless their Grid flags are set (see chromosphere.hpp).
1052 const Vec S = grid.enable_direct_collisional_ionization
1053 ? ionization_rate_S(grid, T_i)
1054 : Vec(grid.ns, arma::fill::zeros); // S_i, direct Voronov (off by default)
1055 const Vec a = recombination_rate_alpha(grid, T_i); // α_r, case-B radiative
1056 const Vec P = photoionization_rate_P(grid); // P_phot
1057 const Vec S_cr = ionization_rate_S_CR(grid, T_i); // multilevel (= κ_c Φ, n_e-free)
1058 const Vec kappa_c = grid.enable_threebody_recombination
1059 ? recombination_rate_kappa_c(grid, T_i)
1060 : Vec(grid.ns, arma::fill::zeros); // κ_c for α_c=κ_c n_e (off by default)
1061
1062 // Backward-Euler cubic a3 f³ + a2 f² + a1 f + a0 = 0 (see header).
1063 const Vec rho_tot = rho_i + rho_n;
1064 const Vec n_tot = rho_tot / m;
1065 const Vec f_old = rho_i / rho_tot;
1066 const Vec Stot = S + S_cr; // total collisional ionization
1067 // a3 = dt n_tot² κ_c — group κ_c in first: n_tot² alone overflows float32
1068 // (n_tot ~ 1e19 → n_tot² ~ 1e38 ≈ FLT_MAX), but n_tot·κ_c is ~1e-18.
1069 const Vec a3v = dt * n_tot % (n_tot % kappa_c);
1070 const Vec a2v = dt * n_tot % (Stot + a);
1071 const Vec a1v = 1.0 + dt * P - dt * n_tot % Stot;
1072 const Vec a0v = -(f_old + dt * P);
1073
1074 Vec f_new(grid.ns);
1075 for (arma::uword i = 0; i < grid.ns; ++i) {
1076 // Seed: κ_c→0 quadratic root a2 f² + a1 f - (-a0) = 0 (the pre-Route-B
1077 // solve). Linear fallback when a2→0 (photoionization-only limit).
1078 const float A = a2v(i), B = a1v(i), C = -a0v(i);
1079 float seed;
1080 if (A < 1e-30f) {
1081 seed = (B > 0.0f) ? (C / B) : f_old(i);
1082 } else {
1083 const float disc = B * B + 4.0f * A * C;
1084 seed = (-B + std::sqrt(std::max(disc, 0.0f))) / (2.0f * A);
1085 }
1086 seed = std::min(std::max(seed, 0.0f), 1.0f);
1087 f_new(i) = solve_ion_cubic(a3v(i), a2v(i), a1v(i), a0v(i), seed);
1088 }
1089
1090 // Keep f strictly interior so neither ρ_i nor ρ_n underflows to zero. At the
1091 // hot top (corona-ghost T) Route B fully ionizes the gas (f→1); a vanishing
1092 // neutral density would make U_new = ρU_n/ρ_n (and cons2prim downstream)
1093 // divide by ~0. F_FLOOR bounds the minority species to a dynamically
1094 // negligible but numerically safe fraction.
1095 constexpr float F_FLOOR = 1.0e-8f;
1096 f_new = arma::clamp(f_new, F_FLOOR, 1.0f - F_FLOOR);
1097
1098 // Integrated channel counts (per m³) over Δt, evaluated at post-step
1099 // densities and lagged T_e (writeup §5.3, plan §3.4 issue d). Collisional
1100 // ionization (S_i, S_CR) uses the n_e·n_n product; photoionization is the
1101 // (1-f)·P branch; recombination (α_r radiative, α_c three-body) uses n_e·n_i.
1102 const Vec rho_i_new = f_new % rho_tot;
1103 const Vec rho_n_new = (1.0 - f_new) % rho_tot;
1104 const Vec n_i_new = rho_i_new / m;
1105 const Vec n_n_new = rho_n_new / m;
1106 const Vec a_c = recombination_rate_alpha_c(grid, T_i, n_i_new); // = κ_c n_e
1107 const Vec Gamma_coll = dt * n_i_new % n_n_new % S; // ground collisional — drains χ_H
1108 const Vec Gamma_mlvl = dt * n_i_new % n_n_new % S_cr; // multilevel collisional — drains χ_H
1109 const Vec Gamma_phot = dt * n_n_new % P; // photoionization — no χ_H drain
1110 const Vec Gamma_ion = Gamma_coll + Gamma_mlvl + Gamma_phot; // total ionizations
1111 const Vec Gamma_rec_rad = dt * n_i_new % n_i_new % a; // radiative — χ_H radiated away
1112 const Vec Gamma_rec_3b = dt * n_i_new % n_i_new % a_c; // three-body — super-elastic, returns χ_H
1113 const Vec Gamma_rec = Gamma_rec_rad + Gamma_rec_3b; // total recombinations
1114
1115 // Momentum bookkeeping (lagged velocities).
1116 const Vec rhoV_i_old = rho_i % V;
1117 const Vec rhoU_n_old = rho_n % U;
1118 const Vec dmom = m * (Gamma_ion % U - Gamma_rec % V);
1119 const Vec rhoV_i_new = rhoV_i_old + dmom;
1120 const Vec rhoU_n_new = rhoU_n_old - dmom;
1121 Vec V_new = rhoV_i_new / rho_i_new;
1122 Vec U_new = rhoU_n_new / rho_n_new;
1123 // When a species is at the F_FLOOR limit its momentum/density ratio is
1124 // ill-conditioned (ρU_n/ρ_n with ρ_n→0). Physically the trace fluid is
1125 // collisionally locked to the dominant one, so set its velocity to the
1126 // dominant fluid's rather than the (numerically noisy) division result.
1127 constexpr float F_GUARD = 1.0e-7f; // a few × F_FLOOR
1128 for (arma::uword i = 0; i < grid.ns; ++i) {
1129 if (rho_n_new(i) < F_GUARD * rho_tot(i)) U_new(i) = V_new(i);
1130 if (rho_i_new(i) < F_GUARD * rho_tot(i)) V_new(i) = U_new(i);
1131 }
1132
1133 // Energy bookkeeping (writeup §5.3, eqs. stageE-ei / stageE-en). We update
1134 // in *thermal* form: convert pressures to thermal energy ε = (3/2) p, apply
1135 // the source-driven Δε, then rebuild pressures from the new ε and KE.
1136 const Vec eps_i_old = grid.inv_gm1() * p_i;
1137 const Vec eps_n_old = grid.inv_gm1() * p_n;
1138 const Vec KE_i_old = 0.5 * rho_i % V % V;
1139 const Vec KE_n_old = 0.5 * rho_n % U % U;
1140 const Vec KE_i_new = 0.5 * rho_i_new % V_new % V_new;
1141 const Vec KE_n_new = 0.5 * rho_n_new % U_new % U_new;
1142
1143 const Vec th_inherit = grid.inv_gm1() * grid.k_b * T_n; // per-particle thermal carried by new ion
1144 const Vec th_leave = grid.inv_gm1() * grid.k_b * T_i; // per-particle thermal leaving with recomb. neutral
1145 const Vec KE_inherit = 0.5 * m * U % U; // per-particle KE inherited by new ion (from neutral)
1146 const Vec KE_leave = 0.5 * m * V % V; // per-particle KE leaving ion fluid
1147
1148 // Δ(total energy_i) = Q^{e_i}_ion · Δt (writeup §2.3 boxed; Route B)
1149 // = Γ_ion^Δ (3/2 k_B T_n + ½ m U²)
1150 // − Γ_rec^Δ (3/2 k_B T_i + ½ m V²)
1151 // − (Γ_coll + Γ_mlvl)^Δ χ_H [collisional ioniz drain]
1152 // + Γ_rec,3b^Δ χ_H [three-body super-elastic]
1153 // χ_H exchange with the electron thermal pool: both collisional ionization
1154 // channels (direct S_i and multilevel S_CR) remove χ_H per event; three-body
1155 // recombination is super-elastic and returns χ_H to the electrons (the
1156 // S_CR drain and α_c return cancel at LTE by detailed balance). Photoioniz-
1157 // ations are powered by absorbed UV photons (χ_H from the radiation field,
1158 // not the plasma); radiative recombination radiates χ_H away (case B,
1159 // optically thin). Thermal/KE inheritance applies to all new ions.
1160 const Vec dE_i = Gamma_ion % (th_inherit + KE_inherit)
1161 - Gamma_rec % (th_leave + KE_leave)
1162 - (Gamma_coll + Gamma_mlvl - Gamma_rec_3b) * grid.chi_H_J;
1163 const Vec dE_n = -Gamma_ion % (th_inherit + KE_inherit)
1164 + Gamma_rec % (th_leave + KE_leave);
1165
1166 // Thermal-energy update: Δε = (Q_ion · Δt) − ΔKE. The boxed Q^{e_i}_ion
1167 // accounts only for thermal+KE flux between species (Meier & Shumlak 2012);
1168 // gravity contributes Δρ_s · φ_g to each species' conserved energy through
1169 // the mass conservation, which is "tracked silently" via ρ_new φ_g in
1170 // prim2cons() and does *not* enter the thermal update.
1171 const Vec eps_i_new = eps_i_old + dE_i - (KE_i_new - KE_i_old);
1172 const Vec eps_n_new = eps_n_old + dE_n - (KE_n_new - KE_n_old);
1173
1174 // Floor pressures at a small positive value if χ_H drain exceeded the
1175 // available ion thermal energy. This is a non-conservative regularization
1176 // (writeup §5.3); the lost energy is the radiative-heating term that a
1177 // future PR will supply.
1178 const float p_floor = 1.0e-15f;
1179 Vec p_i_new = grid.gm1() * eps_i_new;
1180 Vec p_n_new = grid.gm1() * eps_n_new;
1181 for (arma::uword i = 0; i < grid.ns; ++i) {
1182 if (p_i_new(i) < p_floor) p_i_new(i) = p_floor;
1183 if (p_n_new(i) < p_floor) p_n_new(i) = p_floor;
1184 }
1185
1186 // Electron pressure. Single-T baseline: carry it (re-slaved each step). Three-T:
1187 // the χ_H ionization cost is an ELECTRON-pool term (collisional ionization drains
1188 // χ_H from the free electrons; three-body recombination returns it) — already in
1189 // the total-energy dE_i above, so attributing it to P_E leaves the protons
1190 // (p_proton = P_I − P_E) carrying only the thermal/KE exchange with the neutrals.
1191 Vec p_e_new = p_e_old;
1192 if (grid.enable_Te) {
1193 const Vec chi_eps = -(Gamma_coll + Gamma_mlvl - Gamma_rec_3b) * grid.chi_H_J;
1194 p_e_new = p_e_old + grid.gm1() * chi_eps;
1195 for (arma::uword i = 0; i < grid.ns; ++i) {
1196 if (p_e_new(i) < p_floor) p_e_new(i) = p_floor;
1197 if (p_e_new(i) > p_i_new(i) - p_floor) p_e_new(i) = p_i_new(i) - p_floor;
1198 }
1199 }
1200
1201 prim_state.zeros();
1202 prim_state += scalar_to(grid, rho_i_new, prim::RHO_I);
1203 prim_state += scalar_to(grid, rho_n_new, prim::RHO_N);
1204 prim_state += scalar_to(grid, V_new, prim::V);
1205 prim_state += scalar_to(grid, U_new, prim::U);
1206 prim_state += scalar_to(grid, p_i_new, prim::P_I);
1207 prim_state += scalar_to(grid, p_n_new, prim::P_N);
1208 prim_state += scalar_to(grid, p_e_new, prim::P_E);
1209}

1211// Equilibrium-reference well-balancing (chromosphere.hpp::eq_wb): on the first
1212// step, cache R_eq = rhs_explicit_state(eq_state) so it can be subtracted every
1213// step, making eq_state an exact discrete fixed point. Computed here (not in the
1214// const rhs_explicit_state) because it mutates grid; the empty() guard inside
1215// rhs_explicit_state ensures this capture is itself uncorrected. dt_state is set
1216// by broadcast_dt before this runs; R_eq's residual dt-dependence is 2nd-order
1217// small at equilibrium (the predictor half-step is a near-no-op at balance).
1218static void ensure_eq_residual(Grid& grid) {
1219 if (grid.eq_wb && grid.eq_residual.is_empty() && !grid.eq_state.is_empty()) {
1220 grid.eq_residual = rhs_explicit_state(grid, grid.eq_state);
1221 }
1222}

1224// ----------------------------------------------------------------------------
1225// Backward-Euler split integrator (writeup §3.7, §5.3). The operational order is
1226// A: explicit MUSCL/Rusanov hydro RHS; B: drag; C: ion-neutral temperature
1227// equilibration; D: implicit conduction; E: ionization (when enabled); then
1228// optional beam/coronal heating, radiative cooling, and floors/electron slaving.
1229// In particular, Stage D changes pressure at fixed velocity and the next Stage-A
1230// RHS differentiates that updated pressure field.
1231// ----------------------------------------------------------------------------
1232Vec advance_Euler_state(Grid& grid, const Vec& xn_state, const Vec& dt_i) {
1233 if (!grid.eos_gamma_table.empty())
1234 throw std::logic_error(
1235 "the legacy two-fluid integrator cannot advance a Gamma1-table Grid; "
1236 "use mixture_advance");
1237 broadcast_dt(grid, dt_i);
1238 ensure_eq_residual(grid);
1239
1240 Vec U_star = xn_state + grid.dt_state % rhs_explicit_state(grid, xn_state);
1241 Vec prim = cons2prim(grid, U_star);
1242
1243 const float dt = dt_i(0); // uniform by cal_dt_i construction
1244 // Flare: sanitize the explicit predictor before any source stage reads it.
1245 // The MUSCL update can drive p_i/p_n (hence T) negative at the steep
1246 // evaporation front on the coarse grid, which would feed a negative T into
1247 // the conduction tridiagonal and produce NaN. The floor clips densities and
1248 // pressures positive and slaves the trace neutral; no-op for steady runs.
1249 if (grid.floors_active()) apply_flare_floor_stage(grid, prim);
1250 // TRAC: recompute the adaptive cutoff T_c from the current profile so the
1251 // conduction and cooling stages broaden the (currently) unresolved TR.
1252 if (grid.enable_trac) grid.trac_cutoff_T = compute_trac_cutoff_T(grid, prim);
1253 apply_drag_stage(grid, prim, dt);
1254 apply_temperature_stage(grid, prim, dt);
1255 // Stage D heat conduction. Default-on; the isentropic relaxation experiment
1256 // (model_column Stage 1) disables it so the adiabatic atmosphere stays a
1257 // conduction-free Euler steady state (docs/gentle_evaporation_downflow.md).
1258 if (grid.enable_conduction) {
1259 apply_conduction_stage(grid, prim, dt);
1260 }
1261 if (grid.enable_ionization) {
1262 apply_ionization_stage(grid, prim, dt);
1263 }
1264 // Beam heating BEFORE cooling so the heating-vs-radiation competition that
1265 // defines the explosive/gentle threshold (Fisher et al. 1985) is captured:
1266 // the cooling stage then radiates whatever fraction of the deposited heat
1267 // the gas can.
1268 if (grid.enable_beam_heating) {
1269 apply_beam_heating_stage(grid, prim, dt);
1270 }
1271 // Ambient coronal heating BEFORE cooling, like the beam: the RTV balance is
1272 // the competition between this heating, conduction, and the radiative sink.
1273 // Always-on for a steady run (no temporal gate beyond the optional Phase-3
1274 // ramp inside coronal_heating_rate); no-op when enable_coronal_heating is off.
1275 if (grid.enable_coronal_heating) {
1276 apply_coronal_heating_stage(grid, prim, dt);
1277 }
1278 if (grid.enable_radiative_cooling) {
1279 apply_radiative_cooling_stage(grid, prim, dt);
1280 }
1281 // Vacuum floor / trace-neutral slaving — keeps the fully-ionized evaporated
1282 // column well-conditioned (ρ_n→0). No-op for steady scenarios.
1283 if (grid.floors_active()) {
1284 apply_flare_floor_stage(grid, prim);
1285 }
1286
1287 // Single-temperature limit: slave the electron pool to half the charged
1288 // thermal energy (p_e = ½ p_total ⇒ T_e ≡ T_i = T_charged). E_E is the 7th
1289 // variable that is always carried but, with ENABLE_TE=0, never feeds back —
1290 // this overwrite makes the reported T_e equal T_i to round-off and recovers
1291 // the pre-T_e single-temperature baseline exactly.
1292 if (!grid.enable_Te) {
1293 const Vec p_i = get_scalar(grid, prim, prim::P_I);
1294 const Vec p_e = get_scalar(grid, prim, prim::P_E);
1295 prim += scalar_to(grid, 0.5f * p_i - p_e, prim::P_E);
1296 }
1297
1298 return prim2cons(grid, prim);
1299}
