1/*!
2 * @file single_fluid/mixture.cpp
3 * @brief Release solver: state decode, MUSCL reconstruction, the mixture Roe
4 * numerical flux, the geometric source, and the CFL timestep.
5 * @ingroup release_solver
6 *
7 * See single_fluid/mixture.hpp for the governing equations and the software
8 * contract. Nothing here reaches into the two-fluid research solver.
9 *
10 * One explicit RHS evaluation, in order:
11 *
12 * 1. reconstruct `(log rho, u, log p)` with the asymmetric MC3/Koren limiter
13 * (beta = 2) on the non-uniform-mesh-aware slope metrics;
14 * 2. take a MUSCL-Hancock predictor half step that applies the SAME momentum
15 * source (flux-tube pressure term plus gravity) as the corrector;
16 * 3. build both sides of every face through the authoritative equilibrium face
17 * builder, so a face state is on the Saha manifold by construction;
18 * 4. evaluate the 3x3 mixture Roe characteristic flux, falling back per face to
19 * local Rusanov whenever the Roe average is not admissible;
20 * 5. difference the fluxes with the flux-tube area factor and add the source.
21 *
22 * The interior operator is reference-free: it needs no frozen global hydrostatic
23 * reference state. `Grid::eq_wb` exists for the two-fluid research solver only.
24 */

47// Reconstruction slots. Slot 2 carries the THERMAL variable: log(p) with the
48// release reconstruction, log(T) with the reference one. The packing, limiter
49// arithmetic and extrapolation are identical either way; only the decode (which
50// quantity is written) and the face builder (how the state is closed) branch.
51enum MixtureSlot : arma::uword { MIX_LOG_RHO = 0, MIX_U = 1, MIX_THERMAL = 2 };

59// Neumann-mirrored neighbour shift of a cell-centred STATIC mesh metric:
60// shift_up(v)(i) = v(i+1) in the interior and v(ns-1) at the top face;
61// shift_down mirrors at the base. Only mesh metrics are ever shifted this way,
62// so no ghost data is involved and the release solver needs none of the
63// two-fluid packed-state helpers.
64Vec shift_up(const Vec& v) {
65 Vec out(arma::size(v));
66 const arma::uword n = v.n_elem;
67 for (arma::uword i = 0; i + 1 < n; ++i) out(i) = v(i + 1);
68 out(n - 1) = v(n - 1);
69 return out;
70}
71
72Vec shift_down(const Vec& v) {
73 Vec out(arma::size(v));
74 const arma::uword n = v.n_elem;
75 for (arma::uword i = 1; i < n; ++i) out(i) = v(i - 1);
76 out(0) = v(0);
77 return out;
78}

86void MixtureField::require_matches(const Grid& grid, const Vec& state) const {
87 if (source_grid != &grid || source_state != &state
88 || source_memory != state.memptr()
89 || source_elements != state.n_elem)
90 throw std::logic_error(
91 "mixture field cache does not match the immutable conserved state");
92 const Vec* boundaries[] = {&grid.mix_inner_boundary0, &grid.mix_inner_boundary1,
93 &grid.mix_outer_boundary0, &grid.mix_outer_boundary1};
94 for (arma::uword block = 0; block < 4; ++block)
95 for (arma::uword row = 0; row < num_of_mixture_eq; ++row)
96 if (boundary_signature[block*num_of_mixture_eq+row] != (*boundaries[block])(row))
97 throw std::logic_error(
98 "mixture field cache was invalidated by a boundary update");
99}

109void mixture_decode_into(const Grid& grid, const Vec& state, MixtureField& out,
110 std::uint64_t generation, const MixtureField* previous) {
111 ProfileScope timer(ProfileRegion::Decode);
112 if (grid.eos_gamma_table.empty())
113 throw std::logic_error("the mixture solver requires a Gamma1 table");
114 if (state.n_elem != grid.n_mixture_state)
115 throw std::invalid_argument("mixture state size mismatch");
116
117 out.source_grid = &grid;
118 out.source_state = &state;
119 out.source_memory = state.memptr();
120 out.source_elements = state.n_elem;
121 out.state_generation = generation;
122 const Vec* boundaries[] = {&grid.mix_inner_boundary0, &grid.mix_inner_boundary1,
123 &grid.mix_outer_boundary0, &grid.mix_outer_boundary1};
124 for (arma::uword block = 0; block < 4; ++block)
125 for (arma::uword row = 0; row < num_of_mixture_eq; ++row)
126 out.boundary_signature[block*num_of_mixture_eq+row] = (*boundaries[block])(row);
127 out.cells.resize(grid.ns);
128 const arma::uword ext_n = grid.ns+4;
129 out.extended_primitive.set_size(ext_n*num_of_mixture_eq);
130 out.extended_temperature.set_size(ext_n);
131 const auto sz = state_size(grid);
132 const auto ext_size = arma::size(ext_n, num_of_mixture_eq);
133 const bool log_p = grid.pressure_reconstruct;
134 auto put = [&](arma::uword ext_i, double rho, double momentum,
135 double pressure, double temperature) {
136 out.extended_primitive(arma::sub2ind(ext_size, ext_i, MIX_LOG_RHO)) =
137 std::log(rho);
138 out.extended_primitive(arma::sub2ind(ext_size, ext_i, MIX_U)) =
139 momentum/rho;
140 out.extended_primitive(arma::sub2ind(ext_size, ext_i, MIX_THERMAL)) =
141 log_p ? std::log(pressure) : std::log(temperature);
142 out.extended_temperature(ext_i) = temperature;
143 };
144 const bool has_guesses = previous && previous->source_grid == &grid
145 && previous->cells.size() == grid.ns;
146 parallel_for_cells(grid.ns, [&](std::size_t raw_i) {
147 const arma::uword i = static_cast<arma::uword>(raw_i);
148 auto at = [&](arma::uword row) {
149 return static_cast<double>(state(arma::sub2ind(sz, i, row)));
150 };
151 const double guess = has_guesses ? previous->cells[i].T
152 : std::numeric_limits<double>::quiet_NaN();
153 out.cells[i] = decode_equilibrium_mixture(
154 grid.eos_gamma_table, at(mix::RHO), at(mix::MOM), at(mix::ENERGY),
155 mixture_cell_phi(grid, i), guess, grid.eos_gamma_debug_clamp);
156 // Refresh the shared hint, but do NOT read it here: this site has its own
157 // guess (`previous`), validated against source_grid and cell count. Each
158 // worker writes only its own pre-sized hint slot.
159 grid.store_eos_temperature_hint(i, out.cells[i].T);
160 const MixtureThermo& th = out.cells[i];
161 put(i+2, th.rho, at(mix::MOM), th.p, th.T);
162 });
163
164 auto decode_ghost = [&](const Vec& ghost, double phi, arma::uword ext_i) {
165 const MixtureThermo th = decode_equilibrium_mixture(
166 grid.eos_gamma_table, ghost(mix::RHO), ghost(mix::MOM),
167 ghost(mix::ENERGY), phi, std::numeric_limits<double>::quiet_NaN(),
168 grid.eos_gamma_debug_clamp);
169 put(ext_i, th.rho, static_cast<double>(ghost(mix::MOM)), th.p, th.T);
170 };
171 const double phi_inner = grid.phi_g_imh(0);
172 const double phi_outer = grid.phi_g_iph(grid.ns-1);
173 decode_ghost(grid.mix_inner_boundary1, phi_inner, 0);
174 decode_ghost(grid.mix_inner_boundary0, phi_inner, 1);
175 decode_ghost(grid.mix_outer_boundary0, phi_outer, grid.ns+2);
176 decode_ghost(grid.mix_outer_boundary1, phi_outer, grid.ns+3);
177}

179void mixture_pack_ghost(const Grid& grid, Vec& ghost, double rho,
180 double velocity, double temperature, double phi) {
181 if (ghost.n_elem != num_of_mixture_eq) ghost.zeros(num_of_mixture_eq);
182 grid.eos_gamma_table.require_n_h_in_bounds(
183 rho/eos_constants::m_h, grid.eos_gamma_debug_clamp);
184 const double e_int = equilibrium_internal_energy(rho, temperature);
185 ghost(mix::RHO) = static_cast<float>(rho);
186 ghost(mix::MOM) = static_cast<float>(rho*velocity);
187 ghost(mix::ENERGY) = static_cast<float>(
188 mixture_total_energy(rho, rho*velocity, e_int, phi));
189}

207void stencil_view(const Grid& grid, const MixtureField& decoded, int offset,
208 Work& result) {
209 result.set_size(grid.ns*num_of_mixture_eq);
210 const arma::uword ext_n = grid.ns+4;
211 const auto ext_size = arma::size(ext_n, num_of_mixture_eq);
212 const auto packed = state_size(grid);
213 for (arma::uword slot = 0; slot < num_of_mixture_eq; ++slot)
214 for (arma::uword i = 0; i < grid.ns; ++i) {
215 const arma::uword ext_i =
216 static_cast<arma::uword>(static_cast<int>(i)+2+offset);
217 result(arma::sub2ind(packed, i, slot)) =
218 decoded.extended_primitive(arma::sub2ind(ext_size, ext_i, slot));
219 }
220}

242 result.set_size(ratio.n_elem);

249void limit_mc3(const Work& ratio, double beta, bool plus, Work& result) {
250 result.set_size(ratio.n_elem);
251 for (arma::uword i = 0; i < ratio.n_elem; ++i) {
252 const double r = ratio(i);
253 if (!std::isfinite(r)) { result(i) = 0.0; continue; }
254 const double third_order = plus ? (2.0*r + 1.0)/3.0 : (r + 2.0)/3.0;
255 result(i) = std::max(0.0, std::min(std::min(beta*r, beta), third_order));
256 }

259// Build one one-sided face state per cell from the reconstructed variables.
260void build_face_cell(const Grid& grid, const Work& w, const Vec& phi_face,
261 MixtureFaceArrays& out, arma::uword i,
262 const Work* temperature_hint) {
263 const auto sz = state_size(grid);
264 const double log_rho = w(arma::sub2ind(sz, i, MIX_LOG_RHO));
265 const double velocity = w(arma::sub2ind(sz, i, MIX_U));
266 const double thermal = w(arma::sub2ind(sz, i, MIX_THERMAL));
267 const MixtureFaceState face = grid.pressure_reconstruct
268 ? equilibrium_mixture_face_state_from_log_pressure(
269 grid.eos_gamma_table, log_rho, velocity, thermal,
270 static_cast<double>(phi_face(i)), grid.eos_gamma_debug_clamp,
271 temperature_hint ? (*temperature_hint)(i)
272 : std::numeric_limits<double>::quiet_NaN())
273 : equilibrium_mixture_face_state_from_logs(
274 grid.eos_gamma_table, log_rho, velocity, thermal,
275 static_cast<double>(phi_face(i)), grid.eos_gamma_debug_clamp);
276 const std::array<double, num_of_mixture_eq> flux = equilibrium_mixture_flux(face);
277 const double conserved[num_of_mixture_eq] = {
278 face.rho, face.rho*face.velocity, face.energy};
279 for (arma::uword k = 0; k < num_of_mixture_eq; ++k) {
280 out.conserved(arma::sub2ind(sz, i, k)) = conserved[k];
281 out.flux(arma::sub2ind(sz, i, k)) = flux[k];
282 }
283 out.velocity(i) = face.velocity;
284 out.pressure(i) = face.pressure;
285 out.sound_speed(i) = face.sound_speed;
286 out.dp_deint(i) = face.dp_deint_rho;
287}

306// Face-local Rusanov (local Lax-Friedrichs) numerical flux.
307inline void rusanov_face(const Grid& grid, const MixtureFaceArrays& left,
308 const MixtureFaceArrays& right, arma::uword i,
309 double a, Work& output) {
310 const auto sz = state_size(grid);
311 for (arma::uword row = 0; row < num_of_mixture_eq; ++row) {
312 const arma::uword k = arma::sub2ind(sz, i, row);
313 output(k) = 0.5*(left.flux(k) + right.flux(k)
314 - a*(right.conserved(k) - left.conserved(k)));
315 }
316}

318// Mixture Roe characteristic flux on U = (rho, rho u, E).
319//
320// With the Roe-averaged velocity u~, total enthalpy H~ and the general-EOS
321// coefficients b = (dp/de_int)_rho and c~ (the density-weighted acoustic speed),
322// the wave strengths of the three-field decomposition are
323//
324// theta = (b/c~^2)[dE - u~ d(rho u) + (u~^2 - H~) drho]
325// alpha_+/- = 1/2 [drho + theta +/- (d(rho u) - u~ drho)/c~]
326// alpha_0 = -theta
327//
328// and the dissipation is sum_k |lambda_k| alpha_k r_k with
329// r_-/+ = (1, u~ -/+ c~, H~ -/+ u~ c~) and r_0 = (1, u~, H~ - c~^2/b).
330// This is a local characteristic average, not a claim of exact general-EOS Roe
331// Property U; any inadmissible average falls back face-locally to Rusanov.
332void build_roe_flux(const Grid& grid, const MixtureFaceArrays& left,
333 const MixtureFaceArrays& right, const Work& a_face,
334 Work& output) {
335 const auto sz = state_size(grid);
336 for (arma::uword i = 0; i < grid.ns; ++i) {
337 const arma::uword k_rho = arma::sub2ind(sz, i, mix::RHO);
338 const arma::uword k_mom = arma::sub2ind(sz, i, mix::MOM);
339 const arma::uword k_e = arma::sub2ind(sz, i, mix::ENERGY);
340
341 const double rho_l = left.conserved(k_rho), rho_r = right.conserved(k_rho);
342 const double mom_l = left.conserved(k_mom), mom_r = right.conserved(k_mom);
343 const double energy_l = left.conserved(k_e), energy_r = right.conserved(k_e);
344 const double p_l = left.pressure(i), p_r = right.pressure(i);
345 const double c_l = left.sound_speed(i), c_r = right.sound_speed(i);
346 const double b_l = left.dp_deint(i), b_r = right.dp_deint(i);
347 if (!(rho_l > 0.0) || !(rho_r > 0.0) || !(p_l > 0.0) || !(p_r > 0.0)
348 || !(c_l > 0.0) || !(c_r > 0.0) || !(b_l > 0.0) || !(b_r > 0.0)
349 || !std::isfinite(energy_l) || !std::isfinite(energy_r)) {
350 rusanov_face(grid, left, right, i, a_face(i), output);
351 continue;
352 }
353
354 const double v_l = mom_l/rho_l, v_r = mom_r/rho_r;
355 const double h_l = (energy_l + p_l)/rho_l;
356 const double h_r = (energy_r + p_r)/rho_r;
357 const double sr_l = std::sqrt(rho_l), sr_r = std::sqrt(rho_r);
358 const double denom = sr_l + sr_r;
359 const double velocity = (sr_l*v_l + sr_r*v_r)/denom;
360 const double enthalpy = (sr_l*h_l + sr_r*h_r)/denom;
361 const double b = (sr_l*b_l + sr_r*b_r)/denom;
362 const double c2 = (sr_l*c_l*c_l + sr_r*c_r*c_r)/denom;
363 if (!(b > 0.0) || !(c2 > 0.0) || !std::isfinite(enthalpy)) {
364 rusanov_face(grid, left, right, i, a_face(i), output);
365 continue;
366 }
367 const double c = std::sqrt(c2);
368
369 const double d_rho = rho_r - rho_l;
370 const double d_mom = mom_r - mom_l;
371 const double d_energy = energy_r - energy_l;
372 const double theta = b/c2*(d_energy - velocity*d_mom
373 + (velocity*velocity - enthalpy)*d_rho);
374 const double acoustic = (d_mom - velocity*d_rho)/c;
375 const double alpha_minus = 0.5*(d_rho + theta - acoustic);
376 const double alpha_zero = -theta;
377 const double alpha_plus = 0.5*(d_rho + theta + acoustic);
378 const double lambda_minus = std::abs(velocity - c);
379 const double lambda_zero = std::abs(velocity);
380 const double lambda_plus = std::abs(velocity + c);
381
382 const double dissipation[num_of_mixture_eq] = {
383 lambda_minus*alpha_minus + lambda_zero*alpha_zero
384 + lambda_plus*alpha_plus,
385 lambda_minus*alpha_minus*(velocity - c) + lambda_zero*alpha_zero*velocity
386 + lambda_plus*alpha_plus*(velocity + c),
387 lambda_minus*alpha_minus*(enthalpy - velocity*c)
388 + lambda_zero*alpha_zero*(enthalpy - c2/b)
389 + lambda_plus*alpha_plus*(enthalpy + velocity*c)};
390 if (!std::isfinite(dissipation[0]) || !std::isfinite(dissipation[1])
391 || !std::isfinite(dissipation[2])) {
392 rusanov_face(grid, left, right, i, a_face(i), output);
393 continue;
394 }
395 for (arma::uword row = 0; row < num_of_mixture_eq; ++row) {
396 const arma::uword k = arma::sub2ind(sz, i, row);
397 output(k) = 0.5*(left.flux(k) + right.flux(k) - dissipation[row]);
398 }
399 }
400}

402// Momentum source of the field-aligned PDE: the flux-tube pressure term
403// p d_s(ln A) written in the code's B form (A B = const => d_s(ln A) =
404// -d_s(ln B) = B d_s(1/B)) plus the gravitational body force -rho d_s(phi_g).
405// The energy row has no source because E carries the gravitational potential,
406// so the gravitational work is already inside the (E + p) u flux.
407//
408// This is the ONE source the hydrodynamic update uses, and both stages of the
409// MUSCL-Hancock step apply it: the Hancock predictor with weight dt_predictor
410// and the corrector inside the final RHS.
411void build_source(const Grid& grid, const Vec& state,
412 const MixtureField& decoded, Work& source) {
413 source.zeros(grid.ns*num_of_mixture_eq);
414 const auto sz = state_size(grid);
415 for (arma::uword i = 0; i < grid.ns; ++i) {
416 const double g_par = -static_cast<double>(
417 (grid.phi_g_iph(i) - grid.phi_g_imh(i))/grid.ds_i(i));
418 source(arma::sub2ind(sz, i, mix::MOM)) =
419 decoded.cells[i].p * static_cast<double>(grid.B_i(i))
420 * static_cast<double>(grid.dinvB_ds_i(i))
421 + static_cast<double>(state(arma::sub2ind(sz, i, mix::RHO)))*g_par;
422 }
423}

425// Read-only copy-out of the production face reconstruction and the numerical
426// mass flux actually used at the i+1/2 faces. Reads the same arrays the
427// continuity row is differenced from; writes nothing the solver consumes.
428void capture_face_flux(const Grid& grid, const Work& w, const Work& wl,
429 const Work& wr, const MixtureFaceArrays& fl,
430 const MixtureFaceArrays& fr, const Work& a_face,
431 const Work& flux, const Work& r, const Work& r_ip1,
432 const Work& lp_r, const Work& lm_rip1) {
433 MixtureFaceFluxCapture& c = grid.face_flux_capture;
434 c.resize(grid.ns);
435 const auto sz = state_size(grid);
436 for (arma::uword i = 0; i < grid.ns; ++i) {
437 auto W = [&](const Work& v, arma::uword slot) {
438 return v(arma::sub2ind(sz, i, slot));
439 };
440 c.rho_cell[i] = std::exp(W(w, MIX_LOG_RHO));
441 c.v_cell[i] = W(w, MIX_U);
442 c.rho_L[i] = std::exp(W(wl, MIX_LOG_RHO));
443 c.rho_R[i] = std::exp(W(wr, MIX_LOG_RHO));
444 c.v_L[i] = W(wl, MIX_U);
445 c.v_R[i] = W(wr, MIX_U);
446 // Both (T, p) pairs, whichever slot 2 actually carried. The missing member
447 // of the pair comes from the SAME authoritative closure the solver used for
448 // this face, so the mismatch diagnostic is directly comparable between the
449 // two reconstructions. Capture steps only.
450 auto thermal_pair = [&](const Work& v, double rho, double& temperature,
451 double& pressure) {
452 const double slot = W(v, MIX_THERMAL);
453 if (grid.pressure_reconstruct) {
454 pressure = std::exp(slot);
455 temperature = equilibrium_temperature_from_density_pressure(
456 rho, pressure);
457 } else {
458 temperature = std::exp(slot);
459 pressure = equilibrium_caloric_state(rho, temperature).pressure;
460 }
461 };
462 thermal_pair(w, c.rho_cell[i], c.T_cell[i], c.p_cell[i]);
463 thermal_pair(wl, c.rho_L[i], c.T_L[i], c.p_L[i]);
464 thermal_pair(wr, c.rho_R[i], c.T_R[i], c.p_R[i]);
465 c.cs_L[i] = fl.sound_speed(i);
466 c.cs_R[i] = fr.sound_speed(i);
467 c.a_face[i] = a_face(i);
468 const arma::uword k_rho = arma::sub2ind(sz, i, mix::RHO);
469 c.f_central[i] = 0.5*(fl.flux(k_rho) + fr.flux(k_rho));
470 c.f_total[i] = flux(k_rho);
471 c.f_diff[i] = grid.roe_characteristic_flux
472 ? c.f_total[i] - c.f_central[i]
473 : -0.5*c.a_face[i]*(fr.conserved(k_rho) - fl.conserved(k_rho));
474 c.r_rho[i] = W(r, MIX_LOG_RHO);
475 c.phi_plus_rho[i] = W(lp_r, MIX_LOG_RHO);
476 c.r_ip1_rho[i] = W(r_ip1, MIX_LOG_RHO);
477 c.phi_minus_rho[i] = W(lm_rip1, MIX_LOG_RHO);
478 c.r_v[i] = W(r, MIX_U);
479 c.phi_plus_v[i] = W(lp_r, MIX_U);
480 // Slot-2 limiter inputs/outputs: log(p) in the release reconstruction.
481 c.r_T[i] = W(r, MIX_THERMAL);
482 c.phi_plus_T[i] = W(lp_r, MIX_THERMAL);
483 }
484 c.valid = true;
485}

487// Predictor decode: caloric only (the half step never needs the acoustic Gamma1).
488void decode_predictor_into(const Grid& grid, const Vec& state,
489 const MixtureField& previous, MixtureField& output) {
490 if (state.n_elem != grid.n_mixture_state)
491 throw std::invalid_argument("mixture predictor state size mismatch");
492 output.cells.clear();
493 const arma::uword ext_n = grid.ns+4;
494 output.extended_primitive.set_size(ext_n*num_of_mixture_eq);
495 output.extended_temperature.set_size(ext_n);
496 const auto sz = state_size(grid);
497 const auto ext_size = arma::size(ext_n, num_of_mixture_eq);
498 const bool log_p = grid.pressure_reconstruct;
499 auto put = [&](arma::uword ext_i, const CaloricMixtureThermo& th,
500 double momentum) {
501 output.extended_primitive(arma::sub2ind(ext_size, ext_i, MIX_LOG_RHO)) =
502 std::log(th.rho);
503 output.extended_primitive(arma::sub2ind(ext_size, ext_i, MIX_U)) =
504 momentum/th.rho;
505 output.extended_primitive(arma::sub2ind(ext_size, ext_i, MIX_THERMAL)) =
506 log_p ? std::log(th.p) : std::log(th.T);
507 output.extended_temperature(ext_i) = th.T;
508 };
509 parallel_for_cells(grid.ns, [&](std::size_t raw_i) {
510 const arma::uword i = static_cast<arma::uword>(raw_i);
511 auto at = [&](arma::uword row) {
512 return static_cast<double>(state(arma::sub2ind(sz, i, row)));
513 };
514 const CaloricMixtureThermo th = decode_equilibrium_caloric_mixture(
515 grid.eos_gamma_table, at(mix::RHO), at(mix::MOM), at(mix::ENERGY),
516 mixture_cell_phi(grid, i), previous.cells[i].T,
517 grid.eos_gamma_debug_clamp);
518 grid.store_eos_temperature_hint(i, th.T);
519 put(i+2, th, at(mix::MOM));
520 });
521 auto decode_ghost = [&](const Vec& ghost, double phi, arma::uword ext_i) {
522 const CaloricMixtureThermo th = decode_equilibrium_caloric_mixture(
523 grid.eos_gamma_table, ghost(mix::RHO), ghost(mix::MOM),
524 ghost(mix::ENERGY), phi, std::numeric_limits<double>::quiet_NaN(),
525 grid.eos_gamma_debug_clamp);
526 put(ext_i, th, static_cast<double>(ghost(mix::MOM)));
527 };
528 const double phi_inner = grid.phi_g_imh(0);
529 const double phi_outer = grid.phi_g_iph(grid.ns-1);
530 decode_ghost(grid.mix_inner_boundary1, phi_inner, 0);
531 decode_ghost(grid.mix_inner_boundary0, phi_inner, 1);
532 decode_ghost(grid.mix_outer_boundary0, phi_outer, grid.ns+2);
533 decode_ghost(grid.mix_outer_boundary1, phi_outer, grid.ns+3);
534}

538Vec mixture_rhs_explicit(const Grid& grid, const Vec& state,
539 const MixtureField& decoded, double dt_predictor) {
540 ProfileScope timer(ProfileRegion::Rhs);
541 decoded.require_matches(grid, state);
542 MixtureRhsScratch& scratch = grid.mixture_rhs_scratch;
543 auto& m = scratch.work;
544 Work& w=m[0]; Work& w_ip1=m[1]; Work& w_im1=m[2];
545 Work& w_ip2=m[3]; Work& w_im2=m[4];
546 Work& dw_iph=m[5]; Work& dw_imh=m[6];
547 Work& r=m[7]; Work& r_ip1=m[8]; Work& r_im1=m[9];
548 Work& t_hint_0=m[10]; Work& t_hint_ip1=m[11]; Work& t_hint_im1=m[12];
549 Work& wr_iph=m[13]; Work& wl_iph=m[14];
550 Work& wr_imh=m[15]; Work& wl_imh=m[16];
551 Work& wt=m[17]; Work& wt_ip1=m[18]; Work& wt_im1=m[19];
552 Work& lp_r=m[20]; Work& lm_r=m[21];
553 Work& lm_rip1=m[22]; Work& lp_rim1=m[23];
554
555 // Parent-cell temperature hints for the (rho,p)->T face inversion. Built only
556 // when the pressure reconstruction is active; the log(T) path never inverts.
557 const Work* h0 = nullptr;
558 const Work* h_ip1 = nullptr;
559 const Work* h_im1 = nullptr;
560 if (grid.pressure_reconstruct) {
561 temperature_stencil_view(grid, decoded, 0, t_hint_0);
562 temperature_stencil_view(grid, decoded, 1, t_hint_ip1);
563 temperature_stencil_view(grid, decoded, -1, t_hint_im1);
564 h0 = &t_hint_0; h_ip1 = &t_hint_ip1; h_im1 = &t_hint_im1;
565 }
566 stencil_view(grid, decoded, 0, w);
567 stencil_view(grid, decoded, 1, w_ip1);
568 stencil_view(grid, decoded, -1, w_im1);
569 stencil_view(grid, decoded, 2, w_ip2);
570 stencil_view(grid, decoded, -2, w_im2);
571
572 dw_iph = w_ip1 - w;
573 dw_imh = w - w_im1;
574 r = dw_imh/dw_iph;
575 r_ip1 = dw_iph/(w_ip2 - w_ip1);
576 r_im1 = (w_im1 - w_im2)/dw_imh;
577
578 // Non-uniform reconstruction weights and slope-ratio metrics depend only on
579 // the static mesh, so they are built once per mesh generation.
580 const Work& W1 = scratch.W1;
581 const Work& W3 = scratch.W3;
582 const Work& W4 = scratch.W4;
583 if (!grid.uniform_mesh) {
584 if (!scratch.weights_valid
585 || scratch.weights_generation != grid.metrics_generation()
586 || scratch.W1.n_elem != grid.ns*num_of_mixture_eq) {
587 const Vec w1_i = 0.5f*grid.ds_i/grid.ds_iph_i;
588 broadcast_slots(grid, w1_i, scratch.W1);
589 broadcast_slots(grid, shift_up(w1_i), scratch.W3);
590 broadcast_slots(grid, shift_down(w1_i), scratch.W4);
591 broadcast_slots(grid, grid.ds_iph_i/grid.ds_imh_i, scratch.metric_r);
592 broadcast_slots(grid, shift_up(grid.ds_iph_i)/grid.ds_iph_i,
593 scratch.metric_r_ip1);
594 broadcast_slots(grid, grid.ds_imh_i/shift_down(grid.ds_imh_i),
595 scratch.metric_r_im1);
596 scratch.weights_generation = grid.metrics_generation();
597 scratch.weights_valid = true;
598 }
599 r %= scratch.metric_r;
600 r_ip1 %= scratch.metric_r_ip1;
601 r_im1 %= scratch.metric_r_im1;
602 }
603 r.elem(arma::find_nonfinite(r)).zeros();
604 r_ip1.elem(arma::find_nonfinite(r_ip1)).zeros();
605 r_im1.elem(arma::find_nonfinite(r_im1)).zeros();
606
607 const double beta = grid.limiter_beta;
608 auto lim_plus = [&](const Work& q, Work& out) {
609 if (grid.mc3_limiter) limit_mc3(q, beta, true, out);
610 else limit_minmod(q, out);
611 };
612 auto lim_minus = [&](const Work& q, Work& out) {
613 if (grid.mc3_limiter) limit_mc3(q, beta, false, out);
614 else limit_minmod(q, out);
615 };
616 lim_plus(r, lp_r); lim_minus(r, lm_r);
617 lim_minus(r_ip1, lm_rip1); lim_plus(r_im1, lp_rim1);
618
619 // First (predictor) pass. The predictor update differences ONLY
620 // fl_iph.flux - fr_imh.flux, so only those two reconstructions are needed.
621 if (grid.uniform_mesh) {
622 wl_iph = w + 0.5*lp_r%(w_ip1 - w);
623 wr_imh = w - 0.5*lm_r%(w_ip1 - w);
624 } else {
625 wl_iph = w + W1%lp_r%(w_ip1 - w);
626 wr_imh = w - W1%lm_r%(w_ip1 - w);
627 }
628
629 MixtureFaceArrays& fr_iph = scratch.faces[0];
630 MixtureFaceArrays& fl_iph = scratch.faces[1];
631 MixtureFaceArrays& fr_imh = scratch.faces[2];
632 MixtureFaceArrays& fl_imh = scratch.faces[3];
633 {
634 // Both predictor faces are extrapolated from cell i, so cell i's
635 // temperature is the natural Newton seed for both.
636 const FaceRequest requests[2] = {
637 {&wl_iph, &grid.phi_g_iph, &fl_iph, h0},
638 {&wr_imh, &grid.phi_g_imh, &fr_imh, h0}};
639 build_faces(grid, requests, 2);
640 }
641
642 // The ONE momentum source of the PDE (flux-tube pressure term + gravity),
643 // evaluated at U^n. Both MUSCL-Hancock stages consume it: the predictor just
644 // below and the corrector when the final RHS is assembled.
645 Work& source = scratch.source;
646 build_source(grid, state, decoded, source);
647
648 const auto sz = state_size(grid);
649 Vec& predicted = scratch.predicted_state;
650 predicted.set_size(grid.n_mixture_state);
651 for (arma::uword i = 0; i < grid.ns; ++i) {
652 const double scale = dt_predictor/static_cast<double>(grid.ds_i(i));
653 for (arma::uword row = 0; row < num_of_mixture_eq; ++row) {
654 const arma::uword k = arma::sub2ind(sz, i, row);
655 // Hancock half step. It differences cell i's OWN two extrapolated
656 // fluxes over ds_i -- i.e. it applies the pressure gradient -- so it
657 // must apply the source that balances that gradient in the same
658 // stage. Omitting it left the half state of a hydrostatic column
659 // with a spurious upward dt*|g| velocity, which the corrector then
660 // reconstructed into every face: that, not the numerical flux, was
661 // the dominant discrete hydrostatic defect of the release scheme.
662 predicted(k) = static_cast<float>(
663 static_cast<double>(state(k))
664 - scale*(fl_iph.flux(k) - fr_imh.flux(k))
665 + dt_predictor*source(k));
666 }
667 }
668 MixtureField& decoded_predicted = scratch.predicted;
669 decode_predictor_into(grid, predicted, decoded, decoded_predicted);
670 stencil_view(grid, decoded_predicted, 0, wt);
671 stencil_view(grid, decoded_predicted, 1, wt_ip1);
672 stencil_view(grid, decoded_predicted, -1, wt_im1);
673
674 if (grid.uniform_mesh) {
675 wr_iph = 0.5*(w_ip1 + wt_ip1) - 0.5*lm_rip1%(w_ip2 - w_ip1);
676 wl_iph = 0.5*(w + wt) + 0.5*lp_r%(w_ip1 - w);
677 wr_imh = 0.5*(w + wt) - 0.5*lm_r%(w_ip1 - w);
678 wl_imh = 0.5*(w_im1 + wt_im1) + 0.5*lp_rim1%(w - w_im1);
679 } else {
680 wr_iph = 0.5*(w_ip1 + wt_ip1) - W3%lm_rip1%(w_ip2 - w_ip1);
681 wl_iph = 0.5*(w + wt) + W1%lp_r%(w_ip1 - w);
682 wr_imh = 0.5*(w + wt) - W1%lm_r%(w_ip1 - w);
683 wl_imh = 0.5*(w_im1 + wt_im1) + W4%lp_rim1%(w - w_im1);
684 }
685
686 // Corrector parentage, matching the four reconstructions just above:
687 // wr_iph[i] from cell i+1, wl_iph[i] and wr_imh[i] from cell i,
688 // wl_imh[i] from cell i-1.
689 {
690 const FaceRequest requests[4] = {
691 {&wr_iph, &grid.phi_g_iph, &fr_iph, h_ip1},
692 {&wl_iph, &grid.phi_g_iph, &fl_iph, h0},
693 {&wr_imh, &grid.phi_g_imh, &fr_imh, h0},
694 {&wl_imh, &grid.phi_g_imh, &fl_imh, h_im1}};
695 build_faces(grid, requests, 4);
696 }
697
698 Work a_imh(grid.ns), a_iph(grid.ns);
699 for (arma::uword i = 0; i < grid.ns; ++i) {
700 a_imh(i) = std::max(
701 std::abs(fl_imh.velocity(i)) + fl_imh.sound_speed(i),
702 std::abs(fr_imh.velocity(i)) + fr_imh.sound_speed(i));
703 a_iph(i) = std::max(
704 std::abs(fl_iph.velocity(i)) + fl_iph.sound_speed(i),
705 std::abs(fr_iph.velocity(i)) + fr_iph.sound_speed(i));
706 }
707 Work& flux_imh = scratch.flux_imh;
708 Work& flux_iph = scratch.flux_iph;
709 flux_imh.set_size(grid.n_mixture_state);
710 flux_iph.set_size(grid.n_mixture_state);
711 if (grid.roe_characteristic_flux) {
712 build_roe_flux(grid, fl_imh, fr_imh, a_imh, flux_imh);
713 build_roe_flux(grid, fl_iph, fr_iph, a_iph, flux_iph);
714 } else {
715 for (arma::uword i = 0; i < grid.ns; ++i) {
716 rusanov_face(grid, fl_imh, fr_imh, i, a_imh(i), flux_imh);
717 rusanov_face(grid, fl_iph, fr_iph, i, a_iph(i), flux_iph);
718 }
719 }
720
721 Vec& rhs = scratch.rhs;
722 rhs.set_size(grid.n_mixture_state);
723 for (arma::uword i = 0; i < grid.ns; ++i) {
724 const double B_i = grid.B_i(i);
725 const double inv_B_iph = 1.0/static_cast<double>(grid.B_iph(i));
726 const double inv_B_imh = 1.0/static_cast<double>(grid.B_imh(i));
727 const double inv_ds = 1.0/static_cast<double>(grid.ds_i(i));
728 for (arma::uword row = 0; row < num_of_mixture_eq; ++row) {
729 const arma::uword k = arma::sub2ind(sz, i, row);
730 rhs(k) = static_cast<float>(
731 -B_i*(flux_iph(k)*inv_B_iph - flux_imh(k)*inv_B_imh)*inv_ds
732 + source(k));
733 }
734 }
735 if (grid.capture_face_flux)
736 capture_face_flux(grid, w, wl_iph, wr_iph, fl_iph, fr_iph, a_iph,
737 flux_iph, r, r_ip1, lp_r, lm_rip1);
738 return Vec(rhs);
739}

742// Timestep
743// ============================================================================
744
745// The release has no volumetric energy source, so the acoustic CFL condition is
746// the only timestep constraint: the implicit conduction stage is unconditionally
747// stable and imposes none.
748Vec mixture_timestep(const Grid& grid, const Vec& state,
749 const MixtureField& decoded) {
750 ProfileScope timer(ProfileRegion::Cfl);
751 decoded.require_matches(grid, state);
752 const auto sz = state_size(grid);
753 Vec dt_i(grid.ns);
754 for (arma::uword i = 0; i < grid.ns; ++i) {
755 const MixtureThermo& th = decoded.cells[i];
756 const double momentum =
757 static_cast<double>(state(arma::sub2ind(sz, i, mix::MOM)));
758 const double speed = std::abs(momentum/th.rho)
759 + std::sqrt(th.gamma1*th.p/th.rho);
760 dt_i(i) = static_cast<float>(grid.CFL*grid.ds_i(i)/speed);
761 }
762 profile_note_timestep_limiter(TimestepLimiter::Acoustic);
763 return arma::min(dt_i)*arma::ones<Vec>(grid.ns);
764}
