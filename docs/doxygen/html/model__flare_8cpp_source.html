15Vec model_flare_ic(Grid& grid) {
16 // Force the full physical model: the explosive/gentle threshold is the
17 // competition between beam heating and radiative loss, so Stage R cooling
18 // must be active; ionization tracks the evaporated material going ionized.
19 grid.enable_ionization = true;
20 grid.enable_radiative_cooling = true;
21 // The full Stage E network (S_i + S_CR ionization, α_r + α_c recombination)
22 // is the default (see chromosphere.hpp). The flare specifically *requires*
23 // three-body recombination α_c = κ_c n_e: explosive evaporation shock-
24 // compresses the condensation to densities where α_c (volumetric rate ∝
25 // n_e³) is the dominant recombination sink; without it the dense, rapidly-
26 // ionizing condensation has no adequate sink — the ionization fraction
27 // drifts, the ion/neutral pressure partition corrupts, and the hydro goes
28 // NaN a few s after the beam fires. model_c7_ic inverts P_phot over exactly
29 // the active network, so C7 stays a fixed point regardless of the flags.
30
31 // Build the C7 chromosphere + lower-TR column (also sets the inner discrete-
32 // HSE base, the ambient RTV q(T), numerical diffusivity, TRAC). model_c7_ic
33 // reads enable_radiative_cooling to gate q(T)/TRAC, hence the order above.
34 Vec xn = model_c7_ic(grid);
35
36 // --- Flare beam-heating driver (docs/explosive_evaporation_plan.md) ------
37 // Beam energy flux F_e [W/m²]. Default 5×10⁷ W m⁻² = 5×10¹⁰ erg cm⁻² s⁻¹ —
38 // Fisher et al. (1985) middle explosive run (well above F_crit ≈ 7×10⁶ W m⁻²,
39 // peaks at ~500 km/s). Overridable via FLARE_BEAM_FLUX so the gentle control
40 // (1×10⁶ W m⁻² = 1×10⁹ erg cm⁻² s⁻¹, below threshold) reuses this build.
41 auto env_f = [](const char* key, float fallback) -> float {
42 if (const char* e = std::getenv(key)) {
43 try { return std::stof(std::string(e)); } catch (...) {}
44 }
45 return fallback;
46 };
47
48 float beam_flux = env_f("FLARE_BEAM_FLUX", 5.0e7f);
49
50 grid.enable_beam_heating = true;
51 grid.beam_flux = beam_flux;
52 grid.beam_t_on = env_f("FLARE_T_ON", 2.0f); // let IC settle ~1 acoustic time
53 grid.beam_duration = env_f("FLARE_DUR", 10.0f); // flat-top heating (Fisher: a few s)
54 grid.beam_ramp = env_f("FLARE_RAMP", 1.0f); // cosine ramps
55 grid.beam_h_lo_km = env_f("FLARE_H_LO", 1600.0f); // deposition window [km]
56 grid.beam_h_hi_km = env_f("FLARE_H_HI", 2000.0f); // (upper chromosphere, below TR)
57
58 // Open the outer face so the supersonic evaporation upflow can leave the
59 // domain (the C7 Mach-0.05 cap would choke it). Env-toggle for diagnosis.
60 grid.outer_free_outflow = (env_f("FLARE_FREE_OUTFLOW", 1.0f) != 0.0f);
61
62 // Optional extra numerical diffusivity to stabilize the evaporation shock on
63 // the coarse grid (multiplier on the C7 grid-scaled value set by model_c7_ic).
64 grid.numerical_diffusivity_per_length *= env_f("FLARE_DIFF_MULT", 1.0f);
65
66 return xn;
67}
