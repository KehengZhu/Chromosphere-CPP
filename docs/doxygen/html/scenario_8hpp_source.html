27/// One complete initial-condition + boundary-condition source, in the form
28/// chromo_main consumes it.
29///
30/// A Scenario packages the three things the driver needs from ANY IC/BC source,
31/// tabulated or analytic, plus a name for logging and sidecar metadata:
32///
33/// * peek_ns — how many cells the grid should have. Queried BEFORE
34/// Grid::init(), because the Grid must be allocated first.
35/// * ic — fill the freshly allocated Grid (geometry, gravity, ghost
36/// buffers) and return the initial conserved state vector.
37/// * update_bc — refresh the Grid's boundary buffers from the current state,
38/// once per timestep.
39///
40/// Holding them in std::function keeps the driver independent of which scenario
41/// is running; make_scenario() binds the free functions of one scenario
42/// translation unit into this struct.
43struct Scenario {
44 /// Returns the grid resolution this scenario expects (a tabulated
45 /// scenario reads it from its data file; an analytic one returns a
46 /// sensible default). Called before Grid::init().
47 std::function<arma::uword()> peek_ns;
48
49 /// Populate the Grid's cell-centered + boundary fields and return the
50 /// initial conserved-state vector.
51 std::function<Vec(Grid&)> ic;
52
53 /// Refresh the boundary buffers from the current state. Called once
54 /// per timestep before advance_Euler_state.
55 std::function<void(Grid&, const Vec&)> update_bc;
56
57 /// Scenario identifier as spelled on the command line (e.g. "model_column").
58 /// Used for log lines and run-metadata sidecars.
59 std::string name;
60};
61

65Scenario make_scenario(const std::string& name, const std::string& data_path = "");

67/// Open-hyperbolic boundary update shared by all built-in scenarios.
68///
69/// * Inner face (s = s_{-1/2}) — closed reflecting wall:
70/// ρ(-1) = ρ(0), T(-1) = T(0), V(-1) = -V(0)
71/// ρ(-2) = ρ(1), T(-2) = T(1), V(-2) = -V(1)
72/// Anti-symmetric velocity guarantees zero mass flux across s = 0; the
73/// ρ and T mirrors keep the pressure gradient there well posed.
74///
75/// * Outer face (s = s_{ns-1/2}) — pure Neumann outflow on every variable:
76/// ρ(ns) = ρ(ns+1) = ρ(ns-1)
77/// T(ns) = T(ns+1) = T(ns-1)
78/// V(ns) = V(ns+1) = V(ns-1)
79/// U(ns) = U(ns+1) = U(ns-1)
80/// No impedance / dissipation at the face — well-posed only for IC near
81/// HSE or with an added sponge layer; transient runs (e.g. C7 IC under
82/// gravity) may go unstable.
83///
84/// Ghost energies are rebuilt with the same gauge as the IC: φ_g(-1) =
85/// grid.phi_g_imh[0] and φ_g(ns) = grid.phi_g_iph[ns-1].
86void apply_open_bcs(Grid& grid, const Vec& xn);
