40Scenario make_scenario(const std::string& name, const std::string& data_path) {
41 Scenario sc;
42 sc.name = name;
43
44 if (name == "model_c7") {
45 sc.peek_ns = []() -> arma::uword { return 100; };
46 // "New explanation" upper BC (docs/gentle_evaporation_downflow.md):
47 // well-balanced reconstruction + hydrostatic ghost pressure + imposed
48 // TR temperature jump (downward q(T) via the Stage-D Dirichlet ghost-T)
49 // + EOS ghost density. tr_jump_bc=true is exclusive to this bare scenario
50 // — model_flare / analytic_canopy / model_gentle keep the RTV reservoir.
51 sc.ic = [](Grid& g) { return model_c7_ic(g, /*extended=*/false, /*tr_jump_bc=*/true); };
52 sc.update_bc = [](Grid& g, const Vec& xn) { model_c7_update_bc(g, xn); };
53 return sc;
54 }
55 if (name == "model_flare") {
56 // Grid resolution is env-tunable (FLARE_NS, default 100) — a finer grid
57 // resolves the steep condensation-front contact discontinuity and shrinks
58 // the single-cell T = p/n overheating there. Matches the FLARE_* knobs.
59 sc.peek_ns = []() -> arma::uword {
60 arma::uword ns = 100;
61 if (const char* e = std::getenv("FLARE_NS")) { try { ns = std::stoul(e); } catch (...) {} }
62 return ns;
63 };
64 sc.ic = [](Grid& g) { return model_flare_ic(g); };
65 sc.update_bc = [](Grid& g, const Vec& xn) { model_flare_update_bc(g, xn); };
66 return sc;
67 }
68 // The canonical field-aligned chromosphere→corona column (the DEFAULT
69 // scenario) and its historical two-fluid sibling. They share one IC/BC
70 // implementation but NOT a solver: `model_column` always runs the
71 // single-fluid release solver, `model_gentle` always runs the historical
72 // two-fluid research solver. Nothing a user can set moves either one across
73 // that line.
74 if (name == "model_column" || name == "model_gentle") {
75 if (name == "model_column") {
76 // Canonical release model: the validated Saha/Gamma1 N=500/R4
77 // physical-conduction-only single-fluid column. The Gamma1 table is
78 // supplied UNCONDITIONALLY (only the path is overridable), so
79 // `model_column` names exactly one solver. The remaining entries are
80 // override-preserving defaults for the release model/grid, so
81 // selecting model_column means the release configuration without a
82 // launcher-specific collection of ISO_* assignments. No two-fluid,
83 // fixed-gamma, finite-rate-ionization, cooling, TRAC or artificial-
84 // conduction knob appears here: the release solver has none of them,
85 // and model_column_ic rejects them outright if they are set.
86 if (std::getenv("ISO_GAMMA"))
87 throw std::runtime_error(
88 "make_scenario: model_column is the single-fluid release "
89 "scenario and derives every thermodynamic index from the "
90 "Saha/Gamma1 closure; ISO_GAMMA (the historical fixed-gamma "
91 "two-fluid index) has no meaning here. Use model_gentle for "
92 "the historical fixed-gamma column.");
93 set_env_default("GAMMA_TABLE", "data/eos/gamma1_hydrogen_v1.dat");
94 set_env_default("ISO_H_BASE", "1600");
95 set_env_default("ISO_DH", "553");
96 set_env_default("ISO_NS", "500");
97 set_env_default("ISO_HEAT_FLUX", "1");
98 set_env_default("ISO_T_TOP", "22000");
99 set_env_default("ISO_HYDRO_T_DECOUPLE", "1");
100 set_env_default("ISO_REFINE_PROFILE", "outer");
101 set_env_default("ISO_REFINE_FACTOR", "4");
102 set_env_default("ISO_REFINE_S_LO_KM", "500");
103 set_env_default("ISO_REFINE_TRANSITION_KM", "20");
104 }
105 // HISTORICAL two-fluid research preset (NOT the release): the documented
106 // stable full-physics resolved-corona gentle conduction-driven
107 // evaporation configuration — resolved corona + radiative sink +
108 // two-fluid + finite-rate ionization, conduction on. The base is raised
109 // to h = 1003 km (model_c7's validated floor): the Stage-E n² channel
110 // counts and the radiative loss overflow float32 at photospheric density
111 // (n ~ 1e23 ⇒ n² ≫ FLT_MAX). It never loads a Gamma1 table, so it always
112 // selects the seven-row two-fluid solver.
113 if (name == "model_gentle") {
114 set_env_default("ISO_CORONA", "1");
115 set_env_default("ISO_H_BASE", "1003");
116 set_env_default("ISO_HEAT_FLUX", "1");
117 set_env_default("ISO_COOLING", "1");
118 set_env_default("ISO_IONIZATION", "1");
119 set_env_default("ISO_TWO_FLUID", "1");
120 set_env_default("ISO_NS", "600");
121 }
122 sc.peek_ns = []() { return column_peek_ns(); };
123 sc.ic = [](Grid& g) { return model_column_ic(g); };
124 sc.update_bc = [](Grid& g, const Vec& xn) { model_column_update_bc(g, xn); };
125 return sc;
126 }
127 if (name == "analytic_canopy") {
128 sc.peek_ns = []() { return analytic_canopy_peek_ns(); };
129 sc.ic = [](Grid& g) { return analytic_canopy_ic(g); };
130 sc.update_bc = [](Grid& g, const Vec& xn) { analytic_canopy_update_bc(g, xn); };
131 return sc;
132 }
133 if (name == "pfss_field_line") {
134 if (data_path.empty())
135 throw std::runtime_error("make_scenario: pfss_field_line requires a data_path "
136 "(6th positional argument to chromo_main)");
137 sc.peek_ns = [data_path]() { return pfss_peek_ns(data_path); };
138 sc.ic = [data_path](Grid& g) { return pfss_ic(g, data_path); };
139 sc.update_bc = [](Grid& g, const Vec& xn) { pfss_update_bc(g, xn); };
140 return sc;
141 }
142
143 throw std::runtime_error("make_scenario: unknown scenario name '" + name + "'");
144}

146// Reflecting wall at the inner face + zero-gradient outflow at the outer face.
147// See scenario.hpp for the BC math; the implementation just reads cells 0, 1,
148// and ns-1 out of the packed state, mirrors / copies them into the four ghost
149// buffers, and re-runs Grid::broadcast so the packed _state caches stay in sync.
150void apply_open_bcs(Grid& grid, const Vec& xn) {
151 const arma::uword ns = grid.ns;
152 const float k_b = grid.k_b;
153 const float m_i = grid.m_i;
154 const float m_n = grid.m_n;
155
156 auto cell_prim = [&](arma::uword i,
157 float& rho_i, float& rho_n,
158 float& V, float& U,
159 float& T_i, float& T_n) {
160 const auto sz = arma::size(ns, num_of_eq);
161 rho_i = xn(arma::sub2ind(sz, i, cons::RHO_I));
162 rho_n = xn(arma::sub2ind(sz, i, cons::RHO_N));
163 const float momI = xn(arma::sub2ind(sz, i, cons::MOM_I));
164 const float momN = xn(arma::sub2ind(sz, i, cons::MOM_N));
165 const float E_i = xn(arma::sub2ind(sz, i, cons::E_I));
166 const float E_n = xn(arma::sub2ind(sz, i, cons::E_N));
167 V = momI / rho_i;
168 U = momN / rho_n;
169 const float phi_g_cell = 0.5f * (grid.phi_g_imh(i) + grid.phi_g_iph(i));
170 // cons2prim convention (state.cpp): p_i = 2 n_i k_b T_i with electron
171 // quasi-neutrality folded in, p_n = n_n k_b T_n.
172 const float p_i = grid.gm1() * E_i - grid.half_gm1() * rho_i * V * V
173 - grid.gm1() * rho_i * phi_g_cell;
174 const float p_n = grid.gm1() * E_n - grid.half_gm1() * rho_n * U * U
175 - grid.gm1() * rho_n * phi_g_cell;
176 const float n_i = rho_i / m_i;
177 const float n_n = rho_n / m_n;
178 T_i = p_i / (2.0f * n_i * k_b);
179 T_n = p_n / (n_n * k_b);
180 };
181
182 auto pack_ghost = [&](Vec& ob,
183 float rho_i, float rho_n,
184 float V_g, float U_g,
185 float T_i, float T_n,
186 float phi_g_g) {
187 const float n_i = rho_i / m_i;
188 const float n_n = rho_n / m_n;
189 ob(cons::RHO_I) = rho_i;
190 ob(cons::RHO_N) = rho_n;
191 ob(cons::MOM_I) = rho_i * V_g;
192 ob(cons::MOM_N) = rho_n * U_g;
193 ob(cons::E_I) = grid.inv_gm1() * k_b * n_i * 2.0f * T_i
194 + 0.5f * rho_i * V_g * V_g + rho_i * phi_g_g;
195 ob(cons::E_N) = grid.inv_gm1() * k_b * n_n * T_n
196 + 0.5f * rho_n * U_g * U_g + rho_n * phi_g_g;
197 ob(cons::E_E) = grid.inv_gm1() * k_b * n_i * T_i; // electron internal energy (T_e = T_i)
198 };
199
200 // --- Inner reflecting wall (mirror cells 0 and 1) ----------------------
201 {
202 float rho_i, rho_n, V, U, T_i, T_n;
203 const float phi_g_in = grid.phi_g_imh(0);
204 cell_prim(0, rho_i, rho_n, V, U, T_i, T_n);
205 pack_ghost(grid.inner_boundary0_i, rho_i, rho_n, -V, -U, T_i, T_n, phi_g_in);
206 if (ns >= 2) {
207 cell_prim(1, rho_i, rho_n, V, U, T_i, T_n);
208 pack_ghost(grid.inner_boundary1_i, rho_i, rho_n, -V, -U, T_i, T_n, phi_g_in);
209 } else {
210 grid.inner_boundary1_i = grid.inner_boundary0_i;
211 }
212 }
213
214 // --- Outer face -------------------------------------------------------
215 // Default (open line / coronal top): pure Neumann outflow — ρ, T, V, U all
216 // zero-gradient at both outer ghosts. The boundary has no impedance —
217 // outgoing waves see a transparent face and no externally-specified state on
218 // the incoming characteristic. Stable only when the IC is near HSE /
219 // steady-state; transient runs that need boundary damping may go unstable.
220 //
221 // When grid.outer_reflecting (closed-loop apex): mirror cells ns-1, ns-2 with
222 // V,U → −V,−U so the face is a zero-flux symmetry plane — the evaporated
223 // plasma is confined and fills the loop. g_∥→0 at the apex, so the gravity
224 // potential is flat there and reusing phi_g_iph(ns-1) for both ghosts is exact
225 // to leading order.
226 {
227 float rho_i, rho_n, V, U, T_i, T_n;
228 const float phi_g_out = grid.phi_g_iph(ns - 1);
229 cell_prim(ns - 1, rho_i, rho_n, V, U, T_i, T_n);
230 if (grid.outer_reflecting) {
231 pack_ghost(grid.outer_boundary0_i, rho_i, rho_n, -V, -U, T_i, T_n, phi_g_out);
232 if (ns >= 2) {
233 cell_prim(ns - 2, rho_i, rho_n, V, U, T_i, T_n);
234 pack_ghost(grid.outer_boundary1_i, rho_i, rho_n, -V, -U, T_i, T_n, phi_g_out);
235 } else {
236 grid.outer_boundary1_i = grid.outer_boundary0_i;
237 }
238 } else {
239 pack_ghost(grid.outer_boundary0_i, rho_i, rho_n, V, U, T_i, T_n, phi_g_out);
240 pack_ghost(grid.outer_boundary1_i, rho_i, rho_n, V, U, T_i, T_n, phi_g_out);
241 }
242 }
243
244 grid.broadcast();
245}
