32/// `nk` selector for ip1/im1/ip2/im2: treat the input as ONE scalar field of
33/// length ns (one row of the packed state). Scalar shifts are always
34/// zero-gradient at the domain ends — the seven-row ghost buffers do not apply
35/// to an anonymous scalar.
36const arma::uword SLICE = 1;
37/// `nk` selector for ip1/im1/ip2/im2: treat the input as the packed seven-row
38/// state (all `num_of_eq` rows), so the shifts pull the two-fluid ghost buffers
39/// in at the domain ends.
40const arma::uword CUBE = num_of_eq;
41
42/// Outer/inner ghost-cell handling for the packed-state index shifts: `true`
43/// makes every shift zero-gradient (Neumann) at both domain ends, `false` (the
44/// build setting) pulls the two-fluid ghost buffers `Grid::inner_boundary0_i` /
45/// `_boundary1_i` and `Grid::outer_boundary0_i` / `_boundary1_i` instead.
46const bool USE_NEUMANN_BC = false;

70/// Index shifts. `nk = CUBE` treats the input as packed state; `nk = SLICE`
71/// treats it as one scalar field of length ns.
72///
73/// Shift one cell OUTWARD: `ip1(xn)(i) = xn(i+1)` for i < ns-1. The top slot
74/// i = ns-1 is filled from the FIRST outer ghost `Grid::outer_boundary0_i`,
75/// except for a SLICE input or when USE_NEUMANN_BC is set, where it is a
76/// zero-gradient copy of `xn(ns-1)`.
77Vec ip1(const Grid& grid, const Vec& xn, arma::uword nk = CUBE);
78/// Shift one cell INWARD: `im1(xn)(i) = xn(i-1)` for i > 0. The bottom slot
79/// i = 0 is filled from the FIRST inner ghost `Grid::inner_boundary0_i`,
80/// except for a SLICE input or when USE_NEUMANN_BC is set, where it is a
81/// zero-gradient copy of `xn(0)`.
82Vec im1(const Grid& grid, const Vec& xn, arma::uword nk = CUBE);
83/// Shift two cells OUTWARD: ip1 applied twice, then (packed state only) the top
84/// slot i = ns-1 overwritten with the SECOND outer ghost
85/// `Grid::outer_boundary1_i`, which is the value the MUSCL stencil needs two
86/// cells beyond the boundary.
87Vec ip2(const Grid& grid, const Vec& xn, arma::uword nk = CUBE);
88/// Shift two cells INWARD: im1 applied twice, then (packed state only) the
89/// bottom slot i = 0 overwritten with the SECOND inner ghost
90/// `Grid::inner_boundary1_i`.
91Vec im2(const Grid& grid, const Vec& xn, arma::uword nk = CUBE);

97/// Conserved → primitive (writeup eq 38). Maps the seven `cons::` rows
98/// (ρ_i, ρ_n, ρ_i V, ρ_n U, E_I, E_N, E_E) onto the seven `prim::` rows
99/// (ρ_i, ρ_n, V, U, p_I, p_N, p_E) by removing the kinetic and gravitational
100/// parts of the energies with the cell-centred potential
101/// φ_g = ½(phi_g_imh + phi_g_iph) [J/kg]. `p_I` is the TOTAL charged pressure
102/// (protons + electrons, quasi-neutrality n_e = n_i); `p_E` is the electron
103/// partial pressure alone and carries neither kinetic nor gravitational energy.
104/// Densities [kg/m^3], velocities [m/s], pressures [Pa].
105Vec cons2prim(const Grid& grid, const Vec& cons_state);
106/// Primitive → conserved, the exact inverse of cons2prim with the same
107/// φ_g gauge and the same "E_I holds the total charged energy, E_E holds the
108/// electron internal energy alone" convention.
109Vec prim2cons(const Grid& grid, const Vec& prim_state);

111// ============================================================================
112// Numerics: flux, source, spectral radius, MUSCL limiter

115/// Minmod limiter (writeup eq 15): φ(r) = max(0, min(1, r)).
116Vec flux_lim(const Vec& r);
117
118/// MC3 / Koren limiter (BATSRUS ModFaceValue 'mc3') — the ASYMMETRIC third-order
119/// (κ=1/3) monotonized-central limiter. In the code's ratio form r = Δ₋/Δ₊ the two
120/// faces of a cell take DIFFERENT limited slopes (unlike the symmetric minmod):
121/// '+' faces (Lxn = u_i + ½φΔ₊): φ₊(r) = max(0, min(β r, β, (2r+1)/3))
122/// '−' faces (Rxn = u_i − ½φΔ₊): φ₋(r) = max(0, min(β r, β, (r+2)/3))
123/// β = grid.limiter_beta (2 ⇒ classic Koren). Non-finite r ⇒ 0 (first order at
124/// flats/extrema, matching flux_lim). Selected by grid.mc3_limiter in rhs.
125///
126/// The '+' branch φ₊(r) = max(0, min(β r, β, (2r+1)/3)), used for the
127/// right-face (Lxn) reconstruction term.
128Vec flux_lim_mc3_plus (const Vec& r, float beta);
129/// The '−' branch φ₋(r) = max(0, min(β r, β, (r+2)/3)), used for the left-face
130/// (Rxn) reconstruction term. See flux_lim_mc3_plus for the shared derivation.
131Vec flux_lim_mc3_minus(const Vec& r, float beta);

163/// Semi-implicit (backward-Euler) integrator. Explicit MUSCL+Rusanov step
164/// for R_E, then point-implicit drag + frictional heating, point-implicit
165/// ion–neutral temperature equilibration, a tridiagonal heat-conduction
166/// solve per species, and (if grid.enable_ionization) the point-implicit
167/// ionization Stage E (writeup §3.7, §5.3). Mutates grid.dt_state.
168Vec advance_Euler_state(Grid& grid, const Vec& xn_state, const Vec& dt_i);
169
170/// Stage E: backward-Euler ionization / recombination on a single cell, scalar
171/// quadratic solve in ionization fraction f = ρ_i / (ρ_i + ρ_n). Operates on
172/// primitive state in place; intended for the operator-split integrator but
173/// exposed for direct testing. (writeup §5.3.)
174void apply_ionization_stage(const Grid& grid, Vec& prim_state, float dt);
175
176/// Stage R (writeup §2.4): backward-Euler optically-thick chromospheric
177/// radiative cooling on the folded ion+electron thermal pressure. Carlsson
178/// & Leenaarts 2012 recipe summed over H I + Ca II + Mg II; tables in
179/// physics.hpp::cl2012. Activated by Grid::enable_radiative_cooling.
180void apply_radiative_cooling_stage(const Grid& grid, Vec& prim_state, float dt);
181
182/// Flare beam-heating stage: deposits grid.beam_flux into the upper-chromosphere
183/// electron/ion thermal pool (p_i) over a finite layer, gated by the temporal
184/// window around grid.sim_time, to drive chromospheric evaporation (Fisher et
185/// al. 1985). No-op unless grid.enable_beam_heating. Mutates p_i in place.
186void apply_beam_heating_stage(const Grid& grid, Vec& prim_state, float dt);
187
188/// Ambient coronal heating stage: deposits the steady footpoint-anchored
189/// volumetric heating H(s) (physics.hpp::coronal_heating_rate) into the charged
190/// thermal pool (shared with neutrals by heat capacity; electrons when
191/// enable_Te), driving the conductive flux that sustains the corona and, when
192/// ramped up (coronal_heat_enhance > 1), gentle chromospheric evaporation
193/// (Antiochos & Sturrock 1978). No-op unless grid.enable_coronal_heating.
194void apply_coronal_heating_stage(const Grid& grid, Vec& prim_state, float dt);
195
196/// TRAC adaptive cutoff temperature T_c (Johnston et al. 2020 Eq. 8): the
197/// maximum temperature among grid cells where the charged-fluid TR is
198/// under-resolved (L_R/L_T > 1/2, with L_T = T/|dT/ds|, L_R = Δs), clamped to
199/// [grid.trac_T_chrom, 0.2 T_peak]. Used by the conduction and cooling stages to
200/// broaden the unresolved TR. Exposed for testing.
201float compute_trac_cutoff_T(const Grid& grid, const Vec& prim_state);
202
203/// Pure-explicit forward-Euler step — only the MUSCL+Rusanov R_E predictor of
204/// advance_Euler_state, with the implicit drag / temperature / conduction
205/// stages skipped (equivalent to R_I ≡ 0). Mutates grid.dt_state.
206Vec advance_Euler_explicit_state(Grid& grid, const Vec& xn_state, const Vec& dt_i);

208/// Explicit RK4. Mutates grid.dt_state.
