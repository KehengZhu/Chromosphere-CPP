31/// Fixed capacity of every per-thread slot array in the solver (ParallelFailure's
32/// index/exception slots, the profiling counter slots, the EOS operation counters,
33/// the conduction residual and step slots). Those arrays are statically sized, so
34/// a thread index must never reach this value; require_supported_parallel_runtime()
35/// enforces the bound instead of letting a large OMP_NUM_THREADS overrun them.
36constexpr int kMaximumParallelThreads = 256;
37
38/// Upper bound on the team size used for the implicit conduction stage, which
39/// takes min(this, parallel_max_threads()) threads. The stage is a barrier-heavy
40/// nonlinear Newton sequence around a serial Thomas solve, so on production-shaped
41/// meshes it stops scaling and then regresses: measured per 0.1 physical-second
42/// probe at 2638 cells, 2 threads ~0.092 s and 4 threads ~0.087 s, versus ~0.132 s
43/// at 8 and ~0.204 s at 12 (docs/openmp_parallelization_recap.md). Every other
44/// stage still uses the full requested team.
45constexpr int kMaximumConductionThreads = 4;
46

91/// Fail closed if the runtime would hand out more threads than the fixed
92/// per-thread slot arrays can hold. Call it from serial code before entering a
93/// region that writes those slots; it throws std::runtime_error rather than
94/// letting a thread index run past kMaximumParallelThreads.
95inline void require_supported_parallel_runtime() {
96 const int count = parallel_max_threads();
97 if (count < 1 || count > kMaximumParallelThreads)
98 throw std::runtime_error("OpenMP thread count exceeds fixed solver slot capacity");
99}

101/// Deterministic exception collector. Each worker writes only its own slot; after
102/// the region, the caller rethrows the exception from the lowest physical cell.
103class ParallelFailure {
104public:
105 /// Start empty: no slot holds an exception.
106 ParallelFailure() { reset(); }
107
108 /// Drop every recorded exception and reset all cell indices to "none", so the
109 /// collector can be reused for another region.
110 void reset() noexcept {
111 indices_.fill(std::numeric_limits<std::size_t>::max());
112 for (auto& error : errors_) error = std::exception_ptr{};
113 }
114
115 /// Record that physical cell @p cell threw @p error. The calling worker writes
116 /// only its own slot — no lock, no atomic — and keeps just the LOWEST cell
117 /// index it has seen, which is what makes rethrow_lowest() deterministic. A
118 /// thread index outside the slot capacity is dropped rather than written out
119 /// of bounds. Safe to call from a `catch (...)` block: it cannot itself throw.
120 void capture(std::size_t cell, std::exception_ptr error) noexcept {
121 const int tid = parallel_thread_index();
122 if (tid < 0 || tid >= kMaximumParallelThreads) return;
123 const std::size_t slot = static_cast<std::size_t>(tid);
124 if (cell < indices_[slot]) {
125 indices_[slot] = cell;
126 errors_[slot] = error;
127 }
128 }
129
130 /// True if any slot holds an exception. Used inside a region (from an `omp
131 /// single` block) to decide whether the remaining work should be abandoned.
132 bool failed() const noexcept {
133 for (const auto& error : errors_)
134 if (error) return true;
135 return false;
136 }
137
138 /// Rethrow the exception belonging to the lowest physical cell index recorded
139 /// by any thread; do nothing if none was recorded. Call this after the region
140 /// has joined. Because the winner is chosen by cell index and not by which
141 /// thread happened to fail first, the reported error is independent of the
142 /// thread count and matches what a serial run would have thrown.
143 void rethrow_lowest() const {
144 std::size_t best = std::numeric_limits<std::size_t>::max();
145 std::exception_ptr chosen;
146 for (int tid = 0; tid < kMaximumParallelThreads; ++tid) {
147 const std::size_t slot = static_cast<std::size_t>(tid);
148 if (errors_[slot] && indices_[slot] < best) {
149 best = indices_[slot];
150 chosen = errors_[slot];
151 }
152 }
153 if (chosen) std::rethrow_exception(chosen);
154 }
155
156private:
157 /// Lowest failing physical cell index seen by each thread; size_t max means
158 /// "that thread has not failed". Cache-line aligned to avoid false sharing.
159 alignas(64) std::array<std::size_t, kMaximumParallelThreads> indices_;
160 /// The exception each thread captured for its indices_ entry, one slot per
161 /// thread, empty when that thread has not failed.
162 alignas(64) std::array<std::exception_ptr, kMaximumParallelThreads> errors_;
163};

165/// Apply @p function to every cell index in [0, @p count), in parallel when the
166/// build and runtime allow it. A single-thread runtime and the serial build both
167/// take a plain loop, so the callable sees identical arguments either way; the
168/// caller must therefore keep the per-index work independent.
169///
170/// Exceptions are deterministic rather than fatal: OpenMP forbids escaping a
171/// parallel region, so a throwing index is captured per thread and, after the
172/// region joins, the exception from the lowest cell index is rethrown. A parallel
173/// run reports exactly the error a serial run would.
174template <class Function>
175void parallel_for_cells(std::size_t count, Function function) {
176#ifdef CHROMO_USE_OPENMP
177 require_supported_parallel_runtime();
178 if (parallel_max_threads() == 1) {
179 for (std::size_t i = 0; i < count; ++i) function(i);
180 return;
181 }
182 ParallelFailure failure;
183#pragma omp parallel for schedule(static)
184 for (long long raw = 0; raw < static_cast<long long>(count); ++raw) {
185 const std::size_t i = static_cast<std::size_t>(raw);
186 try {
187 function(i);
188 } catch (...) {
189 failure.capture(i, std::current_exception());
190 }
191 }
192 failure.rethrow_lowest();
193#else
194 for (std::size_t i = 0; i < count; ++i) function(i);
195#endif
196}
