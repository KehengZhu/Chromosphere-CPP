1/*!
2 * @file scenarios/data_file_parser.hpp
3 * @brief Parser for the tabulated [META]/[CELLS]/[GHOSTS] scenario data-file format.
4 * @ingroup scenarios
5 *
6 * Parser for the tabulated scenario data-file format produced by
7 * util/extract_field_line.py. Used by any scenario that initialises the grid
8 * from an external file (pfss_field_line for now; Bifrost traces later).
9 *
10 * Format (ASCII, line-oriented):
11 *
12 * # Comments are lines starting with '#'. Blank lines are skipped.
13 * [META]
14 * key=value (one per line, free-form)
15 * [CELLS]
16 * i ds_m B_imh_T B_iph_T phi_g_imh phi_g_iph ne_im3 nn_im3 T_K
17 * (ns rows, indexed 0..ns-1 in column 1; whitespace-separated)
18 * [GHOSTS]
19 * tag B_T phi_g ne nn T T_e_factor
20 * (exactly 4 rows; tag in {outer_0, outer_1, inner_0, inner_1})
21 *
22 * Section order is fixed: [META] first (must declare ns=), then [CELLS],
23 * then [GHOSTS]. peek_ns() only parses [META] for the ns= value.
24 */

38/// One fully parsed scenario data file: the [META] scalars, the ns per-cell
39/// [CELLS] columns, and the four [GHOSTS] rows, in SI units throughout.
40///
41/// This is a faithful transcription of the file, not a Grid: the consumer
42/// (pfss_ic) copies the geometry columns into the Grid and builds the conserved
43/// state from (n_e, n_n, T). Vector members are sized only after a successful
44/// parse_scenario_data_file().
45struct ScenarioDataFile {
46 /// Number of interior cells declared by `ns=` in [META]; the length of every
47 /// [CELLS] column. Must match Grid::ns.
48 arma::uword ns = 0;
49 /// Gravitational acceleration the generator assumed when it integrated the
50 /// φ_g columns [m/s^2] (solar surface value 274). Recorded for provenance;
51 /// the per-face φ_g columns are what the solver actually uses.
52 float g_si = 274.0f;
53 /// Field strength at the outer face of the last cell [T], i.e. B_iph(ns-1).
54 /// Recorded for provenance; B_imh_T / B_iph_T are authoritative.
55 float B_outer_T = 0.0f;
56 /// Additive gauge offset already applied to the φ_g columns [J/kg]. The
57 /// generators write 0, i.e. φ_g = 0 at the inner face of the trace.
58 float phi_g_offset_Jpkg = 0.0f;
59 /// Field-line topology. "closed" ⇒ the outer face is the loop apex (reflecting
60 /// symmetry BC); anything else (default "open") ⇒ coronal outflow at the top.
61 std::string topology = "open";
62 /// Loop half-length L [m] (footpoint→apex arc), written by make_loop_dat.py.
63 /// Used to calibrate the ambient coronal heating E_H0(L, s_H, T_max). 0 if the
64 /// file omits it (older traces) — the heating overlay then falls back to the
65 /// arc-length sum.
66 float loop_half_length_m = 0.0f;
67
68 /// CELLS, all length ns.
69 /// Cell width along the field line [m].
70 Vec ds_m;
71 /// Magnetic field magnitude at the inner (i−½) cell face [T].
72 Vec B_imh_T;
73 /// Magnetic field magnitude at the outer (i+½) cell face [T].
74 Vec B_iph_T;
75 /// Gravitational potential at the inner (i−½) cell face [J/kg], increasing
76 /// outward along the trace.
77 Vec phi_g_imh;
78 /// Gravitational potential at the outer (i+½) cell face [J/kg].
79 Vec phi_g_iph;
80 /// Electron number density [m^-3]; also the ion (proton) number density
81 /// under quasi-neutrality, so ρ_i = ne_im3 · m_i.
82 Vec ne_im3;
83 /// Neutral-hydrogen number density [m^-3], so ρ_n = nn_im3 · m_n.
84 Vec nn_im3;
85 /// Cell temperature [K], shared by ions, neutrals and electrons in the IC.
86 Vec T_K;
87
88 /// GHOSTS, length 4 in order {outer_0, outer_1, inner_0, inner_1}.
89 /// Ghost magnetic field magnitude [T].
90 Vec ghost_B_T;
91 /// Ghost gravitational potential [J/kg] as written in the file. pfss_ic
92 /// overrides it with the adjacent interior face potential (phi_g_iph(ns-1)
93 /// at the outer end, phi_g_imh(0) at the inner end) to match the φ_g gauge
94 /// the RHS assumes.
95 Vec ghost_phi_g_Jpkg;
96 /// Ghost electron/ion number density [m^-3].
97 Vec ghost_ne_im3;
98 /// Ghost neutral-hydrogen number density [m^-3].
99 Vec ghost_nn_im3;
100 /// Ghost temperature [K], before the T_e_factor scaling.
101 Vec ghost_T_K;
102 /// Dimensionless multiplier applied to ghost_T_K when the ghost energies are
103 /// built (T_eff = ghost_T_K · ghost_T_e_factor). 1 leaves the tabulated
104 /// temperature alone; model_c7 uses 2 at the outer ghost to mimic the hotter
105 /// coronal electrons.
106 Vec ghost_T_e_factor;
107};
