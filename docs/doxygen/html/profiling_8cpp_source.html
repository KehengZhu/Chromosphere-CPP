22struct alignas(64) ThreadProfileSlot {
23 std::array<double, static_cast<unsigned>(ProfileRegion::Count)> seconds{};
24 std::array<std::uint64_t, static_cast<unsigned>(TimestepLimiter::Count)> limiters{};
25 EosInversionProfile inversions;
26 PressureInversionProfile pressure_inversions;

34ThreadProfileSlot& current_slot() noexcept {
35 return slots[static_cast<std::size_t>(parallel_thread_index())];

38ThreadProfileSlot merged_profile() {
39 ThreadProfileSlot merged;
40 for (int tid = 0; tid < kMaximumParallelThreads; ++tid) {
41 const ThreadProfileSlot& slot = slots[static_cast<std::size_t>(tid)];
42 for (unsigned i = 0; i < static_cast<unsigned>(ProfileRegion::Count); ++i)
43 merged.seconds[i] += slot.seconds[i];
44 for (unsigned i = 0; i < static_cast<unsigned>(TimestepLimiter::Count); ++i)
45 merged.limiters[i] += slot.limiters[i];
46 merged.inversions.calls += slot.inversions.calls;
47 merged.inversions.initial_guess_accepts += slot.inversions.initial_guess_accepts;
48 merged.inversions.one_update_convergences += slot.inversions.one_update_convergences;
49 merged.inversions.total_iterations += slot.inversions.total_iterations;
50 merged.inversions.maximum_iterations = std::max(
51 merged.inversions.maximum_iterations, slot.inversions.maximum_iterations);
52 merged.inversions.bisection_fallbacks += slot.inversions.bisection_fallbacks;
53 merged.inversions.bracket_evaluations += slot.inversions.bracket_evaluations;
54 merged.pressure_inversions.calls += slot.pressure_inversions.calls;
55 merged.pressure_inversions.hinted_calls += slot.pressure_inversions.hinted_calls;
56 merged.pressure_inversions.evaluations += slot.pressure_inversions.evaluations;
57 merged.pressure_inversions.maximum_evaluations = std::max(
58 merged.pressure_inversions.maximum_evaluations,
59 slot.pressure_inversions.maximum_evaluations);
60 merged.pressure_inversions.bisection_fallbacks +=
61 slot.pressure_inversions.bisection_fallbacks;
62 merged.conduction_calls += slot.conduction_calls;
63 merged.conduction_total_iterations += slot.conduction_total_iterations;
64 merged.conduction_maximum_iterations = std::max(
65 merged.conduction_maximum_iterations, slot.conduction_maximum_iterations);
66 }
67 return merged;
68}

70const char* region_name(ProfileRegion region) {
71 switch (region) {
72 case ProfileRegion::Cfl: return "cfl";

174void print_runtime_profile(std::ostream& out) {
175 if (!enabled) return;
176 const ThreadProfileSlot merged = merged_profile();
177 out << "\n===== CHROMO runtime profile =====\n";
178 out << std::fixed << std::setprecision(6);
179 for (unsigned i = 0; i < static_cast<unsigned>(ProfileRegion::Count); ++i)
180 out << "time." << region_name(static_cast<ProfileRegion>(i)) << "_s="
181 << merged.seconds[i] << '\n';
182 for (unsigned i = 0; i < static_cast<unsigned>(TimestepLimiter::Count); ++i)
183 out << "limiter." << limiter_name(static_cast<TimestepLimiter>(i)) << '='
184 << merged.limiters[i] << '\n';
185 const EosInversionProfile& inversions = merged.inversions;
186 const PressureInversionProfile& pressure = merged.pressure_inversions;
187 const double average = inversions.calls
188 ? static_cast<double>(inversions.total_iterations)/inversions.calls : 0.0;
189 out << "inversion.calls=" << inversions.calls << '\n'
190 << "inversion.initial_guess_accepts=" << inversions.initial_guess_accepts << '\n'
191 << "inversion.one_update_convergences=" << inversions.one_update_convergences << '\n'
192 << "inversion.average_iterations=" << average << '\n'
193 << "inversion.maximum_iterations=" << inversions.maximum_iterations << '\n'
194 << "inversion.bisection_fallbacks=" << inversions.bisection_fallbacks << '\n'
195 << "inversion.bracket_evaluations=" << inversions.bracket_evaluations << '\n'
196 << "pressure_inversion.calls=" << pressure.calls << '\n'
197 << "pressure_inversion.hinted_calls=" << pressure.hinted_calls << '\n'
198 << "pressure_inversion.average_evaluations="
199 << (pressure.calls
200 ? static_cast<double>(pressure.evaluations)/pressure.calls : 0.0) << '\n'
201 << "pressure_inversion.maximum_evaluations=" << pressure.maximum_evaluations << '\n'
202 << "pressure_inversion.bisection_fallbacks=" << pressure.bisection_fallbacks << '\n'
203 << "conduction.calls=" << merged.conduction_calls << '\n'
204 << "conduction.average_newton_updates="
205 << (merged.conduction_calls
206 ? static_cast<double>(merged.conduction_total_iterations)/merged.conduction_calls
207 : 0.0) << '\n'
208 << "conduction.maximum_newton_updates=" << merged.conduction_maximum_iterations << '\n'
209 << "==================================\n";
210}
