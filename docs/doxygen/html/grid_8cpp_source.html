104// Exact (bit-for-bit) equality of a cached geometry snapshot against the live
105// array. A mismatch in length counts as changed.
106bool same_vector(const Vec& cached, const Vec& live) {
107 if (cached.n_elem != live.n_elem) return false;
108 for (arma::uword i = 0; i < live.n_elem; ++i)
109 if (!(cached[i] == live[i])) return false;
110 return true;
111}

114bool Grid::static_metrics_current() const {
115 return metrics_valid

117 && s_face.n_elem == ns + 1

130void Grid::force_rebuild_metrics() {
131 // --- static-mesh metric caches --------------------------------------------
132 // Rebuild the center-to-center distances and canonical coordinates from ds_i,
133 // taking cell centers at face midpoints. Mirror the boundary cell width into
134 // the ghost (Neumann) at both ends so ds_iph_i[ns-1]/ds_imh_i[0] reproduce the
135 // legacy 0.5*(ds_i + ip1/im1(ds_i,SLICE)) expressions used by rhs/conduction.
136 if (ds_i.n_elem == ns && ns > 0) {
137 const float dmin = ds_i.min();
138 const float dmax = ds_i.max();
139 // Reject a genuinely broken mesh. The all-zero state right after init()
140 // (dmax == 0) is treated as "not yet populated" and skipped, not rejected.
141 if (dmax > 0.0f) {
142 if (!ds_i.is_finite() || dmin <= 0.0f)
143 throw std::runtime_error(
144 "Grid::broadcast: ds_i must be strictly positive and finite");
145 // Relative spread tolerance: 1 part in 10^6 counts as uniform.
146 uniform_mesh = (dmax - dmin) <= 1.0e-6f * dmax;
147
148 s_face(0) = 0.0f;
149 for (arma::uword i = 0; i < ns; ++i)
150 s_face(i + 1) = s_face(i) + ds_i(i);
151 for (arma::uword i = 0; i < ns; ++i) {
152 s_i(i) = 0.5f * (s_face(i) + s_face(i + 1));
153 const float ds_ip1 = (i + 1 < ns) ? ds_i(i + 1) : ds_i(i); // mirror
154 const float ds_im1 = (i > 0) ? ds_i(i - 1) : ds_i(i); // mirror
155 ds_iph_i(i) = 0.5f * (ds_i(i) + ds_ip1);
156 ds_imh_i(i) = 0.5f * (ds_im1 + ds_i(i));
157 }
158 }
159 }
160
161 // Packed (n_state) broadcasts. Previously built as
162 // dst.zeros(n_state); for k: dst += scalar_to(*this, src, k);
163 // which allocated num_of_eq n_state-sized temporaries per field. The direct
164 // fill below writes the same values; the explicit `0.0f +` reproduces the
165 // accumulation's one signed-zero difference (0.0f + -0.0f == +0.0f) so the
166 // result is bit-for-bit what the old loop produced.
167 auto pack = [&](Vec& dst, const Vec& src) {
168 dst.set_size(n_state);
169 for (arma::uword k = 0; k < num_of_eq; ++k) {
170 float* out = dst.memptr() + k * ns;
171 for (arma::uword i = 0; i < ns; ++i) out[i] = 0.0f + src[i];
172 }
173 };
174 pack(B_state_imh, B_imh);
175 pack(B_state_iph, B_iph);
176 pack(B_state, B_i);
177 pack(ds_state, ds_i);
178 pack(dinvB_ds_state, dinvB_ds_i);
179 pack(ds_iph_state, ds_iph_i);
180 pack(ds_imh_state, ds_imh_i);
181
182 metrics_ds_i = ds_i;
183 metrics_B_i = B_i;
184 metrics_B_imh = B_imh;
185 metrics_B_iph = B_iph;
186 metrics_dinvB_ds_i = dinvB_ds_i;
187 metrics_valid = true;
188 ++metrics_generation_;
189}
