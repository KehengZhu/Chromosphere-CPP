22Vec analytic_canopy_ic(Grid& grid) {
23 // Reuse the full C7 setup (thermodynamics, gravity, ghost cells, conserved
24 // state). model_c7_ic leaves B=1, dinvB_ds=0; we now override.
25 Vec xn = model_c7_ic(grid);
26
27 // Walk up the field line using grid.ds_i to get the height at each face.
28 // Field is vertical here, so arc length s == height above the C7 base.
29 float s_imh = 0.0f;
30 for (arma::uword i = 0; i < grid.ns; ++i) {
31 const float s_iph = s_imh + grid.ds_i(i);
32 const float Bimh = kBinf_T + (kB0_T - kBinf_T) * std::exp(-s_imh / kHB_m);
33 const float Biph = kBinf_T + (kB0_T - kBinf_T) * std::exp(-s_iph / kHB_m);
34 grid.B_imh(i) = Bimh;
35 grid.B_iph(i) = Biph;
36 grid.B_i (i) = 0.5f * (Bimh + Biph);
37 grid.dinvB_ds_i(i) = (1.0f / Biph - 1.0f / Bimh) / grid.ds_i(i);
38 s_imh = s_iph;
39 }
40
41 grid.broadcast(); // re-sync the packed _state caches after B/dinvB changes
42 return xn;
43}
