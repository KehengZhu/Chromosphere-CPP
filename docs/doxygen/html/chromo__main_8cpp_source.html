1/*!
2 * @file chromo_main.cpp
3 * @brief Command-line driver: parse arguments and environment, build the
4 * scenario, run the timestep loop, and write snapshots and sidecars.
5 * @ingroup driver
6 *
7 * The driver owns no physics. It selects a solver, steps it, and writes files:
8 *
9 * 1. make_scenario() builds the requested IC/BC pair.
10 * 2. Grid::init() allocates the mesh; the scenario IC fills the geometry,
11 * gravity, ghost buffers and the initial conserved state (and may resize
12 * the Grid when it generates a locally refined mesh).
13 * 3. A loaded `Gamma1` table selects the RELEASE single-fluid solver
14 * (mixture_advance()); without one the driver runs the historical
15 * two-fluid solver (advance_Euler_state() / advance_Euler_explicit_state()).
16 * The release path rejects `SINGLE_FLUID`, `ENABLE_TE`, `ISO_TWO_FLUID`,
17 * `ISO_IONIZATION` and finite-rate ionization outright.
18 * 4. Each step refreshes the boundary, computes the CFL timestep, advances,
19 * and writes a snapshot when the OutputSchedule says it is due.
20 * 5. On exit it prints `termination=end_time|step_cap|other` together with the
21 * requested end time, final time, final step and effective cap.
22 *
23 * The full command-line and environment reference is on the
24 * @ref configuration "Configuration reference" page; the file formats written
25 * here are described on the @ref io_formats "Output formats" page.
26 */

64std::string sha256_file(const std::string& path) {
65 std::ifstream input(path, std::ios::binary);
66 if (!input) throw std::runtime_error("cannot hash Gamma1 table: "+path);
67 std::vector<uint8_t> bytes((std::istreambuf_iterator<char>(input)), {});
68 const uint64_t bit_length = static_cast<uint64_t>(bytes.size())*8;
69 bytes.push_back(0x80);
70 while (bytes.size()%64 != 56) bytes.push_back(0);
71 for (int shift = 56; shift >= 0; shift -= 8)
72 bytes.push_back(static_cast<uint8_t>(bit_length >> shift));
73 static constexpr uint32_t k[64] = {
74 0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,
75 0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,
76 0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,
77 0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,
78 0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,
79 0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,0xd192e819,0xd6990624,0xf40e3585,0x106aa070,
80 0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,
81 0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2};
82 std::array<uint32_t,8> h = {0x6a09e667,0xbb67ae85,0x3c6ef372,0xa54ff53a,
83 0x510e527f,0x9b05688c,0x1f83d9ab,0x5be0cd19};
84 for (std::size_t off = 0; off < bytes.size(); off += 64) {
85 uint32_t w[64];
86 for (int i = 0; i < 16; ++i)
87 w[i] = (uint32_t(bytes[off+4*i])<<24)|(uint32_t(bytes[off+4*i+1])<<16)
88 | (uint32_t(bytes[off+4*i+2])<<8)|uint32_t(bytes[off+4*i+3]);
89 for (int i = 16; i < 64; ++i) {
90 const uint32_t s0 = rotr(w[i-15],7)^rotr(w[i-15],18)^(w[i-15]>>3);
91 const uint32_t s1 = rotr(w[i-2],17)^rotr(w[i-2],19)^(w[i-2]>>10);
92 w[i] = w[i-16]+s0+w[i-7]+s1;
93 }
94 uint32_t a=h[0],b=h[1],c=h[2],d=h[3],e=h[4],f=h[5],g=h[6],hh=h[7];
95 for (int i = 0; i < 64; ++i) {
96 const uint32_t s1=rotr(e,6)^rotr(e,11)^rotr(e,25);
97 const uint32_t ch=(e&f)^((~e)&g);
98 const uint32_t t1=hh+s1+ch+k[i]+w[i];
99 const uint32_t s0=rotr(a,2)^rotr(a,13)^rotr(a,22);
100 const uint32_t maj=(a&b)^(a&c)^(b&c);
101 const uint32_t t2=s0+maj;
102 hh=g; g=f; f=e; e=d+t1; d=c; c=b; b=a; a=t1+t2;
103 }
104 h[0]+=a;h[1]+=b;h[2]+=c;h[3]+=d;h[4]+=e;h[5]+=f;h[6]+=g;h[7]+=hh;
105 }
106 std::ostringstream out;
107 out << std::hex << std::setfill('0');
108 for (uint32_t word : h) out << std::setw(8) << word;
109 return out.str();

114/// Command-line entry point.
115///
116/// Reads the positional arguments and the `CHROMO_*` / `ISO_*` environment,
117/// builds the requested Scenario, allocates and initialises the Grid, then runs
118/// the timestep loop — boundary refresh, CFL timestep, solver advance, snapshot
119/// when due — until the end time or the step cap is reached, writing the
120/// snapshot, log and sidecar files on the way and the `termination=` summary at
121/// the end. See the @ref configuration "Configuration reference" for the full
122/// argument and environment list.
123///
124/// @param argc Argument count.
125/// @param argv `[output_path] [mode] [ionization] [scenario] [data_path]
126/// [time_mult] [cooling]`, each optional and positional; the
127/// defaults are documented at the top of the function body.
128/// @return 0 on a completed run; non-zero after a fatal configuration or
129/// solver error, whose message is printed to stderr.

136 // ionization: "ionization" or "no-ionization" (default: no-ionization for
137 // the default model_column release, ionization otherwise)
138 // scenario: "model_column" (default — the single-fluid RELEASE column) |
139 // "model_gentle" | "model_c7" | "model_flare" |
140 // "pfss_field_line" | "analytic_canopy" (all two-fluid)
141 // data_path: required for tabulated scenarios (e.g. pfss_field_line);

445 << "# columns=cell_km face_km rho_cell v_cell T_cell "
446 "rho_L rho_R v_L v_R T_L T_R cs_L cs_R a_face "
447 "f_central f_diff f_total "
448 "r_rho phi_plus_rho r_ip1_rho phi_minus_rho "
449 "r_v phi_plus_v r_T phi_plus_T "
450 "p_cell p_L p_R\n";
451 face_flux.precision(10);
452 }
453
454 // Outer-face conduction sidecar. One row per captured step, holding the outer-face
455 // quantities of the FINAL CONVERGED Newton iteration of that step's conduction
456 // solve. t/step label the START of the step, matching the .faceflux convention.
457 std::ofstream outer_cond;
458 int outer_cond_stride = face_flux_stride;
459 if (const char* e = std::getenv("CHROMO_OUTER_COND_STRIDE")) {
460 try { outer_cond_stride = std::max(1, std::stoi(e)); } catch (...) {}
461 }
462 if (write_outer_cond) {
463 if (!release_mode)
464 throw std::runtime_error(
465 "CHROMO_OUTER_COND_DIAG requires the release solver (the capture "
466 "lives in mixture_apply_conduction)");
467 outer_cond.open(out_path+".outercond");
468 if (!outer_cond) throw std::runtime_error("cannot open outer-conduction sidecar");
469 outer_cond << "# outer-face conduction diagnostic (read-only capture of the "
470 "final converged mixture_apply_conduction iteration)\n"
471 << "# ns=" << grid.ns << " ds_km=" << (grid.ds_i(grid.ns-1)*1.0e-3f)
472 << " uniform_mesh=" << grid.uniform_mesh
473 << " impose_outer_heat_flux=" << grid.impose_outer_heat_flux << '\n'
474 << "# release conduction is PHYSICAL only (kappa_e + kappa_n; no "
475 "TRAC, no artificial diffusivity)\n"
476 << "# ds_face = half the top cell when an external face "
477 "temperature is imposed, else the centre-to-ghost-centre span\n"
478 << "# q_face = kappa_face*(T_wall - T_top)/ds_face"
479 " [W m^-2, positive = into the top cell]\n"
480 << "# columns=t step T_top T_wall kappa_face "
481 "ds_face area_ratio q_face\n";
482 outer_cond.precision(10);
483 }
484
485 auto write_outer_cond_record = [&](float t_now, long long step_now) {
486 const OuterConductionCapture& oc = grid.outer_conduction_capture;
487 if (!outer_cond.is_open() || !oc.valid) return;
488 outer_cond << t_now << ' ' << step_now << ' '
489 << oc.T_top << ' ' << oc.T_wall << ' '
490 << oc.kappa_face << ' '
491 << oc.ds_face << ' ' << oc.area_ratio << ' '
492 << oc.q_face << '\n';
493 };
494
495 auto write_face_record = [&](float t_now, long long step_now) {
496 const MixtureFaceFluxCapture& c = grid.face_flux_capture;
497 if (!face_flux.is_open() || !c.valid) return;
498 {
499 face_flux << "# t = " << t_now << " step = " << step_now << '\n';
500 for (arma::uword i = face_flux_lo; i < grid.ns; ++i) {
501 face_flux << face_cell_km(i) << ' ' << face_face_km(i) << ' '
502 << c.rho_cell[i] << ' ' << c.v_cell[i] << ' ' << c.T_cell[i] << ' '
503 << c.rho_L[i] << ' ' << c.rho_R[i] << ' '
504 << c.v_L[i] << ' ' << c.v_R[i] << ' '
505 << c.T_L[i] << ' ' << c.T_R[i] << ' '
506 << c.cs_L[i] << ' ' << c.cs_R[i] << ' ' << c.a_face[i] << ' '
507 << c.f_central[i] << ' ' << c.f_diff[i] << ' ' << c.f_total[i] << ' '
508 << c.r_rho[i] << ' ' << c.phi_plus_rho[i] << ' '
509 << c.r_ip1_rho[i] << ' ' << c.phi_minus_rho[i] << ' '
510 << c.r_v[i] << ' ' << c.phi_plus_v[i] << ' '
511 << c.r_T[i] << ' ' << c.phi_plus_T[i] << ' '
512 << c.p_cell[i] << ' ' << c.p_L[i] << ' ' << c.p_R[i] << '\n';
513 }
514 }
515 };
516
517 float time = 0.0f;
518 long long step = 0;
519 std::uint64_t state_generation = 0;
520 MixtureField decoded_storage[2];
521 MixtureField* decoded=&decoded_storage[0];
522 const MixtureField* previous_decoded=nullptr;
523 if (schedule.due(step, time)) {
524 write_frame(time, step);
525 schedule.note_written(step, time);
526 }
527
528 while (time < total_time && step < step_cap) {
529 grid.sim_time = time; // expose current time to time-dependent terms (beam window)
530 {
531 ProfileScope boundary_timer(ProfileRegion::Boundary);
532 sc.update_bc(grid, xn);
533 }
534 Vec dt;
535 if (release_mode) {
536 mixture_decode_into(grid,xn,*decoded,state_generation,previous_decoded);
537 dt = mixture_timestep(grid,xn,*decoded);
538 } else {
539 dt = cal_dt_i(grid,xn);
540 }
541 const float dt_avg = arma::mean(dt);
542 // Arm the read-only face-flux capture for THIS step's single RHS call. dt is
543 // already known here, so the final step of the run is always captured.
544 if (write_face_flux)
545 grid.capture_face_flux = (step % face_flux_stride == 0)
546 || (time + dt_avg >= total_time) || (step + 1 >= step_cap);
547 const bool capture_cond_now = write_outer_cond
548 && ((step % outer_cond_stride == 0)
549 || (time + dt_avg >= total_time) || (step + 1 >= step_cap));
550 grid.capture_outer_conduction = capture_cond_now;
551
552 if (step % progress_stride == 0) {
553 std::cout << "[" << mode << "] step = " << step
554 << " dt = " << dt_avg
555 << " time = " << time
556 << " T_c = " << grid.trac_cutoff_T << std::endl;
557 }
558
559 if (release_mode)
560 xn = mixture_advance(grid,xn,dt,*decoded);
561 else
562 xn = explicit_only ? advance_Euler_explicit_state(grid,xn,dt)
563 : advance_Euler_state(grid,xn,dt);
564 if (grid.capture_face_flux) {
565 write_face_record(time, step); // t/step of the state the RHS was taken at
566 grid.capture_face_flux = false;
567 }
568 if (capture_cond_now) {
569 write_outer_cond_record(time, step);
570 grid.capture_outer_conduction = false;
571 }
572 time += dt_avg;
573 ++step;
574 ++state_generation;
575 if (release_mode) {
576 previous_decoded=decoded;
577 decoded = decoded == &decoded_storage[0]
578 ? &decoded_storage[1] : &decoded_storage[0];
579 }
580
581 if (schedule.due(step, time)) {
582 write_frame(time, step);
583 schedule.note_written(step, time);
584 }
585 }
586 // Always capture the final state, even when it does not land on the cadence.
587 if (!schedule.already_written(step)) {
588 write_frame(time, step);
589 schedule.note_written(step, time);
590 }
591
592 std::cout << "[" << mode << "] END! step=" << step << " time=" << time << std::endl;
593 const Termination termination =
594 classify_termination(time, total_time, step, step_cap);
595 std::cout << "termination=" << termination_name(termination) << '\n'
596 << "requested_end_time=" << total_time << '\n'
597 << "final_physical_time=" << time << '\n'
598 << "final_step=" << step << '\n'
599 << "effective_step_cap=" << step_cap
600 << " (source=" << step_cap_source_name(cap_sel.source) << ")" << std::endl;
601 if (termination == Termination::StepCap) {
602 std::cerr << "WARNING: run stopped at the step cap (" << step_cap
603 << ") after " << time << " of the requested " << total_time
604 << " physical seconds. Raise or unset CHROMO_STEP_CAP." << std::endl;
605 }
606 print_runtime_profile(std::cout);
607 if (eos_counting_on) {
608 const EosOperationCounts counts = eos_operation_counts();
609 std::cout << "eos.gamma1_queries=" << counts.gamma1_queries << '\n'
610 << "eos.temperature_logs=" << counts.temperature_logs << '\n'
611 << "eos.n_h_logs=" << counts.n_h_logs << std::endl;
612 }
613 return 0;
614}
