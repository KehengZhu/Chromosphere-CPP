21Vec pfss_ic(Grid& grid, const std::string& data_path) {
22 const ScenarioDataFile f = parse_scenario_data_file(data_path);
23 if (f.ns != grid.ns)
24 throw std::runtime_error("pfss_ic: data file ns=" + std::to_string(f.ns)
25 + " differs from Grid ns=" + std::to_string(grid.ns));
26
27 // Topology sets the outer-face closure. Both "closed" (half loop: outer face
28 // is the apex, a reflecting symmetry plane) and "full" (whole loop: outer face
29 // is footpoint B, a reflecting chromospheric wall) want a reflecting outer face;
30 // the inner face is always reflecting. An "open" line keeps the coronal outflow.
31 grid.outer_reflecting = (f.topology == "closed" || f.topology == "full");
32
33 // Field-line geometry from the file.
34 for (arma::uword i = 0; i < grid.ns; ++i) {
35 grid.ds_i (i) = f.ds_m (i);
36 grid.B_imh (i) = f.B_imh_T(i);
37 grid.B_iph (i) = f.B_iph_T(i);
38 grid.B_i (i) = 0.5f * (f.B_imh_T(i) + f.B_iph_T(i));
39 grid.phi_g_imh(i) = f.phi_g_imh(i);
40 grid.phi_g_iph(i) = f.phi_g_iph(i);
41 grid.dinvB_ds_i(i) = (1.0f / f.B_iph_T(i) - 1.0f / f.B_imh_T(i)) / f.ds_m(i);
42 }
43
44 // Build the conserved-variable state. Energy includes ρ·φ_g so that
45 // cons2prim (state.cpp:108) recovers the right pressure.
46 Vec xn = arma::zeros<Vec>(grid.n_state);
47 for (arma::uword i = 0; i < grid.ns; ++i) {
48 const float n_e = f.ne_im3(i);
49 const float n_n = f.nn_im3(i);
50 const float T = f.T_K (i);
51 const float phi_g = 0.5f * (grid.phi_g_imh(i) + grid.phi_g_iph(i));
52 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::RHO_I)) = n_e * grid.m_i;
53 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::RHO_N)) = n_n * grid.m_n;
54 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::MOM_I)) = 0.0f;
55 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::MOM_N)) = 0.0f;
56 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::E_I))
57 = grid.inv_gm1() * grid.k_b * n_e * 2.0f * T + n_e * grid.m_i * phi_g;
58 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::E_N))
59 = grid.inv_gm1() * grid.k_b * n_n * T + n_n * grid.m_n * phi_g;
60 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, cons::E_E))
61 = grid.inv_gm1() * grid.k_b * n_e * T; // electron internal energy, T_e = T
62 }
63
64 // Ghost cells. φ_g convention matches model_c7_ic (and rhs.cpp:154,178):
65 // outer ghost uses phi_g_iph[ns-1], inner ghost uses phi_g_imh[0]. The
66 // T_e_factor column scales the ghost's effective temperature (model_c7
67 // uses 2.0 at the outer ghost to mimic coronal T_e).
68 auto fill_ghost = [&](Vec& ob, arma::uword idx, float phi_g_eff) {
69 const float n_e = f.ghost_ne_im3 (idx);
70 const float n_n = f.ghost_nn_im3 (idx);
71 const float T_eff = f.ghost_T_K (idx) * f.ghost_T_e_factor(idx);
72 ob(cons::RHO_I) = n_e * grid.m_i;
73 ob(cons::RHO_N) = n_n * grid.m_n;
74 ob(cons::MOM_I) = 0.0f;
75 ob(cons::MOM_N) = 0.0f;
76 ob(cons::E_I) = grid.inv_gm1() * grid.k_b * n_e * 2.0f * T_eff + n_e * grid.m_i * phi_g_eff;
77 ob(cons::E_N) = grid.inv_gm1() * grid.k_b * n_n * T_eff + n_n * grid.m_n * phi_g_eff;
78 ob(cons::E_E) = grid.inv_gm1() * grid.k_b * n_e * T_eff; // T_e = T_eff at the ghost
79 };
80
81 const float phi_g_outer = grid.phi_g_iph(grid.ns - 1);
82 const float phi_g_inner = grid.phi_g_imh(0);
83 fill_ghost(grid.outer_boundary0_i, /*idx=*/0, phi_g_outer);
84 fill_ghost(grid.outer_boundary1_i, /*idx=*/1, phi_g_outer);
85 fill_ghost(grid.inner_boundary0_i, /*idx=*/2, phi_g_inner);
86 fill_ghost(grid.inner_boundary1_i, /*idx=*/3, phi_g_inner);
87
88 // --- Optional flare beam-heating overlay (Phase-3 de-risk) -------------
89 // Opt-in via PFSS_FLARE=1: run the model_flare physics (Fisher, Canfield &
90 // McClymont 1985 nonthermal-beam explosive evaporation) on THIS field line's
91 // PFSS geometry + open BC instead of the uniform-B C7 column. Mirrors
92 // model_flare_ic: force ionization + radiative cooling (the radiative sink
93 // sets the explosive threshold), then add the support physics model_c7_ic
94 // configures but pfss_ic otherwise omits — grid-scaled numerical diffusivity
95 // (shock stabilization), TRAC (under-resolved TR), and the RTV (1978) coronal
96 // conductive flux q(T) — and switch the beam on. All knobs share the FLARE_*
97 // env names with model_flare_ic so one build serves both. Default off ⇒ the
98 // plain pfss_field_line scenario (and the test suite) is unchanged.
99 //
100 // NOTE (de-risk shortcut): the C7 background here is NOT re-calibrated to be a
101 // Stage-E fixed point (model_c7_ic does that by inverting P_phot over the
102 // active network; pfss_ic uses the scalar fallback ~1e-4 s⁻¹). Over the ~15 s
103 // beam-driven run the resulting quiet-background ionization drift is ~0.1% and
104 // negligible against the beam, but the per-line ensemble (Phase 2) will need
105 // the full height-dependent calibration to keep non-flaring lines steady.
106 auto env_f = [](const char* key, float fallback) -> float {
107 if (const char* e = std::getenv(key)) {
108 try { return std::stof(std::string(e)); } catch (...) {}
109 }
110 return fallback;
111 };
112 if (env_f("PFSS_FLARE", 0.0f) != 0.0f) {
113 grid.enable_ionization = true;
114 grid.enable_radiative_cooling = true;
115
116 grid.numerical_diffusivity_per_length =
117 2.0e3f * env_f("FLARE_DIFF_MULT", 1.0f);
118 grid.enable_trac = true;
119 grid.trac_T_chrom = 2.0e4f;
120 { // RTV F_c = (2/7) κ₀ T_cor^{7/2} / L (quiet-Sun defaults, as model_c7_ic).
121 // Per-line L from the PFSS trace is a Phase-2 refinement.
122 const float kappa0 = 1.0e-11f, T_cor = 8.0e5f, L_cor = 4.5e7f;
123 grid.outer_heat_flux = (2.0f / 7.0f) * kappa0 * std::pow(T_cor, 3.5f) / L_cor;
124 grid.impose_outer_heat_flux = true;
125 }
126
127 grid.enable_beam_heating = true;
128 grid.beam_flux = env_f("FLARE_BEAM_FLUX", 5.0e7f); // W/m² (explosive default)
129 grid.beam_t_on = env_f("FLARE_T_ON", 2.0f);
130 grid.beam_duration = env_f("FLARE_DUR", 10.0f);
131 grid.beam_ramp = env_f("FLARE_RAMP", 1.0f);
132 grid.beam_h_lo_km = env_f("FLARE_H_LO", 1600.0f);
133 grid.beam_h_hi_km = env_f("FLARE_H_HI", 2000.0f);
134 // Self-consistent thick-target beam (Emslie 1978): inject at the loop apex
135 // and attenuate down each leg by collisional stopping, with the stopping
136 // depth rising as evaporation fills the loop. On by default for the loop
137 // (the fixed window is unphysical here); FLARE_THICK_TARGET=0 reverts to it.
138 grid.beam_thick_target = (env_f("FLARE_THICK_TARGET", 1.0f) != 0.0f);
139 grid.beam_E_cut_keV = env_f("FLARE_E_CUT", 20.0f); // low-energy cutoff [keV]
140 grid.beam_delta = env_f("FLARE_DELTA", 5.0f); // spectral index δ
141 // Informational here: pfss_update_bc's apply_open_bcs is already a
142 // transparent (Neumann) outflow, so the supersonic upflow can exit
143 // regardless. Set for parity with model_flare and any future BC swap.
144 grid.outer_free_outflow = (env_f("FLARE_FREE_OUTFLOW", 1.0f) != 0.0f);
145 }
146
147 // --- Optional ambient-coronal-heating overlay (GENTLE conduction-driven
148 // evaporation; docs/gentle_evaporation_plan.md) -------------------------
149 // Opt-in via GENTLE=1 (the alternative to PFSS_FLARE — gentle vs explosive).
150 // Adds the steady footpoint-anchored heating H(s) that turns the resolved
151 // chromosphere→TR→corona column into a true steady state — the RTV (1978) /
152 // Aschwanden & Schrijver (2002) static balance the prescribed-T loop .dat
153 // otherwise lacks — together with conduction + radiative cooling. There is NO
154 // beam: gentle evaporation (Antiochos & Sturrock 1978) is driven by a slow,
155 // sub-threshold time RAMP of the heating (GENTLE_ENHANCE > 1), keeping v ≪ c_s.
156 // Because the corona is heated INSIDE the domain, no boundary q(T) crutch is
157 // imposed (unlike model_c7). Default off ⇒ plain pfss_field_line scenario.
158 if (env_f("GENTLE", 0.0f) != 0.0f) {
159 grid.enable_ionization = true;
160 grid.enable_radiative_cooling = true;
161 grid.enable_vacuum_floor = true; // fully-ionized corona: n_n→0 conditioning
162
163 // Well-balanced explicit reconstruction (chromosphere.hpp::well_balanced,
164 // docs/gentle_evaporation_downflow.md). Undoes the φ_g inconsistency in the
165 // MUSCL slope that otherwise gives every hydrostatic atmosphere a spurious
166 // (γ−1)g downforce — the resolution-independent root of the persistent
167 // chromosphere/TR downflow. Validated on model_column and ported to
168 // model_c7; enabling it here removes that artifact from the resolved
169 // chromosphere→TR→corona loop so the relaxed preflare baseline sits closer
170 // to V≈0 and the gentle (Antiochos & Sturrock 1978) upflow is measured
171 // against a quieter zero. The genuine energy-budget downflow (a corona kept
172 // up only by conduction) is physical and survives the fix. Default on for
173 // GENTLE; GENTLE_WELL_BALANCED=0 reverts to the old reconstruction.
174 grid.well_balanced = (env_f("GENTLE_WELL_BALANCED", 1.0f) != 0.0f);
175
176 grid.numerical_diffusivity_per_length =
177 2.0e3f * env_f("GENTLE_DIFF_MULT", 1.0f);
178 grid.enable_trac = true;
179 grid.trac_T_chrom = 2.0e4f;
180
181 // Loop half-length L [m]: from the .dat [META] if present, else the
182 // arc-length sum (a full loop spans 2L, so halve it).
183 float L = f.loop_half_length_m;
184 if (L <= 0.0f) {
185 const float arc = arma::sum(grid.ds_i);
186 L = (f.topology == "full") ? 0.5f * arc : arc;
187 }
188
189 // RTV (1978) + Aschwanden & Schrijver (2002) heating calibration (cgs),
190 // then →SI. Inverts the static scaling for the footpoint heating E_H0 that
191 // relaxes a loop of half-length L to apex temperature T_max:
192 // p₀ = (T_max/1400)³ / L_cm [dyne cm⁻²] (RTV eq 4.3)
193 // E_unif = 9.8e4 · p₀^{7/6} · L_cm^{−5/6} [erg cm⁻³ s⁻¹](RTV eq 4.4)
194 // E_H0 = E_unif · exp(½ L/s_H) [peak, footpoint; Serio 1981]
195 // →SI × 0.1 W m⁻³.
196 // s_H/L ≈ 0.2–0.5 (Aschwanden 2001: s_H ≈ 12 ± 5 Mm); keep ≳ 0.3 to stay
197 // above the Serio/Martens static-existence limit (no apex-max loop below
198 // ~L/3). T_max is the relaxed OUTCOME — verify it post-relaxation and nudge
199 // GENTLE_E0 if the corona settles off target.
200 const float T_max = env_f("GENTLE_TMAX_MK", 2.0f) * 1.0e6f;
201 const float sH_frac = env_f("GENTLE_SH_FRAC", 0.4f);
202 const float s_H = sH_frac * L;
203 const float L_cm = L * 100.0f;
204 const float p0 = std::pow(T_max / 1400.0f, 3.0f) / L_cm;
205 const float E_unif = 9.8e4f * std::pow(p0, 7.0f / 6.0f) * std::pow(L_cm, -5.0f / 6.0f);
206 const float E_H0_SI = 0.1f * E_unif * std::exp(0.5f * L / s_H);
207
208 grid.enable_coronal_heating = true;
209 grid.coronal_heat_E0 = env_f("GENTLE_E0", E_H0_SI); // W/m³ (override allowed)
210 grid.coronal_heat_sH = s_H;
211 grid.coronal_heat_s0 = env_f("GENTLE_S0_MM", 0.0f) * 1.0e3f;
212 grid.coronal_heat_two_sided = (f.topology == "full");
213 // Phase-3 driver. enhance = 1 (default) ⇒ pure steady relaxation; a modest
214 // 2–5× rise over ≫ one sound-crossing time stays in the gentle regime.
215 grid.coronal_heat_enhance = env_f("GENTLE_ENHANCE", 1.0f);
216 grid.coronal_heat_t_on = env_f("GENTLE_T_ON", 0.0f);
217 grid.coronal_heat_ramp = env_f("GENTLE_RAMP", 30.0f);
218
219 // Let the gentle upflow leave an open top (no Mach cap in apply_open_bcs
220 // anyway; reflecting apex for closed/full confines and fills the loop).
221 grid.outer_free_outflow = (env_f("GENTLE_FREE_OUTFLOW", 1.0f) != 0.0f);
222 }
223
224 grid.broadcast();
225 return xn;
226}
