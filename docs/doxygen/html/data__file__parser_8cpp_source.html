18// Trim leading/trailing ASCII whitespace in-place.
19std::string trim(std::string s) {
20 const auto isws = [](unsigned char c) { return c == ' ' || c == '\t' || c == '\r'; };
21 while (!s.empty() && isws((unsigned char)s.back())) s.pop_back();
22 std::size_t i = 0;
23 while (i < s.size() && isws((unsigned char)s[i])) ++i;
24 return s.substr(i);
25}

32bool is_section_header(const std::string& line, std::string* name_out) {
33 const auto t = trim(line);
34 if (t.size() < 3 || t.front() != '[' || t.back() != ']') return false;
35 if (name_out) *name_out = t.substr(1, t.size() - 2);
36 return true;
37}

49arma::uword parse_uword(const std::string& s, const std::string& ctx) {
50 try {
51 std::size_t pos = 0;
52 const long long v = std::stoll(s, &pos);
53 if (v < 0) throw std::runtime_error("negative integer '" + s + "' (" + ctx + ")");
54 return static_cast<arma::uword>(v);

60// Read meta key=value pairs. Returns true if ns was found.
61bool parse_meta_block(std::ifstream& in, ScenarioDataFile& out, std::string& next_section) {
62 bool ns_seen = false;
63 std::string line;
64 while (std::getline(in, line)) {
65 if (is_skippable(line)) continue;
66 std::string sec;
67 if (is_section_header(line, &sec)) { next_section = sec; return ns_seen; }
68 const auto t = trim(line);
69 const auto eq = t.find('=');
70 if (eq == std::string::npos)
71 throw std::runtime_error("scenario data file: malformed META line '" + t + "'");
72 const auto k = trim(t.substr(0, eq));
73 const auto v = trim(t.substr(eq + 1));
74 if (k == "ns") { out.ns = parse_uword(v, "META ns"); ns_seen = true; }
75 else if (k == "g_si") { out.g_si = parse_float(v, "META g_si"); }
76 else if (k == "B_outer_T") { out.B_outer_T = parse_float(v, "META B_outer_T"); }
77 else if (k == "phi_g_offset_Jpkg") { out.phi_g_offset_Jpkg = parse_float(v, "META phi_g_offset"); }
78 else if (k == "topology") { out.topology = v; }
79 else if (k == "loop_half_length_m") { out.loop_half_length_m = parse_float(v, "META loop_half_length_m"); }
80 // unknown keys are silently ignored — keeps the format extensible.
81 }
82 next_section.clear();
83 return ns_seen;
84}

86void parse_cells_block(std::ifstream& in, ScenarioDataFile& out, std::string& next_section) {
87 if (out.ns == 0)
88 throw std::runtime_error("scenario data file: [CELLS] before ns= in [META]");
89
90 out.ds_m .set_size(out.ns);
91 out.B_imh_T .set_size(out.ns);
92 out.B_iph_T .set_size(out.ns);
93 out.phi_g_imh .set_size(out.ns);
94 out.phi_g_iph .set_size(out.ns);
95 out.ne_im3 .set_size(out.ns);
96 out.nn_im3 .set_size(out.ns);
97 out.T_K .set_size(out.ns);
98
99 arma::uword rows_seen = 0;
100 std::string line;
101 while (std::getline(in, line)) {
102 if (is_skippable(line)) continue;
103 std::string sec;
104 if (is_section_header(line, &sec)) { next_section = sec; goto done; }
105 std::istringstream ss(line);
106 std::vector<std::string> toks;
107 for (std::string t; ss >> t; ) toks.push_back(t);
108 if (toks.size() != 9)
109 throw std::runtime_error("scenario data file: [CELLS] expects 9 columns, got "
110 + std::to_string(toks.size()) + " in row '" + trim(line) + "'");
111 const arma::uword i = parse_uword(toks[0], "[CELLS] index");
112 if (i != rows_seen)
113 throw std::runtime_error("scenario data file: [CELLS] row index "
114 + std::to_string(i) + " out of order, expected "
115 + std::to_string(rows_seen));
116 out.ds_m (i) = parse_float(toks[1], "[CELLS] ds_m");
117 out.B_imh_T (i) = parse_float(toks[2], "[CELLS] B_imh_T");
118 out.B_iph_T (i) = parse_float(toks[3], "[CELLS] B_iph_T");
119 out.phi_g_imh(i) = parse_float(toks[4], "[CELLS] phi_g_imh");
120 out.phi_g_iph(i) = parse_float(toks[5], "[CELLS] phi_g_iph");
121 out.ne_im3 (i) = parse_float(toks[6], "[CELLS] ne_im3");
122 out.nn_im3 (i) = parse_float(toks[7], "[CELLS] nn_im3");
123 out.T_K (i) = parse_float(toks[8], "[CELLS] T_K");
124 ++rows_seen;
125 }
126 next_section.clear();
127done:
128 if (rows_seen != out.ns)
129 throw std::runtime_error("scenario data file: [CELLS] had "
130 + std::to_string(rows_seen) + " rows, expected "
131 + std::to_string(out.ns));
132}

134void parse_ghosts_block(std::ifstream& in, ScenarioDataFile& out, std::string& next_section) {
135 out.ghost_B_T .set_size(4);
136 out.ghost_phi_g_Jpkg.set_size(4);
137 out.ghost_ne_im3 .set_size(4);
138 out.ghost_nn_im3 .set_size(4);
139 out.ghost_T_K .set_size(4);
140 out.ghost_T_e_factor.set_size(4);
141
142 const std::vector<std::string> tag_order = {"outer_0", "outer_1", "inner_0", "inner_1"};
143 arma::uword rows_seen = 0;
144 std::string line;
145 while (std::getline(in, line)) {
146 if (is_skippable(line)) continue;
147 std::string sec;
148 if (is_section_header(line, &sec)) { next_section = sec; goto done; }
149 std::istringstream ss(line);
150 std::vector<std::string> toks;
151 for (std::string t; ss >> t; ) toks.push_back(t);
152 if (toks.size() != 7)
153 throw std::runtime_error("scenario data file: [GHOSTS] expects 7 columns, got "
154 + std::to_string(toks.size()) + " in row '" + trim(line) + "'");
155 if (rows_seen >= tag_order.size())
156 throw std::runtime_error("scenario data file: [GHOSTS] has more than 4 rows");
157 if (toks[0] != tag_order[rows_seen])
158 throw std::runtime_error("scenario data file: [GHOSTS] row "
159 + std::to_string(rows_seen) + " has tag '" + toks[0]
160 + "', expected '" + tag_order[rows_seen] + "'");
161 out.ghost_B_T (rows_seen) = parse_float(toks[1], "[GHOSTS] B_T");
162 out.ghost_phi_g_Jpkg(rows_seen) = parse_float(toks[2], "[GHOSTS] phi_g");
163 out.ghost_ne_im3 (rows_seen) = parse_float(toks[3], "[GHOSTS] ne");
164 out.ghost_nn_im3 (rows_seen) = parse_float(toks[4], "[GHOSTS] nn");
165 out.ghost_T_K (rows_seen) = parse_float(toks[5], "[GHOSTS] T");
166 out.ghost_T_e_factor(rows_seen) = parse_float(toks[6], "[GHOSTS] T_e_factor");
167 ++rows_seen;
168 }
169 next_section.clear();
170done:
171 if (rows_seen != 4)
172 throw std::runtime_error("scenario data file: [GHOSTS] had "
173 + std::to_string(rows_seen) + " rows, expected 4");
174}

178arma::uword peek_ns_from_file(const std::string& path) {
179 std::ifstream in(path);
180 if (!in) throw std::runtime_error("scenario data file: cannot open '" + path + "'");
181 std::string line;
182 bool in_meta = false;
183 while (std::getline(in, line)) {
184 if (is_skippable(line)) continue;
185 std::string sec;
186 if (is_section_header(line, &sec)) {
187 in_meta = (sec == "META");
188 if (!in_meta) continue;
189 continue;
190 }
191 if (!in_meta) continue;
192 const auto t = trim(line);
193 const auto eq = t.find('=');
194 if (eq == std::string::npos) continue;
195 if (trim(t.substr(0, eq)) == "ns") return parse_uword(trim(t.substr(eq + 1)), "META ns");
196 }
197 throw std::runtime_error("scenario data file: ns= not found in [META] section of '" + path + "'");
198}

200ScenarioDataFile parse_scenario_data_file(const std::string& path) {
201 std::ifstream in(path);
202 if (!in) throw std::runtime_error("scenario data file: cannot open '" + path + "'");
203
204 ScenarioDataFile out;
205 std::string line, current_section, next_section;
206
207 // Scan until first section header.
208 while (std::getline(in, line)) {
209 if (is_skippable(line)) continue;
210 if (is_section_header(line, &current_section)) break;
211 throw std::runtime_error("scenario data file: data before first section header");
212 }
213
214 while (!current_section.empty()) {
215 next_section.clear();
216 if (current_section == "META") parse_meta_block (in, out, next_section);
217 else if (current_section == "CELLS") parse_cells_block (in, out, next_section);
218 else if (current_section == "GHOSTS") parse_ghosts_block(in, out, next_section);
219 else throw std::runtime_error("scenario data file: unknown section [" + current_section + "]");
220 current_section = next_section;
221 }
222
223 if (out.ns == 0) throw std::runtime_error("scenario data file: missing [META] ns=");
224 if (out.ds_m.n_elem == 0) throw std::runtime_error("scenario data file: missing [CELLS]");
225 if (out.ghost_B_T.n_elem != 4) throw std::runtime_error("scenario data file: missing [GHOSTS]");
226 return out;
227}
