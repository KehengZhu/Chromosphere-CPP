26/// Single source of truth for the classical pure-hydrogen EOS constants (SI).
27/// EosGammaTable::load() rejects any Gamma1 table whose metadata header does not
28/// declare exactly these five values, so a loaded table and this closure are
29/// guaranteed to have been built from the same physical constants.
30namespace eos_constants {
31constexpr double k_b = 1.380649e-23; ///< Boltzmann constant [J/K]
32constexpr double m_e = 9.1093837015e-31;///< electron rest mass [kg]
33/// Mass per hydrogen nucleus [kg]. The pure-H mixture carries all of its mass in
34/// the nuclei, so the hydrogen-nuclei density is n_H = rho_total / m_h.
35constexpr double m_h = 1.6726219e-27;
36constexpr double h = 6.62607015e-34; ///< Planck constant [J s]
37/// Hydrogen ground-state ionization potential chi_H [J] (13.5984 eV). It sets the
38/// Saha exponential and is the per-ionization energy stored in e_int.
39constexpr double chi_h = 2.179872361e-18;
40} // namespace eos_constants

69/// Algebraic inverse of the pure-H Saha pressure closure. With
70/// S(T) = (2 pi m_e k_B T / h^2)^{3/2} exp(-chi_H / k_B T),
71/// x^2/(1-x) = S/n_H, p = (1+x) n_H k_B T,
72/// eliminating n_H gives x = sqrt(S/(S+y)) and n_H = y/(1+x) for y = p/(k_B T);
73/// the returned total mass density is m_H n_H. Evaluated in the log domain so it
74/// is exact in both the fully-neutral (y >> S) and fully-ionized (y << S) limits.
75/// Throws std::domain_error unless pressure and temperature are positive/finite.
76double equilibrium_density_from_pressure(double pressure, double temperature);

78/// Equilibrium indices at one total-density/temperature state. gamma_energy is
79/// analytic from the same Saha caloric closure; gamma_sound comes from CRASH.
80struct GammaState {
81 double x_eq; ///< Saha equilibrium ionization fraction n_e/n_H [-]
82 /// Caloric index gamma_E = 1 + p/e_int [-], with e_int the total internal
83 /// energy density INCLUDING the ionization energy. This is the analytic index
84 /// that relates pressure to internal energy; it is NOT the acoustic index.
85 double gamma_energy;
86 /// CRASH-tabulated adiabatic (acoustic) index Gamma1 [-], interpolated from
87 /// EosGammaTable. It sets the sound speed c = sqrt(Gamma1 p / rho) and is a
88 /// distinct quantity from gamma_energy.
89 double gamma_sound;
90};

92/// Equilibrium thermodynamics of one mixture state, decoded from (rho, rho u, E).
93/// Every carrier quantity below (x, n_e, n_HI, p_e) is derived from Saha
94/// equilibrium at (rho, T); none of them is an independent conserved variable.
95struct MixtureThermo {
96 double rho; ///< total mass density [kg/m^3] (the conserved row)
97 double T; ///< common mixture temperature [K], from the caloric inversion
98 double x; ///< Saha ionization fraction n_e/n_H [-]
99 double n_H; ///< hydrogen-nuclei density rho/m_H [m^-3]
100 double n_e; ///< electron number density = x n_H [m^-3]
101 double n_HI; ///< neutral-hydrogen number density = (1-x) n_H [m^-3]
102 double p; ///< total pressure (1+x) n_H k_B T [Pa]
103 double p_e; ///< electron partial pressure x n_H k_B T [Pa]
104 double gamma1; ///< CRASH adiabatic (acoustic) index [-]
105 double internal_energy; ///< e_int(rho,T) including the ionization energy [J/m^3]
106};

122/// Fully decoded, on-manifold face state of the release solver. Flux and
123/// wave-speed evaluation use these fields directly, so no temperature inversion
124/// is needed after the numerical flux.
125struct MixtureFaceState {
126 double rho; ///< total mass density at the face [kg/m^3]
127 double velocity; ///< field-aligned velocity at the face [m/s]
128 double temperature; ///< common mixture temperature at the face [K]
129 double pressure; ///< total p [Pa]
130 double internal_energy; ///< e_int(rho,T) including ionization energy [J/m^3]
131 double energy; ///< E = e_int + 1/2 rho v^2 + rho phi [J/m^3]
132 double sound_speed; ///< sqrt(Gamma1 p / rho) [m/s]
133 double gamma1; ///< CRASH adiabatic (acoustic) index Gamma1 [-]
134 /// General-EOS pressure response at fixed total density: the partial
135 /// derivative of p with respect to e_int at fixed rho, b = (dp/de_int)_rho
136 /// [dimensionless]. It is evaluated from the same Saha closure as the ratio
137 /// (dp/dT)_rho / (de_int/dT)_rho. This is distinct from Gamma1: Gamma1 fixes
138 /// the isentropic acoustic speed, while b fixes the energy component of the
139 /// Euler characteristic vectors.
140 double dp_deint_rho;
141};

221/// Every caloric quantity of the pure-H equilibrium closure at one (rho, T),
222/// produced by a SINGLE Saha evaluation. Each field is computed with exactly the
223/// expression of the standalone function it replaces, so a caller that used to
224/// invoke saha_ionization_fraction_n_h, equilibrium_internal_energy and
225/// equilibrium_heat_capacity separately gets bit-for-bit the same values.
226struct CaloricState {
227 double n_h; ///< hydrogen-nuclei density rho/m_H [m^-3]
228 double x; ///< Saha ionization fraction n_e/n_H [-]
229 double n_e; ///< electron number density = x n_H [m^-3]
230 double n_hi; ///< neutral-hydrogen number density = (1-x) n_H [m^-3]
231 double pressure; ///< total pressure (1+x) n_H k_B T [Pa]
232 /// Total internal energy density e_int = 3p/2 + x n_H chi_H [J/m^3]; the
233 /// second term is the ionization energy stored in the mixture.
234 double internal_energy;
235 /// Effective heat capacity at constant density C_V^eff = d e_int/dT [J/m^3/K].
236 /// It includes the d x/dT ionization contribution, which dominates across the
237 /// ionization zone, and is the Newton derivative of the caloric inversion.
238 double heat_capacity;
239};

255/// Invert the equilibrium pressure closure p(rho,T) = (1+x(rho,T)) n_H k_B T for
256/// the temperature at a KNOWN total density. This is the counterpart of
257/// equilibrium_density_from_pressure and uses exactly the same authoritative
258/// pressure as MixtureFaceState::pressure.
259///
260/// The closure supplies its own exact bracket: with T1 = p m_H/(rho k_B) the
261/// fully-neutral temperature, x in (0,1) gives T in (T1/2, T1] for every physical
262/// state, and dp/dT|_rho = n_H k_B (1 + x + T dx/dT) > 0 makes p(T) strictly
263/// monotone there. The solver is therefore a safeguarded Newton iteration that can
264/// never leave a valid factor-of-two bracket and falls back to bisection on any
265/// rejected step. Throws std::domain_error unless rho and p are positive/finite,
266/// and std::runtime_error if the iteration fails to converge.
267double equilibrium_temperature_from_density_pressure(
268 double rho_total, double pressure,
269 double temperature_guess = std::numeric_limits<double>::quiet_NaN());
