139inline double saha_log_c() {
140 static const double value = 1.5 * std::log(2.0 * PI * eos_constants::m_e
141 * eos_constants::k_b /
142 (eos_constants::h * eos_constants::h));
143 return value;
144}

146double saha_from_coordinates(const EosCoordinates& c) {
147 const double log_a = saha_log_c() + 1.5 * c.log_temperature
148 - eos_constants::chi_h /
149 (eos_constants::k_b * c.temperature)
150 - c.log_n_h;
151 if (log_a >= 0.0)
152 return 2.0 / (1.0 + std::sqrt(1.0 + 4.0 * std::exp(-log_a)));
153 const double u = std::exp(0.5 * log_a);
154 return 2.0 * u / (u + std::sqrt(u * u + 4.0));
155}

157double saha_dx_d_temperature(double x, double temperature) {
158 const double theta = eos_constants::chi_h /
159 (eos_constants::k_b * temperature);
160 return x * (1.0 - x) * (1.5 + theta) /
161 ((2.0 - x) * temperature);
162}

164// ---------------------------------------------------------------------------
165// One authoritative caloric evaluation. EosCoordinates carries the validated
166// log-space coordinates, so Saha, density guards and Gamma1 reuse the same two
167// logarithms instead of recomputing them independently.
168// ---------------------------------------------------------------------------
169struct CaloricEval {
170 double n_h;
171 double log_n_h;
172 double log_temperature;
173 double x;
174 double pressure;
175 double internal_energy;
176 double heat_capacity; // NaN unless produced by caloric_eval_full
177 double n_e() const { return x * n_h; }
178 double n_hi() const { return (1.0 - x) * n_h; }
179};

181CaloricEval caloric_eval_from_coordinates(const EosCoordinates& c) {
182 CaloricEval ev;
183 ev.n_h = c.n_h;
184 ev.log_n_h = c.log_n_h;
185 ev.log_temperature = c.log_temperature;
186 ev.x = saha_from_coordinates(c);
187 ev.pressure = (1.0 + ev.x) * ev.n_h * eos_constants::k_b * c.temperature;
188 ev.internal_energy = 1.5 * ev.pressure + ev.x * ev.n_h * eos_constants::chi_h;
189 ev.heat_capacity = std::numeric_limits<double>::quiet_NaN();
190 return ev;
191}

212// The heat capacity depends on the Saha fraction only through x, so it can be
213// completed from an energy-only evaluation without a second Saha solve.
214double caloric_capacity(double n_h, double x, double temperature) {
215 const double dx_d_t = saha_dx_d_temperature(x, temperature);
216 return n_h * (1.5 * eos_constants::k_b * (1.0 + x)
217 + (1.5 * eos_constants::k_b * temperature
218 + eos_constants::chi_h) * dx_d_t);
219}

289double equilibrium_density_from_pressure(double pressure, double temperature) {
290 require_finite_positive(pressure, "pressure");
291 require_finite_positive(temperature, "temperature");
292
293 // log S(T), written with exactly the same constant and term order as the
294 // log_a expression in saha_ionization_fraction_n_h above (log_a = log S - log n_H).
295 const double log_s = saha_log_c() + 1.5 * temperature_log(temperature)
296 - eos_constants::chi_h / (eos_constants::k_b * temperature);
297 const double y = pressure / (eos_constants::k_b * temperature);
298 const double d = std::log(y) - log_s;
299
300 // x = sqrt(S/(S+y)) = sqrt(1/(1+e^d)). Two algebraically identical branches,
301 // each chosen so its exponential argument is non-positive: no overflow for
302 // either y >> S (deep neutral) or y << S (fully ionized).
303 double x;
304 if (d <= 0.0) {
305 x = 1.0 / std::sqrt(1.0 + std::exp(d));
306 } else {
307 const double e = std::exp(-d); // = S/y
308 x = std::sqrt(e / (1.0 + e));
309 }
310 const double n_h = y / (1.0 + x);
311 return n_h * eos_constants::m_h;
312}

321/// Safeguarded Newton/bisection inversion of p(rho,T) at fixed rho, returning the
322/// caloric evaluation AT the converged temperature so the caller never repeats the
323/// Saha solve. `n_h`/`log_n_h` must already describe the validated rho.
324PressureInversion invert_pressure_eval(double rho_total, double pressure,
325 double n_h, double log_n_h,
326 double temperature_guess) {
327 // Exact bracket from 0 < x < 1 in p = (1+x) n_H k_B T (see eos.hpp).
328 const double t_neutral = pressure / (n_h * eos_constants::k_b);
329 double lower = 0.5 * t_neutral;
330 double upper = t_neutral;
331 if (!(lower > 0.0) || !std::isfinite(upper))
332 throw std::domain_error("equilibrium temperature inversion: unusable bracket");
333
334 double temperature = temperature_guess;
335 // A hint taken from a neighbouring cell can land marginally above T_neutral
336 // of THIS face: in the near-neutral limit x -> 0 the root sits at the upper
337 // bracket end, so any reconstruction perturbation of (rho,p) moves it across.
338 // Clamping is still a valid starting point inside the bracket, whereas
339 // discarding the hint would throw away the best available estimate exactly
340 // where it is most accurate.
341 if (std::isfinite(temperature)) {
342 if (temperature > upper) temperature = upper;
343 else if (temperature <= lower) temperature = lower * (1.0 + 1.0e-9);
344 }
345 const bool hinted = std::isfinite(temperature) && temperature > lower
346 && temperature <= upper;
347 if (!hinted) temperature = 0.75 * t_neutral;
348
349 const double tolerance = 4.0 * std::numeric_limits<double>::epsilon() * pressure;
350 std::uint64_t bisections = 0;
351 for (int iteration = 0; iteration < 128; ++iteration) {
352 const CaloricEval ev = caloric_eval_with_log_n(
353 rho_total, temperature, n_h, log_n_h);
354 const double residual = ev.pressure - pressure;
355 if (std::abs(residual) <= tolerance) {
356 profile_note_pressure_inversion(
357 static_cast<std::uint64_t>(iteration)+1, hinted, bisections);
358 return PressureInversion{temperature, ev};
359 }
360 if (residual > 0.0) upper = temperature;
361 else lower = temperature;
362 if (upper - lower <= 4.0 * std::numeric_limits<double>::epsilon() * upper) {
363 profile_note_pressure_inversion(
364 static_cast<std::uint64_t>(iteration)+1, hinted, bisections);
365 return PressureInversion{temperature, ev};
366 }
367
368 // dp/dT|_rho = n_H k_B (1 + x + T dx/dT); strictly positive, so a Newton
369 // step is always well defined. Reject it only if it leaves the bracket.
370 const double dp_dt = n_h * eos_constants::k_b
371 * (1.0 + ev.x + temperature * saha_dx_d_temperature(ev.x, temperature));
372 double candidate = temperature - residual / dp_dt;
373 if (!std::isfinite(candidate) || !(candidate > lower) || !(candidate < upper)) {
374 candidate = 0.5 * (lower + upper);
375 ++bisections;
376 }
377 temperature = candidate;
378 }
379 std::ostringstream message;
380 message << "equilibrium temperature inversion failed to converge: rho="
381 << rho_total << " kg/m^3, p=" << pressure << " Pa, bracket=["
382 << lower << ',' << upper << "] K";
383 throw std::runtime_error(message.str());
384}

481InversionResult invert_caloric(const EosGammaTable& table, double rho_total,
482 double internal_energy,
483 double temperature_guess, bool debug_clamp) {
484 if (table.empty()) throw std::logic_error("Gamma1 table is empty");
485 require_finite_positive(rho_total, "rho_total");
486 require_finite_positive(internal_energy, "internal_energy");
487 const double n_h = rho_total / eos_constants::m_h;
488 require_finite_positive(n_h, "n_H");
489 const double log_n_h = n_h_log(n_h);
490 table.require_n_h_in_bounds_from_log(n_h, log_n_h, debug_clamp);
491
492 profile_note_inversion_call();
493 const double scale = std::max(std::numeric_limits<double>::min(), internal_energy);
494
495 double lower = table.min_temperature();
496 double upper = table.max_temperature();
497
498 // ---- zero/one-update fast path -------------------------------------------
499 // Measured on the production column, ~97% of all inversions converge with at
500 // most ONE Newton update from the supplied guess, yet each still paid for the
501 // two table-wide bracket-endpoint energies (a further ~470M Saha solves per
502 // 20 s run) and evaluated energy and heat capacity as two separate Saha
503 // solves. Here the guess is evaluated ONCE (energy + capacity fused), the
504 // step is taken with exactly the safeguards the full solver applies, and the
505 // candidate is evaluated once. Anything else falls through to the untouched
506 // safeguarded Newton/bisection solver below.
507 //
508 // The fast path is gated on the guess AND the accepted candidate lying inside
509 // a 1e-9-relative inset of the table temperature range. That margin is many
510 // orders above the 64-eps window in which the bracket block below returns the
511 // endpoint temperature itself or throws out_of_range, so no call that the full
512 // path would have terminated at an endpoint (or rejected) can be answered here
513 // instead — the fast path only short-circuits interior roots.
514 const double inset_lower = lower * (1.0 + 1.0e-9);
515 const double inset_upper = upper * (1.0 - 1.0e-9);
516 if (std::isfinite(temperature_guess)
517 && temperature_guess > inset_lower && temperature_guess < inset_upper) {
518 const CaloricEval ev = caloric_eval_full_with_log_n(
519 rho_total, temperature_guess, n_h, log_n_h);
520 const double residual = ev.internal_energy - internal_energy;
521 if (std::abs(residual) <= 2.0e-13*scale) {
522 profile_note_inversion_initial_guess();
523 profile_note_inversion_iterations(0);
524 return InversionResult{temperature_guess, true, ev};
525 }
526 // The full solver's first iteration reproduces exactly this state: it
527 // narrows one bracket end to the guess, then takes T - residual/C_V.
528 double fast_lower = lower, fast_upper = upper;
529 if (residual > 0.0) fast_upper = temperature_guess;
530 else fast_lower = temperature_guess;
531 const double candidate = temperature_guess - residual/ev.heat_capacity;
532 const double edge = 4.0*std::numeric_limits<double>::epsilon()
533 * std::max(1.0, std::abs(temperature_guess));
534 const bool accepted =
535 std::isfinite(candidate)
536 && candidate > fast_lower + edge
537 && candidate < fast_upper - edge
538 && (fast_upper - fast_lower
539 > 2.0e-13*std::max(1.0, std::abs(temperature_guess)))
540 && std::abs(2.0*residual) < std::abs((upper - lower)*ev.heat_capacity)
541 && candidate > inset_lower && candidate < inset_upper;
542 if (accepted) {
543 const CaloricEval next = caloric_eval_with_log_n(
544 rho_total, candidate, n_h, log_n_h);
545 if (std::abs(next.internal_energy - internal_energy) <= 2.0e-13*scale) {
546 profile_note_inversion_iterations(1);
547 return InversionResult{candidate, true, next};
548 }
549 }
550 }
551
552 // ---- safeguarded Newton / bisection (unchanged) ---------------------------
553 // A supplied temperature is used two ways: as an exact-hit fast path here, and
554 // (below) as the Newton seed over the table-wide bracket. Seeding the BRACKET
555 // from the guess as well was tried and is a pessimisation — a ~6%-wide bracket
556 // leaves Newton no room, so any slight overshoot is forced to bisect: 9.8
557 // iterations and 7.5 bisections per call, versus 1.13 iterations and ZERO
558 // bisections for the same guess used only as a seed. The wide bracket is a
559 // safety net; it should stay wide.
560 if (std::isfinite(temperature_guess)
561 && temperature_guess > table.min_temperature()
562 && temperature_guess < table.max_temperature()) {
563 // NB: the guess evaluation is not a bracket-endpoint evaluation and is
564 // deliberately not counted as one — the accepted fast path must still
565 // report bracket_evaluations == 0.
566 const CaloricEval guessed = caloric_eval_with_log_n(
567 rho_total, temperature_guess, n_h, log_n_h);
568 if (std::abs(guessed.internal_energy-internal_energy) <= 2.0e-13*scale) {
569 profile_note_inversion_initial_guess();
570 profile_note_inversion_iterations(0);
571 return InversionResult{temperature_guess, true, guessed};
572 }
573 }
574
575 {
576 profile_note_inversion_bracket_evaluations(2);
577 const CaloricEval ev_lower = caloric_eval_with_log_n(
578 rho_total, lower, n_h, log_n_h);
579 const CaloricEval ev_upper = caloric_eval_with_log_n(
580 rho_total, upper, n_h, log_n_h);
581 const double e_lower = ev_lower.internal_energy;
582 const double e_upper = ev_upper.internal_energy;
583 const double lower_tolerance = 64.0 * std::numeric_limits<double>::epsilon()
584 * std::max(internal_energy, e_lower);
585 const double upper_tolerance = 64.0 * std::numeric_limits<double>::epsilon()
586 * std::max(internal_energy, e_upper);
587 if (internal_energy < e_lower - lower_tolerance
588 || internal_energy > e_upper + upper_tolerance) {
589 std::ostringstream message;
590 message << "EOS temperature inversion out of bounds: rho=" << rho_total
591 << " kg/m^3, e_int=" << internal_energy << " J/m^3; valid e_int=["
592 << e_lower << ',' << e_upper << "] over T=[" << lower << ','
593 << upper << "] K";
594 throw std::out_of_range(message.str());
595 }
596 if (std::abs(internal_energy - e_lower) <= lower_tolerance) {
597 profile_note_inversion_iterations(0);
598 return InversionResult{lower, true, ev_lower};
599 }
600 if (std::abs(internal_energy - e_upper) <= upper_tolerance) {
601 profile_note_inversion_iterations(0);
602 return InversionResult{upper, true, ev_upper};
603 }
604 }
605
606 double temperature = temperature_guess;
607 if (!std::isfinite(temperature) || temperature <= lower || temperature >= upper) {
608 temperature = internal_energy / (1.5 * n_h * eos_constants::k_b);
609 temperature = std::max(lower, std::min(upper, temperature));
610 if (temperature <= lower || temperature >= upper)
611 temperature = 0.5 * (lower + upper);
612 }
613
614 double last_heat_capacity = std::numeric_limits<double>::quiet_NaN();
615 double last_candidate = std::numeric_limits<double>::quiet_NaN();
616 // Safeguarded Newton (Numerical Recipes "rtsafe"). Track the bracket width from
617 // the PREVIOUS iteration so the slow-progress test can require each Newton step
618 // to beat a bisection; that is what makes the bounce-forever case impossible
619 // without also rejecting good steps.
620 double previous_bracket_width = upper - lower;
621 for (int iteration = 0; iteration < 128; ++iteration) {
622 // Energy and heat capacity from ONE Saha evaluation. The residual test
623 // below can still short-circuit before the capacity is needed, but the
624 // extra work in that case is a handful of flops — no second Saha solve.
625 const CaloricEval ev = caloric_eval_full_with_log_n(
626 rho_total, temperature, n_h, log_n_h);
627 const double value = ev.internal_energy;
628 const double residual = value - internal_energy;
629 if (std::abs(residual) <= 2.0e-13 * scale) {
630 profile_note_inversion_iterations(static_cast<std::uint64_t>(iteration));
631 return InversionResult{temperature, true, ev};
632 }
633 if (residual > 0.0) upper = temperature;
634 else lower = temperature;
635
636 if (upper - lower <= 2.0e-13 * std::max(1.0, std::abs(temperature))) {
637 profile_note_inversion_iterations(static_cast<std::uint64_t>(iteration));
638 return InversionResult{0.5 * (lower + upper), false, CaloricEval()};
639 }
640
641 const double heat_capacity = ev.heat_capacity;
642 double candidate = temperature - residual / heat_capacity;
643 const double bracket_width = upper - lower;
644 // Reject a Newton step only if it leaves the bracket (up to a few ulp of the
645 // endpoints — the roundoff bounce this guard was added for) or if it fails to
646 // beat halving the previous bracket.
647 //
648 // The previous guard rejected any candidate inside the outer 10% of the
649 // bracket on EITHER side. e_int(T) is solved over the whole table range, and
650 // the root sits in the lowest ~0.04% of it (T ~ 7 kK in [3.2e3, 1e7] K), so
651 // that test threw away essentially every early Newton step and forced ~10
652 // bisections just to shrink the bracket enough for Newton to be allowed to
653 // run. Measured on the h1600 ns=1000 column: 29.0 iterations and 25.8
654 // bisections per call, 21.4M calls in 30 s of simulated time — the inversion
655 // was ~100% of runtime across decode/rhs/projection/conduction.
656 const double edge = 4.0 * std::numeric_limits<double>::epsilon()
657 * std::max(1.0, std::abs(temperature));
658 const bool out_of_bracket = !std::isfinite(candidate)
659 || candidate <= lower + edge
660 || candidate >= upper - edge;
661 // Slow-progress test: require the Newton step to beat halving the previous
662 // bracket.
663 const bool too_slow = !(std::abs(2.0 * residual)
664 < std::abs(previous_bracket_width * heat_capacity));
665 if (out_of_bracket || too_slow) {
666 profile_note_inversion_bisection();
667 candidate = 0.5 * (lower + upper);
668 }
669 previous_bracket_width = bracket_width;
670 last_heat_capacity = heat_capacity;
671 last_candidate = candidate;
672 temperature = candidate;
673 }
674 std::ostringstream message;
675 message << "EOS temperature inversion did not converge: rho=" << rho_total
676 << " kg/m^3, e_int=" << internal_energy << " J/m^3, T="
677 << temperature << " K, bracket=[" << lower << ',' << upper
678 << "] K, residual="
679 << (equilibrium_internal_energy(rho_total, temperature)-internal_energy)
680 << " J/m^3, Cv=" << last_heat_capacity << " J/m^3/K, candidate="
681 << last_candidate << " K";
682 throw std::runtime_error(message.str());
683}

694DecodedCaloricResult decode_caloric_impl(
695 const EosGammaTable& table, double rho, double momentum, double energy,
696 double phi_of_this_state, double temperature_guess, bool debug_clamp,
697 CaloricState* caloric_at_temperature) {

700 || !std::isfinite(phi_of_this_state))

712 *caloric_at_temperature = caloric_state_of(ev, temperature);
713 const double n_h = ev.n_h;
714 const double p_e = ev.x * n_h * eos_constants::k_b * temperature;
715 const CaloricMixtureThermo thermo{
716 rho, temperature, ev.x, n_h, ev.x * n_h, (1.0 - ev.x) * n_h,
717 ev.pressure, p_e, internal_energy};
718 return DecodedCaloricResult{thermo, ev};

749MixtureFaceState face_state_from_eval(
750 const EosGammaTable& table, double rho_total, double velocity,
751 double temperature, double phi_of_face, bool debug_clamp,
752 const CaloricEval& ev) {
753 const GammaState gamma = gamma_state_from_eval(
754 table, ev, temperature, debug_clamp);
755 const double pressure = ev.pressure;
756 const double e_int = ev.internal_energy;
757 const double energy = e_int + rho_total * (0.5 * velocity * velocity
758 + phi_of_face);
759 const double sound_speed = std::sqrt(gamma.gamma_sound * pressure / rho_total);
760 // b = (dp/de_int)_rho, the general-EOS energy coefficient of the Euler
761 // characteristic vectors. Both partials come from the same Saha closure:
762 // (dp/dT)_rho = n_H k_B (1 + x + T dx/dT), (de_int/dT)_rho = C_V^eff.
763 const double dx_d_t = saha_dx_d_temperature(ev.x, temperature);
764 const double dp_d_t_rho = ev.n_h * eos_constants::k_b
765 * (1.0 + ev.x + temperature * dx_d_t);
766 const double de_d_t_rho = caloric_capacity(ev.n_h, ev.x, temperature);
767 return MixtureFaceState{rho_total, velocity, temperature, pressure, e_int,
768 energy, sound_speed, gamma.gamma_sound,
769 dp_d_t_rho / de_d_t_rho};
770}

814MixtureFaceState equilibrium_mixture_face_state_from_log_pressure(
815 const EosGammaTable& table, double log_rho_total, double velocity,
816 double log_pressure, double phi_of_face, bool debug_clamp,
817 double temperature_guess) {
818 if (!std::isfinite(log_rho_total) || !std::isfinite(log_pressure))
819 throw std::domain_error("mixture face logarithms must be finite");
820 const double rho_total = std::exp(log_rho_total);
821 const double pressure = std::exp(log_pressure);
822 require_finite_positive(rho_total, "rho_total");
823 require_finite_positive(pressure, "pressure");
824 const double n_h = rho_total / eos_constants::m_h;
825 const double log_n_h = n_h_log(n_h);
826 table.require_n_h_in_bounds_from_log(n_h, log_n_h, debug_clamp);
827 // The inversion returns the caloric evaluation AT the recovered temperature,
828 // so no Saha solve is repeated after convergence — but every Newton step
829 // inside it costs one. `temperature_guess` is a pure solver hint: any value
830 // outside the exact bracket (including NaN) falls back to 0.75*T_neutral, and
831 // the bracket remains the safety authority either way, so the converged root
832 // cannot depend on it.
833 const PressureInversion inverted = invert_pressure_eval(
834 rho_total, pressure, n_h, log_n_h, temperature_guess);
835 validate_face_inputs(table, rho_total, velocity, inverted.temperature,
836 phi_of_face);
837 return face_state_from_eval(
838 table, rho_total, velocity, inverted.temperature, phi_of_face,
839 debug_clamp, inverted.eval);
840}

849EosGammaTable EosGammaTable::load(const std::string& path) {
850 std::ifstream input(path.c_str());
851 if (!input) throw std::runtime_error("Gamma1 table: cannot open '" + path + "'");
852
853 std::map<std::string, std::string> metadata;
854 std::vector<Row> rows;
855 std::string line;
856 std::size_t line_number = 0;
857 while (std::getline(input, line)) {
858 ++line_number;
859 const std::string text = trim(line);
860 if (text.empty()) continue;
861 if (text[0] == '#') {
862 const std::string field = trim(text.substr(1));
863 const std::size_t equals = field.find('=');
864 if (equals == std::string::npos || equals == 0 || equals + 1 >= field.size())
865 throw std::runtime_error("Gamma1 table: malformed metadata at " + path
866 + ":" + std::to_string(line_number));
867 const std::string key = trim(field.substr(0, equals));
868 const std::string value = trim(field.substr(equals + 1));
869 if (key.empty() || value.empty() || !metadata.emplace(key, value).second)
870 throw std::runtime_error("Gamma1 table: duplicate or empty metadata key '"
871 + key + "' at " + path + ":"
872 + std::to_string(line_number));
873 continue;
874 }
875 std::istringstream parser(text);
876 Row row;
877 std::string extra;
878 if (!(parser >> row.log_t_10 >> row.log_n_10 >> row.gamma1) || (parser >> extra)) {
879 throw std::runtime_error("Gamma1 table: expected exactly 3 numeric columns at "
880 + path + ":" + std::to_string(line_number));
881 }
882 if (!std::isfinite(row.log_t_10) || !std::isfinite(row.log_n_10)
883 || !(row.gamma1 > 0.0) || !std::isfinite(row.gamma1)) {
884 throw std::runtime_error("Gamma1 table: non-finite axis or non-positive Gamma1 at "
885 + path + ":" + std::to_string(line_number));
886 }
887 rows.push_back(row);
888 }
889 for (const auto& item : expected_metadata()) {
890 const auto found = metadata.find(item.first);
891 if (found == metadata.end())
892 throw std::runtime_error("Gamma1 table: missing metadata key '" + item.first
893 + "' in '" + path + "'");
894 if (found->second != item.second)
895 throw std::runtime_error("Gamma1 table: unsupported metadata " + item.first
896 + "='" + found->second + "' in '" + path + "'");
897 }
898 const std::size_t declared_n_t = parse_dimension(metadata, "nT", path);
899 const std::size_t declared_n_n = parse_dimension(metadata, "nN", path);
900 if (metadata.size() != expected_metadata().size() + 2)
901 throw std::runtime_error("Gamma1 table: unexpected metadata key in '" + path + "'");
902 if (rows.empty()) throw std::runtime_error("Gamma1 table: no data rows in '" + path + "'");
903
904 // The generator writes density-major blocks with temperature ascending.
905 std::vector<double> log_t_10;
906 for (const Row& row : rows) {
907 if (!log_t_10.empty() && row.log_n_10 != rows.front().log_n_10) break;
908 push_unique(log_t_10, row.log_t_10);
909 }
910 if (log_t_10.size() < 2)
911 throw std::runtime_error("Gamma1 table: temperature axis needs at least 2 points");
912 std::vector<double> log_n_10;
913 for (std::size_t offset = 0; offset < rows.size(); offset += log_t_10.size())
914 push_unique(log_n_10, rows[offset].log_n_10);
915 if (log_n_10.size() < 2 || rows.size() != log_t_10.size() * log_n_10.size())
916 throw std::runtime_error("Gamma1 table: incomplete rectangular grid in '" + path + "'");
917 if (log_t_10.size() != declared_n_t || log_n_10.size() != declared_n_n)
918 throw std::runtime_error("Gamma1 table: metadata dimensions disagree with data in '"
919 + path + "'");
920 if (!std::is_sorted(log_t_10.begin(), log_t_10.end())
921 || std::adjacent_find(log_t_10.begin(), log_t_10.end(),
922 std::greater_equal<double>()) != log_t_10.end()
923 || !std::is_sorted(log_n_10.begin(), log_n_10.end())
924 || std::adjacent_find(log_n_10.begin(), log_n_10.end(),
925 std::greater_equal<double>()) != log_n_10.end()) {
926 throw std::runtime_error("Gamma1 table: axes must be strictly increasing");
927 }
928 for (std::size_t i_n = 0; i_n < log_n_10.size(); ++i_n) {
929 for (std::size_t i_t = 0; i_t < log_t_10.size(); ++i_t) {
930 const Row& row = rows[i_n * log_t_10.size() + i_t];
931 if (row.log_n_10 != log_n_10[i_n] || row.log_t_10 != log_t_10[i_t])
932 throw std::runtime_error("Gamma1 table: inconsistent rectangular row order");
933 }
934 }
935
936 EosGammaTable table;
937 table.log_temperature_.reserve(log_t_10.size());
938 table.log_n_h_.reserve(log_n_10.size());
939 for (double value : log_t_10) table.log_temperature_.push_back(value * LN_10);
940 for (double value : log_n_10) table.log_n_h_.push_back(value * LN_10);
941 table.gamma1_.reserve(rows.size());
942 for (const Row& row : rows) table.gamma1_.push_back(row.gamma1);
943 return table;
944}

1021double EosGammaTable::gamma1_from_logs(
1022 double temperature, double n_h, double log_temperature, double log_n_h,
1023 bool debug_clamp) const {
1024 if (empty()) throw std::logic_error("Gamma1 table is empty");
1025 require_finite_positive(temperature, "temperature");
1026 require_finite_positive(n_h, "n_H");
1027 if (!std::isfinite(log_temperature) || !std::isfinite(log_n_h))
1028 throw std::domain_error("Gamma1 logarithmic coordinates must be finite");
1029 if (eos_operation_counting_enabled) ++eos_operation_counter().gamma1_queries;
1030 double log_t = log_temperature;
1031 double log_n = log_n_h;
1032 const double tolerance = 64.0 * std::numeric_limits<double>::epsilon()
1033 * std::max({1.0, std::abs(log_t), std::abs(log_n)});
1034 const bool outside = log_t < log_temperature_.front() - tolerance
1035 || log_t > log_temperature_.back() + tolerance
1036 || log_n < log_n_h_.front() - tolerance
1037 || log_n > log_n_h_.back() + tolerance;
1038 if (outside && !debug_clamp) {
1039 std::ostringstream message;
1040 message << "Gamma1 table query out of bounds: T=" << temperature
1041 << " K, n_H=" << n_h << " m^-3; valid T=[" << min_temperature()
