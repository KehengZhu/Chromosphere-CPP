14Vec cal_flux_state(const Grid& grid, const Vec& xn_state) {
15 Vec F_state(arma::size(xn_state), arma::fill::zeros);
16 const Vec rho_i = get_scalar(grid, xn_state, cons::RHO_I);
17 const Vec rho_n = get_scalar(grid, xn_state, cons::RHO_N);
18 const Vec rhoV_i = get_scalar(grid, xn_state, cons::MOM_I);
19 const Vec rhoU_n = get_scalar(grid, xn_state, cons::MOM_N);
20 const Vec e_i = get_scalar(grid, xn_state, cons::E_I);
21 const Vec e_n = get_scalar(grid, xn_state, cons::E_N);
22 const Vec e_e = get_scalar(grid, xn_state, cons::E_E);
23
24 const Vec V = rhoV_i / rho_i;
25 const Vec U = rhoU_n / rho_n;
26 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
27 // p_i = total charged pressure (protons + electrons): the momentum flux and
28 // the charged total-energy flux are unchanged by the T_e split.
29 const Vec p_i = grid.gm1() * e_i - grid.half_gm1() * rho_i % V % V - grid.gm1() * rho_i % phi_g;
30 const Vec p_n = grid.gm1() * e_n - grid.half_gm1() * rho_n % U % U - grid.gm1() * rho_n % phi_g;
31
32 F_state += scalar_to(grid, rhoV_i, cons::RHO_I);
33 F_state += scalar_to(grid, rhoU_n, cons::RHO_N);
34 F_state += scalar_to(grid, rho_i % V % V + p_i, cons::MOM_I);
35 F_state += scalar_to(grid, rho_n % U % U + p_n, cons::MOM_N);
36 F_state += scalar_to(grid, (e_i + p_i) % V, cons::E_I);
37 F_state += scalar_to(grid, (e_n + p_n) % U, cons::E_N);
38 // Electron internal energy advects at the charged-fluid velocity V (electron
39 // inertia negligible). The −p_e ∇·V compression work is an explicit source
40 // (cal_source_state); here just the advective flux ε_e V.
41 F_state += scalar_to(grid, e_e % V, cons::E_E);
42 return F_state;
43}

45Vec cal_spectral_radius_state(const Grid& grid, const Vec& xn_state) {
46 Vec res_state(arma::size(xn_state), arma::fill::zeros);
47 const Vec rho_i = get_scalar(grid, xn_state, cons::RHO_I);
48 const Vec rho_n = get_scalar(grid, xn_state, cons::RHO_N);
49 const Vec rhoV_i = get_scalar(grid, xn_state, cons::MOM_I);
50 const Vec rhoU_n = get_scalar(grid, xn_state, cons::MOM_N);
51 const Vec e_i = get_scalar(grid, xn_state, cons::E_I);
52 const Vec e_n = get_scalar(grid, xn_state, cons::E_N);
53
54 const Vec V = rhoV_i / rho_i;
55 const Vec U = rhoU_n / rho_n;
56 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
57 Vec p_i = grid.gm1() * e_i - grid.half_gm1() * rho_i % V % V - grid.gm1() * rho_i % phi_g;
58 Vec p_n = grid.gm1() * e_n - grid.half_gm1() * rho_n % U % U - grid.gm1() * rho_n % phi_g;
59
60 // Flare scenario: in the fast (~10²–10³ km/s), hot (~10⁷ K) evaporated column
61 // the float32 cancellation p = ⅔E − ⅓ρv² − ⅔ρφ can leave p slightly NEGATIVE,
62 // making c_s = √(γp/ρ) a NaN that poisons the Rusanov flux. Clamp p ≥ 0 for
63 // the sound speed only. Gated on beam heating so steady runs / tests are unchanged.
64 if (grid.enable_beam_heating) {
65 p_i = arma::clamp(p_i, 0.0f, arma::datum::inf);
66 p_n = arma::clamp(p_n, 0.0f, arma::datum::inf);
67 }
68
69 const Vec cs_i = arma::sqrt(grid.gamma_mono * p_i / rho_i);
70 const Vec cs_n = arma::sqrt(grid.gamma_mono * p_n / rho_n);
71
72 Vec spc_r_i = arma::max(arma::abs(V) + arma::abs(cs_i),
73 arma::abs(U) + arma::abs(cs_n));
74 for (arma::uword i = 0; i < num_of_eq; ++i) {
75 res_state += scalar_to(grid, spc_r_i, i);
76 }
77 return res_state;
78}

80// Explicit source (writeup §4.3):
81// S_E = [0, 0, p_i B D - ρ_i G, p_n B D - ρ_n G, 0, 0]^T
82// with D = ∂(1/B)/∂s and G = ∂φ_g/∂s.
83Vec cal_source_state(const Grid& grid, const Vec& xn_state) {
84 Vec S_state(arma::size(xn_state), arma::fill::zeros);
85 const Vec rho_i = get_scalar(grid, xn_state, cons::RHO_I);
86 const Vec rho_n = get_scalar(grid, xn_state, cons::RHO_N);
87 const Vec rhoV_i = get_scalar(grid, xn_state, cons::MOM_I);
88 const Vec rhoU_n = get_scalar(grid, xn_state, cons::MOM_N);
89 const Vec e_i = get_scalar(grid, xn_state, cons::E_I);
90 const Vec e_n = get_scalar(grid, xn_state, cons::E_N);
91
92 const Vec V = rhoV_i / rho_i;
93 const Vec U = rhoU_n / rho_n;
94 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
95 const Vec p_i = grid.gm1() * e_i - grid.half_gm1() * rho_i % V % V - grid.gm1() * rho_i % phi_g;
96 const Vec p_n = grid.gm1() * e_n - grid.half_gm1() * rho_n % U % U - grid.gm1() * rho_n % phi_g;
97
98 // Gravity at cell center from the face potentials.
99 const Vec gb = -(grid.phi_g_iph - grid.phi_g_imh) / grid.ds_i;
100
101 S_state += scalar_to(grid, p_i % grid.B_i % grid.dinvB_ds_i + rho_i % gb, cons::MOM_I);
102 S_state += scalar_to(grid, p_n % grid.B_i % grid.dinvB_ds_i + rho_n % gb, cons::MOM_N);
103
104 // Electron compression work −p_e ∇·V (three-temperature model only). In the
105 // 1.5D field-aligned geometry (cross-section A ∝ 1/B) the velocity divergence
106 // consistent with the flux differencing is ∇·V = B [V_{i+½}/B_{i+½} −
107 // V_{i-½}/B_{i-½}]/Δs with face velocities V_{i±½} the cell-center averages.
108 // Adding this to ε_e (advected as ε_e V in cal_flux_state) gives the correct
109 // adiabatic electron response dε_e/dt = −(5/2) p_e ∇·V. The complementary
110 // proton compression is automatic: p_proton = p_total − p_e and p_total
111 // (from the conservatively-evolved E_I) carries the full −p_total ∇·V work.
112 // Gated on enable_Te so the single-temperature baseline is untouched.
113 if (grid.enable_Te) {
114 const Vec V = rhoV_i / rho_i;
115 const Vec e_e = get_scalar(grid, xn_state, cons::E_E);
116 const Vec p_e = grid.gm1() * e_e;
117 const Vec V_ip1 = ip1(grid, V, SLICE);
118 const Vec V_im1 = im1(grid, V, SLICE);
119 // Face velocity by interpolation between cell centers. Uniform mesh: the
120 // midpoint average. Refined mesh: distance-weighted so the face value is a
121 // proper linear interpolant (metric-weighted), keeping ∇·V second-order.
122 Vec Vf_iph, Vf_imh;
123 if (grid.uniform_mesh) {
124 Vf_iph = 0.5 * (V + V_ip1);
125 Vf_imh = 0.5 * (V_im1 + V);
126 } else {
127 const Vec ds_ip1 = ip1(grid, grid.ds_i, SLICE);
128 const Vec ds_im1 = im1(grid, grid.ds_i, SLICE);
129 Vf_iph = (0.5 * ds_ip1 % V + 0.5 * grid.ds_i % V_ip1) / grid.ds_iph_i;
130 Vf_imh = (0.5 * grid.ds_i % V_im1 + 0.5 * ds_im1 % V) / grid.ds_imh_i;
131 }
132 const Vec divV = grid.B_i % (Vf_iph / grid.B_iph - Vf_imh / grid.B_imh) / grid.ds_i;
133 S_state += scalar_to(grid, -p_e % divV, cons::E_E);
134 }
135 return S_state;
136}
