1/*!
2 * @file single_fluid/integrator.cpp
3 * @brief Release solver: implicit physical conduction and the release timestep.
4 * @ingroup release_solver
5 *
6 * U^n --MUSCL/Roe hydro--> U* --implicit physical conduction--> U^{n+1}
7 *
8 * Those are the only two stages, composed by first-order Lie splitting. There is
9 * no equilibrium projection between them (U = (rho, rho u, E) is already the
10 * authoritative state, and Saha equilibrium enters only through the EOS closure
11 * that decodes it), and there are no volumetric energy stages: TRAC, nonthermal
12 * beam heating, volumetric coronal heating, radiative cooling and
13 * artificial/numerical conduction are NOT part of the release and live only in
14 * the historical two-fluid research solver (src/two_fluid/).
15 *
16 * The conduction stage is a nonlinear backward-Euler solve on the mixture energy
17 * row alone: Newton on the cell temperatures with a tridiagonal Jacobian, the
18 * effective heat capacity and the conductivity taken from one fused Saha
19 * evaluation per cell per pass. Mass and momentum are untouched, and the accepted
20 * energy is the flux-updated internal energy plus the unchanged kinetic and
21 * gravitational parts, so the stage is conservative by construction.
22 */

52double mixture_kappa_e(double n_e, double n_hi, double temperature) {
53 return 9.2048e-12*n_e*std::pow(temperature, 2.5)
54 / (n_e + 2.836e-11*n_hi*temperature*temperature);
55}
56
57double mixture_kappa_n(double n_e, double n_hi, double temperature) {
58 return 0.0342006*n_hi*temperature
59 / (1.20613*n_e*std::sqrt(2.0*temperature)
60 + 1.70573*n_hi*std::sqrt(temperature));
61}

63// The release conductivity is PHYSICAL only: Spitzer electron conduction plus
64// neutral-hydrogen conduction. No TRAC broadening and no mesh-scaled artificial
65// diffusivity enter the release operator anywhere.
66double mixture_conductivity(double n_e, double n_hi, double temperature) {
67 return mixture_kappa_e(n_e, n_hi, temperature)
68 + mixture_kappa_n(n_e, n_hi, temperature);
69}

71void thomas_solve_double_inplace(
72 const std::vector<double>& a, std::vector<double>& b,
73 const std::vector<double>& c, std::vector<double>& d,
74 std::vector<double>& x) {
75 const std::size_t n = b.size();
76 for (std::size_t i = 1; i < n; ++i) {
77 const double m = a[i]/b[i-1];
78 b[i] -= m*c[i-1];
79 d[i] -= m*d[i-1];
80 }
81 x[n-1] = d[n-1]/b[n-1];
82 for (std::size_t i = n-1; i > 0; --i)
83 x[i-1] = (d[i-1]-c[i-1]*x[i])/b[i-1];
84}

94// The outer state as the CONDUCTION rows see it. When the hydro and conduction
95// temperatures are decoupled, the imposed temperature lives at the PHYSICAL
96// boundary face, not at a fictitious ghost centre. Reconstruct the material
97// state at that face from the externally imposed hydro back-pressure and the
98// fixed face temperature, then evaluate kappa there. Without the override this
99// remains the decoded hydro ghost state.
100struct OuterConductionWall { double T, n_e, n_HI; };
101
102OuterConductionWall outer_conduction_wall(const Grid& grid,
103 const CaloricMixtureThermo& ghost) {
104 if (!grid.outer_conduction_temperature_override)
105 return {ghost.T, ghost.n_e, ghost.n_HI};
106 const double T = static_cast<double>(grid.outer_conduction_temperature);
107 const double rho = equilibrium_density_from_pressure(ghost.p, T);
108 const double n_h = rho/eos_constants::m_h;
109 const double x = saha_ionization_fraction_n_h(n_h, T);
110 return {T, x*n_h, (1.0-x)*n_h};
111}

122Vec mixture_set_internal_energy(
123 const Grid& grid, const Vec& state,
124 const std::vector<double>& target_internal_energy) {
125 if (target_internal_energy.size() != grid.ns)
126 throw std::invalid_argument("mixture internal-energy target has wrong size");
127 Vec updated = state;
128 const auto sz = state_size(grid);
129 for (arma::uword i = 0; i < grid.ns; ++i) {
130 const double rho = static_cast<double>(state(arma::sub2ind(sz, i, mix::RHO)));
131 const double momentum =
132 static_cast<double>(state(arma::sub2ind(sz, i, mix::MOM)));
133 const double phi = mixture_cell_phi(grid, i);
134 const double target = target_internal_energy[i];
135 if (!std::isfinite(target))
136 throw std::domain_error(
137 "non-finite mixture internal-energy target at cell "
138 + std::to_string(i) + ", time=" + std::to_string(grid.sim_time)
139 + ", rho=" + std::to_string(rho));
140 const double e_min = equilibrium_internal_energy(
141 rho, grid.eos_gamma_table.min_temperature());
142 const double e_max = equilibrium_internal_energy(
143 rho, grid.eos_gamma_table.max_temperature());
144 const double tolerance = 64.0*std::numeric_limits<double>::epsilon()
145 * std::max({std::abs(target), e_min, e_max});
146 if (target < e_min-tolerance || target > e_max+tolerance)
147 throw std::out_of_range(
148 "mixture energy target outside the EOS caloric domain at cell "
149 + std::to_string(i) + ", time=" + std::to_string(grid.sim_time)
150 + ", rho=" + std::to_string(rho)
151 + ", target_e=" + std::to_string(target)
152 + ", valid_e=[" + std::to_string(e_min) + ","
153 + std::to_string(e_max) + "]");
154 // Round the packed float back into the caloric domain: independently
155 // rounding E can otherwise place e_int = E - KE - rho phi a few ulp
156 // outside the table edge and make the next decode throw.
157 const double carried = 0.5*momentum*momentum/rho + rho*phi;
158 float energy = static_cast<float>(target + carried);
159 while (static_cast<double>(energy) - carried < e_min)
160 energy = std::nextafter(energy, std::numeric_limits<float>::infinity());
161 while (static_cast<double>(energy) - carried > e_max)
162 energy = std::nextafter(energy, -std::numeric_limits<float>::infinity());
163 if (!std::isfinite(energy))
164 throw std::runtime_error(
165 "failed to pack the mixture energy inside the EOS domain");
166 updated(arma::sub2ind(sz, i, mix::ENERGY)) = energy;
167 }
168 return updated;
169}

173// ============================================================================
174// Nonlinear backward-Euler physical conduction
175// ============================================================================
176//
177// Solved for T with the analytic effective heat capacity C_V^eff = de_int/dT and
178// only d(kappa)/dT lagged (a quasi-Newton scheme):
179//
180// C_V^eff dT + dt [ g_L (dT_i - dT_{i-1}) - g_R (dT_{i+1} - dT_i) ]
181// = -( e_int(rho,T) - e_old - dt div q ),
182// g_{L,R} = (B_i / ds_i) (kappa_face / B_face) / ds_face.
183//
184// Mass and momentum never enter, so the stage conserves them exactly; the
185// accepted energy is E = e_target + 1/2 rho u^2 + rho phi.
186Vec mixture_apply_conduction(const Grid& grid, const Vec& state, double dt) {
187 if (!(dt > 0.0)) return state;
188 ProfileScope timer(ProfileRegion::Conduction);
189 const arma::uword ns = grid.ns;
190 MixtureConductionScratch& s = grid.mixture_conduction_scratch;
191 s.resize(ns);
192 const auto sz = state_size(grid);
193
194 ParallelFailure failure;
195 std::array<double, kMaximumParallelThreads> residual_max_slots{};
196 std::array<double, kMaximumParallelThreads> step_max_slots{};
197 int team_size = 1;
198 bool stop = false;
199 bool converged = false;
200 bool caloric_seeded = true;
201 bool previous_step_was_small = false;
202 int converged_after_updates = 0;
203 CaloricMixtureThermo inner{};
204 CaloricMixtureThermo outer{};
205 OuterConductionWall wall{};
206 double k_inner = 0.0;
207 double k_outer = 0.0;
208 double diag_kr_top = 0.0;
209 const int conduction_threads = std::min(
210 kMaximumConductionThreads, parallel_max_threads());
211
212 auto face_k = [&](double kh, double kt, double dh, double dtw) {
213 if (grid.uniform_mesh) return 0.5*(kh+kt);
214 kh = std::max(kh, 1.0e-30);
215 kt = std::max(kt, 1.0e-30);
216 return (dh+dtw)/(dh/kh+dtw/kt);
217 };
218
219 // One Grid owns one mutable conduction scratch set and is intentionally not
220 // concurrently reentrant. All physical-cell writes below are index-disjoint.
221#pragma omp parallel if(conduction_threads > 1) num_threads(conduction_threads) shared(stop,converged,caloric_seeded,previous_step_was_small,converged_after_updates,inner,outer,wall,k_inner,k_outer,diag_kr_top,team_size,failure,residual_max_slots,step_max_slots)
222 {
223 const int tid = parallel_thread_index();
224
225#pragma omp for schedule(static)
226 for (long long raw_i = 0; raw_i < static_cast<long long>(ns); ++raw_i) {
227 const arma::uword i = static_cast<arma::uword>(raw_i);
228 try {
229 auto at = [&](arma::uword row) {
230 return static_cast<double>(state(arma::sub2ind(sz,i,row)));
231 };
232 const double phi = mixture_cell_phi(grid, i);
233 CaloricState seeded;
234 const CaloricMixtureThermo th = decode_equilibrium_caloric_mixture(
235 grid.eos_gamma_table, at(mix::RHO), at(mix::MOM),
236 at(mix::ENERGY), phi, grid.eos_temperature_hint(i),
237 grid.eos_gamma_debug_clamp, &seeded);
238 grid.store_eos_temperature_hint(i, th.T);
239 s.rho[i] = th.rho;
240 s.e_old[i] = th.internal_energy;
241 s.temperature[i] = th.T;
242 s.n_e[i] = seeded.n_e;
243 s.n_hi[i] = seeded.n_hi;
244 s.e_at_T[i] = seeded.internal_energy;
245 s.capacity[i] = seeded.heat_capacity;
246 } catch (...) {
247 failure.capture(i, std::current_exception());
248 }
249 }
250
251#pragma omp single
252 {
253 team_size = parallel_team_size();
254 if (failure.failed()) {
255 stop = true;
256 } else {
257 try {
258 inner = decode_ghost(grid, grid.mix_inner_boundary0,
259 grid.phi_g_imh(0));
260 outer = decode_ghost(grid, grid.mix_outer_boundary0,
261 grid.phi_g_iph(ns-1));
262 wall = outer_conduction_wall(grid, outer);
263 k_inner = mixture_conductivity(inner.n_e, inner.n_HI, inner.T);
264 k_outer = mixture_conductivity(wall.n_e, wall.n_HI, wall.T);
265 } catch (...) {
266 failure.capture(ns, std::current_exception());
267 stop = true;
268 }
269 }
270 }
271
272 if (!stop) {
273 for (int iteration = 0; iteration <= 40; ++iteration) {
274#pragma omp for schedule(static)
275 for (long long raw_i = 0; raw_i < static_cast<long long>(ns); ++raw_i) {
276 const arma::uword i = static_cast<arma::uword>(raw_i);
277 try {
278 if (!caloric_seeded) {
279 const CaloricState cs = equilibrium_caloric_state(
280 s.rho[i], s.temperature[i]);
281 s.n_e[i] = cs.n_e;
282 s.n_hi[i] = cs.n_hi;
283 s.e_at_T[i] = cs.internal_energy;
284 s.capacity[i] = cs.heat_capacity;
285 }
286 s.conductivity[i] = mixture_conductivity(
287 s.n_e[i], s.n_hi[i], s.temperature[i]);
288 } catch (...) {
289 failure.capture(i, std::current_exception());
290 }
291 }
292
293#pragma omp single
294 {
295 caloric_seeded = false;
296 if (failure.failed()) stop = true;
297 }
298 if (stop) break;
299
300#pragma omp for schedule(static)
301 for (long long raw_i = 0; raw_i < static_cast<long long>(ns); ++raw_i) {
302 const arma::uword i = static_cast<arma::uword>(raw_i);
303 try {
304 const double kl = i == 0
305 ? face_k(s.conductivity[i],k_inner,grid.ds_i(i),grid.ds_i(i))
306 : face_k(s.conductivity[i],s.conductivity[i-1],
307 grid.ds_i(i),grid.ds_i(i-1));
308 // The imposed outer temperature is a PHYSICAL-FACE datum:
309 // the wall conductivity is used directly and the flux
310 // distance is the half top cell, not a ghost-centre span.
311 const bool physical_outer_face =
312 i+1 == ns && grid.outer_conduction_temperature_override;
313 const double kr = i+1 == ns
314 ? (physical_outer_face
315 ? k_outer
316 : face_k(s.conductivity[i],k_outer,grid.ds_i(i),grid.ds_i(i)))
317 : face_k(s.conductivity[i],s.conductivity[i+1],
318 grid.ds_i(i),grid.ds_i(i+1));
319 const double ds_right = physical_outer_face
320 ? 0.5*static_cast<double>(grid.ds_i(i))
321 : static_cast<double>(grid.ds_iph_i(i));
322 if (i+1 == ns) diag_kr_top = kr;
323 s.g_left[i] = grid.B_i(i)/grid.ds_i(i)
324 * kl/grid.B_imh(i)/grid.ds_imh_i(i);
325 s.g_right[i] = grid.B_i(i)/grid.ds_i(i)
326 * kr/grid.B_iph(i)/ds_right;
327 } catch (...) {
328 failure.capture(i, std::current_exception());
329 }
330 }
331
332#pragma omp single
333 {
334 if (failure.failed()) stop = true;
335 }
336 if (stop) break;
337
338 double local_residual_max = 0.0;
339#pragma omp for schedule(static)
340 for (long long raw_i = 0; raw_i < static_cast<long long>(ns); ++raw_i) {
341 const arma::uword i = static_cast<arma::uword>(raw_i);
342 try {
343 const double tl = i==0 ? inner.T : s.temperature[i-1];
344 const double tr = i+1==ns ? wall.T : s.temperature[i+1];
345 double divergence = s.g_right[i]*(tr-s.temperature[i])
346 -s.g_left[i]*(s.temperature[i]-tl);
347 if (i == 0 && grid.inner_conduction_neumann) {
348 divergence += s.g_left[i]*(s.temperature[i]-tl);
349 s.g_left[i] = 0.0;
350 }
351 if (i+1 == ns && grid.impose_outer_heat_flux) {
352 divergence = -s.g_left[i]*(s.temperature[i]-tl)
353 + grid.outer_heat_flux/grid.ds_i(i);
354 s.g_right[i] = 0.0;
355 }
356 s.target[i] = s.e_old[i]+dt*divergence;
357 const double residual = s.e_at_T[i]-s.target[i];
358 s.a[i] = -dt*s.g_left[i];
359 s.b[i] = s.capacity[i]+dt*(s.g_left[i]+s.g_right[i]);
360 s.c[i] = -dt*s.g_right[i];
361 s.rhs[i] = -residual;
362 local_residual_max = std::max(local_residual_max,
363 std::abs(residual)/std::max(s.e_old[i],1.0e-30));
364 } catch (...) {
365 failure.capture(i, std::current_exception());
366 }
367 }
368 residual_max_slots[static_cast<std::size_t>(tid)] = local_residual_max;
369#pragma omp barrier
370
371#pragma omp single
372 {
373 if (failure.failed()) {
374 stop = true;
375 } else {
376 s.a[0]=0.0;
377 s.c[ns-1]=0.0;
378 double max_scaled_residual = 0.0;
379 for (int slot = 0; slot < team_size; ++slot)
380 max_scaled_residual = std::max(
381 max_scaled_residual,
382 residual_max_slots[static_cast<std::size_t>(slot)]);
383 if (max_scaled_residual < 2.0e-11) {
384 converged = true;
385 converged_after_updates = iteration;
386 stop = true;
387 } else if (previous_step_was_small) {
388 failure.capture(ns, std::make_exception_ptr(std::runtime_error(
389 "mixture conduction stagnated before residual convergence: "
390 "scaled_residual="+std::to_string(max_scaled_residual))));
391 stop = true;
392 } else if (iteration == 40) {
393 stop = true;
394 } else {
395 try {
396 thomas_solve_double_inplace(s.a,s.b,s.c,s.rhs,s.delta);
397 } catch (...) {
398 failure.capture(ns, std::current_exception());
399 stop = true;
400 }
401 }
402 }
403 }
404 if (stop) break;
405
406 double local_step_max = 0.0;
407#pragma omp for schedule(static)
408 for (long long raw_i = 0; raw_i < static_cast<long long>(ns); ++raw_i) {
409 const arma::uword i = static_cast<arma::uword>(raw_i);
410 try {
411 double candidate = s.temperature[i]+s.delta[i];
412 if (!std::isfinite(candidate))
413 throw std::runtime_error(
414 "mixture conduction Newton produced a non-finite T");
415 const double tmin = grid.eos_gamma_table.min_temperature();
416 const double tmax = grid.eos_gamma_table.max_temperature();
417 if (candidate<=tmin)
418 candidate=0.5*(s.temperature[i]+tmin);
419 if (candidate>=tmax)
420 candidate=0.5*(s.temperature[i]+tmax);
421 local_step_max = std::max(local_step_max,
422 std::abs(candidate-s.temperature[i])/
423 std::max(s.temperature[i],1.0));
424 s.temperature[i]=candidate;
425 } catch (...) {
426 failure.capture(i, std::current_exception());
427 }
428 }
429 step_max_slots[static_cast<std::size_t>(tid)] = local_step_max;
430#pragma omp barrier
431
432#pragma omp single
433 {
434 if (failure.failed()) {
435 stop = true;
436 } else {
437 double max_relative_step = 0.0;
438 for (int slot = 0; slot < team_size; ++slot)
439 max_relative_step = std::max(
440 max_relative_step,
441 step_max_slots[static_cast<std::size_t>(slot)]);
442 previous_step_was_small = max_relative_step < 2.0e-11;
443 }
444 }
445 if (stop) break;
446 }
447 }
448 }
449
450 failure.rethrow_lowest();
451 if (!converged)
452 throw std::runtime_error("mixture conduction Newton did not converge");
453 profile_note_conduction_iterations(
454 static_cast<std::uint64_t>(converged_after_updates));
455
456 if (grid.capture_outer_conduction) {
457 OuterConductionCapture& oc = grid.outer_conduction_capture;
458 oc.T_top = s.temperature[ns-1];
459 oc.T_wall = wall.T;
460 oc.kappa_face = diag_kr_top;
461 oc.ds_face = grid.outer_conduction_temperature_override
462 ? 0.5*static_cast<double>(grid.ds_i(ns-1))
463 : static_cast<double>(grid.ds_iph_i(ns-1));
464 oc.area_ratio = grid.B_i(ns-1)/grid.B_iph(ns-1);
465 oc.q_face = oc.kappa_face*(oc.T_wall-oc.T_top)/oc.ds_face;
466 oc.imposed_neumann = grid.impose_outer_heat_flux;
467 oc.valid = true;
468 }
469
470 return mixture_set_internal_energy(grid, state, s.target);
471}

473double mixture_conduction_residual_max(const Grid& grid, const Vec& before,
474 const Vec& after, double dt) {
475 if (grid.eos_gamma_table.empty())
476 throw std::logic_error("mixture conduction residual requires a Gamma1 table");
477 const arma::uword ns = grid.ns;
478 const auto sz = state_size(grid);
479 std::vector<double> conductivity(ns), e_old(ns), e_now(ns), temperature(ns);
480 for (arma::uword i = 0; i < ns; ++i) {
481 const double phi = mixture_cell_phi(grid, i);
482 auto decode = [&](const Vec& st) {
483 return decode_equilibrium_caloric_mixture(
484 grid.eos_gamma_table,
485 static_cast<double>(st(arma::sub2ind(sz, i, mix::RHO))),
486 static_cast<double>(st(arma::sub2ind(sz, i, mix::MOM))),
487 static_cast<double>(st(arma::sub2ind(sz, i, mix::ENERGY))), phi);
488 };
489 const CaloricMixtureThermo old_th = decode(before);
490 const CaloricMixtureThermo new_th = decode(after);
491 e_old[i] = old_th.internal_energy;
492 e_now[i] = new_th.internal_energy;
493 temperature[i] = new_th.T;
494 conductivity[i] = mixture_conductivity(new_th.n_e, new_th.n_HI,
495 new_th.T);
496 }
497 const CaloricMixtureThermo inner = decode_ghost(
498 grid, grid.mix_inner_boundary0, grid.phi_g_imh(0));
499 const CaloricMixtureThermo outer = decode_ghost(
500 grid, grid.mix_outer_boundary0, grid.phi_g_iph(ns-1));
501 const double k_inner = mixture_conductivity(inner.n_e, inner.n_HI, inner.T);
502 const OuterConductionWall wall = outer_conduction_wall(grid, outer);
503 const double k_outer = mixture_conductivity(wall.n_e, wall.n_HI, wall.T);
504 auto face_k = [&](double kh, double kt, double dh, double dtw) {
505 if (grid.uniform_mesh) return 0.5*(kh+kt);
506 kh=std::max(kh,1.0e-30); kt=std::max(kt,1.0e-30);
507 return (dh+dtw)/(dh/kh+dtw/kt);
508 };
509 double maximum = 0.0;
510 for (arma::uword i = 0; i < ns; ++i) {
511 const double kl = i == 0
512 ? face_k(conductivity[i],k_inner,grid.ds_i(i),grid.ds_i(i))
513 : face_k(conductivity[i],conductivity[i-1],grid.ds_i(i),grid.ds_i(i-1));
514 const bool physical_outer_face =
515 i+1 == ns && grid.outer_conduction_temperature_override;
516 const double kr = i+1 == ns
517 ? (physical_outer_face
518 ? k_outer
519 : face_k(conductivity[i],k_outer,grid.ds_i(i),grid.ds_i(i)))
520 : face_k(conductivity[i],conductivity[i+1],grid.ds_i(i),grid.ds_i(i+1));
521 double gl = grid.B_i(i)/grid.ds_i(i)*kl/grid.B_imh(i)/grid.ds_imh_i(i);
522 const double ds_right = physical_outer_face
523 ? 0.5*static_cast<double>(grid.ds_i(i))
524 : static_cast<double>(grid.ds_iph_i(i));
525 double gr = grid.B_i(i)/grid.ds_i(i)*kr/grid.B_iph(i)/ds_right;
526 const double tl = i ? temperature[i-1] : inner.T;
527 const double tr = i+1 < ns ? temperature[i+1] : wall.T;
528 double divergence = gr*(tr-temperature[i])-gl*(temperature[i]-tl);
529 if (i == 0 && grid.inner_conduction_neumann)
530 divergence += gl*(temperature[i]-tl);
531 if (i+1 == ns && grid.impose_outer_heat_flux)
532 divergence = -gl*(temperature[i]-tl)+grid.outer_heat_flux/grid.ds_i(i);
533 const double residual = e_now[i] - e_old[i] - dt*divergence;
534 maximum = std::max(maximum, std::abs(residual)/std::max(e_old[i],1.0e-30));
535 }
536 return maximum;
537}

540// Release timestep
541// ============================================================================
542
543Vec mixture_advance(Grid& grid, const Vec& state, const Vec& dt_i,
544 const MixtureField& decoded) {
545 if (grid.eos_gamma_table.empty())
546 throw std::logic_error("the mixture solver requires a Gamma1 table");
547 decoded.require_matches(grid, state);
548 const double dt = dt_i(0); // uniform by mixture_timestep construction
549
550 Vec next = state;
551 {
552 const Vec rhs = mixture_rhs_explicit(grid, state, decoded, dt);
553 const auto sz = state_size(grid);
554 for (arma::uword i = 0; i < grid.ns; ++i)
555 for (arma::uword row = 0; row < num_of_mixture_eq; ++row) {
556 const arma::uword k = arma::sub2ind(sz, i, row);
557 next(k) = state(k) + dt_i(i)*rhs(k);
558 }
559 }
560 if (grid.enable_conduction)
561 next = mixture_apply_conduction(grid, next, dt);
562 return next;
563}
