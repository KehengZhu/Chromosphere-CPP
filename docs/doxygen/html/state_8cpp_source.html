17Vec get_scalar(const Grid& grid, const Vec& xn_state, arma::uword index) {
18 Vec scalar(grid.ns);
19 for (arma::uword i = 0; i < grid.ns; ++i) {
20 scalar(i) = xn_state(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, index));
21 }
22 return scalar;
23}

25Vec scalar_to(const Grid& grid, const Vec& xn_i, arma::uword index) {
26 Vec xn_state = arma::zeros<Vec>(grid.n_state);
27 for (arma::uword i = 0; i < grid.ns; ++i) {
28 xn_state(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, index)) = xn_i(i);
29 }
30 return xn_state;
31}

33// ============================================================================
34// Index shifts. ip1(xn)(i) = xn(i+1) in the interior; the i = ns-1 slot
35// is filled from outer_boundary0_i (or kept Neumann if USE_NEUMANN_BC).
36// Likewise im1 fills i = 0 from inner_boundary0_i.
37// ============================================================================
38
39Vec ip1(const Grid& grid, const Vec& xn, arma::uword nk) {
40 Vec xn_ip1(arma::size(xn));
41 for (arma::uword i = 0; i < grid.ns - 1; ++i) {
42 for (arma::uword k = 0; k < nk; ++k) {
43 xn_ip1(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, k)) =
44 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i + 1, k));
45 }
46 }
47 for (arma::uword k = 0; k < nk; ++k) {
48 xn_ip1(arma::sub2ind(arma::size(grid.ns, num_of_eq), grid.ns - 1, k)) =
49 grid.outer_boundary0_i[k];
50 if (USE_NEUMANN_BC || nk == SLICE) {
51 xn_ip1(arma::sub2ind(arma::size(grid.ns, num_of_eq), grid.ns - 1, k)) =
52 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), grid.ns - 1, k));
53 }
54 }
55 return xn_ip1;
56}

58Vec im1(const Grid& grid, const Vec& xn, arma::uword nk) {
59 Vec xn_im1(arma::size(xn));
60 for (arma::uword i = 1; i < grid.ns; ++i) {
61 for (arma::uword k = 0; k < nk; ++k) {
62 xn_im1(arma::sub2ind(arma::size(grid.ns, num_of_eq), i, k)) =
63 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), i - 1, k));
64 }
65 }
66 for (arma::uword k = 0; k < nk; ++k) {
67 xn_im1(arma::sub2ind(arma::size(grid.ns, num_of_eq), 0, k)) =
68 grid.inner_boundary0_i[k];
69 if (USE_NEUMANN_BC || nk == SLICE) {
70 xn_im1(arma::sub2ind(arma::size(grid.ns, num_of_eq), 0, k)) =
71 xn(arma::sub2ind(arma::size(grid.ns, num_of_eq), 0, k));
72 }
73 }
74 return xn_im1;
75}

97// ============================================================================
98// Equation of state: cons ↔ prim (writeup eq 38, electron quasi-neutrality
99// folds T_e = T_i and n_e = n_i into the ion pressure with an extra factor of 2).
100// ============================================================================
101
102Vec cons2prim(const Grid& grid, const Vec& cons_state) {
103 Vec prim_state(arma::size(cons_state), arma::fill::zeros);
104 const Vec rho_i = get_scalar(grid, cons_state, cons::RHO_I);
105 const Vec rho_n = get_scalar(grid, cons_state, cons::RHO_N);
106 const Vec rhoV_i = get_scalar(grid, cons_state, cons::MOM_I);
107 const Vec rhoU_n = get_scalar(grid, cons_state, cons::MOM_N);
108 const Vec e_i = get_scalar(grid, cons_state, cons::E_I);
109 const Vec e_n = get_scalar(grid, cons_state, cons::E_N);
110 const Vec e_e = get_scalar(grid, cons_state, cons::E_E);
111
112 const Vec V = rhoV_i / rho_i;
113 const Vec U = rhoU_n / rho_n;
114 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
115 // p_i is the TOTAL charged pressure (protons + electrons); see cons::E_I note.
116 const Vec p_i = grid.gm1() * e_i - grid.half_gm1() * rho_i % V % V - grid.gm1() * rho_i % phi_g;
117 const Vec p_n = grid.gm1() * e_n - grid.half_gm1() * rho_n % U % U - grid.gm1() * rho_n % phi_g;
118 const Vec p_e = grid.gm1() * e_e; // electron partial pressure (no KE / gravity)
119
120 prim_state += scalar_to(grid, rho_i, prim::RHO_I);
121 prim_state += scalar_to(grid, rho_n, prim::RHO_N);
122 prim_state += scalar_to(grid, V, prim::V);
123 prim_state += scalar_to(grid, U, prim::U);
124 prim_state += scalar_to(grid, p_i, prim::P_I);
125 prim_state += scalar_to(grid, p_n, prim::P_N);
126 prim_state += scalar_to(grid, p_e, prim::P_E);
127 return prim_state;
128}

130Vec prim2cons(const Grid& grid, const Vec& prim_state) {
131 Vec cons_state(arma::size(prim_state), arma::fill::zeros);
132 const Vec rho_i = get_scalar(grid, prim_state, prim::RHO_I);
133 const Vec rho_n = get_scalar(grid, prim_state, prim::RHO_N);
134 const Vec V = get_scalar(grid, prim_state, prim::V);
135 const Vec U = get_scalar(grid, prim_state, prim::U);
136 const Vec p_i = get_scalar(grid, prim_state, prim::P_I);
137 const Vec p_n = get_scalar(grid, prim_state, prim::P_N);
138 const Vec p_e = get_scalar(grid, prim_state, prim::P_E);
139
140 const Vec rhoV_i = rho_i % V;
141 const Vec rhoU_n = rho_n % U;
142 const Vec phi_g = 0.5 * (grid.phi_g_imh + grid.phi_g_iph);
143 // E_I is the TOTAL charged energy (p_i already includes the electron pressure
144 // p_e); E_E carries the electron internal energy alone. See cons::E_I note.
145 const Vec e_i = grid.inv_gm1() * p_i + 0.5 * rho_i % V % V + rho_i % phi_g;
146 const Vec e_n = grid.inv_gm1() * p_n + 0.5 * rho_n % U % U + rho_n % phi_g;
147 const Vec e_e = grid.inv_gm1() * p_e;
148
149 cons_state += scalar_to(grid, rho_i, cons::RHO_I);
150 cons_state += scalar_to(grid, rho_n, cons::RHO_N);
151 cons_state += scalar_to(grid, rhoV_i, cons::MOM_I);
152 cons_state += scalar_to(grid, rhoU_n, cons::MOM_N);
153 cons_state += scalar_to(grid, e_i, cons::E_I);
154 cons_state += scalar_to(grid, e_n, cons::E_N);
155 cons_state += scalar_to(grid, e_e, cons::E_E);
156 return cons_state;
157}

163Vec flux_lim(const Vec& r) {
164 Vec one (arma::size(r), arma::fill::ones);
165 Vec zero(arma::size(r), arma::fill::zeros);
166 Vec res = arma::max(zero, arma::min(one, r));
167 res(arma::find_nan(res)).zeros();
168 return res;
169}

171// ============================================================================
172// MC3 / Koren limiter (BATSRUS ModFaceValue 'mc3'): the asymmetric third-order
173// (κ=1/3) monotonized-central limiter. The '+' (right-face) and '−' (left-face)
174// reconstruction terms take DIFFERENT central branches — (2r+1)/3 and (r+2)/3 —
175// which is why minmod's single flux_lim(r) cannot express it. Both clip to the
176// TVD envelope min(β r, β) and floor at 0 (⇒ first order where Δ₊,Δ₋ disagree in
177// sign, i.e. r<0, and at flats where r is non-finite). β = limiter_beta.
178// ============================================================================
179
180Vec flux_lim_mc3_plus(const Vec& r, float beta) {
181 Vec zero(arma::size(r), arma::fill::zeros);
182 Vec bcap(arma::size(r)); bcap.fill(beta);
183 Vec res = arma::min(beta * r, bcap);
184 res = arma::min(res, (2.0f * r + 1.0f) / 3.0f);
185 res = arma::max(zero, res);
186 res(arma::find_nan(res)).zeros();
187 return res;
188}
189
190Vec flux_lim_mc3_minus(const Vec& r, float beta) {
191 Vec zero(arma::size(r), arma::fill::zeros);
192 Vec bcap(arma::size(r)); bcap.fill(beta);
193 Vec res = arma::min(beta * r, bcap);
194 res = arma::min(res, (r + 2.0f) / 3.0f);
195 res = arma::max(zero, res);
196 res(arma::find_nan(res)).zeros();
197 return res;
198}
