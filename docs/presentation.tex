\section{Ionization \& Recombination Mechanisms}
% =====================================================================
\begin{frame}{The chromosphere bridges two regimes}
  \[
    f \equiv \frac{\rho_i}{\rho_\mathrm{tot}}:\quad
    \lesssim10^{-4}\ (T_\mathrm{min}) \;\longrightarrow\; \sim1\ (\text{TR top})
  \]
  \begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \textbf{Lower} — dense, collisional, \alert{LTE / Saha}
    \begin{itemize}
      \item collisional ioniz.\ $\rightleftharpoons$ \alert{three-body recomb.}
      \item $\dfrac{n_e n_i}{n_n}=\Phi(T_e)$
    \end{itemize}
  \end{column}
  \begin{column}{0.5\textwidth}
    \textbf{Mid / upper} — rarefied, \alert{out of equilibrium}
    \begin{itemize}
      \item $\tau_\mathrm{ion}>\tau_\mathrm{dyn}$, $f$ set by \emph{history}
      \item photoioniz.\ $\rightleftharpoons$ radiative recomb.
    \end{itemize}
  \end{column}
  \end{columns}
  \vfill
  \cit{Carlsson \& Stein 2002; Leenaarts et al. 2007}
\end{frame}

\begin{frame}{Ionization channels}
  \[
    \Gamma_\mathrm{ion}
      = \underbrace{n_e n_n\,(S_i+S_\mathrm{CR})}_{\text{collisional}}
      \;+\; \underbrace{n_n\,P_\mathrm{phot}}_{\text{photo}}
  \]
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Direct (Voronov):}
      $S_i=2.91\times10^{-14}\,\dfrac{U^{0.39}e^{-U}}{0.232+U}\ \mathrm{m^3 s^{-1}}$,
      $\;U=\chi_H/k_BT_e$.\\
      $T\!\sim\!5$--$6$\,kK $\Rightarrow U\!\sim\!26$--$31$: $e^{-U}$ tiny, \alert{negligible until TR}.
      \cit{Voronov 1997}
    \item \textbf{Multilevel (Rydberg ladder):} $S_\mathrm{CR}=\alpha_c\,\Phi/n_e$;
      no Boltzmann penalty, $\sim10^{4}$--$10^{5}\times$ ground state $\Rightarrow$ reaches Saha.
      \cit{Bates et al. 1962; Johnson 1972}
    \item \textbf{Photoionization} (Ly$\alpha$+Balmer): \alert{dominant ionizer};
      radiation-fed $\Rightarrow$ no $\chi_H$ drain. \cit{Carlsson \& Stein 2002}
  \end{itemize}
\end{frame}

\begin{frame}{Recombination channels}
  \[
    \Gamma_\mathrm{rec}
      = \underbrace{n_e^2\,\alpha_r}_{\text{radiative}}
      \;+\; \underbrace{n_e^2\,\alpha_c}_{\text{three-body}}
  \]
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Radiative} $\mathrm{H^+}\!+e\to\mathrm H+\gamma$ (case-B):
      $\alpha_r=2.7\times10^{-19}(T_e/10^4)^{-0.75}\ \mathrm{m^3 s^{-1}}$; one Ly-continuum photon lost.
      \cit{Hummer 1994}
    \item \textbf{Three-body} $\mathrm{H^+}\!+e+e\to\mathrm H(n)+e$:
      $\alpha_c\approx 5.6\times10^{-27}(k_BT_e[\mathrm{eV}])^{-9/2}n_e\ \mathrm{cm^3 s^{-1}}$ —
      \alert{$\propto n_e$}, $\sim5\times10^4\times$ ground state.
      \cit{Hinnov \& Hirschberg 1962; Stevefelt et al. 1975}
    \item \textbf{Neutral / molecular} ($\propto n_e n_i n_n,\ n_i n_n^2$):
      $\sim\alpha_r$ only at the dense base; carried as uncertainty.
      \cit{Drawin 1968; Cretu et al. 2022}
  \end{itemize}
\end{frame}

\begin{frame}{Equilibrium: one formulation, two limits}
  $df/dt=0$ (drop $P_\mathrm{phot}$):
  \[
    (1-f)\,S_i = f\big[\alpha_r + \alpha_{3b}\,f\,n_\mathrm{tot}\big]
  \]
  \begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \textbf{Low density} ($\alpha_c\ll\alpha_r$)
    \[ \frac{f}{1-f}=\frac{S_i}{\alpha_r} \]
    kinetic, density-independent.
  \end{column}
  \begin{column}{0.5\textwidth}
    \textbf{High density} ($\alpha_c\gg\alpha_r$), $\alpha_{3b}=S_i/\Phi$
    \[ \boxed{\ \frac{f^2}{1-f}=\frac{\Phi(T_e)}{n_\mathrm{tot}}\ } \]
    \alert{Saha} — thermodynamics.
  \end{column}
  \end{columns}
  \vfill
  Detailed balance level-by-level ($\alpha_{3b}(n)=S_i(n)/\Phi_n$) $\Rightarrow$ the
  $S_\mathrm{CR}=\alpha_c\Phi/n_e$ closure recovers LTE \emph{by construction}.
\end{frame}

\section{Photoionization}
% =====================================================================
\begin{frame}{Photoionization — frozen field, inverted from C7}
  \[
    \mathrm H + \gamma \to \mathrm H^+ + e,\qquad
    \Gamma_\mathrm{ion}^\mathrm{phot}=n_n\,P_\mathrm{phot}
  \]
  \begin{itemize}\setlength\itemsep{3pt}
    \item No radiative transfer: field $\approx$ steady ($T_\mathrm{rad}$ varies $\lesssim1$--$2\%$ through shocks)
      $\Rightarrow$ \alert{freeze $R_{ik}$ as $f(z)$}. \cit{Carlsson \& Stein 2002; Sollum 1999}
  \end{itemize}
  \textbf{C7-context inversion} — make C7 an equilibrium fixed point:
  \[
    P_\mathrm{phot}=\frac{n_p}{n_n}\,(\alpha_r+\alpha_c)\,n_e-(S_i+S_\mathrm{CR})\,n_e,
    \quad n_p=n_\mathrm H-n_\mathrm{H1}\ \ \cit{Avrett \& Loeser 2008, Tab.~26}
  \]
  \begin{itemize}\setlength\itemsep{3pt}
    \item Blend $z\!\gtrsim\!1500$\,km $\to$ frozen FAL-C $R_{ik}(z)$. \cit{Chae 2021}
    \item $R_{ik}:\ 7\times10^{-8}\,\mathrm{s^{-1}}\ (T_\mathrm{min}) \to 7\times10^{-3}\,\mathrm{s^{-1}}$ (top).
  \end{itemize}
\end{frame}

\section{Cooling}
% =====================================================================
\begin{frame}{Energy sinks: ionization potential + radiation}
  \textbf{1. Ionization-potential bookkeeping} ($\chi_H=13.6$\,eV):
  \[
    Q_\chi = \big(\Gamma^\mathrm{col}_\mathrm{rec}-\Gamma^\mathrm{col}_\mathrm{ion}\big)\,\chi_H
    \;\xrightarrow[\text{detailed balance}]{\text{LTE}}\; 0
  \]
  \begin{itemize}\setlength\itemsep{2pt}
    \item collisional ioniz.\ \alert{drains} $e^-$ pool ($-\chi_H$); photoioniz.\ does not (radiation-fed)
    \item three-body recomb.\ \alert{superelastic} ($+\chi_H\to$ heat)
    \item $\Rightarrow$ no spurious sink in the deep chromosphere
  \end{itemize}
  \vfill
  \textbf{2. Radiative recomb.:} one $\chi_H$ Ly-continuum photon lost (optically-thin case-B).
  \vfill
  Cooling \alert{on by default}; \texttt{no-cooling} isolates ionization energetics.
\end{frame}

\begin{frame}{Radiative cooling: optically thick + optically thin}
  \textbf{2a. Optically-thick chromospheric lines} (escape prob.\ $\ll1$)
  \cit{Carlsson \& Leenaarts 2012}
  \[
    Q^X_\mathrm{rad}=L_X(T_e)\,E_X(\tau_X)\,\frac{N_{X^\star}}{N_X}(T_e)\,A_X\,\frac{N_\mathrm H}{\rho}\,n_e\,\rho,
    \qquad X\in\{\mathrm{H\,I,Ca\,II,Mg\,II}\}
  \]
  effective for $T\lesssim3\times10^4$\,K.
  \vfill
  \textbf{2b. Optically-thin TR / coronal loss} \cit{Klimchuk et al. 2008}
  \[
    Q_\mathrm{thin}=n_e n_H\,\Lambda(T),\qquad
    \Lambda_\mathrm{peak}\approx4\times10^{-35}\ \mathrm{W\,m^3}\ (\log T\approx5.1)
  \]
  stitched on above $\sim2\times10^4$\,K $\Rightarrow$ lower TR re-radiates coronal flux $q(T)$.
  \vfill
  Both \alert{implicit} (Stage~R): $\tau_\mathrm{rad}\!\sim\!1$--$10$\,s $<$ CFL step.
\end{frame}

\section{Chromospheric Evaporation}
% =====================================================================
\begin{frame}{Explosive evaporation — the physics to reproduce}
  \cit{Fisher, Canfield \& McClymont 1985}
  \begin{itemize}\setlength\itemsep{4pt}
    \item \textbf{Flux threshold} (heating $>$ radiable at $10^5$\,K peak):
      $F_\mathrm{crit}\approx7\times10^9\,\mathrm{erg\,cm^{-2}s^{-1}}=7\times10^6\,\mathrm{W\,m^{-2}}$
      — above $\to$ \alert{explosive}, below $\to$ gentle.
    \item \textbf{Timescale:} $3kT/Q\lesssim L_0/c_s$ ($\sim$10\,s).
    \item \textbf{Velocity ceiling:} $v\le 2.35\,c_s$ (evaporated, coronal $T$).
    \item \textbf{Driver:} thick-target electron beam $\to$ upflow + condensation recoil.
  \end{itemize}
  \vfill
  \emph{Requires the radiative sink active} — the threshold \alert{is} heating vs.\ radiation.
\end{frame}

\begin{frame}{Where to place the upper boundary: the conduction crossover}
  \begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \centering
    \includegraphics[width=\linewidth]{heat-conductivities-c7-narrow-range.png}
  \end{column}
  \begin{column}{0.48\textwidth}\footnotesize
    Boundary must lie in the \alert{electron-conduction regime} (RTV $q(T)$ valid).
    \[ \kappa_e \propto T_e^{5/2} \]
    \begin{itemize}\setlength\itemsep{3pt}
      \item chromosphere (mostly neutral): $\kappa_n>\kappa_e$
      \item TR ($T$ ramps): $\kappa_e$ overtakes
      \item crossover $\kappa_e=\kappa_n$ at $h\approx2149$\,km (C7)
    \end{itemize}
    \vfill
    \[ \boxed{\,h\approx2153\,\mathrm{km},\ T\approx2.2\times10^4\,\mathrm{K}\,} \]
    first \alert{$\kappa_e\gg\kappa_n$} height $\Rightarrow$ lower-TR/coronal, not chromospheric.
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Setup: boundary, initial, and deposited-energy conditions}
  \begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \textbf{Initial:} Model~C7 column \cit{Avrett \& Loeser 2008}; discrete-HSE base, settle $1\,t_\mathrm{ac}$.
    \par\medskip
    \textbf{Boundaries}
    \begin{itemize}\footnotesize
      \item lower: physical $V=0$ + numerical diffusion
      \item upper: \alert{opened} at crossover ($\kappa_e\!=\!\kappa_n$, $\sim$2153\,km); Neumann $V$, supersonic outflow
      \item coronal flux $q(T)$, RTV (negligible vs.\ beam)
    \end{itemize}
  \end{column}
  \begin{column}{0.5\textwidth}
    \textbf{Beam heating} (volumetric, not a hot wall)
    \begin{itemize}\footnotesize
      \item deposit over $\Delta h_\mathrm{beam}\!\approx\!80$\,km, $\propto n_\mathrm{tot}$, $\int\!=\!F_e$
      \item into $e/i$ pool, just before cooling
      \item $t_\mathrm{on}=2$\,s, $\tau=10$\,s, 1\,s cosine ramps
    \end{itemize}
    \par\medskip
    \textbf{Two runs}
    \begin{itemize}\footnotesize
      \item \alert{Explosive}: $F_e=5\times10^7\,\mathrm{W\,m^{-2}}$
      \item Gentle: $F_e=1\times10^6\,\mathrm{W\,m^{-2}}$
    \end{itemize}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Neutral-profile oscillation — a trace-species effect}
  \[
    T=\frac{p}{2 n k_B}\ \xrightarrow{\ n\to0\ \text{faster than}\ p\ }\ \text{blows up}
  \]
  \textbf{Seen:} evaporation drives $f\to1\Rightarrow$ \alert{neutral becomes trace} ($n_n\to0$);
  $n_n$ oscillates, $n_\mathrm{tot}$ smooth. Both edges:
  \begin{itemize}\setlength\itemsep{2pt}
    \item evaporation: $f\to1$, $n_n\to0$ (trace neutral)
    \item condensation: $f\to0$, $n_i\to0$ (trace ion) — mirror $T_i$ spike
  \end{itemize}
  $n_\mathrm{tot}$ smooth while trace density oscillates $\Rightarrow$ \alert{not} a hydro artifact,
  but trace-species overheating via the cell-local Stage-E solve.
  \vfill
  \textbf{Fix:} downward-only cap of trace pressure to dominant-fluid $T$ (2$\times$ deadband)
  $\Rightarrow$ kills oscillation, preserves $-184$\,km\,s$^{-1}$ flow, never touches $V$.
\end{frame}

\section{Extended Boundary: PFSS Field Lines in the Corona}
% =====================================================================
\begin{frame}{The truncated column: why we outgrow this boundary}
  Old upper BC at the lower TR ($h\approx2153$\,km): three corona-derived closures
  \cit{Rosner et al.\ 1978; Bradshaw \& Emslie 2020}
  \[
    \rho(T)\propto\frac1T,\qquad
    V=M\,c_s(T)\ (M\!\le\!0.05),\qquad
    q=\frac{2}{7}\frac{\kappa_0 T_\mathrm{cor}^{7/2}}{L}\approx90\,\mathrm{W\,m^{-2}}
  \]
  \begin{itemize}\setlength\itemsep{4pt}
    \item $q=\text{const}$ — prescribed, not a dynamic corona; stable only for $q\in[{\sim}30,90]\,\mathrm{W\,m^{-2}}$
    \item open top $\rho V>0$: the evaporation cycle $\oint$ never closes (no filling / draining)
    \item $B=\text{const}\Rightarrow A=\text{const}$ (no geometry);\ \ $\rho_n/\rho\to0$ (stiff TR $\to$ slaving + TRAC)
  \end{itemize}
\end{frame}

\begin{frame}{Extend the domain along observed PFSS field lines}
  Full loop $s\in[0,L]$, boundaries at the \alert{photosphere}; geometry from a PFSS
  extrapolation of AR\,13768. \cit{Schrijver \& De Rosa 2003}
  \[
    B(s)\,A(s)=\text{const}\ \Rightarrow\ A(s)\propto\frac{1}{|B(s)|},\qquad
    \text{closed: }\partial_s X|_\mathrm{apex}=0,\quad \text{open: outflow top}
  \]
  Self-consistent thick-target beam injected at the apex \cit{Brown 1971; Emslie 1978}:
  \[
    \phi(s)\propto n_\mathrm{tot}(s)\Big(1+\frac{N(s)}{N_c}\Big)^{-\delta/2},\quad
    N(s)=\!\int_\mathrm{apex}^{s}\!\! n_\mathrm{tot}\,ds',\quad
    N_c=\frac{E_c^{2}}{2K}
  \]
  $\Rightarrow$ corona resolved; beam heats \emph{both} feet (two-ribbon), loop fills self-consistently.
\end{frame}

\begin{frame}{Three temperatures: $T_e \neq T_i \neq T_n$}
  Add the electron internal energy $\varepsilon_e=\tfrac32 p_e$ (advected at $V$):
  \begin{align*}
    \partial_t\varepsilon_e+\partial_s(\varepsilon_e V)+p_e\,\partial_s V
      &=\partial_s(\kappa_e\,\partial_s T_e)+Q_\mathrm{beam}-n_e n_H\Lambda(T_e)\\
      &\quad-\chi_H\dot{n}_\mathrm{ion}+\nu_{ei}C_e(T_i-T_e)+\nu_{en}C_e(T_n-T_e)
  \end{align*}
  \[
    \kappa_e\propto T_e^{5/2},\qquad
    \nu_{ei}\propto\frac{n_e\ln\Lambda}{T_e^{3/2}},\qquad
    \nu_{ei}\!\to\!\infty\ (\texttt{ENABLE\_TE=0})\Rightarrow T_e=T_i
  \]
  \begin{itemize}\setlength\itemsep{2pt}
    \item beam $\to$ electrons; slow $\nu_{ei}$ at the tenuous loop-top $\Rightarrow T_e\sim3\text{--}8\,T_i$ (onset)
    \item dense chromosphere: $\nu_{ei}$ large $\Rightarrow T_e\to T_i$ as the loop fills
  \end{itemize}
  \cit{Spitzer 1962; Bradshaw 2006; Manchester 2012; Sokolov 2021}
\end{frame}
